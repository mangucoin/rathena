-	script	Ice Cream Maker::IceCreamer	-1,{
	mes "[Ice Cream Maker]";
	mes "Helado fresco hecho con nieve de Lutie!";
	mes "Disfrutalo ahora, no estara en venta por mucho tiempo!";
	mes "^3355FF100 Zeny^000000 el Helado,";
	mes "Helado!";
	next;
	if(select("Dame Helado!:Cancelar Intercambio") == 2) {
		mes "Estas seguro de que no quieres?";
		mes "No lo vendre por mucho tiempo,";
		mes "y una vez que se acabe,";
		mes "no habra mas!!!";
		close;
	}
	mes "[Ice Cream Maker]";
	mes "Helado fresco hecho con nieve de Lutie!";
	mes "Todos quieren nuestro delicioso helado, ";
	mes "pero tenemos una cantidad limitada,";
	mes "asi que solo puedes comprar 5 a la vez!!";
	next;
	while(true) {
		input .@input;
		if(.@input < 1) {
			mes "[Ice Cream Maker]";
			mes "Si no quieres comprar ninguno,";
			mes "podrias por favor dejar que el siguiente cliente";
			mes "haga su compra?";
			mes "Gracias.";
			close;
		}
		if(.@input > 5) {
			mes "[Ice Cream Maker]";
			mes "Auch";
			mes "Esperas demasiado.";
			mes "Querido cliente,";
			mes "Si comes mas de 5 helados,";
			mes "Podrias tener que hacer muchos viajes";
			mes "al bano esta noche.";
			next;
			continue;
		}
		break;
	}
	if(Zeny<.@input*100) {
		mes "[Ice Cream Maker]";
		mes "Querido cliente, tu billetera parece estar ligera.";
		mes "El precio es ^3355FF100 Zeny^000000 por helado.";
		close;
	}
	if(checkweight(536,.@input) == 0) {
		mes "[Ice Cream Maker]";
		mes "Querido cliente, parece que llevas mucho encima.";
		mes "El Helado esta bien,";
		mes "pero debes considerar tu peso";
		mes "antes de hacer una compra.";
		close;
	}
	set Zeny,Zeny-100*.@input;
	getitem 536,.@input;	// Ice_Cream
	close;
}

// Alberta
alberta,120,45,2	duplicate(IceCreamer)	Ice Cream Maker#1	85
