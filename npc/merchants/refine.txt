//============================================================
//= Main Refiner Function
//============================================================
//= To allow auto safe refining/multiple refining set the
//= second argument to '1' in the function call.
//= If you enable this function, be sure to edit the value of
//= .@safe to the max safe refine in refine.yml as well.
//============================================================
function	script	refinemain	{
	disable_items;
	.@npc_name$ = getarg(0);
	.@features = getarg(1);
	mes "["+ .@npc_name$ +"]";
	mes "Soy el Herrero.";
	mes "Puedo refinar todo tipo de armas, armaduras y equipamiento, asi que dime";
	mes "que quieres que refine.";
	next;

	setarray .@indices[1], EQI_HEAD_TOP, EQI_ARMOR, EQI_HAND_L, EQI_HAND_R, EQI_GARMENT, EQI_SHOES, EQI_ACC_L, EQI_ACC_R, EQI_HEAD_MID, EQI_HEAD_LOW;
	for(.@i = 1; .@i<getarraysize(.@indices); ++.@i) {
		if(getequipisequiped(.@indices[.@i])) {
			.@menu$ = .@menu$ + F_getpositionname(.@indices[.@i]) + "-[" + getequipname(.@indices[.@i]) + "]";
			.@equipped = 1;
		}
		.@menu$ = .@menu$ + ":";
	}
	if (.@equipped == 0) {
		mes "["+ .@npc_name$ +"]";
		mes "No creo que pueda refinar ningun objeto que tengas...";
		close;
	}
	.@part = .@indices[select(.@menu$)];

	if(!getequipisequiped(.@part)) { //custom check
		mes "["+ .@npc_name$ +"]";
		mes "No llevas puesto";
		mes "nada ahi que";
		mes "pueda refinar.";
		emotion ET_FRET;
		close;
	}
	//Check if the item is refinable...
	if(!getequipisenableref(.@part)) {
		mes "["+ .@npc_name$ +"]";
		mes "No creo que pueda";
		mes "refinar este objeto...";
		close;
	}
	//Check to see if the items is already +10
	if(getequiprefinerycnt(.@part) >= 10) {
		mes "["+ .@npc_name$ +"]";
		mes "No puedo refinar esto";
		mes "mas. Esta tan refinado";
		mes "como se puede!";
		close;
	}
	.@refineitemid = getequipid(.@part); // save id of the item
	.@refinerycnt = getequiprefinerycnt(.@part); //save refinery count
	setarray .@card[0], getequipcardid(.@part,0), getequipcardid(.@part,1), getequipcardid(.@part,2), getequipcardid(.@part,3);
	.@price = getequiprefinecost(.@part, REFINE_COST_NORMAL, REFINE_ZENY_COST);
	.@material = getequiprefinecost(.@part, REFINE_COST_NORMAL, REFINE_MATERIAL_ID);
	.@itemtype = getiteminfo( .@refineitemid, ITEMINFO_TYPE );

	if( .@itemtype == IT_ARMOR ){
		.@equip_lv = getequiparmorlv( .@part );

		switch( .@equip_lv ){
			case 1:
				.@safe = 4;
				break;
			default:
				// TODO:
				close;
		}

		// If the VIP system is enabled, the prices for non-VIP players are considerably higher.
		if( VIP_SCRIPT && !vip_status(VIP_STATUS_ACTIVE) ){
			switch( .@equip_lv ){
				case 1:
					.@price = .@price * 10;
					break;
				default:
					// TODO:
					close;
			}
		}
	}else if( .@itemtype == IT_WEAPON ){
		.@equip_lv = getequipweaponlv( .@part );

		switch( .@equip_lv ){
			case 1:
				.@safe = 7;
				break;
			case 2:
				.@safe = 6;
				break;
			case 3:
				.@safe = 5;
				break;
			case 4:
				.@safe = 4;
				break;
			default:
				// TODO:
				close;
		}

		// If the VIP system is enabled, the prices for non-VIP players are considerably higher.
		if( VIP_SCRIPT && !vip_status(VIP_STATUS_ACTIVE) ){
			switch( .@equip_lv ){
				case 1:
					.@price = .@price * 40;
					break;
				case 2:
					.@price = .@price * 50;
					break;
				case 3:
					.@price = .@price * 2;
					break;
				case 4:
					.@price = .@price * 2;
					break;
				default:
					// TODO:
					close;
			}
		}
	}else{
		.@safe = 4;
	}

	// TODO: What was this? Maybe shadow?
	// case 5: .@price = .@price * 10; break;

	if(.@features != 1) {
		mes "["+ .@npc_name$ +"]";
		mes "Para refinar esto necesito";
		mes "un ^003366"+getitemname(.@material)+"^000000 y";
		mes "una tarifa de servicio de " + .@price + " Zeny.";
		mes "Realmente deseas continuar?";
		next;
		if(select("Si:No") == 2){
			mes "["+ .@npc_name$ +"]";
			mes "Si...";
			mes "No hay necesidad de";
			mes "apurarse. Tomate tu tiempo.";
			close;
		}
		if(getequippercentrefinery(.@part) < 100) {
			mes "["+ .@npc_name$ +"]";
			mes "Oh no! Si sigo";
			mes "refinando esto, hay un riesgo de que podria";
			switch(.@material) {
			case 985:
				mes "destruirse! Eso significa que ^FF0000este equipo^000000, y ^FF0000cualquier carta^000000 o propiedad especial agregada a esta armadura, ^FF0000desapareceran^000000.";
				break;
			default:
				mes "destruirse, y ^FF0000perderas el arma^000000, cualquier ^FF0000carta en el arma^000000,";
				mes "o cualquier propiedad especial agregada.";
				break;
			}
			next;
			mes "["+getarg(0)+"]";
			mes "No puedo ser mas claro.";
			mes "Una vez que un arma se destruye,";
			mes "no hay forma de recuperarla.";
			mes "Realmente tienes la posibilidad";
			mes "de ^FF0000perder esta arma^000000 para siempre.";
			mes "Aun quieres refinar?";
			next;
			if(select("Si:No") == 2){
				mes "["+ .@npc_name$ +"]";
				mes "Estoy completamente de acuerdo...";
				mes "Puede que sea un gran refinador, pero a veces incluso yo cometo errores.";
				close;
			}
		}
		if((countitem(.@material) < 1) || (Zeny < .@price)) {
			mes "["+ .@npc_name$ +"]";
			mes "No parece que tengas";
			mes "suficiente Zeny o "+getitemname(.@material)+"...";
			mes "Ve a conseguir mas. Estare";
			mes "aqui todo el dia si me necesitas.";
			close;
		}
		Zeny = Zeny-.@price;
		delitem .@material,1;

		// anti-hack
		if (callfunc("F_IsEquipIDHack", .@part, .@refineitemid) || callfunc("F_IsEquipCardHack", .@part, .@card[0], .@card[1], .@card[2], .@card[3]) ||
		    callfunc("F_IsEquipRefineHack", .@part, .@refinerycnt)) {
			mes "["+ .@npc_name$ +"]";
			emotion ET_FRET;
			mes "Espera un segundo...";
			mes "Crees que soy tonto?!";
			mes "Cambiaste el objeto mientras no miraba! Fuera de aqui!";
			close;
		}

		if(getequippercentrefinery(.@part) <= rand(100)) {
			failedrefitem .@part;
			mes "["+ .@npc_name$ +"]";
			emotion (!rand(5))?ET_MONEY:ET_HUK;
			.@lose = rand(1,3);
			if (.@lose == 1) {
				mes "OH! DIOS MIO!";
				mes "Maldicion! No otra vez!";
				mes "Lo siento terriblemente, pero sabes que la practica hace al maestro.";
				mes "Um, verdad? Je je...";
			} else if(.@lose == 2) {
				mes "Nooooooo!";
				mes "Se rompio!";
				mes "L-Lo siento!";
			} else {
				mes "Rayos!";
				mes "No pudo soportar";
				mes "mas templado!";
				mes "Lo siento mucho...";
			}
			close;
		}
		mes "["+getarg(0)+"]";
		successrefitem .@part;
		emotion ET_SMILE;
		.@win = rand(1,3);
		if (.@win == 1) {
			mes "Perfecto!";
			mes "Je je!";
			mes "Una vez mas,";
			mes "un trabajo impecable";
			mes "del maestro~";
		} else if(.@win == 2) {
			mes "Exito...!";
			mes "Una vez mas, mi increible";
			mes "talento verdaderamente deslumbra";
			mes "y brilla hoy.";
		} else {
			mes "Je je!";
			mes "Ya termine.";
			mes "Sin duda, mi trabajo es";
			mes "de tu satisfaccion.";
			mes "Pura y total perfeccion~";
		}
		close;
	}

// New Refining Functions ========================
	if (.@refinerycnt < .@safe) {
		mes "["+ .@npc_name$ +"]";
		mes "Puedo refinar esto hasta el limite seguro o una cantidad deseada de veces. Tu eliges.";
		next;
		.@menu2 = select("Hasta el limite seguro, por favor.","Yo decidire cuantas veces.","Cambie de opinion...");
	} else
		.@menu2 = 2;
	switch(.@menu2){
	case 1:
		.@refinecnt = .@safe - .@refinerycnt;
		break;
	case 2:
		next;
		mes "["+ .@npc_name$ +"]";
		mes "Cuantas veces te gustaria que refinara tu objeto?";
		next;
		input .@refinecnt;
		.@refinecheck = .@refinecnt + .@refinerycnt;
		if (.@refinecnt < 1 || .@refinecheck > 10) {
			mes "["+ .@npc_name$ +"]";
			mes "No puedo refinar este objeto tantas veces.";
			close;
		}
		if(.@refinecheck > .@safe) {
			.@refinecheck = .@refinecheck - .@safe;
			mes "["+ .@npc_name$ +"]";
			mes "Esto intentara refinar el equipo " + .@refinecheck + " veces mas alla del limite seguro. Tu equipo podria destruirse... esta bien?";
			next;
			if(select("Si...","No...") == 2){
				mes "["+ .@npc_name$ +"]";
				mes "Tu lo has dicho... Que asi sea.";
				close;
			}
		}
		break;
	case 3:
		next;
		mes "["+ .@npc_name$ +"]";
		mes "Tu lo has dicho... Que asi sea.";
		close;
	}
	.@fullprice = .@price * .@refinecnt;
	mes "["+ .@npc_name$ +"]";
	mes "Eso te costara " + .@refinecnt + " " + getitemname(.@material) + " y " + .@fullprice + " Zeny. Esta bien?";
	next;
	if(select("Si","No...") == 2){
		mes "["+ .@npc_name$ +"]";
		mes "Tu lo has dicho... Que asi sea.";
		close;
	}
	if(countitem(.@material) < .@refinecnt || Zeny < .@fullprice) {
		mes "["+ .@npc_name$ +"]";
		mes "Es todo lo que tienes? Desafortunadamente no puedo trabajar para ti a un precio mas bajo. Intenta ponerte en mis zapatos.";
		close;
	}
	Zeny = Zeny - .@fullprice;
	delitem .@material,.@refinecnt;
	while(.@refinecnt){
		if (getequipisequiped(.@part) == 0) {
			mes "["+ .@npc_name$ +"]";
			mes "Mira aqui... no tienes ningun objeto puesto...";
			close;
		}
		// anti-hack
		if (callfunc("F_IsEquipIDHack", .@part, .@refineitemid) || callfunc("F_IsEquipCardHack", .@part, .@card[0], .@card[1], .@card[2], .@card[3]) ||
				callfunc("F_IsEquipRefineHack", .@part, .@refinerycnt) || (.@menu2 == 1 && getequippercentrefinery(.@part) < 100)) {
			mes "["+ .@npc_name$ +"]";
			mes "Clang... No, pero imaginaste que podria ser tan tonto?!";
			mes "Lo cambiaste...";
			mes "Fuera antes de que te aturda con mi Martillo!!";
			close;
		}
		mes "Clang, clang!!!";
		if(.@menu2 == 2 && getequippercentrefinery(.@part) <= rand(100)) {
			failedrefitem .@part;
			emotion ET_HUK;
			mes "["+ .@npc_name$ +"]";
			mes "WAHHHH!!! Lo siento tanto... Te adverti que esto podia pasar...";
			.@refinecnt = .@refinecnt - 1;
			if(.@refinecnt == 0) close;
			mes "Aqui tienes el Zeny y los materiales sin usar de vuelta...";
			getitem .@material,.@refinecnt;
			.@fullprice = .@refinecnt * .@price;
			Zeny = Zeny + .@fullprice;
			close;
		}
		successrefitem .@part;
		emotion ET_BEST;
		.@refinecnt = .@refinecnt - 1;
		.@refinerycnt = getequiprefinerycnt(.@part);
		next;
	}
	mes "["+ .@npc_name$ +"]";
	mes "Todo terminado... Vuelve pronto.";
	close;
}

			break;
		}

	if (Zeny < .@sell) {
		mes "["+ .@npc_name$ +"]";
		mes "Err...";
		mes "No tienes";
		mes "suficiente Zeny para comprar";
		mes ""+ .@input +" de ellos.";
		close;

	.@repairprice = 5000;

	mes "["+ .@npc_name$ +"]";
	mes "Hey!";
	mes "Quieres que";
	mes "repare algun objeto?";
	mes "Puedes contar conmigo";
	mes "para reparaciones de objetos!";
	next;
	switch(select("De hecho, si tengo algunos objetos...:Ninguno por el momento.")) {
	case 1:
		.@checkitem = 1;
		while (1) {
			if (getbrokenid(.@checkitem) == 0) {
				break;
			}
			.@checkitem = .@checkitem+1;
		}
		.@checkitem = .@checkitem-1;
		if (!.@checkitem) {
			mes "["+ .@npc_name$ +"]";
			mes "Oh wow, esto es increible!";
			mes "Debes cuidar muy bien tus cosas. Ninguno de tus objetos esta danado!";
			next;
			mes "["+ .@npc_name$ +"]";
			mes "Si todos fueran como tu, me quedare desempleado!! Jaja~!";
			close;
		}
		mes "["+ .@npc_name$ +"]";
		mes "Hmm...";
		mes "Veamos...";
		mes "De todos tus objetos,";
		mes "" + .@checkitem + " estan danados.";
		mes "Te gustaria reparar?";
		next;
		.@totalcost = .@repairprice*.@checkitem;
		mes "["+ .@npc_name$ +"]";
		mes "Cada reparacion cuesta " + .@repairprice + " Zeny. Asi que reparar todos tus objetos danados costaria " + .@totalcost + " Zeny! Te gustaria reparar los objetos?";
		next;
		switch(select("Si:No")) {
		case 1:
			if (Zeny < .@totalcost) {
				mes "["+ .@npc_name$ +"]";
				mes "Para para...";
				mes "Revisa tu billetera antes de recibir la cuenta de reparacion! No puedo reparar nada porque no tienes suficiente Zeny.";
				close;
			}
			.@checkitem2 = 1;
			while (1) {
				if (getbrokenid(.@checkitem2) == 0) {
					break;
				}
				.@checkitem2 = .@checkitem2+1;
			}
			.@checkitem2 = .@checkitem2-1;
			if (.@checkitem == .@checkitem2) {
				Zeny = Zeny-.@totalcost;
				while (.@checkitem) {
					repair(.@checkitem);
					.@checkitem = .@checkitem-1;
				}
				mes "["+ .@npc_name$ +"]";
				mes "Listo! Todo hecho. Ahora, intenta ser un poco mas cuidadoso. Los objetos tambien tienen vida, sabes.";
				close;
			}
			else {
				mes "["+ .@npc_name$ +"]";
				mes "Mmm? Algo esta mal. Espera... Equipa los objetos que necesitas reparar y luego vuelve a verme.";
				close;
			}
		case 2:
			mes "["+ .@npc_name$ +"]";
			mes "Bueno, no es asunto mio, pero no es bueno dejar objetos danados. Deberias repararlos lo mas pronto posible!";
			close;
		}
	case 2:
		mes "["+ .@npc_name$ +"]";
		mes "Jojojo...";
		mes "No tienes";
		mes "ningun asunto conmigo";
		mes "si no tienes ningun";
		mes "objeto que reparar.";
		close;
	}
}
