// PecoPeco Breeder (for Knights) :: breeder
//============================================================
prontera,55,350,5	script	Peco Peco Breeder#knt	105,{
	if(Upper==0) set .@price,2500; //Normal Peco - default price
	if(Upper==1) set .@price,2500; //Armored Peco
	if(Upper==2) set .@price,2500; //Baby Peco

	mes "[Peco Peco Breeder]";
	if (BaseJob == Job_Knight && Class < Job_Rune_Knight) {
		mes "Bienvenido.";
		mes "Honorable Knight,";
		mes "te gustaria alquilar";
		mes "un Peco Peco? La tarifa de";
		mes "alquiler es "+.@price+" zeny.";
		next;
		switch(select("Alquilar Peco Peco:Cancelar")) {
		case 1:
			if (Zeny < .@price) {
				mes "[Peco Peco Breeder]";
				mes "No tienes";
				mes "suficiente zeny.";
				mes "Acaso estas...";
				mes "en bancarrota?";
				close;
			}
			else if(getskilllv("KN_RIDING") == 0) {
				mes "[Peco Peco Breeder]";
				mes "Lo siento, pero no";
				mes "eres elegible para este";
				mes "servicio. Por favor ve a aprender";
				mes "la habilidad Peco Peco Ride primero.";
				close;
			}
			else if(checkriding()) {
				mes "[Peco Peco Breeder]";
				mes "Ya estas";
				mes "montado en un";
				mes "Peco Peco.";
				close;
			}
			else if(ismounting()) {
				mes "[Peco Peco Breeder]";
				mes "Por favor quita tu montura de cash.";
				close;
			}
			set Zeny, Zeny-.@price;
			setriding;
			close;
		case 2:
			mes "[Peco Peco Breeder]";
			mes "Ya veo.";
			mes "Bueno entonces,";
			mes "que tengas un buen dia.";
			close;
		}
	} else {
		mes "Lo siento, pero estos";
		mes "Peco Pecos solo estan";
		mes "disponibles para Knights";
		mes "y Lord Knights.";
		close;
	}
}

// Grand PecoPeco Breeder (for Crusaders)
//============================================================
prontera,232,318,3	script	Peco Peco Breeder#cru	105,{
	if(Upper==0) set .@price,3500; //Normal Peco - default price
	if(Upper==1) set .@price,3500; //Armored Peco
	if(Upper==2) set .@price,3500; //Baby Peco

	mes "[Peco Peco Breeder]";
	if (BaseJob == Job_Crusader && Class < Job_Rune_Knight) {
		if(Upper != 1 ) mes "Bienvenido, Crusader.";
		else mes "Bienvenido, Paladin.";
		mes "Tenemos un Peco Peco";
		mes "especial preparado";
		mes "para ti. Alquilar uno";
		mes "costara "+.@price+" zeny.";
		next;
		switch(select("Alquilar un PecoPeco:Salir")) {
		case 1:
			if (Zeny < .@price) {
				mes "[Peco Peco Breeder]";
				mes "No tienes";
				mes "suficiente zeny.";
				mes "Si quieres";
				mes "un Peco Peco por favor";
				mes "trae "+.@price+" zeny...";
				close;
			}
			else if(getskilllv("KN_RIDING") == 0) {
				mes "[Peco Peco Breeder]";
				mes "Primero debes aprender";
				mes "a montar un PecoPeco antes";
				mes "de que pueda alquilarte uno.";
				close;
			}
			else if(checkriding()) {
				mes "[Peco Peco Breeder]";
				mes "Ya estas";
				mes "montado en un Peco Peco.";
				close;
			}
			else if(ismounting()) {
				mes "[Peco Peco Breeder]";
				mes "Por favor quita tu montura de cash.";
				close;
			}
			set Zeny, Zeny-.@price;
			setriding;
			close;
		case 2:
			mes "[Peco Peco Breeder]";
			mes "Nos vemos.";
			close;
		}
	} else {
		mes "Que puedo hacer por ti?";
		mes "Ten en cuenta que";
		mes "este servicio de alquiler";
		mes "de Peco Peco es estrictamente para";
		mes "Crusaders y Paladins.";
		close;
	}
}

// Falcon Master :: breeder
//============================================================
hu_in01,381,304,5	script	Falcon Breeder#hnt	105,{
	if(Upper==0) set .@price,2500; //Normal Falcon - default price
	if(Upper==1) set .@price,2500; //Scarf Falcon
	if(Upper==2) set .@price,2500; //Baby Falcon

	mes "[Falcon Breeder]";
	if (BaseJob == Job_Hunter) {
		if (checkwug()) {
			mes "Um...";
			mes "No puedes alquilar un Falcon";
			mes "hasta que despidas a tu warg primero!";
			close;
		}
		mes "Necesitas un Falcon?";
		mes "Puedes alquilar tu propia";
		mes "ave de presa confiable por una";
		mes "tarifa de solo "+.@price+" zeny~";
		next;
		switch(select("Alquilar Falcon:Cancelar")) {
		case 1:
			if (checkwug()) {
				mes "[Falcon Breeder]";
				mes "Um...";
				mes "No puedes alquilar un Falcon";
				mes "hasta que despidas a tu warg primero!";
				close;
			}
			if (Zeny < .@price) {
				mes "[Falcon Breeder]";
				mes "Que es esto?";
				mes "No tienes";
				mes "suficiente zeny?!";
				mes "Mejor empieza a";
				mes "cazar dinero";
				mes "en vez de monstruos~";
				close;
			}
			else if(getskilllv("HT_FALCON") == 0) {
				mes "[Falcon Breeder]";
				mes "Cielos~";
				mes "Ve a aprender como";
				mes "manejar un Falcon";
				mes "primero! No puedo alquilarte uno";
				mes "si no puedes";
				mes "manejarlo, sabes.";
				close;
			}
			else if(checkfalcon()) {
				mes "[Falcon Breeder]";
				mes "Um...";
				mes "Ya tienes";
				mes "un Falcon. Esta justo";
				mes "ahi, no puedes verlo?";
				close;
			}
			set Zeny, Zeny-.@price;
			setfalcon;
			close;
		case 2:
			mes "[Falcon Breeder]";
			mes "E-espera, a donde";
			mes "vas? Estos";
			mes "Falcons son de primera,";
			mes "te lo garantizo! Vamos, oye~";
			close;
		}
	} else {
		mes "Joven tonto!";
		mes "Los Falcons solo pueden";
		mes "ser usados por Hunters";
		mes "y Snipers, entendiste?";
		mes "...Je je, celoso?";
		close;
	}
}
