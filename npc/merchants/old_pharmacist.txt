alberta_in,16,28,4	script	Pharmacist	61,{
	if (checkweight(1201,1) == 0) {
		mes "- Espera un momento! -";
		mes "- Actualmente estas llevando -";
		mes "- demasiados objetos contigo. -";
		mes "- Por favor vuelve despues -";
		mes "- de guardar algunos objetos en el almacen Kafra. -";
		close;
	}
	mes "[Old Pharmacist]";
	mes "Ummmm...";
	mes "Que te trae por aqui...?";
	next;
	switch(select("Hacer Pocion:Hablar.:Informacion de Mezcla:Cancelar")) {
	case 1:
		if (MaxWeight - Weight < 5000) {
			mes "[Old Pharmacist]";
			mes "Por que llevas tantas cosas!";
			mes "No seas codicioso, lleva solo lo que necesites!";
			next;
			mes "[Old Pharmacist]";
			mes "Estas demasiado pesado para recibir pociones de mi...";
			mes "Ve a guardar algunos objetos en tu almacen primero!";
			close;
		} else {
			mes "[Old Pharmacist]";
			mes "Tienes todo el material listo, verdad? Cual te gustaria?";
			next;
			switch(select("Red Potion.:Orange Potion.:Yellow Potion.:White Potion.:Blue Potion.:Green Potion.:En realidad, no quiero nada.")) {
			case 1:
				callsub L_Making,507,3,501;
			case 2:
				mes "[Old Pharmacist]";
				if (countitem(507) < countitem(713)) {
					set .@max,countitem(507);
				}
				else if (countitem(508) < countitem(713)) {
					set .@max,countitem(508);
				}
				else {
					set .@max,countitem(713);
				}
				if ((countitem(507) < 1) || (countitem(508) < 1) || (countitem(713) == 0)) {
					mes "Sinverguenza! Que esperabas?! Viniendo aqui sin nada. Tsk!";
					mes "Largate!";
					close;
				}
				if (Zeny < 3) {
					mes "Sinverguenza! Que esperabas?! Viniendo aqui sin nada. Tsk!";
					mes "Largate!";
					close;
				}
				mes "Cuantas?";
				next;
				switch(select("Hacer tantas como pueda.:Quiero elegir la cantidad.:En realidad, no quiero nada.")) {
				case 1:
					if ((countitem(507) < .@max) || (countitem(508) < .@max) || (countitem(713) < .@max) || (Zeny < .@max*3)) {
						mes "[Old Pharmacist]";
						mes "Sinverguenza! Ni siquiera tienes todos los materiales y quieres que te haga pociones?!";
						close;
					}
					set Zeny,Zeny-(.@max*5);
					delitem 507,.@max; //Red_Herb
					delitem 508,.@max; //Yellow_Herb
					delitem 713,.@max; //Empty_Bottle
					getitem 502,.@max; //Orange_Potion
					break;
				case 2:
					mes "[Old Pharmacist]";
					mes "Entonces elige un numero menor a 100. Si no quieres ninguna, solo ingresa '0'. Con los materiales que tienes, puedes hacer unas "+.@max+" pociones.";
					next;
					input .@amount,0,101;
					if (.@amount == 0) {
						mes "[Old Pharmacist]";
						mes "Decidete de una vez!";
						close;
					}
					if (.@amount > 100) {
						mes "[Old Pharmacist]";
						mes "Estas sordo? Dije menos de 100!";
						close;
					}
					if ((countitem(507) < .@amount) || (countitem(508) < .@amount) || (countitem(713) < .@amount) || (Zeny < .@amount*3)) {
						mes "[Old Pharmacist]";
						mes "Sinverguenza! Ni siquiera tienes todos los materiales y quieres que te haga pociones?!";
						close;
					}
					next;
					set Zeny,Zeny-(.@amount*5);
					delitem 507,.@amount; //Red_Herb
					delitem 508,.@amount; //Yellow_Herb
					delitem 713,.@amount; //Empty_Bottle
					getitem 502,.@amount; //Orange_Potion
					break;
				case 3:
					mes "[Old Pharmacist]";
					mes "Que?!";
					mes "Grrr...";
					mes "Bah!";
					mes "Largate!";
					close;
				}
				mes "[Old Pharmacist]";
				mes "Aqui tienes. Esta todo listo asi que puedes llevartelo. Pero recuerda! Abusar de la medicina no es bueno.";
				close;
			case 3:
				callsub L_Making,508,10,503;
			case 4:
				callsub L_Making,509,20,504;
			case 5:
				callsub L_Making,510,30,505;
			case 6:
				callsub L_Making,511,3,506;
			case 7:
				mes "[Old Pharmacist]";
				mes "Que?!";
				mes "Grrr...";
				mes "Bah!";
				mes "Largate!";
				close;
			}
		}
	case 2:
		mes "[Old Pharmacist]";
		mes "Con la medicina, puedes aumentar la capacidad de regeneracion de una persona. Pero, solo son buenos hasta cierto punto. *Suspiro* Estoy empezando a pensar en los dias cuando era joven. Debo estar envejeciendo.";
		next;
		mes "[Old Pharmacist]";
		mes "De todas formas, una pocion es simplemente una pocion. Nada mas y nada menos.";
		close;
	case 3:
		mes "[Old Pharmacist]";
		mes "Hrrrmm...";
		mes "Ustedes los jovenes pueden ser bastante molestos. Pero, ya que preguntaste, te explicare.";
		next;
		mes "[Old Pharmacist]";
		mes "Las hierbas funcionan bien por si solas, pero si usas mis tecnicas y habilidades especiales para hacer pociones con ellas, el efecto es mucho mucho mayor.";
		next;
		mes "[Old Pharmacist]";
		mes "Si pides con ganas y educadamente, las hare para ti. Pero, no gratis... No te preocupes, solo cobro una pequena tarifa, asi que no es tan caro.";
		next;
		mes "[Old Pharmacist]";
		mes "Red Potion - ^0098E52 Red Herbs, 1 Empty Bottle, 2 zeny de tarifa.^000000";
		mes "Orange Potion - ^0098E51 Red Herb, 1 Yellow Herb, 1 Empty Bottle, 5 zeny de tarifa.^000000";
		mes "Yellow Potion - ^0098E52 Yellow Herbs, 1 Empty Bottle, 10 zeny.^000000";
		next;
		mes "[Old Pharmacist]";
		mes "White Potion - ^0098E52 White Herbs, 1 Empty Bottle, 20 zeny de tarifa.^000000";
		mes "Blue Potion - ^0098E52 Blue Herbs, 1 Empty Bottle, 30 zeny de tarifa.^000000";
		mes "Green Potion - ^0098E52 Green Herbs, 1 Empty Bottle, 3 zeny de tarifa.^000000";
		close;
	case 4:
		mes "[Old Pharmacist]";
		mes "Que persona tan aburrida. Si tienes algo que decir, por que no lo dices?!";
		close;
	}

L_Making:
	.@item_req = getarg(0);
	.@req_amount = getarg(1);

	mes "[Old Pharmacist]";
	if ((countitem(.@item_req)/2) < countitem(713)) {
		.@max = countitem(.@item_req) /2;
	} else {
		.@max = countitem(713);
	}
	if (countitem(.@item_req) < 2 || countitem(713) == 0) {
		mes "Sinverguenza! Que esperabas?! Viniendo aqui sin nada. Tsk!";
		mes "Largate!";
		close;
	}
	if (Zeny < 3) {
		mes "Sinverguenza! Que esperabas?! Viniendo aqui sin nada. Tsk!";
		mes "Largate!";
		close;
	}
	mes "Cuantas?";
	next;
	switch(select("Hacer tantas como pueda.:Quiero elegir la cantidad.:En realidad, no quiero nada.")) {
	case 1:
		if ((countitem(.@item_req) < .@max*2) || countitem(713) < .@max || (Zeny < .@max*.@req_amount)) {
			mes "[Old Pharmacist]";
			mes "Sinverguenza! Ni siquiera tienes todos los materiales y quieres que te haga pociones?!";
			close;
		}
		Zeny = Zeny - (.@max*.@req_amount);
		delitem .@item_req,.@max*2;
		delitem 713,.@max; //Empty_Bottle
		getitem getarg(2),.@max;
		break;
	case 2:
		mes "[Old Pharmacist]";
		mes "Entonces elige un numero menor a 100. Si no quieres ninguna, solo ingresa '0'. Con los materiales que tienes, puedes hacer unas "+.@max+" pociones.";
		next;
		input .@amount;
		if (.@amount == 0) {
			mes "[Old Pharmacist]";
			mes "Decidete de una vez!";
			close;
		}
		if (.@amount > 100) {
			mes "[Old Pharmacist]";
			mes "Estas sordo? Dije menos de 100!";
			close;
		}
		if ((countitem(.@item_req) < .@amount*2) || countitem(713) < .@amount || (Zeny < .@amount*.@req_amount)) {
			mes "[Old Pharmacist]";
			mes "Sinverguenza! Ni siquiera tienes todos los materiales y quieres que te haga pociones?!";
			close;
		}
		Zeny = Zeny - (.@max*.@req_amount);
		delitem .@item_req,.@amount*2;
		delitem 713,.@amount; //Empty_Bottle
		getitem getarg(2),.@amount;
		break;
	case 3:
		mes "[Old Pharmacist]";
		mes "Que?!";
		mes "Grrr...";
		mes "Bah!";
		mes "Largate!";
		close;
	}
	mes "[Old Pharmacist]";
	mes "Aqui tienes. Esta todo listo asi que puedes llevartelo. Pero recuerda! Abusar de la medicina no es bueno.";
	close;
}
