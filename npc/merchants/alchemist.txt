alde_alche,24,188,3	script	Guild Dealer	740,{
	if (checkweight(1201,1) == 0) {
		mes "- Espera un momento! -";
		mes "- Actualmente estas llevando -";
		mes "- demasiados objetos contigo. -";
		mes "- Por favor vuelve -";
		mes "- despues de guardar algunos objetos en el almacen Kafra. -";
		close;
	}
	mes "[Gever Al Sharp]";
	mes "Bienvenido a la";
	mes "Union de Alquimistas.";
	mes "En que puedo ayudarte hoy?";
	next;
	switch(select("Comprar materiales.:Comprar un manual de produccion.:Salir.")) {
	case 1:
		mes "[Gever Al Sharp]";
		mes "Que te gustaria?";
		next;
		if (select("Medicine Bowl - 8 Zeny:Cancelar.") == 1) {
			mes "[Gever Al Sharp]";
			mes "Cuantos quieres?";
			mes "Ingresa '0' si quieres salir.";
			next;
			while(1) {
				input .@input,0,2001;
				if (.@input == 0) {
					mes "[Gever Al Sharp]";
					mes "El trato fue cancelado.";
					mes "Vuelve en otra ocasion.";
					close;
				}
				else if (.@input < 1 || .@input > 2000) {
					mes "[Gever Al Sharp]";
					mes "El numero debe";
					mes "ser menor a 2000.";
					next;
				}
				else {
					break;
				}
			}
			set .@sell,.@input * 8;
			set .@item_weight,.@input * 10;
			if (Zeny < .@sell) {
				mes "[Gever Al Sharp]";
				mes "No tienes";
				mes "suficiente zeny.";
				mes "Revisa cuanto zeny";
				mes "tienes primero.";
				close;
			}
			if ((MaxWeight - Weight) < .@item_weight) {
				mes "[Gever Al Sharp]";
				mes "No parece que";
				mes "puedas cargar todo.";
				mes "Por favor revisa el espacio";
				mes "en tu inventario.";
				close;
			}
			set Zeny, Zeny-.@sell;
			getitem 7134,.@input; //Medicine_Bowl
			mes "[Gever Al Sharp]";
			mes "Gracias.";
			mes "Vuelve pronto.";
			close;
		}
		mes "[Gever Al Sharp]";
		mes "Bueno entonces,";
		mes "vuelve cuando";
		mes "necesites comprar";
		mes "materiales relacionados con";
		mes "la Alquimia, de acuerdo?";
		close;
	case 2:
		mes "[Gever Al Sharp]";
		mes "Que necesitas?";
		mes "Los manuales cuestan generalmente 100,000 zeny. Pero hay un par de manuales especiales que costaran mas.";
		next;
		switch(select("Guia de Creacion de Pociones:Guia de Creacion de Alcohol:Guia de Creacion de Bottle Grenade:Guia de Creacion de Acid Bottle:Guia de Creacion de Plant Bottle:Guia de Creacion de Marine Sphere Bottle:Guia de Creacion de Glistening Coat:Guia de Creacion de Condensed Potion:Cancelar Trato.")) {
		case 1:
			callsub S_SellManual,7144,100000; //Normal_Potion_Book
			break;
		case 2:
			callsub S_SellManual,7127,100000; //Alcol_Create_Book
			break;
		case 3:
			callsub S_SellManual,7128,100000; //FireBottle_Create_Book
			break;
		case 4:
			callsub S_SellManual,7129,100000; //Acid_Create_Book
			break;
		case 5:
			callsub S_SellManual,7130,100000; //Plant_Create_Book
			break;
		case 6:
			callsub S_SellManual,7131,100000; //Mine_Create_Book
			break;
		case 7:
			callsub S_SellManual,7132,100000; //Coating_Create_Book
			break;
		case 8:
			callsub S_SellManual,7133,240000; //Slim_Potion_Create_Book
			break;
		case 9:
			mes "[Gever Al Sharp]";
			mes "Bueno entonces...";
			mes "Vuelve si alguna vez";
			mes "necesitas comprar";
			mes "un manual de produccion.";
			close;
		}
	case 3:
		mes "[Gever Al Sharp]";
		mes "Muy bien entonces,";
		mes "que tengas un buen dia.";
		close;
	}

S_SellManual:
	.@item_id = getarg(0);
	.@zeny_req = getarg(1);

	if (Zeny < .@zeny_req) {
		mes "[Gever Al Sharp]";
		mes "No tienes";
		mes "suficiente zeny.";
		mes "Revisa cuanto zeny";
		mes "tienes primero.";
		close;
	}
	mes "[Gever Al Sharp]";
	mes getitemname(.@item_id) +"?";
	mes "Seran "+ .@zeny_req +" zeny.";
	next;
	if (select("Comprar.:Salir.") == 1) {
		Zeny = Zeny - .@zeny_req;
		getitem .@item_id,1;
		mes "[Gever Al Sharp]";
		mes "Gracias por";
		mes "tu compra.";
		close;
	}
	mes "[Gever Al Sharp]";
	mes "Vuelve en";
	mes "otra ocasion.";
	close;
}
