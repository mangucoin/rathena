prontera,165,60,2	script	Apprentice Craftsman	73,{
	disable_items;
	mes "[Apprentice Craftsman]";
	if (Zeny >= 400000) {
		mes "He estado estudiando formas de mejorar una armadura para maximizar su capacidad.";
		next;
		mes "[Apprentice Craftsman]";
		mes "Encantar es una habilidad increible que infunde misteriosos poderes de estado en el socket oculto de la armadura.";
		next;
		mes "[Apprentice Craftsman]";
		mes "Sin embargo, debes tener en cuenta que si hay dos armaduras del mismo tipo en tu posesion, el Encantamiento se aplicara en el orden en que estan colocadas en tu inventario.";
		next;
		mes "[Apprentice Craftsman]";
		mes "En ese caso, el Encantamiento podria aplicarse a un objeto que no querias Encantar. Asi que solo trae ^5555ffUNA Armadura^000000 que quieras encantar para estar seguro...";
		next;
		mes "[Apprentice Craftsman]";
		mes "No me hago responsable de lo que pasaria si tienes mas de una del mismo tipo en tu inventario.";
		next;
		switch(select("Armadura Sin Slot.:Armadura Con Slot.:Armadura de Alto Grado.:Quiza la proxima vez.")) {
		case 1:
			setarray .@Items[0],2307,2309,2314,2316,2321,2325,2327,2328,2330,2332,2334,2335,2341,2344,2346,2348,2350,2337,2386,2394,2395,2396;
			set .@j,50;
			break;
		case 2:
			setarray .@Items[0],2311,2318,2319,2320,2308,2310,2315,2317,2322,2324,2326,2331,2333,2336,2342,2345,2347,2349,2351;
			set .@j,55;
			set .@k,1;
			break;
		case 3:
			setarray .@Items[0],2364,2365,2391,2374,2375,2376,2377,2378,2379,2380,2381,2382,2387,2388,2389,2390;
			set .@j,60;
			break;
		case 4:
			mes "[Apprentice Craftsman]";
			mes "Por favor vuelve cuando tengas interes en encantar tu armadura.";
			close;
		}
		set .@menu$,"";
		for(set .@i,0; .@i<getarraysize(.@Items); set .@i,.@i+1)
			set .@menu$, .@menu$+getitemname(.@Items[.@i])+((.@k)?" [1]":"")+":";
		callsub S_EnchantArmor, .@Items[select(.@menu$)-1], .@j;
		end;
	}
	mes "Estoy a cargo de Encantar Armaduras. En pocas palabras, he estado estudiando formas de potenciar armaduras.";
	next;
	mes "[Apprentice Craftsman]";
	mes "Si por casualidad quieres encantar tu armadura, traeme 400,000 zeny y la armadura que quieras encantar y listo.";
	close;

S_EnchantArmor:
	set .@itemid,getarg(0);
	set .@failrate,getarg(1);
	mes "[Apprentice Craftsman]";
	if (countitem(.@itemid) == 1) {
		mes "El encantamiento de socket te costara 400,000 zeny. Y se encantara una opcion aleatoria. Por supuesto, hay una posibilidad de que tu armadura se rompa.";
		next;
		mes "[Apprentice Craftsman]";
		mes "Primero y mas importante.";
		mes "^ff5555El Nivel de Refinamiento Existente de la Armadura";
		mes "y las Cartas se PERDERAN.^000000";
		mes "Aun asi quieres intentar un Encantamiento?";
		next;
		if(select("Hmm... Dejame pensarlo.:Adelante.") == 1) {
			mes "[Apprentice Craftsman]";
			mes "Bueno, no puedo culparte. La seguridad primero, eh?";
			mes "Que tengas un buen dia.";
			close;
		}
		mes "[Apprentice Craftsman]";
		mes "Todo un aventurero, eh? Bueno, empezamos?";
		close2;
		specialeffect2 EF_MAPPILLAR;
		if (Zeny < 400000) {
			mes "[Apprentice Craftsman]";
			mes "Lo siento, pero no tienes suficiente zeny.";
			close;
		}
		progressbar "ffff00",7;
		set Zeny, Zeny-400000;
		delitem .@itemid,1;
		switch (rand(1,.@failrate)) {
			case 1: set .@addpart,4702;break;
			case 2: set .@addpart,4712;break;
			case 3: set .@addpart,4722;break;
			case 4: set .@addpart,4732;break;
			case 5: set .@addpart,4742;break;
			case 6: set .@addpart,4752;break;
			case 7:
			case 8: set .@addpart,4701;break;
			case 9:
			case 10: set .@addpart,4711;break;
			case 11:
			case 12: set .@addpart,4721;break;
			case 13:
			case 14: set .@addpart,4731;break;
			case 15:
			case 16: set .@addpart,4741;break;
			case 17:
			case 18: set .@addpart,4751;break;
			case 19:
			case 20:
			case 21: set .@addpart,4700;break;
			case 22:
			case 23:
			case 24: set .@addpart,4710;break;
			case 25:
			case 26:
			case 27: set .@addpart,4720;break;
			case 28:
			case 29:
			case 30: set .@addpart,4730;break;
			case 31:
			case 32:
			case 33: set .@addpart,4740;break;
			case 34:
			case 35:
			case 36: set .@addpart,4750;break;
			default:
				specialeffect2 EF_PHARMACY_FAIL;
				mes "[Apprentice Craftsman]";
				mes "Bueno, que lastima.";
				mes "El equipo solicitado fallo al encantarse.";
				close;
		}
		getitem2 .@itemid, 1, 1, 0, 0, 0, 0, 0, .@addpart;
	}
	else {
		mes "Hmm? No hay nada que encantar!";
		mes "Por favor vuelve con solo UN equipo para encantar.";
		close;
	}
}
