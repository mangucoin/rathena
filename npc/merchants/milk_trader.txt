	if (checkweight(1201,1) == 0) {
		mes "^3355FFUn momento!";
		mes "No puedo ofrecerte ninguno de";
		mes "mis servicios porque";
		mes "estas cargando demasiadas";
		mes "cosas. Guarda tus objetos extra en";
		mes "el Almacen Kafra y vuelve~";
		close;
	}
	mes "[Milk Vendor]";
	mes "Hey, hey...";
	mes "Si me traes";
	mes "1 Empty Bottle y";
	mes "15 Zeny, te los intercambio";
	mes "por 1 Milk. Que";
	mes "te parece?";
	next;
	if (select("Intercambiar todas las botellas vacias.:Cancelar") == 1) {
		if (countitem(713) <= 0) {
			mes "[Milk Vendor]";
			mes "Hey...";
			mes "No tienes";
			mes "ninguna Empty Bottle.";
			mes "Realmente no puedo darte";
			mes "esta leche de otra";
			mes "forma, sabes...";
			close;
		}
		set .@bottles,countitem(713);
		set .@total_weight,.@bottles * 50;
		set .@total_cost,.@bottles * 15;
		if (Zeny < .@Total_cost) {
			mes "[Milk Vendor]";
			mes "Oh, wow~!";
			mes "No tienes suficiente";
			mes "zeny para intercambiar todas";
			mes "estas Empty Bottles por";
			mes "Milk. Necesitas tener";
			mes "al menos " + .@total_cost + " zeny.";
			close;
		}
		if (MaxWeight - Weight < .@total_weight) {
			mes "[Milk Vendor]";
			mes "Hmm...";
			mes "Podrias hacer";
			mes "un poco mas de espacio";
			mes "en tu inventario";
			mes "antes de que te de";
			mes "toda esta leche?";
			close;
		}
		set Zeny, Zeny-.@total_cost;
		delitem 713,.@bottles; //Empty Bottles
		getitem 519,.@bottles; //Milk
		close;
	}
	close;
}
