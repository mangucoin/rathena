prontera,123,102,5	script	Merchant#pron	85,{
	if (MaxWeight - Weight < 6301 || checkweight(1201,1) == 0) {
		mes "[Merchant]";
		mes "Jaja!";
		mes "Que eres, superhumano?";
		mes "Llevas demasiadas cosas encima!";
		mes "Mejor guarda algo de eso";
		mes "en el Almacen Kafra~";
		close;
	}
	mes "[Merchant]";
	mes "Buen dia!";
	mes "Has recolectado";
	mes "objetos como Shells o Fluff?";
	mes "Que tal Jellopies? Oh si,";
	mes "los necesito para algo.";
	next;
	mes "[Merchant]";
	mes "Por supuesto, no te pedire";
	mes "que me des eso gratis. Que tal si intercambio una Red Potion por 5 Shells, 10 Fluff,";
	mes "o 10 Jellopies?";
	next;
	switch(select("Claro!:Creo que es un robo!:No, gracias.")) {
	case 1:
		mes "[Merchant]";
		mes "Entonces que objeto";
		mes "quieres traerme?";
		mes "Shells, Fluffs, o Jellopies?";
		next;
		switch(select("Shells:Jellopies:Fluff:Cancelar")) {
		case 1: callfunc "F_PotExchange",935,5;
		case 2: callfunc "F_PotExchange",909,10;
		case 3: callfunc "F_PotExchange",914,10;
		case 4:
			mes "[Merchant]";
			mes "Claro~";
			mes "No hay problema.";
			close;
		}
	case 2:
		mes "[Merchant]";
		mes "Un robo...?";
		mes "Si revisas el valor de mercado";
		mes "de los objetos intercambiados, en realidad yo soy el que sale";
		mes "perdiendo aqui.";
		close;
	case 3:
		mes "[Merchant]";
		mes "Esta bien,";
		mes "no hay problema.";
		mes "Pero vuelve a verme";
		mes "si cambias de opinion.";
		close;
	}
}

function	script	F_PotExchange	{
	.@item_req = getarg(0);
	.@req_amount = getarg(1);

	if (countitem(.@item_req) < .@req_amount) {
		mes "[Merchant]";
		mes "Hey, donde estan";
		mes "todos esos "+ getitemname(.@item_req) +"";
		mes "que prometiste?";
		mes "Dame "+ getitemname(.@item_req) +"!";
		close;
	}
	else {
		mes "[Merchant]";
		mes "Ok, dejame revisar";
		mes "cuantos "+ getitemname(.@item_req) +" tienes";
		mes "contigo. Hmm...";
		next;
		mes "[Merchant]";
		mes "Tienes";
		mes "un total de "+ countitem(.@item_req) +" "+ getitemname(.@item_req) +".";
		mes "Puedo darte un total";
		mes "de "+ (countitem(.@item_req)/.@req_amount) +" Red Potions por esos.";
		next;
		mes "[Merchant]";
		mes "Que dices?";
		mes "Tenemos un trato?";
		next;
		if (select("Trato.:No hay trato.") == 1) {
			mes "[Merchant]";
			mes "Sabes el numero exacto";
			mes "de Red Potions que";
			mes "quieres recibir por";
			mes "esos "+ getitemname(.@item_req) +", verdad?";
			next;
			mes "[Merchant]";
			mes "Ahora, puedo darte";
			mes "un minimo de 1 Red Potion";
			mes "y un maximo de 100 Red Potions";
			mes "a la vez. Si cambias de";
			mes "opinion, solo ingresa '0' para cancelar.";
			next;
			while(1) {
				input .@input,0,101;
				if (.@input == 0) {
					mes "[Merchant]";
					mes "Que...?";
					mes "Para que demonios";
					mes "te molestas en hablarme?";
					mes "Bastante indeciso, no crees?";
					close;
				}
				else if (.@input > 100) {
					mes "[Merchant]";
					mes "No puedo darte mas";
					mes "de 100 Red Potions";
					mes "a la vez. Intentemos";
					mes "de nuevo.";
					next;
				}
				else {
					break;
				}
			}
			.@put_out_item = .@input * .@req_amount;
			if (countitem(.@item_req) < .@put_out_item) {
				mes "[Merchant]";
				mes "Uh oh, el numero que ingresaste no parece correcto. Mejor revisa de nuevo el numero de Red Potions que puedes intercambiar.";
				close;
			}
			mes "[Merchant]";
			mes "Aqui tienes!";
			mes "Revisa cuantas Red Potions";
			mes "te di, deberia estar bien. Gracias, fue un buen trato~";
			delitem .@item_req,.@put_out_item; //Shell
			getitem 501,.@input; //Red_Potion
			close;
		}
		mes "[Merchant]";
		mes "Huh~";
		mes "Esta bien.";
		mes "Aunque no son";
		mes "las Red Potions mas utiles";
		mes "para un aventurero como tu?";
		close;
	}
}
