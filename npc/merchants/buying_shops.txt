// Black Marketeer (Buy Licenses - Non-Merchant Classes) :: qskill_buy_store
//============================================================
que_job01,68,84,1	script	Black Marketeer#Buying	881,{
	if (checkweight(1201,1) == 0 || MaxWeight - Weight < 2400) {
		mes "[Mr. Jass]";
		mes "Eres demasiado codicioso, incluso comparado con alguien como yo!";
		mes "Por que no vas a aligerar tu bolsa primero?";
		close;
	}
	if (getskilllv("ALL_BUYING_STORE") == 1) {
		mes "[Mr. Jass]";
		mes "Hey, ya hiciste un contrato con Hugh.";
		mes "No tengo ningun negocio contigo.";
		close;
	}
	mes "[Mr. Jass]";
	mes "Debes necesitar algo urgentemente para venir a buscarme.";
	mes "Que quieres?";
	next;
	switch(select("Licencia de Tienda de Compra:Quien eres?:Nada, nada!")) {
	case 1:
		mes "[Mr. Jass]";
		mes "Lo sabia!";
		mes "Claro, puedo hacertela.";
		mes "La mia se ve igual que la autentica que los Mercaderes obtienen de ese bastardo del Sr. Hugh!";
		next;
		mes "[Mr. Jass]";
		mes "Y mi licencia es mejor porque no necesitas aprender ninguna habilidad.";
		mes "Cuantas quieres?";
		next;
		mes "[Mr. Jass]";
		mes "Solo para que sepas, solo puedo hacerlas en pequenas cantidades, hasta 10 a la vez.";
		mes "Costara 500 zeny cada una.";
		next;
		while(1) {
			input .@input;
			mes "[Mr. Jass]";
			if (.@input == 0) {
				mes "No necesitas esas licencias?";
				close;
			} else if (.@input > 10) {
				mes "Solo puedo hacer hasta 10 a la vez, sabes.";
				next;
			} else {
				mes "Costara "+(.@input*500)+" zeny.";
				if (Zeny < (.@input*500)) {
					mes "pero no tienes suficiente dinero.";
					next;
					mes "[Mr. Jass]";
					mes "No conoces lo basico de los negocios? Todo tiene un precio.";
					mes "Si quieres algo, tienes que pagar por ello.";
				} else {
					mes "Ja... Ja ja ja!";
					mes "Sr. Hugh, me quedare con tu negocio de licencias. Ya veras!";
					mes "*Risita Risita*";
					getitem 12548,.@input; //Buy_Market_Permit2
					set Zeny,Zeny-(.@input*500);
				}
				close;
			}
		}
	case 2:
		mes "[Mr. Jass]";
		mes "Deje mi pueblo natal hace mucho tiempo.";
		mes "No tiene sentido preguntar quien soy porque todo lo que me queda ahora es mi odio.";
		next;
		mes "[Mr. Jass]";
		mes "...";
		mes "Hugh es un comerciante corrupto sin etica de negocios.";
		mes "Mi unico proposito en la vida es destruir a Hugh.";
		next;
		mes "[Mr. Jass]";
		mes "Ay, bebi demasiado... (*Hip*)";
		mes "Es solo el alcohol hablando.";
		mes "Por favor olvida todo lo que dije.";
		close;
	case 3:
		mes "[Mr. Jass]";
		mes "Esta bien, esta bien! No tienes que gritar.";
		mes "Dejame en paz si no tienes ningun asunto conmigo.";
		close;
	}
}

// Purchasing Team (Learn Skill - Merchant Classes)
//============================================================
alberta_in,58,52,4	script	Purchasing Team#Buying	59,{
	if (checkweight(1201,1) == 0 || MaxWeight - Weight < 2400) {
		mes "- No puedes conversar o realizar la mision porque llevas demasiados objetos. -";
		close;
	}
	mes "[Mr. Hugh]";
	if (BaseClass == Job_Merchant && getskilllv("MC_VENDING") >= 1) {
		if (getskilllv("ALL_BUYING_STORE") == 1) {
			mes "Soy Hugh del Equipo de Compras.";
			mes "En que puedo ayudarte hoy?";
			next;
			if(select("Comprar Licencia de Tienda de Compra:Salir") == 2) {
				mes "[Mr. Hugh]";
				mes "No dudes en preguntarme si necesitas alguna Licencia de Tienda de Compra.";
				mes "Vuelve pronto~";
				close;
			}
			mes "[Mr. Hugh]";
			mes "Son 200 zeny por cada Licencia de Tienda de Compra, y puedes comprar hasta 50 a la vez.";
			mes "Cuantas licencias necesitas?";
			next;
			while(1) {
				input .@input;
				mes "[Mr. Hugh]";
				if (.@input == 0) {
					mes "Has cancelado el intercambio.";
					mes "Que tengas un buen dia.";
					close;
				} else if (.@input > 50) {
					mes "Por favor ingresa un valor de 50 o menos.";
					next;
				} else {
					mes "Costara "+(.@input*200)+" zeny por "+.@input+" licencias.";
					if (Zeny < (.@input*200)) {
						mes "pero parece que no tienes suficiente dinero.";
					} else {
						mes "Gracias por tu compra.";
						getitem 6377,.@input; //Buy_Market_Permit
						set Zeny,Zeny-(.@input*200);
					}
					close;
				}
			}
		} else {
			mes "Soy Hugh del Equipo de Compras del Gremio de Mercaderes de Alberta.";
			mes "Tu eres...";
			next;
			mes "[Mr. Hugh]";
			mes "Sabias? Nuestro gremio ha emitido una licencia para permitir a las personas comprar bienes de otros.";
			next;
			select("Nunca he tenido problemas comprando objetos...");
			mes "[Mr. Hugh]";
			mes "Tienes razon, pero piensalo:";
			mes "no te ha costado trabajo comprar al por mayor?";
			mes "Tendrias que buscar y hablar con todos los que tienen un objeto que quieres.";
			mes "Bastante inconveniente, no?";
			next;
			mes "[Mr. Hugh]";
			mes "Ya que comprar al por mayor es un tema importante para nosotros los Mercaderes,";
			mes "he propuesto un plan innovador a nuestro gremio, basado en mis 10 anos de experiencia en compras.";
			next;
			mes "[Mr. Hugh]";
			mes "'Permitir a las Personas Abrir";
			mes "una Tienda de Compra!'";
			mes "Ese es el titulo de mi propuesta.";
			mes "Ya veras, cuando la leas...";
			next;
			mes "[Mr. Hugh]";
			mes "...";
			mes "...(El Sr. Hugh parlotea sin parar con todos los detalles.)";
			next;
			select("Esta bien, cual es tu punto?");
			mes "[Mr. Hugh]";
			mes "Oh, si. En resumen,";
			mes "puedes comprar ciertos objetos al por mayor a traves de Vending.";
			next;
			mes "[Mr. Hugh]";
			mes "Para abrir una Tienda de Compra, necesitas una licencia emitida por el Gremio de Mercaderes.";
			next;
			mes "[Mr. Hugh]";
			mes "La necesitas cada vez que abres la tienda. Esperamos un aumento significativo en las ganancias a traves de este nuevo tipo de licencia.";
			next;
			mes "[Mr. Hugh]";
			mes "Ese es el punto de mi propuesta!";
			mes "Nuestro presidente estaba muy feliz de saber que vamos a ganar mucho dinero!";
			next;
			mes "["+strcharinfo(0)+"]";
			mes "Por favor ve al punto de una vez!";
			next;
			mes "[Mr. Hugh]";
			mes "No seas tan impaciente, si?";
			mes "Mi punto es, podemos dejarte abrir la Tienda de Compra si has aprendido Vending.";
			next;
			mes "[Mr. Hugh]";
			mes "Por supuesto, cobramos 10,000 zeny como tarifa unica de registro.";
			mes "Necesitaras la ^4A4AFFLicencia de Tienda de Compra^000000 cada vez que abras la tienda.";
			next;
			mes "[Mr. Hugh]";
			mes "Dime, te gustaria registrarte ahora?";
			mes "Si lo haces, te ensenare como abrir la Tienda de Compra.";
			next;
			if(select("Aprender a abrir Tienda de Compra:Cancelar") == 2) {
				mes "[Mr. Hugh]";
				mes "Vaya, que decepcion!";
				mes "Usar esta opcion de compra al por mayor puede beneficiar tu negocio de muchas formas, pero es tu decision.";
				mes "Siempre estoy disponible para consultas!";
				close;
			}
			mes "[Mr. Hugh]";
			if (Zeny < 10000) {
				mes "La tarifa de registro es 10,000 zeny.";
				mes "Por favor ten la tarifa lista primero.";
				close;
			} else {
				mes "Has tomado una buena decision.";
				mes "Por favor dame la tarifa de registro, y firma aqui mismo....";
				next;
				input .@name$;
				mes "[Mr. Hugh]";
				mes .@name$+"....";
				mes "Me gusta tu letra.";
				mes "Listo, ahora estas aprobado para abrir la Tienda de Compra.";
				set Zeny,Zeny-10000;
				getitem 6377,5; //Buy_Stall_Permit
				skill "ALL_BUYING_STORE",1,SKILL_PERM_GRANT;
				next;
				mes "[Mr. Hugh]";
				mes "Actualmente, solo objetos normales ^8C2121EXCEPTO^000000 equipamiento, ciertas pociones y objetos hechos a mano pueden ser comprados al por mayor, pero esto puede ser muy beneficioso para ti, dependiendo de como lo uses.";
				next;
				mes "[Mr. Hugh]";
				mes "Ah, y necesitas al menos uno del objeto que quieres comprar en tu inventario porque tienes que mostrarselo a otros a traves de tu tienda.";
				next;
				mes "[Mr. Hugh]";
				mes "Tu habilidad deberia estar registrada en tu ventana de habilidades. Si no puedes verla probablemente tengas que minimizar tu Lista de Habilidades y revisar la pestana de 3er Job.";
				next;
				mes "[Mr. Hugh]";
				mes "Te he dado 5 Licencias de Tienda de Compra para tu prueba.";
				mes "Disfruta comprando!";
				close;
			}
		}
	}
	mes "Soy Hugh del Equipo de Compras del Gremio de Mercaderes de Alberta.";
	mes "Me encantaria charlar, pero estoy muy ocupado en este momento.";
	next;
	mes "[Mr. Hugh]";
	mes "Mi tiempo esta dedicado exclusivamente a nuestros clientes de la industria de Mercaderes.";
	close;
}
