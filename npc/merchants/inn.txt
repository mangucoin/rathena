	mes "Bienvenido al";
	mes "Royal Dragon,";

			mes "[Hotel Employee]";

			mes "necesitas 5,000 zeny";
			mes "para poder registrarte.";
			close;
		}
		set Zeny,Zeny - 5000;
		mes "[Hotel Employee]";
		mes "Gracias~";
		mes "Espero que disfrutes";
		mes "tu estadia en el";
		mes "Royal Dragon.";
		close2;
		warp "lhz_in02",219,150;
		percentheal 100,100;
		npcskill "AL_BLESSING",10,99,99;
		end;
	case 3:
		mes "[Hotel Employee]";
		mes "Gracias y";
		mes "que tengas un buen dia.";
		close;
	}
}

bra_in01,27,24,3	script	Hotel Keeper#bra1	478,{
	mes "[Hotel Keeper]";
	mes "Bienvenido al hermoso Hotel de Brasilis.";
	next;
	switch(select("Guardar:Descansar -5000 zeny:Cancelar")) {
	case 1:
		mes "[Hotel Keeper]";
		mes "Quieres guardar aqui en el Hotel de Brasilis?";
		next;
		switch(select("No, gracias.:Absolutamente.")) {
		case 1:
			mes "[Hotel Keeper]";
			mes "Esta bien entonces, disfruta tu estadia.";
			close;
		case 2:
			mes "[Hotel Keeper]";
			mes "Tu reaparicion ha sido guardada aqui en el hotel. Espero que disfrutes tu estadia aqui en Brasilis.";
			savepoint "bra_in01",144,69,1,1;
			close;
		}
		end;
	case 2:
		if (Zeny > 4999) {
			mes "[Hotel Keeper]";
			mes "Te mostrare una gran habitacion.";
			close2;
			if (Zeny < 5000)
				end;
			Zeny -= 5000;
			percentheal 100,100;
			warp "bra_in01",144,69;
			end;
		}
		mes "[Hotel Keeper]";
		mes "Lo siento, pero el cargo por servicio es 5,000 zeny por noche.";
	case 3:
		close;
	}
}

//======================= Inn Function ==============================
function	script	F_InnMaid	{
	.@npc_name$ = getarg(0);

	mes .@npc_name$;
	mes "Bienvenido a";
	mes getarg(1) + ".";
	mes "En que puedo ayudarte?";
	next;
	switch(select("Guardar:Descansar -> 5000 zeny:Cancelar")) {
	case 1:
		mes .@npc_name$;
		mes "Tu punto de reaparicion";
		mes "ha sido guardado.";
		mes "Gracias,";
		mes "por favor vuelve pronto.";
		savepoint getarg(2),getarg(3),getarg(4),1,1;
		close;
	case 2:
		mes .@npc_name$;
		if(Zeny < 5000){
			mes "Lo siento, pero el cargo por servicio es 5,000 zeny. Por favor asegurate de tener suficiente dinero para registrarte la proxima vez, si?";
			close;
		}
		mes "Gracias.";
		mes "Espero que";
		mes "disfrutes tu descanso~";
		close2;
		set Zeny,Zeny - 5000;
		percentheal 100,100;
		return;
	case 3:
		close;
	}
}
