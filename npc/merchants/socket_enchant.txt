function	script	Func_Socket	{
	.@item_id = getarg(0);
	.@zeny_req = getarg(4);
	setarray .@item_req[0], getarg(5), getarg(7,0);
	setarray .@req_amount[0], getarg(6), getarg(8,0);

	disable_items;
	mes "[Seiyablem]";
	mes "Quieres agregar un Slot a un "+ getitemname(.@item_id) +"?";
	mes "De acuerdo, por favor traeme ^FF0000"+ .@req_amount[0] +" "+ getitemname(.@item_req[0]) + (.@req_amount[0] > 1 && .@item_req[0] != 999 ? "s":"") +"^000000, "+ (.@item_req[1] != 0 && .@req_amount[1] != 0 ? ("^FF0000"+ .@req_amount[1] +" "+ getitemname(.@item_req[1]) + (.@req_amount[1] > 1 && .@item_req[1] != 999 ? "s":"") + "^000000 ") : "") + "y mi ^FF0000" + (.@zeny_req >= 1000 ? (.@zeny_req /1000 +",000") : .@zeny_req) +",000 zeny^000000 de tarifa de servicio.";
	mes "Ah, y no olvides traer ese "+ getitemname(.@item_id) +"!";
	next;
	mes "[Seiyablem]";
	if (getiteminfo(.@item_id, ITEMINFO_LOCATIONS) & EQP_HAND_R) // weapon
	{
		mes "Puedo intentar agregar un slot ahora si tienes los objetos y zeny requeridos.";
		mes "Sin embargo, debes saber que hay una posibilidad de que pueda fallar.";
		mes "Por lo tanto, debo darte una advertencia justa...";
		next;
		mes "[Seiyablem]";
		mes "Si este intento de agregar un Slot a tu Arma falla, entonces el ^FF0000Arma^000000, ^FF0000y cualquier Carta insertada en ella sera destruida^000000.";
	}
	else // armor
	{
		mes "Si tienes todos los materiales requeridos, mi tarifa de servicio en zeny y la Armadura, entonces podemos proceder con el intento de agregar Slot.";
		mes "Pero antes de eso, debo advertirte del riesgo.";
		next;
		mes "[Seiyablem]";
		mes "Si este intento de agregar un Slot a tu Armadura falla, entonces la ^FF0000Armadura^000000, ^FF0000sus mejoras^000000 ^FF0000y cualquier Carta insertada en ella sera destruida^000000.";
	}
	mes "Aun quieres intentar agregar un Slot?";
	next;
	switch(select("Intentar Agregar Slot:Cancelar"))
	{
		case 1:
			if ((Zeny >= .@zeny_req*1000) && countitem(.@item_req[0]) >= .@req_amount[0] && countitem(getarg(7,512)) >= .@req_amount[1] && countitem(.@item_id) > 0)// if there's no getarg(7) and getarg(8) provided, check itemcount against 0 (we still need a valid item ID though)
			{
				mes "[Seiyablem]";
				mes "Muy bien entonces, que comience el trabajo!";
				mes "Mejor reza por un resultado exitoso.";
				next;
				set .@a, rand(1,100);
				if ((.@a > getarg(2)) && (.@a < getarg(3)))
				{
					specialeffect getarg(3) == 51 ? EF_LORD : EF_SANCTUARY; // only S Class items have Lord of Vermillion effect
					mes "[Seiyablem]";
					mes "Genial, parece haber sido exitoso.";
					mes "Se ve bastante bien hecho. Felicidades!";
					delitem .@item_id,1;
					delitem .@item_req[0],.@req_amount[0];
					if (.@item_req[1] != 0 && .@req_amount[1] != 0)
						delitem .@item_req[1],.@req_amount[1];
					Zeny = Zeny - .@zeny_req *1000;
					getitem getarg(1),1;
					next;
					mes "[Seiyablem]";
					mes "Nos vemos, amigo!";
					close;
				}
				else
				{
					specialeffect EF_SUI_EXPLOSION;
					mes "[Seiyablem]";
					mes "Wah! ...Lo siento mucho, fallo.";
					mes "Sin embargo, soy completamente inocente.";
					mes "Esta es tu suerte, y esta destinada por dios, ok?";
					mes "No te decepciones tanto, e intenta la proxima vez.";
					delitem .@item_id,1;
					delitem .@item_req[0],.@req_amount[0];
					if (.@item_req[1] != 0 && .@req_amount[1] != 0)
						delitem .@item_req[1],.@req_amount[1];
					Zeny = Zeny - .@zeny_req *1000;
					next;
					mes "[Seiyablem]";
					mes "Te deseo buena suerte la proxima vez!";
					close;
				}
			}
			else
			{
				mes "[Seiyablem]";
				mes "Me gustaria proceder con este intento de agregar Slot, pero te faltan algunas cosas.";
				mes "Estas seguro de que tienes el equipo, los materiales requeridos y el zeny?";
				close;
			}
		case 2:
			mes "[Seiyablem]";
			mes "Necesitas tiempo para pensarlo, eh?";
			mes "De acuerdo, puedo entender.";
			mes "Solo recuerda que la vida no es divertida si siempre juegas a lo seguro~";
			close;
	}
}
