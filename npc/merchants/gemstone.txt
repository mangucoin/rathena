payon,173,238,5	script	Jade#pay	754,{
	if (checkweight(1201,1) == 0) {
		mes "^3355FFEspera un segundo! Ahora mismo, llevas demasiados objetos contigo. Por favor vuelve despues de guardar algunas cosas en el Almacen Kafra.^000000";
		close;
	}
	mes "[Jade]";
	mes "Traeme dos";
	mes "Gemas del";
	mes "mismo color, y las";
	mes "cambiare por Gemas";
	mes "de un color diferente.";
	next;
	switch(select("Blue Gemstones a Red!:Red Gemstones a Yellow!:Yellow Gemstones a Blue!")) {
	case 1: callsub S_TradeGems,717,716;
	case 2: callsub S_TradeGems,716,715;
	case 3: callsub S_TradeGems,715,717;
	}

S_TradeGems:
	.@item_req = getarg(0);
	.@item_id = getarg(1);

	if (countitem(.@item_req) < 2) {
		mes "[Jade]";
		mes "Ja...!";
		mes "Me estas tomando el pelo, verdad?";
		mes "No puedo darte este";
		mes "servicio si no me";
		mes "das al menos";
		mes "2 "+ getitemname(.@item_req) +"s!";
		close;
	}
	else {
		.@gems = countitem(.@item_req) /2;
		mes "[Jade]";
		mes "Creo que puedo crear";
		mes "un total de "+ .@gems +" "+ getitemname(.@item_id) +"s";
		mes "usando las "+ getitemname(.@item_req) +"s";
		mes "que tienes actualmente.";
		mes "Que quieres hacer?";
		next;
		switch(select("Dame tantas como puedas.:Quiero elegir la cantidad.:Me retiro.")) {
		case 1:
			delitem .@item_req,.@gems * 2;
			getitem .@item_id,.@gems;
			break;
		case 2:
			mes "[Jade]";
			mes "Entonces cuantas";
			mes "quieres?";
			mes "El numero maximo";
			mes "que puedes ingresar es 100.";
			next;
			while(1) {
				input .@input,0,101;
				if (.@input == 0) {
					mes "[Jade]";
					mes "Ninguna?";
					mes "Supongo que";
					mes "cambiaste de opinion...";
					close;
				}
				else if (.@input > 100) {
					mes "[Jade]";
					mes "Errm...";
					mes "Te pedi que ingresaras";
					mes "una cantidad no mayor";
					mes "de 100, recuerdas...?";
					next;
				}
				else if (.@gems < .@input) {
					// Custom dialogue
					mes "[Jade]";
					mes "Errm...";
					mes "No tienes tantas";
					mes "gemas para intercambiar...";
					next;
				}
				else break;
			}
			delitem .@item_req,.@input * 2;
			getitem .@item_id,.@input;
			break;
		case 3:
			mes "[Jade]";
			mes "Claro, no hay problema.";
			mes "Vuelve cuando quieras.";
			close;
		}
		mes "[Jade]";
		mes "Aqui tienes.";
		mes "Sientete libre de volver";
		mes "cuando quieras.";
		mes "Hm, que es esa mirada?";
		mes "Hay algo en mi cara?";
		close;
	}
}
