que_ng,72,29,3	script	Kunai Merchant Kashin	83,{
	if(BaseClass == Job_Ninja) {
		mes "[Kashin]";
		mes "Soy Kashin, distribuidor";
		mes "de Kunai para Ninjas. Echa";
		mes "un vistazo y avisame";
		mes "si te interesa";
		mes "alguno de mis productos.";
		next;
		switch(select("10 Fell Poison Kunai:10 Icicle Kunai:10 High Wind Kunai:10 Black Earth Kunai:10 Heat Wave Kunai:Cancelar")) {
		case 1: callfunc "Kunai_Trade",13250,20,7524,1,13259; break;
		case 2: callfunc "Kunai_Trade",13251,8,7522,2,13255; break;
		case 3: callfunc "Kunai_Trade",13252,4,7523,2,13257; break;
		case 4: callfunc "Kunai_Trade",13253,2,7524,1,13256; break;
		case 5: callfunc "Kunai_Trade",13254,1,7521,2,13258; break;
		case 6:
			mes "[Kashin]";
			mes "Bueno entonces, gracias";
			mes "por visitar mi tienda.";
			mes "Por favor ven a verme cuando";
			mes "necesites comprar algunos";
			mes "Kunais. Adios por ahora~";
			close;
		}
	}
	mes "[Kashin]";
	mes "Soy Kashin, distribuidor";
	mes "de Kunai para Ninjas. Si";
	mes "tienes amigos que sean";
	mes "Ninjas, entonces podrias";
	mes "querer contarles sobre mi.";
	close;
}

function	script	Kunai_Trade	{
	setarray .@item_req[0], getarg(0), getarg(2);
	setarray .@req_amount[0], getarg(1), getarg(3);
	.@item_id = getarg(4);

	mes "[Kashin]";
	mes "Puedes intercambiar";
	mes .@req_amount[0] +" "+ getitemname(.@item_req[0]) +" y";
	mes .@req_amount[1] +" "+ getitemname(.@item_req[1]) +" por cada";
	mes "set de 10 "+ getitemname(.@item_id) +".";
	next;
	mes "[Kashin]";
	mes "Solo puedo darte un maximo de 500 sets de Kunais a la vez.";
	mes "Si quieres cancelar el intercambio, entonces por favor ingresa ''0.'' Cuantos";
	mes "sets de Kunai te gustarian?";
	next;
	input .@amount;
	if(.@amount < 1 || .@amount > 500) {
		mes "[Kashin]";
		mes "Eh? Lo siento, pero";
		mes "no puedo darte tantos";
		mes "sets de Kunai. Por favor";
		mes "ingresa un valor menor a 500.";
		close;
	}
	if (countitem(.@item_req[0]) < .@amount*.@req_amount[0] || countitem(.@item_req[1]) < .@amount*.@req_amount[1]) {
		mes "[Kashin]";
		mes "Hmm, no tienes";
		mes "suficientes objetos para este";
		mes "intercambio de Kunai. Por favor";
		mes "revisa tus objetos de nuevo.";
		close;
 	}
	if (checkweight(.@item_id, .@amount*10) == 0) {
		mes "[Kashin]";
		mes "Hmm, parece que tu";
		mes "Inventario no tiene";
		mes "suficiente espacio para guardar";
		mes "mas objetos. Mejor";
		mes "libera algo de espacio primero.";
		close;
	}
	mes "[Kashin]";
	mes "Genial, todo esta en";
	mes "orden, asi que procedamos";
	mes "a completar este intercambio.";
	mes "Estoy seguro de que estaras";
	mes "muy satisfecho con";
	mes "estos Kunais.";
	delitem .@item_req[0],.@req_amount[0]*.@amount;
	delitem .@item_req[1],.@req_amount[1]*.@amount;
	getitem .@item_id,10*.@amount;
	close;
}
