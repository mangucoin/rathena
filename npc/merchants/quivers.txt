//= Turns arrows into Arrow Quivers.
//= Breakdown of Subroutine "S_BuyQuiver"
//= arg(0): Type of Arrow to be packaged (item ID).
//= arg(1): How many of each 'getarg(0)' arrow per quiver.
//= arg(2): The cost of making a 'getarg(0)' quiver.
//= arg(3): The quiver given by the NPC (item ID).
//===== Additional Comments: =================================
//= 07/06/05 : Added 1st Version. [Muad_Dib]

payon_in01,5,134,5	script	Inventor Jaax	89,{
	if (checkweight(1201,1) == 0 || MaxWeight - Weight < 2000) {
		mes "[Inventor Jaax]";
		mes "Hey, llevas demasiadas";
		mes "cosas encima. Por que no las guardas en el Almacen Kafra? Podemos hablar despues de eso, verdad?";
		close;
	}
	mes "[Inventor Jaax]";
	mes "Mi nombre es Jaax.";
	mes "Sin ego, puedo";
	mes "decir que soy quizas el";
	mes "^663300mayor inventor de nuestro tiempo^000000.";
	next;
	mes "[Inventor Jaax]";
	mes "Esta vez, he";
	mes "creado algo";
	mes "verdaderamente extraordinario. Los llamo... ^663300Carcajes Magicos^000000!! Esto sera recordado en la historia como una";
	mes "revolucion de flechas!";
	emotion ET_BEST;
	next;
	mes "[Inventor Jaax]";
	mes "He estudiado magia y carcajes por anos, trabajando dia y noche hasta que finalmente descubri como condensar flechas con magia! Con los carcajes magicos, llevaras mas flechas, pero con menos peso!";
	next;
	mes "[Inventor Jaax]";
	mes "Te gustaria probar uno de mis carcajes de flechas? No tengo duda de que alguien como tu puede apreciar mi genialidad!";
	next;
	switch(select("Quiver:Iron Arrow Quiver:Steel Arrow Quiver:Oridecon Arrow Quiver:Fire Arrow Quiver:Silver Arrow Quiver:Wind Arrow Quiver:Stone Arrow Quiver:Crystal Arrow Quiver:Shadow Arrow Quiver:Immaterial Arrow Quiver:Rusty Arrow Quiver")) {
	case 1: callsub S_BuyQuiver,1750,500,500,12004;
	case 2: callsub S_BuyQuiver,1770,500,500,12005;
	case 3: callsub S_BuyQuiver,1753,500,500,12006;
	case 4: callsub S_BuyQuiver,1765,500,500,12007;
	case 5: callsub S_BuyQuiver,1752,500,500,12008;
	case 6: callsub S_BuyQuiver,1751,500,500,12009;
	case 7: callsub S_BuyQuiver,1755,500,500,12010;
	case 8: callsub S_BuyQuiver,1756,500,500,12011;
	case 9: callsub S_BuyQuiver,1754,500,500,12012;
	case 10: callsub S_BuyQuiver,1767,500,500,12013;
	case 11: callsub S_BuyQuiver,1757,500,500,12014;
	case 12: callsub S_BuyQuiver,1762,500,500,12015;
	}

S_BuyQuiver:
	.@item_id = getarg(0);
	.@amount = getarg(1);
	.@zeny_req = getarg(2);

	if (countitem(.@item_id) >= .@amount) {
		mes "[Inventor Jaax]";
		mes "Excelente!";
		mes "Llevas Flechas contigo? Te proporcionare un carcaj que puede llevar "+ .@amount +" de tus "+ getitemname(.@item_id) +"s por solo ^FF3131"+ .@zeny_req +" Zeny^000000.";
		next;
		switch(select("Guardar tantas Flechas en carcajes como sea posible:Comprar 1 carcaj:Cancelar")) {
		case 1:
			.@arrows = countitem(.@item_id);
			set .@quiver,.@arrows / .@amount;
			set .@arrows_used,.@quiver * .@amount;
			.@arrow_zeny01 = .@quiver * .@zeny_req;
			mes "Numero de";
			mes "Flechas: ^3131FF"+.@arrows+" ^000000";
			mes "Numero Maximo";
			mes "de Carcajes";
			mes "Comprables: ^3131FF"+.@quiver+" ^000000";
			mes "Zeny requerido: ^3131FF"+.@arrow_zeny01+" Zeny^000000";
			next;
			mes "[Inventor Jaax]";
			mes "Te gustaria comprar";
			mes "tantos Carcajes";
			mes "como puedas con las Flechas";
			mes "que llevas actualmente?";
			next;
			if (select("Si:Cancelar") == 1) {
				if (.@arrow_zeny01 < Zeny) {
					mes "[Inventor Jaax]";
					mes "Aqui tienes!";
					mes "Solo recuerda, ^FF0000no podras usar el Carcaj cuando tu peso cargado sea el 90% de tu limite de peso maximo^000000.";
					set Zeny, Zeny-.@arrow_zeny01;
					delitem .@item_id,.@arrows_used;// Arrow
					getitem getarg(3),.@quiver; //Quiver
					next;
					mes "[Inventor Jaax]";
					mes "Asi que...";
					mes "Solo lleva la cuenta";
					mes "de cuanto estas";
					mes "cargando de vez";
					mes "en cuando y deberias";
					mes "estar bien.";
					close;
				}
				else {
					mes "[Inventor Jaax]";
					mes "Lo siento, pero no tienes suficiente Zeny. No puedo simplemente regalar estos despues de trabajar anos";
					mes "para desarrollar esta tecnologia revolucionaria!";
					close;
				}
			}
			mes "[Inventor Jaax]";
			mes "Cambiaste de opinion?";
			mes "Cuando la gloria de tener";
			mes "un carcaj esta tan cerca?";
			close;
		case 2:
			if (Zeny > .@zeny_req) {
				mes "[Inventor Jaax]";
				mes "Aqui tienes!";
				mes "Solo recuerda, ^FF0000no podras usar el Carcaj cuando tu peso cargado sea el 90% de tu limite de peso maximo^000000.";
				Zeny = Zeny -.@zeny_req;
				delitem .@item_id,.@amount;// Arrow
				getitem getarg(3),1; //Quiver
				next;
				mes "[Inventor Jaax]";
				mes "Asi que...";
				mes "Solo lleva la cuenta";
				mes "de cuanto estas";
				mes "cargando de vez";
				mes "en cuando y deberias";
				mes "estar bien.";
				close;
			}
			else {
				mes "[Inventor Jaax]";
				mes "Ni siquiera";
				mes "tienes "+ .@zeny_req +" Zeny?";
				mes "Lo siento mucho. No tenia";
				mes "idea de que estuvieras tan...";
				mes "^333333En la ruina^000000.";
				close;
			}
		case 3:
			mes "[Inventor Jaax]";
			mes "Que...?";
			mes "No ves que este invento puede cambiar para siempre la forma en que se llevan las Flechas?! El futuro es ahora!";
			close;
		}
	}
	else {
		mes "[Inventor Jaax]";
		mes "Puedes llevar un maximo de "+ .@amount +" Flechas dentro de este carcaj. Fue hecho usando mi metodo secreto,";
		mes "asi que el peso total de las Flechas y el Carcaj es menor que llevar las Flechas solas.";
		next;
		mes "[Inventor Jaax]";
		mes "Es un milagro de la ciencia! Uno que puedes experimentar por ti mismo si me traes al menos "+ .@amount +" Flechas y "+ .@zeny_req +" Zeny por cada Carcaj.";
		close;
	}
}
