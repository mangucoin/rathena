// Manuk & Splendide
//============================================================
-	script	::merchant_13_2	-1,{
	set .@n$, "["+strnpcinfo(1)+"]";
	mes .@n$;
	if (isequipped(2782) && ep13_2_rhea == 100) {
		if (!checkweight(1201,1)) {
			mes "Parece que llevas demasiadas cosas.";
			mes "Por que no guardas algunos de tus objetos en el almacen y vuelves?";
			close;
		}
		mes "Hola.";
		mes "Que estas buscando?";
		next;
		if (strnpcinfo(1) == "Merchant of Manuk") {
			setarray .@Items[2],12342,12343,12348;
			set .@Book,11019; //Manuk_Selling_Item
			set .@Coin,6080; //Manuk_Coin
			set .@i, select("Ver descripcion del objeto:Manuk's Opportunity:Manuk's Courage:Manuk's Faith:Cancelar");
		}
		else {
			setarray .@Items[2],12344,12345,12349;
			set .@Book,11018; //Splendide_Selling_Item
			set .@Coin,6081; //Splendide_Coin
			set .@i, select("Ver descripcion del objeto:Comprar Pinguicula's Fruit Jam:Comprar Luciola's Honey Jam:Comprar Cornus' Tears:No hacer nada");
		}
		mes .@n$;
		if (.@i == 1) {
			mes "Aqui estan las descripciones de los objetos.";
			close2;
			readbook .@Book,1;
			end;
		}
		else if (.@i < 5) {
			mes "Puedo venderte 3 "+getitemname(.@Items[.@i])+" por ^3131FF1 moneda^000000.";
			next;
			switch(select("Comprar:No comprar")) {
			case 1:
				if (countitem(.@Coin)) {
					mes .@n$;
					mes "Gracias por venir.";
					delitem .@Coin,1;
					getitem .@Items[.@i],3;
					close;
				}
				else {
					mes .@n$;
					mes "No tienes suficientes monedas.";
					close;
				}
			case 2:
				mes .@n$;
				break;
			}
		}
		mes "Vuelve si cambias de opinion.";
		close;
	}
	else {
		if (strnpcinfo(1) == "Merchant of Manuk") {
			mes "Rtt od d";
			mes "Qwo hd is d irr";
		}
		else {
			mes "BurWehAla";
			mes "tasnarAndu Ie Ru";
		}
		close;
	}
}
man_in01,286,16,1	duplicate(merchant_13_2)	Merchant of Manuk	454
spl_in01,110,326,5	duplicate(merchant_13_2)	Merchant of Splendide	439
