function	script	Func_Socket2	{
	.@item_id = getarg(0);
	.@zeny_req = getarg(4);
	setarray .@item_req[0], getarg(5), getarg(7,0);
	setarray .@req_amount[0], getarg(6), getarg(8,0);

	disable_items;
	mes "[Leablem]";
	if (.@item_req[1] != 0 && .@req_amount[1] != 0)
	{
		mes getitemname(.@item_id) +"... Ok, entonces necesitas traerme";
		mes "^FF0000"+ (.@zeny_req >= 1000 ? (.@zeny_req /1000 +",000") : .@zeny_req) +",000 zeny como cargo de servicio,";
		mes "^FF0000"+ .@req_amount[0] +" "+ getitemname(.@item_req[0]) +" y "+ .@req_amount[1] +" "+ getitemname(.@item_req[1]) +".";
		mes "^FF0000Por supuesto, necesitas un "+ getitemname(.@item_id) +".^000000";
	}
	else
	{
		mes getitemname(.@item_id) +"? Ok, entonces necesitas traerme";
		mes "^FF0000"+ (.@zeny_req >= 1000 ? (.@zeny_req/1000 +",000") : .@zeny_req) +",000 zeny como cargo de servicio y "+ .@req_amount[0] +" "+ getitemname(.@item_req[0]) + (.@req_amount[0] > 1 && .@item_req[0] != 999 ? "s":"") +" como requerimiento.";
		mes "^FF0000Por supuesto, no olvides traerme un "+ getitemname(.@item_id) +".";
		mes "^FF0000Debes tener todos los objetos.^000000";
	}
	next;
	mes "[Leablem]";
	mes "Ya trajiste todo?";
	mes "Para tu informacion, si fallas al crear un slot,";
	mes "perderas todos los materiales requeridos asi como el/la "+ ((getiteminfo(.@item_id, ITEMINFO_LOCATIONS)&EQP_HAND_R) ? "arma" : "armadura") +" objetivo.";
	mes "Tambien recuerda, si el/la "+ ((getiteminfo(.@item_id, ITEMINFO_LOCATIONS)&EQP_HAND_R) ? "arma" : "armadura") +" ha sido mejorada y tiene una carta insertada,";
	mes "las perderas incluso si tienes exito al crear un slot.";
	next;
	switch(select("Pedir creacion de slot.:Intentar la proxima vez."))
	{
		case 1:
			if ((Zeny >= .@zeny_req*1000) && countitem(.@item_req[0]) >= .@req_amount[0] && countitem(getarg(7,512)) >= .@req_amount[1] && countitem(.@item_id) > 0)// if there's no .@req_amount[0] and .@req_amount[1] provided, check itemcount against 0 (we still need a valid item ID though)
			{
				mes "[Leablem]";
				mes "Muy bien entonces, que comience el trabajo!";
				mes "Mejor reza por un resultado exitoso.";
				next;
				set .@a, rand(1,100);
				if ((.@a > getarg(2)) && (.@a < getarg(3)))
				{
					specialeffect EF_LORD;
					mes "[Leablem]";
					mes "Genial, parece haber sido exitoso.";
					mes "Se ve bastante bien hecho. Felicidades!";
					delitem .@item_id,1;
					delitem .@item_req[0],.@req_amount[0];
					if (.@item_req[1] != 0 && .@req_amount[1] != 0)
						delitem .@item_req[1],.@req_amount[1];
					Zeny = Zeny - .@zeny_req *1000;
					getitem getarg(1),1;
					next;
					mes "[Leablem]";
					mes "Nos vemos, amigo!";
					close;
				}
				else
				{
					specialeffect EF_SUI_EXPLOSION;
					mes "[Leablem]";
					mes "Wah! ...Lo siento mucho, fallo.";
					mes "Sin embargo, soy completamente inocente.";
					mes "Esta es tu suerte, y esta destinada por dios, ok?";
					mes "No te decepciones tanto,";
					mes "e intenta la proxima vez.";
					delitem .@item_id,1;
					delitem .@item_req[0],.@req_amount[0];
					if (.@item_req[1] != 0 && .@req_amount[1] != 0)
						delitem .@item_req[1],.@req_amount[1];
					Zeny = Zeny - .@zeny_req *1000;
					next;
					mes "[Leablem]";
					mes "Nos vemos, amigo!";
					close;
				}
			}
			else
			{
				mes "[Leablem]";
				mes "Eres tonto o que? No trajiste todos los objetos requeridos!";
				mes "Ve a buscarlos rapido!";
				close;
			}
		case 2:
			mes "[Leablem]";
			mes "Nos vemos la proxima vez.";
			close;
	}
