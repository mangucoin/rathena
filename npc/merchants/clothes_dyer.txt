function Dyes {
	setarray .@DyeName$[0],"el color por defecto","Naranja","Violeta","Rojo","Negro","Verde","Azul","Blanco","Amarillo";
	setarray .@DyeItems[0],0,980,981,975,983,979,978,982,976;
	set .@menu$,"- Por Defecto:";
	for(set .@i,0; .@i<getargcount(); set .@i,.@i+2)
		set .@menu$, .@menu$+"- "+.@DyeName$[getarg(.@i)]+":";
	set .@menu$, .@menu$+"- ^777777Cancelar^000000";
	set .@s, select(.@menu$)-1;
	if (.@s == (getargcount()/2)+1) goto L_End;
	mes .n$;
	if (.@s == 0) {
		mes "Puedo cambiar tu tinte a";
		mes "el color por defecto gratis.";
		mes "Estas seguro?";
		next;
		switch(select("Si:No")) {
		case 1:
			mes .n$;
			setlook 7,0;
			mes "Tu ropa ha sido tenida a "+.@DyeName$[0]+".";
			close;
		case 2:
			goto L_End;
		}
	}
	if (!countitem(.@DyeItems[getarg((.@s-1)*2)])) {
		mes "Para tenir tu ropa, necesitare los Dyestuffs apropiados. Por favor revisa mi lista de precios para la informacion.";
		close;
	}
	if (Zeny < 10000) {
		mes "Lo siento, pero no tienes suficiente dinero.";
		close;
	}
	delitem .@DyeItems[getarg((.@s-1)*2)],1;
	set Zeny, Zeny-10000;
	setlook 7, ((.@s)?(getarg((.@s-1)*2+1)):0);
	mes "Tu ropa ha sido tenida de "+((.@s)?.@DyeName$[getarg((.@s-1)*2)]:.@DyeName$[0])+".";
	close;
}
