// KvM Officer - Guillaume
//============================================================
bat_room,164,178,5	script	KVM Mercenary Officer#1	418,{
	if (getbattleflag("feature.bgqueue"))
		end;
	/*
	set .@kvm,checkquest(6026,PLAYTIME);
	if ((.@kvm == 0) || (.@kvm == 1)) {
		mes "[Croix Mercenary Officer]";
		mes "I know that you've already signed with the Croix. Go back and join their team. We will not accept traitors to fight for us.";
		close;
	}
	else if (.@kvm == 2) erasequest 6026;
	*/
	set .@permit,checkquest(6025,PLAYTIME);
	if ((.@permit == 0) || (.@permit == 1)) {
		mes "[Guillaume Mercenary Officer]";
		mes "Parece que recientemente participaste en una batalla KVM. Debes esperar 5 minutos antes de registrarte de nuevo.";
		close;
	}
	else if (.@permit == 2) erasequest 6025;
	mes "[Guillaume Mercenary Officer]";
	mes "Que conozcan el verdadero poder de Guillaume!";
	next;
	switch(select("Luchare contigo.:Terminar Conversacion.")) {
	case 1:
		mes "[Guillaume Mercenary Officer]";
		mes "Muestrales lo fuertes que somos.";
		mes "Hoy, todos escucharan el grito de triunfo de Guillaume!";
		close2;
		//setquest 6025;
		if (BaseLevel > 79)  warp "bat_room",169,223;
		else if (BaseLevel < 60) warp "bat_room",197,223;
		else warp "bat_room",225,223;
		end;
	case 2:
		mes "[Guillaume Mercenary Officer]";
		mes "Ganaremos!";
		close;
	}
}

// Guillaume Knight - KvM
//============================================================
bat_room,167,178,5	script	Guillaume Knight#kvm	417,{
	mes "[Guillaume Knight]";
	mes "Hola.";
	mes "^Que quieres saber?";
	next;
	switch(select("Aplicar para KVM.:^Que es KVM?:^Como participo en KVM?:Quiero saber mis Puntos Kreiger.")) {
	case 1:
		mes "[Guillaume Knight]";
		if (getbattleflag("feature.bgqueue"))
			mes "Si estas listo para la batalla, haz clic en el ^0000FFboton de batalla de la interfaz^000000 para aplicar a la batalla KVM!";
		else {
			mes "Las aplicaciones aun no estan disponibles.";
			mes "Para aplicar, necesitas ir con un Oficial Mercenario KVM.";
		}
		close;
	case 2:
		mes "[Guillaume Knight]";
		mes "KVM es la abreviatura de Kreiger Von Midgard.";
		mes "Aventurero, ^sabias que el camino al nuevo mundo ha sido abierto?";
		next;
		mes "[Guillaume Knight]";
		mes "Nosotros, la Administracion Guillaume, tenemos varios planes para avanzar al nuevo mundo.";
		mes "Y KVM es uno de ellos.";
		next;
		mes "[Guillaume Knight]";
		mes "Significa que nosotros, uno de los muchos paises en el continente Midgard,";
		mes "hemos decidido emplear muchos aventureros para el avance inmediato al nuevo mundo.";
		next;
		mes "[Guillaume Knight]";
		mes "Por lo tanto, para seleccionar a los mejores aventureros, estamos realizando el KVM.";
		mes "Esperamos la respuesta de muchos aventureros.";
		next;
		mes "[Guillaume Knight]";
		mes "Y les daremos recompensas por su participacion!";
		mes "En cuanto a las recompensas, por favor contacta a un Oficial de Logistica KVM.";
		close;
	case 3:
		mes "[Guillaume Knight]";
		if (getbattleflag("feature.bgqueue")) {
			mes "Basicamente, KVM es una batalla de 3 contra 3.";
			mes "Aplica haciendo clic en el ^0000FFboton de batalla de la interfaz^000000, en grupo o individualmente.";
		} else {
			mes "Basicamente, KVM es una batalla de 5 contra 5.";
			mes "Primero, aplicas con un recepcionista de KVM, en grupo o individualmente.";
		}
		next;
		mes "[Guillaume Knight]";
		if (getbattleflag("feature.bgqueue"))
			mes "Las aplicaciones grupales son para cuando quieras entrar al KVM con miembros de tu party o guild,";
		else
			mes "Las aplicaciones grupales son para cuando quieras entrar al KVM con miembros de tu party,";
		mes "y una aplicacion personal es para cuando quieras entrar al KVM individualmente.";
		next;
		mes "[Guillaume Knight]";
		if (getbattleflag("feature.bgqueue"))
			mes "Ten en cuenta que una vez canceles, ^0000FFno podras aplicar de nuevo por 1 minuto^000000.";
		else {
			mes "Por favor aplica con un oficial KVM, y el te contactara despues cuando estes en Prontera.";
			next;
			mes "[Guillaume Knight]";
			mes "Entonces entras y sigues las instrucciones en el campo de batalla.";
			next;
			mes "[Guillaume Knight]";
			mes "Sin embargo, ten en cuenta que a menos que estes en la oficina KVM, el no podra contactarte.";
		}
		close;
	case 4:
		mes "[Guillaume Knight]";
		mes "Tus Puntos Kreiger son:";
		mes ""+kvm_point+".";
		close;
	}
}
