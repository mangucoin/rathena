	mes "[Logistics]";

	if (!.@item_type)
		mes "Has seleccionado ^ff0000"+ getitemname(.@item_id) +"^000000.";
	else if (.@item_type == 1)

	else if (.@item_type == 2) {

		mes "La armadura para quien tenga Nivel mayor a 80 es: ^ff0000"+ getitemname(.@item_id) +"^000000.";
	}
	else if (.@item_type == 3) {
		mes "Has seleccionado 'Produccion en Masa "+ callfunc("F_GetArmorType",.@item_id) +"'.";
		mes "Solo hay una Produccion en Masa "+ callfunc("F_GetArmorType",.@item_id) +": ^ff0000"+ getitemname(.@item_id) +"^000000.";
	}
	else if (.@item_type == 4) {
		mes "Has seleccionado 'Popularizado "+ callfunc("F_GetArmorType",.@item_id) +"'.";
		mes "Solo hay un Popularizado "+ callfunc("F_GetArmorType",.@item_id) +": ^ff0000"+ getitemname(.@item_id) +"^000000.";
	}

	if (!.@req_setting)
		mes "Requiere ^0000ff2,000^000000 Puntos KVM, y "+ .@name$ +", tus puntos son: ^580080"+ kvm_point +"^000000.";
	else if (.@req_setting == 4)
		mes "Requiere ^0000ff1,200^000000 Puntos KVM, y "+ .@name$ +", tus puntos son: ^580080"+ kvm_point +"^000000.";
	else
		mes "Requiere ^0000ff"+ .@prices[.@req_setting] +"^000000 Puntos KVM, y "+ .@name$ +", tus puntos son: ^580080"+ kvm_point +"^000000.";

	mes "^Estas seguro de que quieres este objeto?";
	next;
	switch(select("No, no lo comprare.:Si, lo comprare.")) {
	case 1:
		mes "[Logistics]";
		mes "Has seleccionado 'No lo comprare'.";
		mes "Al comprar un objeto, ten cuidado ya que no hay reembolsos.";
		break;
	case 2:
		if (kvm_point >= .@prices[.@req_setting]) {
			kvm_point = kvm_point -.@prices[.@req_setting];
			getitem .@item_id,1;
			mes "[Logistics]";
