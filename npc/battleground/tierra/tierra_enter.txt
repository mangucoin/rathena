bat_room,124,178,5	script	Tierra Gorge Officer#01a	418,{
	if (getbattleflag("feature.bgqueue"))
		end;
	if (checkweight(1201,1) == 0) {
		mes "- Espera un momento !! -";
		mes "- Actualmente estas cargando -";
		mes "- demasiados objetos contigo. -";
		mes "- Por favor intenta de nuevo -";
		mes "- despues de reducir tu peso. -";
		close;
	}
	mes "[Guillaume Army Officer]";
	mes "Mostremos el poder del Ejercito de Guillaume a esos apestosos Croix!";
	next;
	switch(select("Quiero unirme a tu ejercito!:Terminar Conversacion")) {
	case 1:
		if ((BaseJob == Job_Novice) || (BaseJob == Job_Super_Novice)) {
			mes "[Guillaume Army Officer]";
			mes "Me alegra mucho que quieras unirte a nosotros y luchar contra el Ejercito Croix, pero lo siento: no puedo enviar a un novato como tu a morir en el cruel campo de batalla.";
		}
		else {
			if (BaseLevel < 80) {
				mes "[Guillaume Army Officer]";
				mes "Me alegra mucho que quieras unirte a nosotros y luchar contra el Ejercito Croix, pero lo siento: no puedo enviar a un novato como tu a morir en el cruel campo de batalla.";
			}
			else {
				set .@chk_urtime,checkquest(2069,PLAYTIME);
				if ((.@chk_urtime == 0)|| (.@chk_urtime == 1)) {
					mes "[Guillaume Army Officer]";
					mes "Parece que acabas de regresar del campo de batalla.";
					mes "Es muy pronto para que vuelvas. Ve a descansar, y deja a los enemigos a nosotros!";
				}
				else {
					if (.@chk_urtime == 2)
						erasequest 2069;
					set .@mapcount,getmapusers("bat_a01");
					if (.@mapcount > 0) {
						mes "[Guillaume Army Officer]";
						mes "He recibido un reporte informandome que un cuerpo de elite ya ha sido enviado al campo de batalla contra los Croix.";
						mes "^Por que no esperas un rato? Te sugiero que afiles tus armas y prepares tus suministros hasta entonces.";
					}
					else {
						if ($@TierraBG1) {
							mes "[Guillaume Army Officer]";
							mes "Un cuerpo de elite ya esta listo para ser enviado al campo de batalla.";
							mes "^Por que no esperas un rato? Te sugiero que afiles tus armas y prepares tus suministros hasta entonces.";
						}
						else {
							mes "[Guillaume Army Officer]";
							mes "Definitivamente pareces estar listo para la batalla!";
							mes "Ve y muestrale a los Croix lo que realmente significa el miedo!";
							mes "Hoy, nuestro grito de victoria resonara por todo el campo de batalla!";
							close2;
							warp "bat_room",57,223;
							end;
						}
					}
				}
			}
		}
		break;
	case 2:
		mes "[Guillaume Army Officer]";
		mes "Hoy seremos victoriosos!";
		break;
	}
	close;
}

bat_room,127,178,5	script	Guillaume Knight#1	417,{
	mes "[Guillaume Knight]";
	mes "Tierra Gorge consiste en dos lados empinados colocados verticalmente, y tiene depositos de raciones para los Ejercitos Guillaume y Croix en las direcciones de las 11 y las 7 en punto.";
	next;
	mes "[Guillaume Knight]";
	mes "La batalla comienza en el barco de tu ejercito, y el objetivo es avanzar y destruir el deposito de raciones enemigo mas rapido de lo que ellos pueden destruir el tuyo.";
	next;
	mes "[Guillaume Knight]";
	mes "El curandero del campo de batalla resucitara soldados una vez cada 25 segundos para que puedan reincorporarse a la batalla.";
	mes "El ejercito que capture la bandera neutral en el centro del campo de batalla sera recompensado con puntos de regeneracion extra, lo que significa que sus soldados resucitaran mas que los del otro lado, dandoles una ventaja.";
	next;
	mes "[Guillaume Knight]";
	mes "Esos dos depositos de raciones estan conectados entre si por caminos cortos y estrechos, pero las puertas estan bloqueadas con barricadas.";
	mes "Piensa y muevete lo mas rapido que puedas. La victoria de tu ejercito depende de tu contribucion.";
	next;
	mes "[Guillaume Knight]";
	if (getbattleflag("feature.bgqueue"))
		mes "Si estas listo, comienza haciendo clic en el ^0000FFboton de batalla de la interfaz^000000.";
	else {
		mes "Por favor recuerda: cuando te unas a una batalla, recibiras un token que indica la duracion establecida durante la cual no puedes participar en el mismo tipo de batalla.";
		mes "Puedes verificar la duracion presionando las teclas Alt+U.";
	}
	next;
	mes "[Guillaume Knight]";
	if (getbattleflag("feature.bgqueue"))
		mes "Ten en cuenta que una vez canceles, ^0000FFno podras aplicar de nuevo por 1 minuto^000000!";
	else
		mes "^Estas listo para la batalla? Entonces registrate con el reclutador a mi lado!";
	close;
}
