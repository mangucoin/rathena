	set .@i, (select(.@menu$)-1)*2;
	mes "[Erundek]";
	mes "Elegiste ^3131FF"+getitemname(.@Items[.@i])+"^000000.";
	switch(.@Items[.@i]) {
		case 2720: mes "Este objeto es solo para la clase Swordman y Taekwon Master."; break;
		case 2721: mes "Este objeto es solo para la clase Thief."; break;
		case 2722: mes "Este objeto es solo para la clase Acolyte."; break;
		case 2723: mes "Este objeto es solo para la clase Magician."; break;
		case 2724: mes "Este objeto es solo para la clase Archer."; break;
		case 2725: mes "Este objeto es solo para la clase Merchant."; break;
		case 2733: mes "Este objeto es solo para Gunslinger."; break;
		default: break;
	}
	mes "Puedes intercambiar por este objeto con ^FF0000"+.@Items[.@i+1]+" "+getitemname(7828)+" o "+.@Items[.@i+1]+" "+getitemname(7829)+"^000000.";
	mes "^Deseas intercambiar?";
	next;
	switch(select("No intercambiar:Intercambiar")) {
	case 1:
		mes "[Erundek]";
		mes "^Necesitas mas tiempo para revisar los objetos?";
		break;
	case 2:
		mes "[Erundek]";
		mes "^Que Insignia quieres intercambiar?";
		mes "Necesitas ^3131FF"+.@Items[.@i+1]+" Insignias^000000 para intercambiar.";
		next;
		if (.@item[0] < 12269 || .@item[0] > 12273 ) {
			mes "[Erundek]";
			mes "Recuerda, los objetos de recompensa del Campo de Batalla estan ^FF0000ligados al personaje^000000. ^Estas seguro de que quieres este objeto?";
			next;
		}
		set .@j, select("Insignia de Bravura:Insignia de Valor:Cancelar");
		mes "[Erundek]";
		if (.@j == 3) {
			mes "Cancelaste el intercambio.";
			break;
		}
		set .@cost, ((.@j==1)?7828:7829);
		if (countitem(.@cost) >= .@Items[.@i+1]) {
			mes "Gracias por el intercambio.";
			delitem .@cost, .@Items[.@i+1];
			getitem .@Items[.@i],1;
		}
		else mes "No tienes suficientes "+getitemname(.@cost)+"s.";
		break;
	}
	close;
}

// Badge Reward Function
//============================================================
function	script	F_BG_Badge	{
	.@team$ = getarg(1);
	.@bg_type$ = getarg(2);

	if (.@bg_type$ == "Tierra") {
		set .@badge,7828; //BF_Badge1
		set .@amount_win,3;
		set .@amount_lose,1;
	} else if (.@bg_type$ == "Flavius") {
		set .@badge,7829; //BF_Badge2
		set .@amount_win,9;
		set .@amount_lose,3;
	} else
		return;

	if (getarg(0) == 1) {
		set .@amount, .@amount_win;
		if (.@team$ == "Guillaume") {
			mes "[Axl Rose]";
			mes "Bendito Guillaume!";
		} else if (.@team$ == "Croix") {
			mes "[Swandery]";
			mes "Bendito Croix!";
		}
		mes "Disfrutemos de nuestra gloriosa victoria!";
		mes strcharinfo(0)+", es una senal que refleja la victoria.";
	} else {
		set .@amount, .@amount_lose;
		if (.@team$ == "Guillaume") {
			mes "[Axl Rose]";
			mes "Perdiste, pero te dedicaste a esta batalla.";
			mes "Esta es una recompensa por tu gran dedicacion de parte de Guillaume Marollo!";
			mes "Solo toma esta derrota como una leccion, y la proxima vez definitivamente ganaras.";
		} else if (.@team$ == "Croix") {
			mes "[Swandery]";
			mes "Oh, "+strcharinfo(0)+" No estes triste.";
			mes "Aunque no ganamos, dimos lo mejor de nosotros.";
			mes "Este es un regalo Real de Croix, y por favor no olvides esta batalla. Ganaremos la proxima.";
		}
	}
	close2;

	// If the VIP system is enabled, VIP players receive an extra 2 badges.
	if (VIP_SCRIPT && vip_status(VIP_STATUS_ACTIVE))
		set .@amount, .@amount+2;

	set .@medal_gap, 500 - countitem(.@badge);
	if (.@medal_gap >= .@amount)
		getitem .@badge,.@amount;
	else
		getitem .@badge,.@medal_gap;
	return;
}
