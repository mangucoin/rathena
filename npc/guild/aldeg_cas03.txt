// Guild Flag (Outside Castle)
//============================================================
-	script	Nuernberg#1::OutsideFlagsA3	722,{

	set .@GID, GetCastleData("aldeg_cas03",1);

	if (.@GID == 0) {
		mes "[ Edicto del Divino Reino de Rune-Midgarts ]";
		mes " ";
		mes "1. Siguiendo la ordenanza del Divino Reino de Rune-Midgarts, ";
		mes "Declaramos que";
		mes "no existe un dueno formal de este castillo.";
		mes " ";
		mes "2. A aquel que pueda ";
		mes "superar todas las pruebas";
		mes "y destruir el Emperium,";
		mes "el rey le otorgara";
		mes "la propiedad de este castillo.";
		close;
	}
	if (getcharid(2) == .@GID) {
		mes "[ Voz Resonante ]";
		mes "Valientes...";
		mes "Desean regresar a su honorable lugar?";
		next;
		if(select("Regresar al castillo del guild.:Salir.") == 1) {
			close2;
			if (getcharid(2) == GetCastleData("aldeg_cas03",1)) {
				warp "aldeg_cas03",205,186;
				end;
			}
			end;
		}
		close;
	}
	mes "[ Edicto del Divino Reino de Rune-Midgarts ]";
	mes " ";
	mes "1. Siguiendo la ordenanza del Divino Reino de Rune-Midgarts, ";
	mes "aprobamos que este lugar esta en";
	mes "posesion privada del Guild ^ff0000"+GetGuildName(.@GID)+"^000000.";
	mes " ";
	mes "2. El Maestro del Guild ^ff0000"+GetGuildName(.@GID)+"^000000 es";
	mes "^ff0000"+GetGuildMaster(.@GID)+"^000000";
	mes "Si alguien se opone a esto,";
	mes "demuestra tu fuerza y honor con una espada de acero en tu mano.";
	close;

OnRecvCastleA03:
	FlagEmblem GetCastleData("aldeg_cas03",1);
	end;
}
