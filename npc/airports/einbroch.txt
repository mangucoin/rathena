airport,143,43,5	script	Airport Staff#airport1a::airport1	90,{
	mes "[Airport Staff]";
	mes "Bienvenido al";
	mes "Aeropuerto de Einbroch,";
	mes "donde ofrecemos vuelos";
	mes "directos a las ciudades de";
	mes "Juno, Lighthalzen y Hugel.";
	next;
	if (select("Abordar la Aeronave:Cancelar") == 1) {
		mes "[Airport Staff]";
		mes "La tarifa de abordaje de";
		mes "la Aeronave es 1,200 zeny, pero";
		mes "si tienes un Free Ticket for Airship,";
		mes "la tarifa sera eximida. Deseas";
		mes "abordar la Aeronave?";
		next;
		if (select("Si:No") == 1) {
			if (countitem(7311) > 0) {
				delitem 7311,1; //Free_Flying_Ship_Ticket
				warp "airport",148,51;
				end;
			}
			if (Zeny >= 1200) {
				set Zeny, Zeny - 1200;
				warp "airport",148,51;
				end;
			}
			mes "[Airport Staff]";
			mes "Lo siento, pero no tienes";
			mes "un Free Ticket for";
			mes "Airship y no tienes";
			mes "suficiente zeny para abordar";
			mes "la Aeronave. Recuerda, la";
			mes "tarifa de abordaje es 1,200 zeny.";
			close;
		}
	}
	mes "[Airport Staff]";
	mes "Gracias y";
	mes "que tengas un buen dia.";
	close;
}
