lhz_airport,143,43,5	script	Airport Staff#lhz_air1a::lhz_airport1	90,{
	mes "[Airport Staff]";
	mes "Bienvenido al";
	mes "Aeropuerto de Lighthalzen,";
	mes "donde ofrecemos vuelos";
	mes "directos a Einbroch, Juno y Hugel.";
	next;
	if (select("Abordar la Aeronave.:Cancelar.") == 1) {
		mes "[Airport Staff]";
		mes "La tarifa de abordaje es";
		mes "1,200 zeny, pero puedes";
		mes "eximirla si canjeas";
		mes "un Free Ticket for Airship.";
		next;
		if (select("Si:No") == 1) {
			if (countitem(7311) > 0) {
				delitem 7311,1; //Free_Flying_Ship_Ticket
				warp "lhz_airport",148,51;
				end;
			}
			if (Zeny >= 1200) {
				set Zeny, Zeny - 1200;
				warp "lhz_airport",148,51;
				end;
			}
			mes "[Airship Staff]";
			mes "Lo siento, pero no tienes";
			mes "1,200 zeny para pagar";
			mes "la tarifa de abordaje.";
			close;
		}
	}
	mes "[Airport Staff]";
	mes "Gracias y";
	mes "vuelve pronto.";
	mes "Que tengas un buen dia~";
	close;
}

lhz_airport,158,43,5	duplicate(lhz_airport1)	Airship Staff#lhz_air1b	90

lhz_airport,143,49,3	script	Arrival Staff#lhz_air2a::lhz_airport2	90,{
	mes "[Arrival Staff]";
	mes "Bienvenido al Aeropuerto de Lighthalzen.";
	mes "Dejame guiarte a la";
	mes "terminal principal si estas llegando de tu vuelo. De lo contrario, por favor";
	mes "aborda la Aeronave de partida para llegar a tu destino previsto.";
	next;
	if (select("Salir a la terminal principal.:Cancelar.") == 1) {
		mes "[Arrival Staff]";
		mes "Una vez en la terminal principal, tendras que pagar la tarifa de nuevo";
		mes "para abordar una Aeronave. Solo";
		mes "deberias salir si Lighthalzen es tu destino previsto. Procedemos";
		mes "a la terminal principal?";
		next;
		if (select("Si:No") == 1) {
			warp "lhz_airport",142,40;
			end;
		}
	}
	mes "[Arrival Staff]";
	mes "Muy bien, gracias";
	mes "por tu preferencia";
	mes "y espero que tengas";
	mes "un vuelo agradable~";
	close;
}
