y_airport,143,43,5	script	Airport Staff#y_air1a::y_airport1	90,{
	mes "[Airport Staff]";
	mes "Bienvenido al Aeropuerto de Juno donde ofrecemos vuelos domesticos a Einbroch, Lighthalzen y Hugel,";
	mes "y vuelos internacionales a Izlude y Rachel.";
	mes "En que puedo servirte?";
	next;
	if (select("Abordar la Aeronave.:Cancelar.") == 1) {
		mes "[Airport Staff]";
		mes "La tarifa de abordaje para todos";
		mes "los vuelos es 1,200 zeny. Si";
		mes "usas un Free Ticket for Airship,";
		mes "la tarifa de abordaje sera eximida. Entonces, deseas partir?";
		next;
		if (select("Si:No") == 1) {
			if (countitem(7311) > 0) {
				delitem 7311,1; //Free_Flying_Ship_Ticket
				warp "y_airport",148,51;
				end;
			}
			if (Zeny >= 1200) {
				set Zeny, Zeny - 1200;
				warp "y_airport",148,51;
				end;
			}
			mes "[Airport Staff]";
			mes "Lo siento, pero no tienes";
			mes "1,200 zeny para pagar";
			mes "la tarifa de abordaje.";
			close;
		}
	}
	mes "[Airport Staff]";
	mes "Gracias y";
	mes "que tengas un buen dia.";
	close;
}
