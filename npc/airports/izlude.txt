-	script	::Airship_Staff_izlude	-1,{
	mes "[Airship Staff]";
	mes "Bienvenido! Te gustaria";
	mes "abordar la Aeronave que";
	mes "parte en el vuelo que hace parada";
	mes "en Juno y Rachel?";
	next;
	if (select("Abordar la Aeronave a Juno/Rachel.:Cancelar.") == 1) {
		mes "[Airship Staff]";
		mes "La tarifa de abordaje es";
		mes "1,200 zeny. Sin embargo, esta";
		mes "tarifa se exime si usas";
		mes "un Free Ticket for Airship. Entonces,";
		mes "aun deseas abordar?";
		next;
		if (select("Si:No") == 1) {
			if (countitem(7311) > 0) {
				delitem 7311,1; //Free_Flying_Ship_Ticket
				warp "airplane_01",244,58;
				end;
			}
			if (Zeny >= 1200) {
				set Zeny, Zeny - 1200;
				warp "airplane_01",244,58;
				end;
			}
			mes "[Airship Staff]";
			mes "Lo siento, pero no tienes";
			mes "1,200 zeny para pagar";
			mes "la tarifa de abordaje.";
			close;
		}
	}
	mes "[Airship Staff]";
	mes "Gracias y";
	mes "vuelve pronto.";
	mes "Que tengas un buen dia~";
	close;
}
