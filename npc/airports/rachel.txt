ra_fild12,295,208,0	script	toairplane#rachel	45,1,1,{
	end;
OnTouch:
	mes "Para usar la aeronave, debes pagar 1,200 zeny o un Free Airship Ticket.";
	mes "Te gustaria usar el servicio?";
	next;
	if (select("Si:No") == 1) {
		if (countitem(7311) > 0) {
			delitem 7311,1; //Free_Flying_Ship_Ticket
			warp "airplane_01",245,60;
			end;
		}
		if (Zeny >= 1200) {
			set Zeny, Zeny - 1200;
			warp "airplane_01",245,60;
			end;
		}
		mes "Lo siento, pero no tienes suficiente dinero.";
		mes "Por favor recuerda, debes pagar 1,200 zeny para usar el servicio.";
		close;
	}
	mes "Gracias, vuelve pronto.";
	close;
}
