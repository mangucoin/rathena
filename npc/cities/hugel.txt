	mes "[Shopkeeper]";
	mes "Bienvenido a la tienda de";
	mes "articulos para fiestas!";
	mes "Por que no disfrutas de unos";
	mes "espectaculares fuegos artificiales";
	mes "con tus amigos?";
	mes "Podemos darte 5 de ellos";
	mes "por 500 zeny.";
	next;
	switch (select("Comprar:Cancelar")) {
		case 1:
			if (Zeny < 500) {
				mes "[Shopkeeper]";
				mes "Lo siento, pero no tienes";
				mes "suficiente dinero~";
				close;
			}
			set Zeny,Zeny-500;
			getitem 12018,5;	// Fire_Cracker
			mes "[Shopkeeper]";
			mes "Aqui tienes!";
			mes "Diviertete con ellos!";
			close;
		case 2:
			mes "[Shopkeeper]";
			mes "Gracias, vuelve pronto.";
			close;
	}
}
