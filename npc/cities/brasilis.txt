-	script	::Crewman_bra2	-1,{
	mes "[Crewman]";
	mes "Oye, ^has escuchado de un lugar llamado Brasilis?";
	mes "Es una ciudad tropical que es caliente como el desierto pero tambien lluviosa. Es un lugar muy misterioso.";
	next;
	mes "[Crewman]";
	mes "Recientemente encontramos una nueva ruta maritima para llegar facilmente.";
	if (VIP_SCRIPT) {
		set .@cost, vip_status(VIP_STATUS_ACTIVE)?1000:10000;
		mes "Son solo 10,000 zeny por un viaje de ida y vuelta, y 1,000 para VIP! ^Quieres ir?"; //custom translation (VIP)
	} else {
		set .@cost,10000;
		mes "Son solo 10,000 zeny por un viaje de ida y vuelta! ^Quieres ir?";
	}
	next;
	switch(select("Llevame a Brasilis!:Me quedo aqui.")) {
	case 1:
		if (Zeny < .@cost) {
			mes "[Crewman]";
			mes "Dije "+callfunc("F_InsertComma",.@cost)+" zeny.";
			close;
		}
		else {
			mes "[Crewman]";
			mes "Genial~!! Vamonos~!";
			set Zeny, Zeny-.@cost;
			close2;
			warp "brasilis",314,60;
			end;
		}
	case 2:
		mes "[Crewman]";
		mes "Bueno, si alguna vez te interesa, avisame y puedo llevarte.";
		close;
	}
}

brasilis,137,77,5	script	Ice-Cream Maker	85,{
	mes "[Ice Cream Maker]";
	mes "Ven~ven~";
	mes "El helado es el bocadillo perfecto para un dia caluroso~";
	mes "Son solo ^3355FF100 Zeny^000000~";
	mes "Helado~";
	mes "Lleva tu Helado!";
	next;
	switch(select("Dame uno!:Helado?:Cancelar.")) {
	case 1:
		mes "[Ice Cream Maker]";
		mes "Como hay tanta gente que quiere un helado fresquito, solo puedes pedir 5 a la vez.";
		mes "Entonces, ^cuantos quieres?";
		next;
		while(1) {
			input .@input; //,1,5;
			if (.@input == 0) {
				mes "[Ice Cream Maker]";
				mes "^Ninguno?";
				mes "Esta bien, sal del camino, tengo clientes que atender.";
				close;
			}
			else if ((.@input < 0) || (.@input > 5)) {
				mes "[Ice Cream Maker]";
				mes "Wow.";
				mes "Pediste demasiado.";
				mes "Si te comes mas de 5 tal vez tengas que pelear con un monstruo en tu estomago. Tranquilo amigo.";
				next;
			}
			else
				break;
		}
		set .@icecream_hap,.@input*100;
		if (Zeny < .@icecream_hap) {
			mes "[Ice Cream Maker]";
			mes "Oye~! No tienes suficiente dinero.";
			mes "Son solo ^3355FF100 Zeny^000000~ En serio!";
			close;
		}
		if (!checkweight(536,.@input)) {
			mes "[Ice Cream Maker]";
			mes "Parece que llevas demasiadas cosas.";
			mes "Aligera tu mochila antes de comprar esto.";
			close;
		}
		set Zeny, Zeny-.@icecream_hap;
		getitem 536,.@input; //Ice_Cream
		close;
	case 2:
		mes "[Ice Cream Maker]";
		mes "El helado es...";
		mes "Espera, ^no sabes";
		mes "que es el Helado?";
		mes "^Debajo de que roca";
		mes "has estado viviendo?";
		next;
		mes "[Ice Cream Maker]";
		mes "Ni voy a empezar con lo raro que suena eso.";
		mes "En fin, lleva tu Helado aqui mientras esta bien frio.";
		close;
	case 3:
		mes "[Ice Cream Maker]";
		mes "No pierdas tu oportunidad de comer el mejor Helado de toda la tierra~!";
		close;
	}
}
