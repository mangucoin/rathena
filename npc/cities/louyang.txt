	mes "[Hong Miao]";
	mes "Bienvenido.";
	next;
	mes "[Hong Miao]";
	mes "Este es un elevador que lleva";
	mes "a la Torre de Observacion. Te ofrecemos un servicio de transporte seguro y rapido por un precio accesible. Te gustaria usar este servicio?";
	next;
	switch(select("Informacion.:Si.:Quiza en otra ocasion.")) {
	case 1:
		mes "[Hong Miao]";
		mes "Despues de muchas sugerencias y propuestas enviadas a la oficina de turismo de Luoyang, la Torre de Observacion fue construida para que los turistas puedan disfrutar de las vistas.";
		next;
		mes "[Hong Miao]";
		mes "Debido a las caracteristicas";
		mes "geograficas de Luoyang, es dificil disfrutar de la impresionante vista que nuestra tierra tiene para ofrecer.";
		next;
		mes "[Hong Miao]";
		mes "Puedes subir a la torre tomando el elevador justo aqui. Ofrecemos este servicio de transporte rapido y seguro por 500 zeny por persona.";
		close;
	case 2:
		if (Zeny < 500) {
			mes "[Hong Miao]";
			mes "Lo siento, pero no tienes suficiente zeny. Espero que vuelvas mas tarde para disfrutar de la Torre de Observacion. Que tengas buen dia.";
			close;
		}
		mes "[Hong Miao]";
		mes "Gracias por tu preferencia.";
		mes "Estamos tratando de brindarte el mejor servicio. Por favor";
		mes "vuelve pronto.";
		next;
		set Zeny, Zeny-500;
		warp "lou_in01",17,19;
		end;
	case 3:
		mes "[Hong Miao]";
		mes "Por favor vuelve";
		mes "mas tarde.";
		mes "Que tengas buen dia.";
		close;
	}
}
