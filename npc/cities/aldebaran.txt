		if (countitem(.@item_req) > 0) {
			mes "[Gatekeeper Boy]";

			mes "Ahora parece que mi Especulacion fue exitosa.";
			mes "Por favor, puedes entrar.";
			mes "Que Dios te bendiga ..";
			close2;
			delitem .@item_req,1;
			warp getarg(2),getarg(3),getarg(4);
			end;
		}
		else {
			mes "[Gatekeeper Boy]";
			mes ". . . . . .";
			mes "Desafortunadamente no tienes el privilegio";
			mes "de entrar por esta Puerta ..";
			mes "No podras pasar";
			mes "mientras los Alquimistas Antiguos";
			mes " no te lo concedan.";
			close;
		}
	case 4:
		mes "[Gatekeeper Boy]";
		mes "Esta Clock Tower";
		mes "es el lugar donde los 3 Antiguos Alquimistas Legendarios";
		mes "dejaron sus Espiritus y Habilidades.";

		close;
	}
