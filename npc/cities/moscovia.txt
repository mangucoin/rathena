-	script	::PR_Officer_moscovia	-1,{
	mes "[Moscovia P.R. Officer]";
	mes "Moscovia! Conoces Moscovia?";
	mes "el paraiso que se extiende en los mares infinitos...";
	mes "Bienvenido a Moscovia";
	mes "Es aventurera y mistica.";
	next;
	switch(select("Sobre Moscovia...:Ir a Moscovia.:Cancelar")) {
		case 1:
			mes "[Moscovia P.R. Officer]";
			mes "Moscovia es un hermoso reino";
			mes "en una isla ubicada al norte de Rune-";
			mes "Midgarts.";
			next;
			mes "[Moscovia P.R. Officer]";
			mes "Estoy seguro de que quedaras";
			mes "absolutamente fascinado";
			mes "por los hermosos paisajes de Moscovia";
			mes "y su esplendido palacio.";
			next;
			mes "[Moscovia P.R. Officer]";
			mes "Ahora que nuestro largo invierno ha";
			mes "pasado,";
			mes "estoy feliz de poder mostrarte";
			mes "nuestro hermoso pueblo.";
			next;
			mes "[Moscovia P.R. Officer]";
			mes "Si te dan ganas de visitar Moscovia";
			mes "aprovecha la oportunidad ahora!";
			mes "Te ayudare a tener un buen viaje";
			mes "a Moscovia!";
			close;

		case 2:
			mes "[Moscovia P.R. Officer]";
			mes "Muy bien, empecemos ahora.";
			mes "Debes pagarme 10,000 zeny";
			mes "para ir a Moscovia.";
			mes "Pero cuando regreses,";
			mes "no tendras que pagar.";
			next;
			mes "[Moscovia P.R. Officer]";
			mes "Podemos irnos ya?";
			next;
			if (select("Vamos!:Cancelar") == 2) {
				mes "[Moscovia P.R. Officer]";
				mes "Si estas muy ocupado ahora,";
				mes "por favor dime cuando quieras.";
				mes "Siempre estoy listo para guiar a cualquiera a Moscovia.";
				close;
			}
			if (Zeny < 10000) {
				mes "[Moscovia P.R. Officer]";
				mes "Lo siento pero no tienes";
				mes "suficiente zeny ahora";
				mes "Necesitas 10,000 zeny";
				mes "para ir a Moscovia";
				mes "Gracias.";
				close;
			} else {
				mes "[Moscovia P.R. Officer]";
				mes "Muy bien, nos vamos ahora.";
				close2;
				set Zeny,Zeny - 10000;
				warp "moscovia",163,55;
				end;
			}
		case 3:
			mes "[Moscovia P.R. Officer]";
			mes "Si estas muy ocupado ahora,";
			mes "por favor dime cuando quieras.";
			mes "Siempre estoy listo para guiar a cualquiera a Moscovia.";
			close;
	}
}
