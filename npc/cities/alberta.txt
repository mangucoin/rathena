alberta,189,151,5	script	Fisk	4W_SAILOR,{
	mes "[Fisk]";
	mes "Ahoy companiero,";
	mes "a donde";
	mes "quieres ir?";
	next;
	switch(select("Barco Hundido -> 250 zeny.","Marina de Izlude -> 500 zeny.","No importa.")) {
	case 1:
		if (Zeny < 250) {
			mes "[Fisk]";
			mes "Oye, no intentes enganiarme! Dije 250 zeny!";
			close;
		}
		Zeny = Zeny - 250;
		warp "alb2trea",43,53;
		end;
	case 2:
		if (Zeny < 500) {
			mes "[Fisk]";
			mes "No hay forma de que llegues ahi sin los 500 zeny primero!";
			close;
		}
		Zeny = Zeny - 500;
		if (checkre(0))
			warp "izlude",195,212;
		else
			warp "izlude",176,182;
		end;
	case 3:
		mes "[Fisk]";
		mes "Esta bien...";
		mes "Rata de tierra.";
		close;
	}
}
