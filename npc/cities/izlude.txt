-	script	::Sailor_izlude	-1,{
	mes "[Sailor]";
	mes "Oigan todos!";
	mes "Atencion, atencion!";
	mes "Vengan a navegar con el viento";
	mes "en un fascinante Barco!";
	mes "Rapido, rapido!";
	next;
	switch(select("Isla Byalan -> 150 Zeny.:Alberta Marina  -> 500 Zeny.:Cancelar.")) {
	case 1:
		if (Zeny < 150) {
			mes "[Sailor]";
			mes "150 Zeny!";
			mes "Solo 150 Zeny por el viaje!";
			close;
		}
		set Zeny, Zeny - 150;
		warp "izlu2dun",107,50;
		end;
	case 2:
		if (Zeny < 500) {
			mes "[Sailor]";
			mes "500 Zeny!";
			mes "Solo 500 Zeny por el viaje!";
			close;
		}
		set Zeny, Zeny - 500;
		warp "alberta",188,169;
		end;
	case 3:
		close;
	}
}
