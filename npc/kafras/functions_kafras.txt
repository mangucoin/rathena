//=== Main Function ========================================================
//=   arg(0): Used to determine which welcome message to show.
//=   arg(1): Used to determine which menu to display.
//=   arg(2): Used to determine if the info menu is shown in F_KafInfo.
//=   arg(3): Cost to use Storage
//=   arg(4): Cost to Rent a Pushcart
//==========================================================================
function	script	F_Kafra	{
	callfunc "F_ClearGarbage"; // Clear outdated, unused variables
	.@welcome = getarg(0);
	.@menu_num = getarg(1);

	// Display Kafra Welcome Message
	switch(.@welcome) {
	default:// Default message (obsolete)
		mes "[Kafra Employee]";
		mes "Bienvenido/a a la";
		mes "Corporacion Kafra.";
		mes "Los servicios Kafra";
		mes "siempre estan de tu lado.";
		mes "¿En que puedo ayudarte?";
		break;
	case 1:// Niflheim specific message
		mes "[Kafra Employee]";
		mes "^666666B-biiienve-e-enido/a";
		mes "a la C-coorpor-r-racion";
		mes "K-kaaafrrrra...^000000";
		break;
	case 2:// Guild Castle Kafra message
		mes "[Kafra Employee]";
		mes "Bienvenido/a, miembro de ^ff0000" + getguildname( getcharid(2) ) + "^000000.";
		mes "La Corporacion Kafra estara contigo a donde quiera que vayas.";
		break;
	case 3:// Amatsu specific message (obsolete)
		mes "[Kafra Employee]";
		mes "¿Vienes de una tierra lejana a estudiar nuestra cultura, o solo estas de paseo?";
		mes "En cualquier caso, ¿por que no te quedas un rato?";
		mes "El aire esta eternamente cargado con el";
		mes "aroma de agradables flores silvestres.";
		break;
	case 4:// Luoyang and Ayothaya specific message (obsolete)
		mes "[Kafra Employee]";
		mes "Con nuestras muchas";
		mes "ubicaciones de servicio Kafra, nunca";
		mes "estas lejos de casa.";
		break;
	case 5:// NPC has it's own welcome message. (Display nothing)
		break;
	}
	next;
	if (.@welcome == 2)// Guild Kafra Menu override (free Teleport, free Storage, Free Cart)
		setarray .@K_Menu0$[0], "Usar Almacen", "Usar Almacen de Guild", "Rentar Carreta", "Usar Servicio de Teletransporte", "Cancelar";
	else {
		switch(.@menu_num) {
			// Save and Storage only
			case 1:	setarray .@K_Menu0$[0],"Guardar","Usar Almacen","Cancelar"; break;
			// Storage only
			case 2:	setarray .@K_Menu0$[0],"Usar Almacen","Cancelar"; break;
			// No Teleport (Common)
			case 3: setarray .@K_Menu0$[0],"Guardar","Usar Almacen","Rentar Carreta","Ver Otra Informacion","Cancelar"; break;
			// Case 4 is Einbroch no tele message.
			// No save, or teleport. (Common)
			case 5:	setarray .@K_Menu0$[0],"Usar Almacen","Rentar Carreta","Ver Otra Informacion","Cancelar"; break;
			// Storage and Check Other Information only.
			case 6:	setarray .@K_Menu0$[0],"Usar Almacen","Ver Otra Informacion","Cancelar"; break;
			// Save, Storage, and Pushcart only (Kafra Warehouse)
			case 7:	setarray .@K_Menu0$[0],"Guardar","Usar Almacen","Rentar Carreta","Cancelar"; break;
			// Save, Storage, Other Check information. (Turbo track)
			case 8: setarray .@K_Menu0$[0],"Guardar","Usar Almacen","Ver Otra Informacion","Cancelar"; break;
			// No Save (Rune Knight)
			case 9: setarray .@K_Menu0$[0],"Usar Almacen","Rentar Carreta","Usar Servicio de Teletransporte","Ver Otra Informacion","Cancelar"; break;
			// Storage, Save, and Pushcart (Dewata, reorder of case 7)
			case 10: setarray .@K_Menu0$[0],"Usar Almacen","Guardar","Rentar Carreta","Cancelar"; break;
			// Default message (obsolete)
			default: setarray .@K_Menu0$[0],"Guardar","Usar Almacen","Usar Servicio de Teletransporte","Rentar Carreta","Ver Otra Informacion","Cancelar"; break;
		}
	}
	.@menu$ = implode(.@K_Menu0$,":");
	while (true) {
		.@j = select(.@menu$) - 1;
		if (.@K_Menu0$[.@j] == "Guardar")
			return;
		else if (.@K_Menu0$[.@j] == "Usar Almacen") {
			if (.@welcome == 2)// Do not charge for Guild Storage
				callfunc "F_KafStor",2,0,0;
			else
				callfunc "F_KafStor",0,getarg(3),.@welcome;
			next;
		}
		else if (.@K_Menu0$[.@j] == "Usar Servicio de Teletransporte") {
			if (.@menu_num != 4)// Display Einbroch "No Teleport Service" notice.
				callfunc "F_KafTele",.@welcome;
			else {
				mes "[Kafra Employee]";
				mes "Debido al ^FF0000Acuerdo de";
				mes "Transporte Limitado^000000, la";
				mes "Corporacion Kafra no puede";
				mes "ofrecer Servicios de Teletransporte";
				mes "en la Republica de Schwarzwald.";
				next;
				mes "[Kafra Employee]";
				mes "Te pedimos que por favor";
				mes "uses el Servicio de Aeronave";
				mes "en su lugar. Gracias por tu";
				mes "comprension y cooperacion.";
			}
			next;
		}
		else if (.@K_Menu0$[.@j] == "Rentar Carreta") {
			if (callfunc("F_KafCart",.@welcome,getarg(4)) == 1)
				next;
		}
		else if (.@K_Menu0$[.@j] == "Ver Otra Informacion") {
			callfunc "F_KafInfo",getarg(2);
			next;
		}
		else if (.@K_Menu0$[.@j] == "Cancelar") {
			callfunc "F_KafEnd",.@welcome,0;
			end;
		}
		else if (.@K_Menu0$[.@j] == "Usar Almacen de Guild") {
			callfunc "F_KafStor",1,0;
			next;
		}
	}
}

//===================================================
// Storage Function
//===================================================
function	script	F_KafStor	{
	.@type = getarg(0);
	.@fee = getarg(1);

	// Unable to access Guild Storage (Busy)
	if (.@type == 1) {
		if (guildopenstorage()) {
			mes "[Kafra Employee]";
			mes "Lo siento, pero otro miembro de la guild esta usando el almacen de guild";
			mes "en este momento. Por favor espera a que termine.";
			close3;
		}
		cutin "", 255;
		close;
	}
	// Unable to access Normal Storage (Insufficient Basic Skills)
	if (!callfunc("F_CanOpenStorage")) {
		mes "[Kafra Employee]";
		// Niflheim Specific Message
		if (getarg(2) == 1) {
			mes "^666666L-lo s-siiieento,";
			mes "e-eres un/a N-noovato/a...";
			mes "N-necesitas h-habilidad";
			mes "b-basica n-nivel 6...^000000";
			return;
		}
		mes "Lo siento, pero";
		mes "necesitas la Habilidad";
		mes "Basica de Novato Nivel 6 para";
		mes "usar el Servicio de Almacen.";
		return;
	}
	// Accessing Normal Storage (Skipped if accessing Storage from Guild castle)
	if (.@type != 2) {
		// Consume "Free Ticket for Kafra Storage" if available.
		if (countitem(7059))
			delitem 7059,1;
		else {
			if (Zeny < .@fee) {
				mes "[Kafra Employee]";
				// Niflheim Specific Message
				if (getarg(2) == 1) {
					percentheal -50,-50;
					mes "^666666Zeeeeny...";
					mes "M-mas z-zeny...!";
					mes "N-necesito 150... zeny...";
					mes "Ergh! T-tomando s-sangre~!^000000";
					return;
				}
				// Standard Message
				mes "Lo siento, pero no";
				mes "tienes suficiente zeny para";
				mes "usar el Servicio de Almacen. La";
				mes "tarifa de acceso al Almacen es de " + .@fee + " zeny.";
				return;
			}
			Zeny -= .@fee;
			RESRVPTS += (.@fee/5);
		}
	}

	// Surpress the "End" message if in guild castle.
	if (.@type != 2) {
		// Niflheim specific Message
		if (getarg(2) == 1) {
			percentheal 0,-10;
			mes "[Kafra Employee]";
			for ( .@i = 0; .@i < 5; .@i++ )
				mes "^666666Gracias.. por... usar...^000000";
		}
		// Normal message
		else {
			mes "[Kafra Employee]";
			mes "Aqui, dejame abrir";
			mes "tu Almacen.";
			mes "Gracias por usar";
			mes "el Servicio Kafra.";
		}
	}
	callfunc("F_CheckKafCode");	//check your storage password, if set
	close2;
	openstorage;
	cutin "", 255;
	end;
}

//===================================================
// Teleport Function
//===================================================
function	script	F_KafTele	{
	mes "[Kafra Employee]";
	mes "Por favor elige";
	mes "tu destino.";
	next;
	.@j = select( implode(@wrpC$,":") ) - 1;
	if (@wrpC$[.@j] == "Cancelar")
		return;
	// Consume "Free Ticket for Kafra Transportation" if available.
	// Do not consume if in Guild castle
	if (countitem(7060) > 0 && getarg(0) != 2)
		delitem 7060,1;
	else {
		if (Zeny < @wrpP[.@j]) {
			mes "[Kafra Employee]";
			mes "Lo siento, pero no tienes";
			mes "suficiente zeny para el Servicio";
			mes "de Teletransporte. La tarifa para";
			mes "teletransportarte a " + @wrpD$[.@j] + " es de " + @wrpP[.@j] + " zeny.";
			close3;
		}
		Zeny -= @wrpP[.@j];
		RESRVPTS += (@wrpP[.@j]/16);
	}
	cutin "", 255;
	if (@wrpD$[.@j] == "Al De Baran") warp "aldebaran",168,112;
	else if (@wrpD$[.@j] == "Alberta") warp "alberta", 117, 56;
	else if (@wrpD$[.@j] == "Comodo") warp "comodo", 209, 143;
	else if (@wrpD$[.@j] == "Comodo Pharos Beacon") warp "cmd_fild07", 127, 134;
	else if (@wrpD$[.@j] == "Geffen") warp "geffen", 120, 39;
	else if (@wrpD$[.@j] == "Izlude") {
		if (checkre(0) == true)
			warp "izlude", 128, 98;
		else
			warp "izlude", 91, 105;
	}
	else if (@wrpD$[.@j] == "Juno") warp "yuno", 158, 125;
	else if (@wrpD$[.@j] == "Mjolnir Dead Pit") warp "mjolnir_02", 99, 351;
	else if (@wrpD$[.@j] == "Morocc") warp "morocc", 156, 46;
	else if (@wrpD$[.@j] == "Orc Dungeon") warp "gef_fild10", 52, 326;
	else if (@wrpD$[.@j] == "Payon") warp "payon", 161, 58;
	else if (@wrpD$[.@j] == "Prontera") warp "prontera", 116, 72;
	else if (@wrpD$[.@j] == "Umbala") warp "umbala", 100, 154;
	end;
}


//===================================================
// Cart Function
//===================================================
function	script	F_KafCart	{
	// Ensure that the class wanting to rent a pushcart is a merchant
	if (BaseClass != Job_Merchant) {
		mes "[Kafra Employee]";
		mes "Lo siento, pero el";
		mes "servicio de renta de Carreta";
		if (checkre(0) == true)
			mes "solo esta disponible para clases Mercader.";
		else {
			mes "solo esta disponible para Mercaderes,";
			mes "Herreros, Maestros Herreros,";
			mes "Alquimistas y Bioquimicos.";
		}
		return 1;
	}
	// Make sure the invoking character does not have a cart already
	else if (checkcart() == true) {
		mes "[Kafra Employee]";
		mes "Ya tienes una";
		mes "Carreta equipada.";
		mes "Lamentablemente, no podemos";
		mes "rentar mas de una por";
		mes "cliente a la vez.";
		return 1;
	}
	// Check if the player has the Pushcart skill
	else if (getskilllv("MC_PUSHCART") == 0) {
		mes "[Kafra Employee]";
		mes "Solo puedes rentar una carreta despues de aprender la habilidad \"Push Cart\".";
		return 1;
	}
	// Consume "Free Ticket for the Cart Service" if available.
	// Do not consume if in Guild castle.
	if (countitem(7061) > 0 && getarg(0) != 2)
		delitem 7061,1;
	else {
		.@rental_fee = getarg(1);
		mes "[Kafra Employee]";
		mes "La tarifa de renta de";
		mes "Carreta es de " + .@rental_fee + " zeny. ¿Te";
		mes "gustaria rentar una Carreta?";
		next;
		if (select("Rentar una Carreta.:Cancelar") == 2)
			return 0;
		if (Zeny < .@rental_fee) {
			mes "[Kafra Employee]";
			mes "Lo siento, pero no";
			mes "tienes suficiente";
			mes "zeny para pagar la tarifa";
			mes "de renta de Carreta de " + .@rental_fee + " zeny.";
			return 1;
		}
		Zeny -= .@rental_fee;
		RESRVPTS += 48;
	}
	setcart;
	return 1;
}

//===================================================
// Check Storage Password Function
//===================================================
function	script	F_CheckKafCode	{
	if (!#kafra_code)
		return;
	mes "Ingresa tu contraseña de almacen:";
	input .@code_;
	if (.@code_ != #kafra_code) {
		dispbottom "Contraseña de almacen incorrecta.";
		close3;
	}
	@kafcode_try = 0;
	return;
}

// === Set / Change / Clear Storage Password Function ===
// getarg(0): NPC Name
// getarg(1): Company Name
//=======================================================
function	script	F_SetKafCode	{
	.@npc_name$ = getarg(0);
	.@comp_name$ = getarg(1);

	mes .@npc_name$;
	if (#kafra_code == 0) {
		mes "" + .@comp_name$ + " te presenta con orgullo un nuevo servicio:";
		mes "Proteccion adicional de almacen con contraseña.";
		next;
		if (select( "Establecer nueva contraseña -> 5000z", "Cancelar" ) == 1)
			callsub( S_SET, .@npc_name$, .@comp_name$ );
		close3;
	}
	mes "Tu almacen esta protegido con contraseña. ¿Que deseas hacer ahora?";
	next;
	switch( select( "Cambiar contraseña anterior -> 5000z", "Eliminar contraseña de almacen -> 1000z", "Cancelar" ) ) {
	case 1:
		mes .@npc_name$;
		mes "Primero, por favor ingresa tu ^0000FFcontraseña anterior^000000.";
		.@code = callfunc("F_EntKafCode");
		if (!.@code || .@code != #kafra_code) {
			mes "Contraseña incorrecta. No puedes establecer una nueva contraseña.";
			emotion ET_SCRATCH;
			break;
		}
		next;
		callsub( S_SET, .@npc_name$, .@comp_name$ );
		break;
	case 2:
		mes .@npc_name$;
		mes "Por favor, ingresa tu contraseña antes de eliminarla.";
		.@code = callfunc("F_EntKafCode");
		if (!.@code) {
			mes "La contraseña no ha sido eliminada.";
			emotion ET_SCRATCH;
			break;
		}
		next;
		mes .@npc_name$;
		if (Zeny < 1000) {
			mes "No tienes suficiente zeny.";
			emotion ET_MONEY;
			break;
		}
		Zeny -= 1000;
		if (.@code == #kafra_code) {
			#kafra_code = 0;
			mes "Has eliminado exitosamente tu contraseña de almacen.";
			mes "Gracias por usar " + .@comp_name$ + ".";
			emotion ET_THANKS;
		} else {
			mes "Contraseña incorrecta. No te devolveremos los 1000z.";
			mes "Por favor, la proxima vez ingresa la contraseña correcta.";
			emotion ET_SORRY;
		}
		break;
	case 3:
		break;
	}
	close3;

S_SET:
	.@npc_name$ = getarg(0);
	.@comp_name$ = getarg(1);

	mes .@npc_name$;
	mes "Ahora ingresa tu ^FF0000nueva contraseña^000000 para proteger tu almacen de ladrones.";
	.@code = callfunc("F_EntKafCode");
	if (!.@code) {
		mes "La contraseña no ha sido cambiada.";
		emotion ET_SCRATCH;
		return;
	}
	next;
	mes .@npc_name$;
	if (Zeny < 5000) {
		mes "No tienes suficiente zeny.";
		emotion ET_MONEY;
		return;
	}
	Zeny -= 5000;
	#kafra_code = .@code;
	mes "Has protegido tu almacen con una contraseña secreta.";
	mes "Gracias por usar " + .@comp_name$ + ".";
	emotion ET_THANKS;
	return;
}

//===================================================
// Basic Password Validation Function
//===================================================
function	script	F_EntKafCode	{
	mes "Ingresa un numero de 1000~10000000:";
	@kafcode_try += 1;
	if (@kafcode_try > 10) {
		@kafcode_try = 0;
		logmes "Tried to fit storage password.";
		return 0;
	}
	if (input(.@code_) == 1) {
		mes "No puedes usar una contraseña tan grande.";
		return 0;
	}
	if (.@code_ < 1000) {
		mes "No deberias usar una contraseña tan corta.";
		return 0;
	}
	return .@code_;
}
