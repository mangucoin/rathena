aldeba_in,96,181,4	script	Kafra Service	113,{
	cutin "kafra_05",2;
	mes "[Kafra Leilah]";
	mes "Hm...?";
	mes "Oh, bienvenido/a a la";
	mes "Sede de la Corporacion";
	mes "Kafra. Necesitabas";
	mes "algo?";
	next;
	switch(select("Guardar:Usar Almacen:Rentar Carreta:Cancelar")) {
	case 1:
		mes "[Kafra Leilah]";
		mes "Tu Punto de Reaparicion ha";
		mes "sido guardado aqui, dentro";
		mes "de la Sede de la Corporacion";
		mes "Kafra. Gracias.";
		next;
		savepoint "aldeba_in",96,179,1,1;
		mes "[Kafra Leilah]";
		mes "Por favor haz uso de";
		mes "los Servicios Kafra que estan";
		mes "disponibles a lo largo de todo";
		mes "Midgard. Gracias por";
		mes "visitar la Sede de Kafra.";
		close2;
		break;
	case 2:
		if (Zeny < 20) {
			mes "[Kafra Leilah]";
			mes "Disculpa, pero";
			mes "parece que no tienes";
			mes "los 20 zeny para pagar";
			mes "la tarifa de acceso al Almacen...";
			close3;
		}
		mes "[Kafra Leilah]";
		mes "Aunque esta instalacion esta";
		mes "destinada exclusivamente para";
		mes "el entrenamiento de Empleadas Kafra";
		mes "y funciones administrativas,";
		mes "abrire tu Almacen.";
		next;
		set Zeny, Zeny-20;
		set RESRVPTS, RESRVPTS + (20/5);
		mes "[Kafra Leilah]";
		mes "En el futuro, por favor";
		mes "pidele a la Empleada Kafra de";
		mes "turno si deseas usar";
		mes "alguno de los Servicios Kafra.";
		mes "Gracias por tu preferencia.";
		callfunc("F_CheckKafCode");    //check your storage password, if set
		close2;
		openstorage;
		break;
	case 3:
		mes "[Kafra Leilah]";
		mes "Mis disculpas, pero no";
		mes "estoy de turno. Te ayudaria";
		mes "si pudiera, pero en realidad no";
		mes "tengo Carretas disponibles.";
		mes "Por que no le pides a otra";
		mes "Empleada Kafra que te asista?";
		next;
		mes "[Kafra Leilah]";
		mes "Las Empleadas Kafra estan";
		mes "ubicadas por todo el";
		mes "continente de Midgard,";
		mes "y deberias poder encontrar";
		mes "varias afuera en Al De Baran.";
		close2;
		break;
	case 4:
		mes "[Kafra Leilah]";
		mes "^666666*Uff...*^000000";
		mes "Que bien, porque en";
		mes "realidad estoy en mi";
		mes "descanso ahora mismo. Elegir";
		mes "''Cancelar'' fue una buena";
		mes "decision de tu parte.";
		close2;
		break;
	}
	cutin "",255;
	end;
}
