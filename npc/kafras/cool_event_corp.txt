//callfunc "F_CoolEventCorp","<menu>","<location>","<save map>",<x>,<y>,"<warp map>";
function	script	F_CoolEventCorp	{
	cutin "zonda_01",2;
	mes "[Cool Event Corp. Staff]";
	mes "Bienvenido/a a Cool Event Corp.";
	mes "Nuestro personal siempre trabaja";
	mes "para superar tus expectativas";
	mes "de servicio de calidad. Entonces,";
	mes "¿en que puedo asistirte hoy?";
	next;
	switch(select(getarg(0))) {
	case 1:
		mes "[Cool Event Corp. Staff]";
		mes "Tu Punto de Reaparicion";
		mes "ha sido guardado aqui";
		mes getarg(1)+".";
		mes "Gracias por usar el";
		mes "servicio de Cool Event Corp.~";
		savepoint getarg(2),getarg(3),getarg(4),1,1;
		close2;
		break;
	case 2:
		mes "[Cool Event Corp. Staff]";
		if(!callfunc("F_CanOpenStorage")){
			mes "Lo siento, pero";
			mes "necesitas la Habilidad";
			mes "Basica de Novato Nivel 6 para";
			mes "usar el Servicio de Almacen.";
			close2;
			break;
		}
		if(Zeny < 40){
			mes "Lo siento, pero no tienes";
			mes "suficiente Zeny para usar";
			mes "el Servicio de Almacen. La";
			mes "tarifa de acceso al Almacen es de 40 Zeny.";
			close2;
			break;
		}
		set Zeny, Zeny - 40;
		mes "Dejame abrir tu almacen";
		mes "personal de inmediato.";
		mes "Gracias por apoyar a Cool";
		mes "Event Corp. usando nuestros";
		mes "servicios. ¡Que tengas un buen dia~!";
		callfunc("F_CheckKafCode");	//check your storage password, if set
		close2;
		openstorage;
		break;
	case 3:
		mes "Por favor elige";
		mes "tu destino.";
		next;
		if (VIP_SCRIPT && !vip_status(VIP_STATUS_ACTIVE))
			set .@cost,4400;
		else
			set .@cost,2200;
		.@destination$ = getarg(5);
		if (select(.@destination$ +" -> "+ .@cost +" z:Cancelar") == 1) {
			if ((countitem(7060) != 0) || (Zeny >= .@cost)) {
				if (countitem(7060) != 0) {
					delitem 7060,1; //Warp_Free_Ticket
				}
				else {
					set Zeny,Zeny-.@cost;
				}
				if (.@destination$ == "Veins") warp "veins",205,101;
				else if (.@destination$ == "Rachel") warp "rachel",115,125;
				cutin "",255;
				end;
			}
			else {
				mes "[Cool Event Corp. Staff]";
				mes "Lo siento, pero no tienes";
				mes "suficiente zeny para el Servicio";
				mes "de Teletransporte. La tarifa para";
				mes "teletransportarte a "+ .@destination$ +" es de "+.@cost+" zeny.";
			}
		}
		close2;
		break;
	case 4:
		mes "[Cool Event Corp. Staff]";
		if(BaseClass != Job_Merchant){
			mes "Lo siento, pero el";
			mes "servicio de renta de Carreta";
			if (checkre(0))
				mes "solo esta disponible para clases Mercader.";
			else {
				mes "solo esta disponible para Mercaderes,";
				mes "Herreros, Maestros Herreros,";
				mes "Alquimistas y Creadores.";
			}
			close2;
			break;
		}
		if(getskilllv("MC_PUSHCART") == 0){
			mes "Solo puedes rentar una carreta despues";
			mes "de aprender la habilidad Pushcart.";
			close2;
			break;
		}
		if(checkcart() == 1){
			mes "Ya tienes una";
			mes "Carreta equipada.";
			mes "Lamentablemente, no podemos";
			mes "rentar mas de una por";
			mes "cliente a la vez.";
			close2;
			break;
		}
		if(countitem(7061)){
			delitem 7061,1; //Cart_Free_Ticket
			setcart;
			close2;
			break;
		}
		mes "La tarifa de renta de";
		mes "Carreta es de 800 Zeny. ¿Te";
		mes "gustaria rentar una Carreta?";
		next;
		if(select("Rentar una Carreta:Cancelar") == 1){
			if(Zeny < 800){
				mes "[Cool Event Corp. Staff]";
				mes "Lo siento, pero no";
				mes "tienes suficiente";
				mes "Zeny para pagar la tarifa";
				mes "de renta de Carreta de 800 Zeny.";
			} else {
				set Zeny, Zeny - 800;
				setcart;
			}
		}
		close2;
		break;
	case 5:
		callfunc("F_SetKafCode","[Cool Event Corp. Staff]","Cool Event Corp.");
		//the function closes msg and removes cutins
	case 6:
	default:
		mes "[Cool Event Corp. Staff]";
		mes "Cool Event Corp. siempre se";
		mes "esfuerza por brindar los mejores";
		mes "servicios a nuestros clientes.";
		mes "Ayudanos a ser los mejores";
		mes "brindandonos tus opiniones";
		mes "y comentarios honestos. Gracias.";
		close2;
	}
	cutin "", 255;
	end;
}

		"Guardar:Usar Almacen::Rentar Carreta:Servicio de Contraseña de Almacen:Cancelar",
		"en el pueblo de Lighthalzen","lighthalzen",95,240;
	end;
