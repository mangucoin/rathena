//===== Description: =========================================
//= Click the chicken and try retrieve the item at a low
//= chance. If you fail he will nuke, freeze, stone,
//= stun, or make you fall asleep.
//= The prize is configurable and triggered by the NPC.
//===== Additional Comments: =================================
//= 1.0 First release.
//= 1.1 Using 'switch rand' instead.
//= 1.2 Cleaned and standardized. [Euphy]
//= 1.2a Switched 'atcommand' to 'unitskilluseid'.
//= 1.3 Switched 'unitskilluseid' to 'unitkill'.
//============================================================

prontera,156,219,4	script	Cluckers	800,{

	set .@GMAccess,80;	// GM level required to access NPC

	if (.startcluck) {
		specialeffect2 EF_HIT3;
		switch(rand(15)) {
			case 0: npctalk "Cluckers : ~COCOROCOOOO!!!"; unitkill getcharid(3); skilleffect "NPC_SELFDESTRUCTION",1; emotion ET_HUK; break;
			case 1: npctalk "Cluckers : Cocoroco~"; break;
			case 2: unitkill getcharid(3); skilleffect "NPC_SELFDESTRUCTION",1; break;
			case 3: sc_start SC_Freeze,10000,0; break;
			case 4: npctalk "Cluckers : ~COCOROCOOOOO!!!"; unitkill getcharid(3); skilleffect "NPC_SELFDESTRUCTION",1; emotion ET_HUK; break;
			case 5: sc_start SC_Sleep,10000,0; break;
			case 6: sc_start SC_Stone,10000,0; emotion ET_KIK; break;
			case 7: npctalk "Cluckers : ~COCOROCOOOO!!!"; unitkill getcharid(3); skilleffect "NPC_SELFDESTRUCTION",1; emotion ET_HUK; break;
			case 8: npctalk "Cluckers : ~Coco! ~COCOROCOO!!"; unitkill getcharid(3); skilleffect "NPC_SELFDESTRUCTION",1; emotion ET_HUK; break;
			case 9: sc_start SC_Stun,10000,0; break;
			case 10: sc_start SC_Sleep,10000,0; emotion ET_KIK; break;
			case 11: npctalk "Cluckers : ~Coco! ~Coco!"; break;
			case 12: sc_start SC_Stun,10000,0; break;
			case 13: unitkill getcharid(3); skilleffect "NPC_SELFDESTRUCTION",1; break;
			default:
				if (rand(50) < 4) {
					npctalk "Cluckers : ~GUAU!...........";
					specialeffect2 EF_SPHERE;
					announce "[~Coco! ~Coco! ~Boom!] " + strcharinfo(0) + " ~le saco el premio! ~Bien hecho!",0;
					getitem $cluck_item_id,$cluck_item_amount;
					set .startcluck,0;
				} else {
					npctalk "Cluckers : ~Coco! ~COCOROCOO!!";
					unitkill getcharid(3); skilleffect "NPC_SELFDESTRUCTION",1;
				}
				break;
		}
		end;
	}
	mes "[Cluckers]";
	if (getgmlevel() >= .@GMAccess) {
		mes "~Coco coco! ~Cocorocoo? ^FF0000~Hola GM " + strcharinfo(0) + ", ~quieres jugar hoy?~^000000";
		mes "Coco coco... ~COCO! ^FF0000~Solo dime que hacer!~^000000";
		next;
		while(1) switch(select("Iniciar Evento:Revisar Premio:Configurar Premio:Hoy no Cluckers")) {
			case 1:
				mes "[Cluckers]";
				mes "~COCO! ^FF0000~Claro que si!~^000000";
				emotion ET_OK;
				close2;
				announce "[~Coco! ~Coco! ~Boom!] esta por comenzar en Prontera!",bc_blue;
				initnpctimer;
				end;
			case 2:
				mes "[Cluckers]";
				mes "Coco, coco coco... ~Coco! ^FF0000~El premio actual es^000000 ^008000"+ $cluck_item_amount +" "+ getitemname($cluck_item_id) +".^000000";
				next;
				break;
			case 3:
				mes "[Cluckers]";
				mes "~Coco coco? ~Coco??? ^FF0000~Cual deberia ser el premio por ganar? Por favor ingresa el ID.~^000000";
				input $cluck_item_id;
				next;
				mes "[Cluckers]";
				mes "~Coco? ~cocorocoo? ^FF0000~Cuantos de este item deberia dar?~^000000";
				input $cluck_item_amount;
				next;
				mes "[Cluckers]";
				mes "Coco coco..? Coco. ^FF0000~Entonces, el premio es^000000 ^008000"+ $cluck_item_amount +" "+ getitemname($cluck_item_id) +"^000000? ^FF0000Genial.~^000000";
				emotion ET_OK;
				next;
				break;
			case 4:
				mes "[Cluckers]";
				mes "Coco coco coco...";
				close;
		}
	}
	mes "~Coco coco! ~Cocorocoo?";
	mes "Coco....";
	close;

OnTimer10000:
	announce "~Apurense detras de la fuente si quieren jugar con el pollo loco!",bc_blue;
	end;
OnTimer20000:
	announce "~Cluckers se comio uno de mis items! ~Tengo mucho miedo para recuperarlo!",bc_blue;
	end;
OnTimer30000:
	announce "Haz clic en el pollo loco e intenta sacarle el item, ~si tienes suerte ganaras! ~Estan listos?",bc_blue;
	end;
OnTimer40000:
	announce "~YA! ~Haz clic en el pollo para obtener el premio!",bc_blue;
	stopnpctimer;
	if (!$cluck_item_id) set $cluck_item_id,512;
	if (!$cluck_item_amount) set $cluck_item_amount,1;
	set .startcluck,1;
	end;
}
