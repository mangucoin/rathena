prontera,142,228,6	script	Find the Mushroom	1084,{
	mes "[ Find The Mushroom ]";
	if (.status == 1) {
		.@count = mobcount( .event_map$, strnpcinfo(0) + "::OnMobKilled" );
		if (.@count > 0) {
			mes "Quedan " + .@count + " Hongos en " + .event_map$ + "!";
			mes "~Encuentra y mata los hongos para ganar " + getitemname(.prize) + "!";
			close;
		}
		.event_map$ = "";
		.status = 0;
		mes "~Todos los hongos fueron eliminados?";
	}
	mes "~No hay ningun evento en este momento!";
	if (.status || getgmlevel() < .GM) close;
	mes "~Iniciar el evento?";
	next;
	if (select("- No:- Si") == 1) close;
	donpcevent strnpcinfo(0) + "::OnMinute10";
	mes "[ Find The Mushroom ]";
	mes "~Evento iniciado!";
	close;

OnInit:
	.prize = 512;	// Reward item ID
	.amount = 10;	// Reward item amount
	.GM = 60;		// GM level required to access NPC
	setarray .maps_list$[0],"izlude","geffen","morocc","prontera"; // Possible maps
	end;

OnMinute10:	// Start time (every hour)
	if (.status) end;
	.status = 1;
	.@spawn = rand(1,10);	// How many Mushrooms should spawn?
	.event_map$ = .maps_list$[ rand(getarraysize(.maps_list$)) ];
	.@label$ = strnpcinfo(0) + "::OnMobKilled";
	killmonster .event_map$, .@label$;
	monster .event_map$,0,0,"~Por favor no me mates!",1084,.@spawn, .@label$;
	announce "Encuentra el Hongo: Un total de " + .@spawn + " Hongos han aparecido en " + .event_map$ + "!",0;
	sleep 2500;
	announce "Encuentra el Hongo: ~Cada Hongo que mates te dara " + getitemname(.prize) + "!",0;
	end;

OnMobKilled:
	if (playerattached() == 0)
		end;
	getitem .prize, .amount;
	.@spawn = mobcount( .event_map$, strnpcinfo(0) + "::OnMobKilled" );
	if (.@spawn > 0)
		announce "[ " + strcharinfo(0) + " ] ha matado un Hongo. Ahora quedan " + .@spawn + " Hongo(s).",bc_map;
	else {
		announce "El Evento Encuentra el Hongo ha terminado. Todos los Hongos han sido eliminados.",0;
		.status = 0;
		.event_map$ = "";
	}
	end;
}
