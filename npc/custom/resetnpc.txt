//===== rAthena Script =======================================
//= Reset NPC
//===== Description: =========================================
//= Resets skills, stats, or both.
//===== Additional Comments: =================================
//= 1.0 First Version
//= 1.1 Optimized for the greater good. [Kisuka]
//= 1.2 Cleaning [Euphy]
//= 1.3 All statuses removed upon skill reset. [Euphy]
//= 1.4 Compressed Script, Added limit use option [Stolao]
//=	Changed set -> setarray, Improved text with F_InsertPlural
//= 1.5 Added sc_end_class to reset related status changes [sader1992]
//============================================================
prontera,150,193,4	script	Reset Girl	124,{
	//	 		Skills,	Stats,	Both,	Limit
	setarray .@Reset,	5000,	5000,	9000,	0;
	mes "[Reset Girl]";
	if(.@Reset[3] && reset_limit > .@Reset[3]) {
		mes "Lo siento, solo puedes resetear "+callfunc("F_InsertPlural",.@Reset[3],"time")+" en tu vida.";
		close;
	}
	mes "Soy la Chica del Reset.";
	mes "Reset Stats: "+ callfunc("F_InsertComma",.@Reset[1]) +"z";
	mes "Reset Skills: "+ callfunc("F_InsertComma",.@Reset[0]) +"z";
	mes "Reset Ambos: "+ callfunc("F_InsertComma",.@Reset[2]) +"z";
	if(.@Reset[3]) mes "Solo puedes resetear "+callfunc("F_InsertPlural",.@Reset[3],"time")+", asi que usalo sabiamente.";
	mes "Por favor selecciona el servicio que deseas:";
	next;
	set .@i,(select("^FF3355Reset Skills:Reset Stats:Reset Ambos^000000:Cancelar"));
	if(.@i > 3) close;
	mes "[Reset Girl]";
	if (Zeny < .@Reset[.@i-1]) {
		mes "Lo siento, no tienes suficiente Zeny.";
		close;
	}
	if(.@Reset[3]){
		mes "Solo puedes resetear "+callfunc("F_InsertPlural",.@Reset[3],"time")+" en tu vida, estas seguro?";
		if(select("Dejame pensar:Esta bien") == 1) close;
	}
	set Zeny, Zeny-.@Reset[.@i-1];
	if(.@i&1){
		sc_end_class;
		ResetSkill;
	}
	if(.@i&2) ResetStatus;
	mes "Listo!";
	if(.@Reset[3]) set reset_limit,reset_limit + 1;
	close;
}
