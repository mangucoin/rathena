//= Players buy ninjas to assassinate other players
//===== Additional Comments: =================================
//= 1.1.1 Changed all gmcommand to atcommand as Poki#3 suggested. [Vicious]
//= 1.1.2 Updated WoE Check. [Paradox924X]

morocc,148,86,5	script	Shifty Assassin	725,{
	set .@ninja_price,250000;

	// STARTS THE MENU //
M_Start:
	mes "[Shifty Assassin]";
	mes "~Que quieres?";
	next;
	menu "Comprar Ninjas",M_Buy,"Asesinar a alguien",M_Kill,"Revisar tus Ninjas",M_Check, ( getgmlevel() > 90 ? "Agregar Ninjas" : "" ),-,"Cancelar",M_Exit;

	// GM MENU TO ADD NINJAS //
	mes "[Shifty Assassin]";
	mes "~Cuantos ninjas quieres hacer disponibles?";
	next;
	input .@add;
	set $ninja_avail, $ninja_avail+.@add;
	mes .@add +" ninjas agregados.";
	close;

	// BUY NINJAS //
M_Buy:
	mes "[Shifty Assassin]";
	mes "~Cuantos ninjas quieres comprar?";
	mes "Hay ^0000FF" + $ninja_avail + "^000000 ninjas disponibles.";
	mes "Cuestan ^0000FF" + .@ninja_price + " zeny ^000000cada uno.";

	input .@buy;
	next;
	if ($ninja_avail < 1) goto NoNinjas;
	if ($ninja_avail < .@buy) goto NotEnoughNinjas;
	set .@price, .@buy*.@ninja_price;
	if (Zeny < .@price) goto NoZeny;

	mes "[Shifty Assassin]";
	mes "Eso te costara ^0000FF"+ .@price +" zeny^000000.";
	next;
	menu "Continuar",-,"Cancelar",M_Exit;

	set Zeny, Zeny - .@price;
	set #ninjas, #ninjas + .@buy;
	set $ninja_avail, $ninja_avail - .@buy;

	mes "[Shifty Assassin]";
	mes "Gracias.";
	close;

	// ASSASSINATE SOMEBODY //
M_Kill:
	if (agitcheck()) goto M_Busy;
	mes "[Shifty Assassin]";
	mes "Ingresa el nombre del objetivo.";
	mes "^FF0000Escribe el nombre exactamente, de lo contrario no podre encontrar a la victima.^000000";
	next;
	menu "Continuar",-,"Cancelar",M_Exit;
	input .@name$;
	if ( !getcharid( 3,.@name$ ) ) {
		mes "[Shifty Assassin]";
		mes .@name$ +" no esta conectado.";
		close;
	}
	next;
	mes "[Shifty Assassin]";
	mes "Ninjas Activos: "+#ninjas;
	mes "Ninjas Descansando: "+#ninjasr;
	mes "~Cuantos quieres enviar?";
	input .@number;
	if (.@number < 1) goto NoNinjasSent;
	if (.@number > #ninjas) goto NotEnoughNinjas1;
	if (.@number > 10) goto TooManyNinjas;
	set .@chance, rand(1,12);
	set #ninjas,#ninjas-.@number;
	set #ninjas,#ninjas+#ninjasr;
	set #ninjasr,0;
	if (.@number < .@chance) goto M_Failure;

	// SUCCESSFUL ATTACK //
	mes "Enviando ninjas ahora.";
	next;
	mes "[Shifty Assassin]";
	set .@ninjasurvived, rand(1,.@number);
	set #ninjasr,.@number-.@ninjasurvived;
	mes "Tu ataque fue exitoso pero solo ^FF0000" + #ninjasr + "^000000 Ninjas sobrevivieron.";
	set .@acc_id, getcharid( 3,.@name$ );
	if ( .@acc_id )
		unitkill .@acc_id;
	announce .@name$ +" ha sido asesinado por los Ninjas de " + strcharinfo(0) +".",bc_npc;
	close;

	// FAILED ATTACK //
M_Failure:
	mes "Enviando ninjas ahora.";
	next;
	mes "[Shifty Assassin]";
	set .@ninjasurvived, rand(1,.@number);
	set #ninjasr, .@number-.@ninjasurvived;
	mes "Tu ataque fallo y solo ^FF0000" + #ninjasr + "^000000 Ninjas sobrevivieron.";

	announce .@name$+" ha sobrevivido al ataque Ninja de " + strcharinfo(0) +".",8;
	close;

	// NINJAS BUSY FOR WOE //
M_Busy:
	mes "[Shifty Assassin]";
	mes "Lo siento, todos mis ninjas estan ocupados en la Guerra del Emperium.";
	close;

	// CHECK YOUR NINJAS //
M_Check:
	mes "[Shifty Assassin]";
	mes "Tienes:";
	mes "^FF0000" + #ninjas + "^000000 Ninjas Activos.";
	mes "^0000FF" + #ninjasr + "^000000 Ninjas Descansando.";
	next;
	goto M_Start;


	// LIMIT //
NoNinjasSent:
	mes "[Shifty Assassin]";
	mes "No puedes matar a nadie sin ninjas.";
	next;
	goto M_Start;

TooManyNinjas:
	mes "[Shifty Assassin]";
	mes "Solo puedes enviar 10 ninjas como maximo.";
	next;
	goto M_Start;

NoZeny:
	mes "[Shifty Assassin]";
	mes "No tienes suficiente zeny.";
	close;

NotEnoughNinjas:
	mes "[Shifty Assassin]";
	mes "No hay tantos ninjas para comprar.";
	next;
	goto M_Start;

NoNinjas:
	mes "[Shifty Assassin]";
	mes "No quedan ninjas para comprar.";
	close;

NotEnoughNinjas1:
	mes "[Shifty Assassin]";
	mes "No tienes tantos ninjas.";
	next;
	goto M_Start;

M_Exit:
	mes "[Shifty Assassin]";
	mes "Adios.";
	close;

// TIMER DELAY NINJA ADDER //
OnClock0600:
OnClock1200:
OnClock1500:
OnClock1900:
OnClock2000:
OnClock0000:
	set $ninja_avail,$ninja_avail+2;
	end;
OnClock1800:
	set $ninja_avail,$ninja_avail+3;
	end;
OnInit:
	set $ninja_avail,$ninja_avail+1;
	end;
}
