//===== Description: =========================================
//= Black Jack card game.  Gameplay based off standard casino
//= black jack rules.  Dealer must have at least 17 to stay and will
//= automatically stay at 17 and up.  Player must have at least
//= 13 to stay.  Aces counted as 11 or 1.  Option to "Double Down".
//= Currently does not allow for "insurance", or "splitting"
//= of pairs.
//===== Additional Comments: =================================
//= 1.0 First version.
//============================================================

cmd_in02,188,89,1	script	Black Jack Dealer	57,{
	mes "[Dealer]";
	mes "Hola! Te gustaria jugar una partida de Black Jack?";
M_Menu:
	next;
	menu "Cuales son las reglas?",M_0, "Quiero jugar.",L_Play, "No gracias.",M_End;

M_0:
	mes "[Dealer]";
	mes "Que te gustaria saber?";
sM_Menu0:
	next;
	menu "Lo basico.",sM_0a, "Ganar y perder.",sM_0b, "'Doblar apuesta'.",sM_0c,
	"Valor del As.",sM_0d, "Nada.",M_Menu;

sM_0a:
	mes "[Dealer]";
	mes "Black Jack es un juego de cartas en el que el objetivo es llegar a 21, o lo mas";
	mes "cerca posible a 21 puntos, sin pasarte de 21.";
	next;
	mes "[Dealer]";
	mes "Antes de que empiece el juego, el jugador debe hacer una apuesta. Una vez hecha";
	mes "la apuesta, tanto el dealer como el jugador reciben 2 cartas";
	mes "cada uno. Dependiendo de tus cartas, puedes elegir ^5533FF'plantarte'^000000,";
	mes "o puedes elegir ^5533FF'pedir'^000000.";
	next;
	mes "[Dealer]";
	mes "Cuando eliges 'plantarte', le dices al dealer que no";
	mes "necesitas mas cartas. Esto permite que el dealer pida si";
	mes "quiere. Para plantarte, debes tener ^FF3355al menos 13 puntos^000000. El";
	mes "dealer solo puede plantarse cuando tiene ^FF335517 puntos o mas^000000.";
	next;
	mes "[Dealer]";
	mes "Cuando eliges 'pedir', le dices al dealer que quieres";
	mes "otra carta. Al pedir mas cartas puedes aumentar tu total";
	mes "de puntos. Mientras tengas ^FF3355menos de 21^000000 puntos puedes pedir una";
	mes "carta del mazo.";
	goto sM_Menu0;
sM_0b:
	mes "[Dealer]";
	mes "Hay 3 formas de ganar y perder en Black Jack.";
	next;
	mes "[Dealer]";
	mes "1.) Al final de una ronda de Black Jack, si tienes mas puntos";
	mes "que el dealer ganaras la ronda. Por el contrario, si tienes";
	mes "menos puntos que el dealer perderas.";
	next;
	mes "[Dealer]";
	mes "2.) Si pides una carta que hace que tu total supere 21";
	mes "perderas automaticamente la ronda. Esto se llama ^5533FF'pasarse'^000000. Si";
	mes "el dealer se pasa, tu ganaras la ronda.";
	next;
	mes "[Dealer]";
	mes "3.) Si tienes un total de 21 con las primeras 2 cartas,";
	mes "ganaras automaticamente la ronda. Esto se llama ^5533FF'Black Jack'^000000";
	mes "y ocurre cuando obtienes un 'As' y una carta de 'valor 10'. Si el";
	mes "dealer obtiene un Black Jack, el ganara automaticamente la ronda.";
	next;
	mes "[Dealer]";
	mes "4.) Ademas de ganar y perder, puedes empatar con el dealer. Si";
	mes "tanto tu como el dealer tienen el mismo total de puntos al final de una";
	mes "ronda, resultara en un empate sin ganador ni perdida o ganancia";
	mes "de dinero.";
	next;
	mes "[Dealer]";
	mes "Esto se llama ^5533FF'empate'^000000 con el dealer. Esto tambien";
	mes "aplica cuando tanto tu como el dealer tienen Black Jack al mismo";
	mes "tiempo.";
	goto sM_Menu0;
sM_0c:
	mes "[Dealer]";
	mes "La opcion de 'Doblar apuesta' te permite duplicar tu apuesta actual,";
	mes "pero con la desventaja de que solo podras pedir una";
	mes "carta adicional. Esta opcion solo esta disponible al inicio de";
	mes "cada ronda.";
	next;
	mes "[Dealer]";
	mes "Un ejemplo de cuando doblar es util, es cuando";
	mes "tus primeras 2 cartas te dan un total de 11. Tienes una buena";
	mes "oportunidad de obtener 21 o 20 con la siguiente carta. Esta";
	mes "seria una buena mano para doblar.";
	goto sM_Menu0;
sM_0d:
	mes "[Dealer]";
	mes "El 'As' es una carta unica en el juego de Black Jack porque";
	mes "puede tener 2 valores. Un As puede contar como 11 puntos,";
	mes "o solo 1 punto. Por ejemplo si tuvieras un As y un 4, eso te daria";
	mes "15 o 5 puntos.";
	next;
	mes "[Dealer]";
	mes "Si decides plantarte, el As automaticamente contara como";
	mes "11 puntos para darte 15 puntos en total.";
	next;
	mes "[Dealer]";
	mes "Si hubieras decidido pedir y recibieras un 9, el As";
	mes "automaticamente contaria como 1 punto para darte un total de 14";
	mes "puntos. Si el As contara como 11 puntos, tendrias un";
	mes "total de mas de 21 y habrias perdido.";
	next;
	mes "[Dealer]";
	mes "Es por la flexibilidad que tienes con el 'As' que lo convierte";
	mes "en la carta mas poderosa del juego.";
	goto sM_Menu0;
M_End:
	mes "[Dealer]";
	mes "Vuelve cuando quieras.";
	close;

//================
L_Play:
	mes "[Dealer]";
	mes "Por favor haz tu apuesta...";
	next;
	menu "2z",M_1a, "10z",M_1b, "20z",M_1c, "100z",M_1d, "Muy caro para mi....",M_End;

M_1a:
	if(Zeny < 2) goto sL_NotEnuf;
	set @bet, 2;
	goto L_Cont0;
M_1b:
	if(Zeny < 10) goto sL_NotEnuf;
	set @bet, 10;
	goto L_Cont0;
M_1c:
	if(Zeny < 20) goto sL_NotEnuf;
	set @bet, 20;
	goto L_Cont0;
M_1d:
	if(Zeny < 100) goto sL_NotEnuf;
	set @bet, 100;
	goto L_Cont0;

sL_NotEnuf:
	mes "[Dealer]";
	mes "Lo siento pero no tienes suficiente zeny para esa apuesta.";
	close;

L_Cont0:
	mes "(las cartas se estan repartiendo)";
	next;
	deletearray $@card[0],13;
	set @dealerTurn, 0;
	set @numP, 0;
	set @numD, 0;
	set @pAce, 0;
	set @dAce, 0;
	callsub sF_GetCards, @numP, @playCard[@numP], @playCard$[@numP], @pAce;
	callsub sF_GetCards, @numP, @playCard[@numP], @playCard$[@numP], @pAce;
	callsub sF_GetCards, @numD, @dealCard[@numD], @dealCard$[@numD], @dAce;
	callsub sF_GetCards, @numD, @dealCard[@numD], @dealCard$[@numD], @dAce;

//==============
L_Start:
	callsub sF_GetTot;

	mes "- Estas son las cartas del ^FF5533DEALER^000000:";
	if (@numD==2) callsub sF_D2cards;
	if (@numD==3) callsub sF_D3cards;
	if (@numD==4) callsub sF_D4cards;
	if (@numD==5) callsub sF_D5cards;
	mes "    El DEALER tiene: ^FF5533"+@dealTot+"^000000";
	mes " ";
	mes "- Estas son ^5533FFTUS^000000 cartas:";
	if (@numP==2) callsub sF_P2cards;
	if (@numP==3) callsub sF_P3cards;
	if (@numP==4) callsub sF_P4cards;
	if (@numP==5) callsub sF_P5cards;
	if(@pAce != 1 || @playTot == 21) mes "    TU tienes: ^5533FF" +@playTot+ "^000000";
	if(@pAce == 1 && @playTot != 21) mes "    Tu tienes: ^5533FF" +@playTot+ "^000000, o ^5533FF" +(@playTot-10)+ "^000000";
	next;
	if(@playTot==21 && @dealTot==21) goto sL_Push;
	if(@numP==2 && @playTot == 21) goto sL_Win;
	if(@numD==2 && @dealTot == 21) goto sL_Lose;
	if(@playTot > 21) goto sL_Lose;
	if(@dealTot > 21) goto sL_Win;
	if(@numP==2 && @dealerTurn==0) menu "Pedir carta",M_Hit, "Plantarse",M_Stay, "Doblar apuesta",M_Double;
	if(@dealerTurn == 0) menu "Pedir carta",M_Hit, "Plantarse",M_Stay;

M_Stay:
	mes "[Dealer]";
	if(@playTot < 13) goto sL_PlayToLow;
	if(@dealTot > 16) mes "El Dealer se planta.";
	if(@dealTot > 16 || @numD == 5) goto L_Check;
	mes "El Dealer va a pedir carta.";
	next;
	callsub sF_GetCards, @numD, @dealCard[@numD], @dealCard$[@numD], @dAce;
	set @dealerTurn, 1;
	goto L_Start;

sL_PlayToLow:
	mes "Lo siento pero no tienes un total suficientemente alto para plantarte. Debes pedir.";
	next;
	goto M_Hit;

M_Hit:
	if(@numP == 5) goto M_Stay;
	callsub sF_GetCards, @numP, @playCard[@numP], @playCard$[@numP], @pAce;
	goto L_Start;
M_Double:
	mes "[Dealer]";
	mes "El jugador eligio doblar apuesta. Tu apuesta actual sera";
	mes "duplicada, y solo podras pedir 1 carta extra.";
	next;
	set @dealerTurn, 1;
	set @bet, @bet*2;
	callsub sF_GetCards, @numP, @playCard[@numP], @playCard$[@numP], @pAce;
	goto L_Start;

//=============
L_Check:
	next;
	if(@playTot < @dealTot) goto sL_Lose;
	if(@playTot == @dealTot) goto sL_Push;

sL_Win:
	mes "[Dealer]";
	mes "Felicidades, ganaste!";
	next;
	set Zeny, Zeny + @bet;
	goto L_Play;
sL_Lose:
	mes "[Dealer]";
	mes "Lo siento pero perdiste.";
	set Zeny, Zeny - @bet;
	next;
	goto L_Play;
sL_Push:
	mes "[Dealer]";
	mes "Es un empate. Empataste con el Dealer.";
	next;
	goto L_Play;


//==================================
	// Sub function for dealing/pulling the cards
sF_GetCards:
	set @rnd, rand(1,13);
	if($@card[@rnd] == 4) goto sF_GetCards;
	set $@card[@rnd], $@card[@rnd] + 1;
	set getarg(1), @rnd;
	if(getarg(1) > 10) set getarg(1), 10;
	if(getarg(1) == 1 && getarg(3) < 1) set getarg(1), 11;
	if(getarg(1) == 11) set getarg(3), 1;
	set getarg(2), " " + getarg(1) + " ";
	if(@rnd == 10) set getarg(2), getarg(1);
	if(@rnd == 1) set getarg(2), " A ";
	if(@rnd == 11) set getarg(2), " J ";
	if(@rnd == 12) set getarg(2), " Q ";
	if(@rnd == 13) set getarg(2), " K ";
	set getarg(0), getarg(0) + 1;
	return;

//==================================
	// Sub function for getting the total score for each hand
sF_GetTot:
	set @i, 0;
	set @dealTot, 0;
	set @playTot, 0;

GetDealTot:
	set @dealTot, @dealTot + @dealCard[@i];
	set @i, @i + 1;
	if(@i < @numD) goto GetDealTot;
	if(@dAce == 1 && @dealTot > 21) set @dealTot, @dealTot - 10;
	set @i, 0;
GetPlayTot:
	set @playTot, @playTot + @playCard[@i];
	set @i, @i + 1;
	if(@i < @numP) goto GetPlayTot;
	if(@pAce == 1 && @playTot > 21) set @pAce, 2;
	if(@pAce > 0 && @playTot > 21) set @playTot, @playTot - 10;
	return;

//=======================================
	// Sub function for displaying the Cards
sF_D2cards:
	mes "    .-----.   .-----. ";
	mes "    | "+@dealCard$[0]+" |   | "+@dealCard$[1]+" | ";
	mes "    '-----'   '-----' ";
	return;
sF_P2cards:
	mes "    .-----.   .-----. ";
	mes "    | "+@playCard$[0]+" |   | "+@playCard$[1]+" | ";
	mes "    '-----'   '-----' ";
	return;
sF_D3cards:
	mes "    .-----.   .-----.   .-----. ";
	mes "    | "+@dealCard$[0]+" |   | "+@dealCard$[1]+" |   | "+@dealCard$[2]+" | ";
	mes "    '-----'   '-----'    '-----' ";
	return;
sF_P3cards:
	mes "    .-----.   .-----.   .-----. ";
	mes "    | "+@playCard$[0]+" |   | "+@playCard$[1]+" |   | "+@playCard$[2]+" | ";
	mes "    '-----'   '-----'    '-----' ";
	return;
sF_D4cards:
	mes " .-----.  .-----.  .-----. ";
	mes " | "+@dealCard$[0]+" |  | "+@dealCard$[1]+" |  | "+@dealCard$[2]+" | ";
	mes " '-----'  '-----'   '-----' ";
	mes "    .-----.  ";
	mes "    | "+@dealCard$[3]+" |";
	mes "    '-----'   ";
	return;
sF_P4cards:
	mes " .-----.  .-----.  .-----. ";
	mes " | "+@playCard$[0]+" |  | "+@playCard$[1]+" |  | "+@playCard$[2]+" |";
	mes " '-----'  '-----'   '-----' ";
	mes "    .-----.  ";
	mes "    | "+@playCard$[3]+" |";
	mes "    '-----'   ";
	return;
sF_D5cards:
	mes " .-----.  .-----.  .-----. ";
	mes " | "+@dealCard$[0]+" |  | "+@dealCard$[1]+" |  | "+@dealCard$[2]+" | ";
	mes " '-----'  '-----'   '-----' ";
	mes "    .-----.   .-----. ";
	mes "    | "+@dealCard$[3]+" |   | "+@dealCard$[4]+" | ";
	mes "    '-----'   '-----' ";
	return;
sF_P5cards:
	mes " .-----.  .-----.  .-----. ";
	mes " | "+@playCard$[0]+" |  | "+@playCard$[1]+" |  | "+@playCard$[2]+" | ";
	mes " '-----'  '-----'   '-----' ";
	mes "    .-----.   .-----. ";
	mes "    | "+@playCard$[3]+" |   | "+@playCard$[4]+" | ";
	mes "    '-----'   '-----' ";
	return;
}
