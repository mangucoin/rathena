sec_pri,36,58,1	script	Chief Warder	105,{
	mes "[Saddeus]";
	emotion ET_QUESTION;
	mes (Sex==SEX_MALE?"Sr.":"Sra.")+" Prisionero #"+BaseLevel+JobLevel+",";
	mes "~que es todo ese ruido ahi?";
	next;
	menu "~Dejame libre, pagare!",-, "~Dejame trabajar por mi libertad!",M_Q1, "Nada",M_NO_THANKS;

M_PAY:
	mes "[Saddeus]";
	set .@MUSTPAY,(PRISON+1)*1000000;
	if (.@MUSTPAY<1000000) set .@MUSTPAY,1000000;
	if (.@MUSTPAY>100000000) set .@MUSTPAY,100000000;
	if (PRISON==0) mes "No te recuerdo. ~Es tu primera vez?";
	if (PRISON>0) mes "~Tu de nuevo? Alguien no aprendio la leccion desde la primera visita...";
	mes "OK, podriamos liberarte por ^FF0000"+.@MUSTPAY+"z^000000.";
	next;
	menu "~Pago en efectivo!",M_PAYCASH,"~Aceptan cheques?",M_PAYBANK,"Cambie de opinion.",-;

	mes "[Saddeus]";
	mes "Tienes algo de tiempo para pensarlo...";
	close;

M_PAYCASH:
	if (.@MUSTPAY>Zeny) goto L_NOCASH;
	set Zeny,Zeny-.@MUSTPAY;
	mes "[Saddeus]";
	mes "OK, firma aqui y aqui.";
	goto L_RELEASE;

L_NOCASH:
	mes "[Saddeus]";
	mes "~Que es esto? ~No es suficiente!";
	close;

M_PAYBANK:
	if (.@MUSTPAY>#kafrabank) goto L_NOBANK;
	set #kafrabank,#kafrabank-.@MUSTPAY;
	mes "[Saddeus]";
	mes "OK, firma tu cheque. Y pon tu nombre en mi libro.";
	goto L_RELEASE;

L_NOBANK:
	mes "[Saddeus]";
	if (#kafrabank==0) mes "Tu expediente no dice nada sobre tus cuentas bancarias.";
	if (#kafrabank!=0) mes "Vaya, aqui esta la info de tu cuenta bancaria... ^FF0000"+#kafrabank+"z^000000... en el expediente. No es suficiente.";
	mes "~Deja tus juegos tontos ahora!";
	close;

L_RELEASE:
	set PRISON,PRISON+1;
	next;
	mes "[Saddeus]";
	mes "~Ahora eres libre!";
	next;
	savepoint "izlude",105,112;
	warp "izlude",105,112;
	close;

M_Q1:
	mes "[Saddeus]";
	mes "Nuestras reservas estan cortas de hongos. ~Recuerdas la papilla de esta manana? ~Que tal algo de trabajo en nuestra granja de hongos patrocinada?";
	next;
	menu "OK, ~trabajare por mi libertad!",-, "~Amo esta prision!",M_NO_THANKS;

	mes "[Saddeus]";
	if(checkcart() || checkfalcon() || checkriding()){
		emotion ET_SCRATCH;
		mes "Lo siento, ~debes liberar tu Carreta, PecoPeco o Halcon!";
		close;
	}
	mes "Habla con nuestro capataz, Oliver.";
	next;

	nude;

	set .@rw,rand(1,4);
	if (.@rw==2) goto L_W2;
	if (.@rw==3) goto L_W3;
	if (.@rw==4) goto L_W4;

L_W1:
	savepoint "sec_in02",179,76;
	warp "sec_in02",179,76;
	close;

L_W2:
	savepoint "sec_in02",139,32;
	warp "sec_in02",139,32;
	close;

L_W3:
	savepoint "sec_in02",100,28;
	warp "sec_in02",100,28;
	close;

L_W4:
	savepoint "sec_in02",107,75;
	warp "sec_in02",107,75;
	close;

M_NO_THANKS:
	mes "[Saddeus]";
	if (rand(2)) mes "~Hoy es Navidad, eh?";
	mes "~Ahora callate y retrocede!";
	if (rand(2)) emotion ET_HUK;
	close;
}

sec_in02,137,57,1	script	Overseer	708,{
	mes "[Oliver]";

	delitem 4002,countitem(4002);//Items: Fabre_Card,
	delitem 4009,countitem(4009);//Items: Chonchon_Card,
	delitem 4022,countitem(4022);//Items: Spore_Card,
	delitem 4048,countitem(4048);//Items: Poison_Spore_Card,

	if (PRISON_Q <= 0 ) goto L_GET_Q;

	mes "Quiero que recolectes ^FF0000"+PRISON_Q+" Orange Net Mushroom^000000.";
	if ( countitem(1069)<PRISON_Q ) close;//Items: Orange_Net_Mushroom,
	mes "OK... dame todos los hongos...";
	mes "Dejame ver... "+countitem(1069)+" de los casi comestibles...";//Items: Orange_Net_Mushroom,
	mes "Y "+countitem(1070)+" de los mortales inutiles.";//Items: Orange_Gooey_Mushroom_,
	next;
	mes "[Oliver]";
	mes "Gracias. ~Eres libre!";
	set PRISON_Q,0;
	delitem 1069,countitem(1069);//Items: Orange_Net_Mushroom,
	delitem 1070,countitem(1070);//Items: Orange_Gooey_Mushroom_,
	next;
	savepoint "izlude",105,112;
	warp "izlude",105,112;
	close;

L_GET_Q:
	set PRISON,PRISON+1;
	if (PRISON > 1) mes "Hmm... Ya has estado aqui antes... Vaya-vaya...";

	if ( countitem(1069) || countitem(1070) ) mes "~Suelta esos hongos!! ~Son venenosos!";//Items: Orange_Net_Mushroom, Orange_Gooey_Mushroom_,
	delitem 1069,countitem(1069);//Items: Orange_Net_Mushroom,
	delitem 1070,countitem(1070);//Items: Orange_Gooey_Mushroom_,

	set PRISON_Q, PRISON*3 + BaseLevel/3;
	set PRISON_Q, PRISON_Q * (readparam(bAspd)/55 + 1);
	set PRISON_Q, PRISON_Q+rand(50,60);
	mes "OK, debes traerme ^FF0000"+PRISON_Q+"^000000 comestibles ^FF0000Orange Net Mushroom^000000.";
	if(readparam(bAspd)>=100) mes "~Eres rapido! Tendras que conseguir mas hongos entonces.";
	if (rand(10)<4) mes "Y puedes comerte los mortales para el desayuno.";
	close;
}
