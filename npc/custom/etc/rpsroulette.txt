cmd_in02,182,126,2	script	Crazy Boris	85,{
	set .@counter,1;
	mes "~Oye tu! ~Listo para Piedra Papel Tijera Ruleta?";
	next;
	menu "Dejame jugar.",PLAY,"Explicame las reglas.",RULES,"Irme",LEAVE;
SAME:
	mes "~Empate! ~Otra vez!";
	next;
PLAY:
	mes "Piedra... Papel...";
	set .@opp, rand(1,3);
	menu "^0000FF~PIEDRA!",-,"^FF0000~TIJERA!",SCISSORS,"^00FF00~PAPEL!^000000",PAPER;

  	if (.@lastchoice == 1) set .@opp, rand(1,3);
  	if (.@opp == 1) emotion ET_ROCK;
  	else if (.@opp == 2) emotion ET_SCISSOR;
  	else emotion ET_WRAP;
  	set .@lastchoice,1;
	if (.@opp == 1) goto SAME;
	if (.@opp == 2) goto WIN;
	goto LOSE;

SCISSORS:
	if (.@lastchoice == 2) set .@opp,rand(1,2);
  	if (.@opp == 1) emotion ET_ROCK;
  	else if (.@opp == 2) emotion ET_SCISSOR;
  	else emotion ET_WRAP;
  	set .@lastchoice,2;
	if (.@opp == 1) goto LOSE;
	if (.@opp == 2) goto SAME;
	goto WIN;

PAPER:
	if (.@lastchoice == 3) set .@opp,rand(2,3);
  	if (.@opp == 1) emotion ET_ROCK;
  	else if (.@opp == 2) emotion ET_SCISSOR;
  	else emotion ET_WRAP;
  	set .@lastchoice,3;
	if (.@opp == 1) goto WIN;
	if (.@opp == 2) goto LOSE;
	goto SAME;

WIN:
	mes "~Maldicion, ganaste!";
	emotion ET_PROFUSELY_SWEAT;
	next;
	set .@win, 1;
	goto YOUPULL;

LOSE:
	emotion ET_SMILE;
	mes "~Ja! ~Perdiste!";
	next;
	set .@win, 0;

YOUPULL:
	mes .@counter +" de 6";
	if ( .@counter == 6 )
		mes "Reza tus oraciones";
	set .@pull, rand( 1,( 7 -.@counter ) );
	set .@counter, .@counter +1;
	next;
	if ( .@pull == 1 ) {
		if ( .@win ) goto KILL;
		specialeffect2 EF_SUI_EXPLOSION;
		emotion ET_KIK;
		percentheal -100,-100;
		mes "*^0000FFClic^000000* *^FF0000BANG^000000*";
		mes "~Estas muerto!";
		close;
	}
	emotion ET_HNG;
	mes "*^0000FFClic^000000* uff...";
	goto PLAY;

RULES:
	mes "Ok, aqui van las reglas:";
	mes "Tengo conmigo una pistola de ^FF00006^000000 camaras con ^FF00001^000000 bala. Primero jugamos ^FF0000Piedra ^00FF00Papel ^0000FFTijera^000000. El perdedor jala el gatillo. El ganador es quien salga mejor parado.";
	mes "Venceme para ganar un premio.";
	next;
	menu "Dejame jugar.",-,"No gracias.",LEAVE;
	mes "Ok aqui vamos...";
	next;
	goto PLAY;

KILL:
	specialeffect EF_SUI_EXPLOSION;
	emotion ET_HUK;
	mes "*^0000FFClic^000000* *^FF0000BANG^000000*";
	mes "~OWWW @#$%^!! ~ESO DOLIO COMO EL INFIERNO!!";
	next;
	mes "~Felicidades! Has ganado...";
	switch( rand( 1,10 ) ) {
		case 1: setarray .@reward[0], 10,984; break;
		case 3: setarray .@reward[0],100,601; break;	// 100x Fly Wings
		case 4: setarray .@reward[0],  8,603; break;	// 8x Old Blue Box
		case 5: setarray .@reward[0],  4,617; break;	// 4x Old Violet Box
		case 6: setarray .@reward[0],  1,616; break;	// 1x Old Card Album
		case 7: setarray .@reward[0], 10,604; break;	// 10x Dead Branch
		case 8: setarray .@reward[0],  3,969; break;	// 3x Gold
		case 10: setarray .@reward[0],20,505; break;	// 20x Blue Potion
		case 2:
		case 9:
			setarray .@reward[0], 10,985;
	}
	mes .@reward[0] +"x "+ getitemname( .@reward[1] ) +"!";
	getitem .@reward[1], .@reward[0];
	close;

LEAVE:
	mes "Cobarde.";
	close;
}
