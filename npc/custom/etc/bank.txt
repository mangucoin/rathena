//= the script. To use it simply change the first set command
//= to set the cost variable to whatever you want the fee to be.
//= Version 2.0: Optimized and brought the script up to standard. [Jguy]

prontera,132,217,5	script	Banker	109,{
	set @cost,500;
	mes "[Banker]";
	mes "Bienvenido al Primer Banco de Prontera. ~Como puedo ayudarte hoy?";
	next;
	switch(select("Me gustaria hacer un deposito.:Me gustaria hacer un retiro.:~Cual es mi saldo actual?:Cancelar")) {
	case 1:
		mes "[Banker]";
		mes "Muy bien... ~Cuanto te gustaria depositar? Lo maximo que puedes depositar a la vez es 999,999 Zeny.";
		next;
		if (@cost > 0) {
			mes "[Banker]";
			mes "Ah, y ten en cuenta que hay un cargo de " +@cost + " Zeny en todas las transacciones.";
			next;
		}
		input @deposit;
		if (@deposit < 1) {
			mes "Asegurate de pedirme depositar una cantidad real.";
			close;
		}
		else if (@deposit > Zeny) {
			mes "Parece que no tienes la cantidad de zeny que intentas depositar.";
			close;
		}
		else if (@deposit > (Zeny - @cost)) {
			mes "Necesitas " + @cost + " Zeny para cubrir la comision de la transaccion.";
			close;
		}
		else {
			set Zeny,Zeny - @deposit;
			set Zeny,Zeny - @cost;
			set #bankstorage,#bankstorage + @deposit;
			mes "[Banker]";
			mes "Muchas gracias... Tu zeny esta en buenas manos.";
			close;
		}
	case 2:
		mes "[Banker]";
		mes "Muy bien... ~Cuanto te gustaria retirar? Lo maximo que puedes retirar a la vez es 999,999 Zeny.";
		next;
		if (@cost > 0) {
			mes "[Banker]";
			mes "Ah, y ten en cuenta que hay un cargo de " +@cost + " Zeny en todas las transacciones.";
			next;
		}
		input @withdrawl;
		if (@withdrawl < 1) {
			mes "Por favor no juegues. Necesito una cantidad real para retirar.";
			close;
		}
		else if (@withdrawl > #bankstorage) {
			mes "Solo tienes ^00FF00" + callfunc("F_InsertComma",#bankstorage) +"^000000 zeny en tu cuenta.";
			close;
		}
		else if ((@cost > Zeny) && ((Zeny + @withdrawl) > @cost)) {
			mes "[Banker]";
			mes "No tienes el Zeny para la comision de la transaccion ahora mismo. ~Te gustaria que tome la comision directamente de tu retiro?";
			next;
			switch(select("Si, por favor.:No, gracias")){
			case 1:
				mes "[Banker]";
				mes "Descontando " + @cost + " de tu retiro para cubrir la comision del deposito...";
				set @withdrawl,@withdrawl - @cost;
				set #bankstorage,#bankstorage - @cost;
				set @cost,0;
				next;
				set Zeny,Zeny - @cost;
				set Zeny,Zeny + @withdrawl;
				set #bankstorage,#bankstorage - @withdrawl;
				mes "[Banker]";
				mes "Aqui esta tu Zeny. Que tengas un buen dia.";
				close;
			case 2:
				mes "[Banker]";
				mes "Muy bien... vuelve pronto.";
				close;
			}
		}
		else {
			set Zeny,Zeny - @cost;
			set Zeny,Zeny + @withdrawl;
			set #bankstorage,#bankstorage - @withdrawl;
			mes "[Banker]";
			mes "Aqui esta tu Zeny. Que tengas un buen dia.";
			close;
		}
	case 3:
		mes "[Banker]";
		mes "Hmmmm... dejame revisar unos papeles.";
		next;
		mes "*Rustle, Rustle*";
		next;
		mes "[Banker]";
		mes "Actualmente tienes " + callfunc("F_InsertComma",#bankstorage) + " Zeny en tu cuenta.";
		close;
	case 4:
		mes "[Banker]";
		mes "Muy bien... vuelve pronto.";
		close;
	}
}
