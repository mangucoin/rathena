function SF_Groom {
	if ($wed_bride_progress == 0) {
		mes "["+@name$+"]";
		mes "Parece que tu novia aun no ha llegado a registrarse.";
		close;
	}
	if (SF_equip_check() == 0)
		close;

	switch($wed_groom_progress) {
	case 1:
		SF_AcceptBride();
		break;
	case 2:
		mes "["+@name$+"]";
		mes "Estoy esperando a que tu pareja te acepte para comenzar la ceremonia.";
		close;
	case 3:
		SF_RetrieveRingM();
		break;
	case 4:
		mes "["+@name$+"]";
		mes "El anillo de tu pareja aun no ha sido reclamado. La ceremonia comenzara cuando ambos hayan reclamado sus anillos.";
		close;
	case 5:
		mes "["+@name$+"]";
		SF_StartCeremony();
		break;
	default:
		mes "["+@name$+"]";
		mes "Por favor no me interrumpas ahora.";
		close;
	}
}

function SF_AcceptGroom {
	mes "["+@name$+"]";
	mes $wed_bride$+", "+$wed_groom$+" ha pedido ser tu esposo por el resto de tu vida. Aceptas?";
	next;
	switch(select("Necesito tiempo para pensarlo.","No, no acepto!","Si, acepto!")) {
	case 1:
		mes "["+@name$+"]";
		mes "Que!?";
		mes "err.. *cof* *cof* muy bien... vuelve cuando te hayas decidido.";
		emotion ET_ANGER;
		close;
	case 2:
		mes "["+@name$+"]";
		mes "!!";
		mes "Ah... err... ehm... esta bien. Ustedes dos parecen tener algunas diferencias que resolver primero.";
		close2;
		emotion ET_HUK;
		npctalk @name$ +" : Damas y caballeros, "+$wed_bride$+" ha rechazado casarse con "+$wed_groom$+"!";
		SF_wed_end();
		break;
	case 3:
		set $wed_bride_progress,2;
		if ($wed_groom_progress == 2) {
			SF_RingsAccepted();
			break;
		}
		emotion ET_OK;
		mes "["+@name$+"]";
		mes "Despues de que tu novio apruebe, se les daran sus anillos, la ceremonia comenzara y estaran oficialmente casados.";
		close;
	}
}

function SF_RetrieveRingM {
	mes "["+@name$+"]";
	if ($@wed_ring && countitem($@wed_ring) < 1) {
		mes "Que paso con tu "+getitemname($@wed_ring)+"? No lo perdiste... verdad? Lo necesitamos para continuar con la ceremonia!";
		close;
	}
	if ($wed_bride_sex)
		set @item, 2634;	//Groom's wedding ring
	else
		set @item, 2635;	//Bride's wedding ring
	if (getnameditem(@item,$wed_groom$) == 0) {
		mes "No pareces tener suficiente espacio para llevar el anillo... libera algo de espacio y vuelve a reclamar el anillo de tu pareja.";
		close;
	}
	mes "Aqui esta el anillo de boda para tu novia.";
	if ($@wed_ring) delitem $@wed_ring,1;
	set $wed_groom_progress,4;

	if ($wed_bride_progress == 4)
		SF_StartCeremony();
	else {
		mes "Una vez que tu novia reclame el anillo, la ceremonia comenzara.";
		close;
	}
}

function SF_StartCeremony {
	mes "Ahora comenzare la ceremonia de boda, y seran declarados marido y mujer.";
	set $wed_bride_progress,5;
	set $wed_groom_progress,5;
	set $@msg$,$wed_groom$;
	if (strcharinfo(0) == $wed_groom$)
		set $@msg$,$wed_bride$;
	if (marriage($@msg$) == 0) {
		next;
		mes "["+@name$+"]";
		mes "Donde esta "+$@msg$+"?? No puedo casarlos si uno no esta presente...";
		close;
	}
	set $wed_bride_progress,6;
	set $wed_groom_progress,6;
	initnpctimer;
	close;
}

//Subfunction: Checks that the groom/bride is still wearing their stuff.
function SF_equip_check {
	if (Sex == SEX_MALE && getequipid(EQI_ARMOR) != 7170) {
		mes "["+@name$+"]";
		mes "Hijo, que hiciste con tu "+getitemname(7170)+"?";
		emotion ET_THINK;
		return 0;
	}
	if (Sex == SEX_FEMALE && getequipid(EQI_ARMOR) != 2338) {
		mes "["+@name$+"]";
		mes "Hija, se supone que debes usar un "+getitemname(2338)+" en todo momento durante la ceremonia...";
		emotion ET_THINK;
		return 0;
	}
	if (Sex == SEX_FEMALE && $@wed_veil && getequipid(EQI_HEAD_TOP) != 2206) {
		mes "["+@name$+"]";
		mes "Hija, no puedes quitarte tu "+getitemname(2206)+" todavia....";
		emotion ET_THINK;
		return 0;
	}
	return 1;
}

	function SF_WedProgress;
	function SF_Principles;
	function SF_Procedure;
	function SF_Register;
	function SF_TryRegister;

	if ($wed_progress) {
		SF_WedProgress();
		end;
	}

	do {
		mes "["+@name$+"]";
		mes "El matrimonio... es algo tan hermoso.";
		mes "Te gustaria casarte con alguien?";
		next;
		set @menu, select(
			"Sere soltero para siempre!",
			"Explica los principios del matrimonio.",
			"Explica el procedimiento del matrimonio.",
			"Quiero casarme con alguien."
		);
		switch (@menu) {
		case 1: //Quit
			mes "["+@name$+"]";
			mes "En ese caso, disfruta tu vida de soltero.";
			close;
		case 2: //Principles
			SF_Principles();
			break;
		case 3: //Procedure
			SF_Procedure();
			break;
		case 4: //Register
			SF_Register();
			break;
		}
	} while (@menu > 1);
	end;

function SF_Register {
	if ($@wed_allow) { //Role select
		mes "["+@name$+"]";
		mes "Muy bien, como quien te gustaria registrarte?";
		next;
		set @submenu, select("Novio","Novia","Cancelar");
	} else if (Sex == SEX_MALE) { //Groom
		mes "["+@name$+"]";
		mes "Muy bien, te registraras como el Novio?";
		next;
		if (select("Si","Cambie de opinion.")==1)
			set @submenu, 1;
		else
			set @submenu, 3;
	} else { //Bride
		mes "["+@name$+"]";
		mes "Muy bien, te registraras como la Novia?";
		next;
		if (select("Si","Cambie de opinion.")==1)
			set @submenu, 2;
		else
			set @submenu, 3;
	}
	switch (@submenu) {
	case 1: //Groom
		SF_TryRegister(0);
		set $wed_progress,1;
		mes "["+@name$+"]";
		mes "Ahora estas registrado como el novio.";
		mes "Dile a tu novia que se registre lo antes posible.";
		emotion ET_SCRATCH;
		initnpctimer;
		close;
	case 2: //Bride
		SF_TryRegister(1);
		set $wed_progress,1;
		mes "["+@name$+"]";
		mes "Ahora estas registrada como la novia.";
		mes "Dile a tu novio que se registre lo antes posible.";
		emotion ET_SCRATCH;
		initnpctimer;
		close;
	default: //Cancel
		mes "["+@name$+"]";
		mes "Vuelve cuando estes listo.";
		close;
	}
}

//Subfunction SF_TryRegister (int bride)
function SF_TryRegister {
	set @bride, getarg(0);
	set @type$, "novio";
	if (@bride)
		set @type$, "novia";

	mes "["+@name$+"]";
	mes "Antes de registrarte como "+@type$+", dejame verificar si cumples con todos los requisitos...";
	next;
	if (Upper == 2) {
		mes "["+@name$+"]";
		mes "Oh querido, eres muy joven para estar pensando en matrimonio!";
		emotion ET_SURPRISE;
		close;
	}
	if (Sex == SEX_MALE)
		set @item, 7170;
	else
		set @item, 2338;

	if (getequipid(EQI_ARMOR) != @item) {
		mes "["+@name$+"]";
		mes "Deberias estar usando un "+getitemname(@item)+" si quieres casarte.";
		close;
	}
	if (Sex == SEX_FEMALE && $@wed_veil && getequipid(EQI_HEAD_TOP) != 2206) {
		mes "["+@name$+"]";
		mes "Donde esta tu "+getitemname(2206)+"? Es un complemento necesario para tu vestido.";
		close;
	}
	if ($@wed_ring && countitem($@wed_ring) < 1) {
		mes "["+@name$+"]";
		mes "Donde esta el anillo? Necesitas un "+getitemname($@wed_ring)+" para el intercambio de anillos, querido.";
		close;
	}
	if (@bride)
		set @cost, $@wed_bride_reg;
	else
		set @cost, $@wed_groom_reg;

	if (Zeny < @cost) {
		mes "["+@name$+"]";
		mes "Lo siento, pero no tienes suficiente para pagar la tarifa de registro.";
		mes "Vuelve cuando hayas reunido "+@cost+"z.";
		close;
	}
	set Zeny,Zeny-@cost;
	sc_start SC_Wedding,3600000,1; //Start Wedding Effect (SC_WEDDING)
	if (@bride) {
		set $wed_bride_progress,1;
		set $wed_bride$,strcharinfo(0);
		set $wed_bride_sex, Sex;
		if ($@wedding_effect == 1) //Store account id for effect.
			set $wedding_effect_id, getcharid(3);
	} else {
		set $wed_groom_progress,1;
		set $wed_groom$,strcharinfo(0);
		set $wed_groom_sex, Sex;
		if ($@wedding_effect == 2) //Store account id for effect.
			set $wedding_effect_id, getcharid(3);
	}
}

		mes "Sabes quien es el conyugue?";
		close;
	}

	mes "["+@name$+"]";
	mes $@divorcer$+" ha pedido divorciarse de ti. Si aceptas, y tienes la tarifa de "+$@wed_divorce_fee+"z a mano, procedere a divorciarlos.";

		mes "Espero que puedan resolver las cosas.";

		close;

		mes "Bueno, no puedo tramitar tu divorcio porque no tienes suficiente para la tarifa. Haz que tu pareja te preste algo?";
