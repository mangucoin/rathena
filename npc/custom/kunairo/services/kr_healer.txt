// ============================================================
// = KunaiRO - Healer Premium
// = Proposito: Curacion completa de HP/SP + remocion de debuffs.
// = Dependencias: core/kr_config.txt (KR_Header)
// = Seguro para actualizaciones: No es parte de rAthena oficial.
// ============================================================

-	script	KR_Healer_Core	-1,{
	callfunc "KR_Header", "Healer";
	mes "Bienvenido. Selecciona un servicio.";
	next;

	switch (select("Curar HP y SP:Remover estados negativos:Curacion completa:Cancelar")) {
		case 1:
			percentheal 100, 100;
			specialeffect2 EF_HEAL;
			mes .KR_Name$;
			mes "Tu HP y SP han sido restaurados.";
			close;

		case 2:
			// KunaiRO: Remove common negative statuses
			sc_end SC_POISON;
			sc_end SC_SILENCE;
			sc_end SC_BLIND;
			sc_end SC_CONFUSION;
			sc_end SC_CURSE;
			sc_end SC_HALLUCINATION;
			sc_end SC_BLEEDING;
			sc_end SC_STONE;
			sc_end SC_FREEZE;
			sc_end SC_STUN;
			sc_end SC_SLEEP;
			sc_end SC_STRIPWEAPON;
			sc_end SC_STRIPSHIELD;
			sc_end SC_STRIPARMOR;
			sc_end SC_STRIPHELM;
			specialeffect2 EF_HEAL2;
			mes .KR_Name$;
			mes "Todos los estados negativos han sido removidos.";
			close;

		case 3:
			percentheal 100, 100;
			sc_end SC_POISON;
			sc_end SC_SILENCE;
			sc_end SC_BLIND;
			sc_end SC_CONFUSION;
			sc_end SC_CURSE;
			sc_end SC_HALLUCINATION;
			sc_end SC_BLEEDING;
			sc_end SC_STONE;
			sc_end SC_FREEZE;
			sc_end SC_STUN;
			sc_end SC_SLEEP;
			sc_end SC_STRIPWEAPON;
			sc_end SC_STRIPSHIELD;
			sc_end SC_STRIPARMOR;
			sc_end SC_STRIPHELM;
			specialeffect2 EF_HEAL;
			specialeffect2 EF_HEAL2;
			mes .KR_Name$;
			mes "Curacion completa realizada.";
			mes "HP, SP y estados restaurados.";
			close;

		case 4:
			close;
	}

OnInit:
	.KR_Name$ = "[ ^4444FFHealer^000000 ]";
	end;
}
