// ============================================================
// = KunaiRO - Job Master Premium
// = Proposito: Cambio de clase completo con soporte para
// =   1ra, 2da, Transcendencia, 3ra, 4ta clase,
// =   Expandidas y Baby.
// = Dependencias: core/kr_config.txt (KR_Header)
// = Seguro para actualizaciones: No es parte de rAthena oficial.
// ============================================================

-	script	KR_JobMaster_Core	-1,{

// ---------------------------------------------------------
// Funciones internas
// ---------------------------------------------------------

// Verifica si el jugador cumple el nivel requerido
function	KR_RequireLevel	{
	if (BaseLevel < getarg(0) || JobLevel < getarg(1)) {
		.@blvl = getarg(0) - BaseLevel;
		.@jlvl = getarg(1) - JobLevel;
		mes "^FF0000Requisito de nivel:^000000";
		mes ((getarg(0) > 1) ?
			"^bb0000" + getarg(0) + "^000000 (Base) / " : "") +
			"^00bb00" + getarg(1) + "^000000 (Job)";
		mes "Necesitas " +
			((.@blvl > 0) ? "^bb0000" + .@blvl + "^000000 niveles base mas " +
				((.@jlvl > 0) ? "y " : "") : "") +
			((.@jlvl > 0) ? "^00bb00" + .@jlvl + "^000000 niveles job mas " : "") +
			"para continuar.";
		close;
	}
	return;
}

// Verifica si la clase actual es baby
function	KR_IsBaby	{
	return ((getarg(0, eaclass()) & EAJL_BABY) > 0);
}

// Verifica si puede cambiar a 4ta clase
function	KR_CanFourth	{
	if (!.KR_FourthClass)
		return false;
	if ((eaclass() & (EAJL_THIRD | EAJL_UPPER)) != (EAJL_THIRD | EAJL_UPPER))
		return false;
	if (eaclass() & EAJL_FOURTH)
		return false;
	if (roclass(eaclass() | EAJL_FOURTH) < 0)
		return false;
	return true;
}

// Verifica si puede cambiar a 3ra clase
function	KR_CanThird	{
	if (!.KR_ThirdClass)
		return false;
	if (!(eaclass() & EAJL_2))
		return false;
	if (eaclass() & EAJL_THIRD)
		return false;
	if (roclass(eaclass() | EAJL_THIRD) < 0)
		return false;
	if ((eaclass() & EAJ_UPPERMASK) == EAJ_SUPER_NOVICE)
		return false;
	if (KR_IsBaby() && (!.KR_BabyClass || !.KR_BabyThird))
		return false;
	return true;
}

// Verifica si puede renacer (transcender)
function	KR_CanRebirth	{
	if (!.KR_RebirthClass)
		return false;
	if (!(eaclass() & EAJL_2))
		return false;
	if (eaclass() & (EAJL_UPPER | EAJL_THIRD))
		return false;
	if (roclass(eaclass() | EAJL_UPPER) < 0)
		return false;
	if (KR_IsBaby() && !.KR_BabyClass)
		return false;
	return true;
}

// Verifica si es primera clase
function	KR_IsFirstCls	{
	return (getarg(0) <= EAJ_TAEKWON);
}

// Verifica montura/falcon/carrito
function	KR_CheckRiding	{
	if (checkfalcon() || checkcart() || checkriding() || ismounting()) {
		mes "Por favor retira tu " +
			((checkfalcon()) ? "falcon" : "") +
			((checkcart()) ? "carrito" : "") +
			((checkriding()) ? "Peco" : "") +
			((ismounting()) ? "montura" : "") +
			" antes de continuar.";
		close;
	}
	return;
}

// Verifica puntos de habilidad sin usar
function	KR_CheckSkillPts	{
	if (.KR_SkillPointCheck && SkillPoint) {
		mes "Por favor usa todos tus ^FF0000puntos de habilidad^000000 antes de continuar.";
		close;
	}
	return;
}

// Agrega opciones de clase al array
function	KR_JobOptions	{
	.@argcount = getargcount();
	.@arr_size = getarraysize(getarg(0));
	for (.@i = 1; .@i < .@argcount; .@i++) {
		setarray getelementofarray(getarg(0), .@arr_size++), getarg(.@i);
	}
}

// ---------------------------------------------------------
// Logica principal del NPC
// ---------------------------------------------------------
	callfunc "KR_Header", "Job Master";
	KR_CheckRiding();
	KR_CheckSkillPts();

	.@eac = eaclass();
	.@fourth_possible = KR_CanFourth();
	.@third_possible = KR_CanThird();
	.@rebirth_possible = KR_CanRebirth();
	.@first_eac = .@eac & EAJ_BASEMASK;
	.@second_eac = .@eac & EAJ_UPPERMASK;

	// Opcion de Renacer
	if (.@rebirth_possible) {
		KR_RequireLevel(.KR_ReqRebirth[0], .KR_ReqRebirth[1]);
		KR_JobOptions(.@job_opt, Job_Novice_High);
	}
	// Cambio a 3ra Clase
	if (.@third_possible) {
		KR_RequireLevel(.KR_ReqThird[0], .KR_ReqThird[1]);
		KR_JobOptions(.@job_opt, roclass(.@eac | EAJL_THIRD));
	}
	// Cambio a 4ta Clase
	if (.@fourth_possible) {
		KR_RequireLevel(.KR_ReqFourth[0], .KR_ReqFourth[1]);
		KR_JobOptions(.@job_opt, roclass(.@eac | EAJL_FOURTH));
	}

	// Super Novice Expandido
	if (.KR_SecondExpanded &&
		(.@eac & EAJ_UPPERMASK) == EAJ_SUPER_NOVICE &&
		!(eaclass() & EAJL_THIRD)) {
		if (!KR_IsBaby(.@eac) || (.KR_BabyClass && .KR_BabyExpanded)) {
			KR_RequireLevel(.KR_ReqExpSNovi[0], .KR_ReqExpSNovi[1]);
			KR_JobOptions(.@job_opt, roclass(.@eac | EAJL_THIRD));
		}
	}

	// 4ta clase expandida: Night Watch, Shinkiro/Shiranui, Hyper Novice, Spirit Handler
	if (.KR_FourthExpanded && .@eac == EAJ_REBELLION) {
		KR_RequireLevel(.KR_ReqFourth[0], .KR_ReqFourth[1]);
		KR_JobOptions(.@job_opt, roclass(EAJ_NIGHT_WATCH));
	}
	if (.KR_FourthExpanded && .@eac == EAJ_KAGEROUOBORO) {
		KR_RequireLevel(.KR_ReqFourth[0], .KR_ReqFourth[1]);
		KR_JobOptions(.@job_opt, roclass(EAJ_SHINKIRO_SHIRANUI));
	}
	if (.KR_FourthExpanded && .@eac == EAJ_SUPER_NOVICE_E) {
		KR_RequireLevel(.KR_ReqFourth[0], .KR_ReqFourth[1]);
		KR_JobOptions(.@job_opt, roclass(EAJ_HYPER_NOVICE));
	}
	if (.KR_FourthExpanded && .@eac == EAJ_SUMMONER) {
		KR_RequireLevel(.KR_ReqSHandler[0], .KR_ReqSHandler[1]);
		KR_JobOptions(.@job_opt, roclass(EAJ_SPIRIT_HANDLER));
	}

	// Ninja/Gunslinger expandidos
	if (.KR_SecondExpanded &&
		((.@eac & (~EAJL_BABY)) == EAJ_NINJA ||
		 (.@eac & (~EAJL_BABY)) == EAJ_GUNSLINGER)) {
		if (!KR_IsBaby(.@eac) || (.KR_BabyClass && .KR_BabyExpanded)) {
			KR_RequireLevel(.KR_ReqExpNJGS[0], .KR_ReqExpNJGS[1]);
			KR_JobOptions(.@job_opt, roclass(.@eac | EAJL_2_1));
		}
	}

	// Novice / Novice High / Baby — primera seleccion de clase
	if (.@first_eac == EAJ_NOVICE && .@second_eac != EAJ_SUPER_NOVICE) {
		KR_RequireLevel(.KR_ReqFirst[0], .KR_ReqFirst[1]);
		switch (Class) {
			case Job_Novice:
				KR_JobOptions(.@job_opt, Job_Swordman, Job_Mage, Job_Archer,
					Job_Acolyte, Job_Merchant, Job_Thief,
					Job_Super_Novice, Job_Taekwon, Job_Gunslinger, Job_Ninja);
				if (.KR_BabyNovice)
					KR_JobOptions(.@job_opt, Job_Baby);
				break;
			case Job_Novice_High:
				if (.KR_LastJob && lastJob)
					KR_JobOptions(.@job_opt,
						roclass((eaclass(lastJob) & EAJ_BASEMASK) | EAJL_UPPER));
				else
					KR_JobOptions(.@job_opt, Job_Swordman_High, Job_Mage_High,
						Job_Archer_High, Job_Acolyte_High, Job_Merchant_High,
						Job_Thief_High);
				break;
			case Job_Baby:
				if (!.KR_BabyClass)
					break;
				KR_JobOptions(.@job_opt, Job_Baby_Swordman, Job_Baby_Mage,
					Job_Baby_Archer, Job_Baby_Acolyte, Job_Baby_Merchant,
					Job_Baby_Thief);
				if (.KR_BabyExpanded)
					KR_JobOptions(.@job_opt, Job_Super_Baby, Job_Baby_Taekwon,
						Job_Baby_Gunslinger, Job_Baby_Ninja);
				if (.KR_BabySummoner)
					KR_JobOptions(.@job_opt, Job_Baby_Summoner);
				break;
			default:
				mes "Ha ocurrido un error.";
				close;
		}
	} else if (KR_IsFirstCls(.@eac) ||
			   KR_IsFirstCls(.@eac & (~EAJL_UPPER)) ||
			   (.KR_BabyClass && KR_IsFirstCls(.@eac & (~EAJL_BABY)))) {
		// Primera clase → Segunda clase (2-1 y 2-2)
		.@class1 = roclass(.@eac | EAJL_2_1);
		.@class2 = roclass(.@eac | EAJL_2_2);
		if (.KR_LastJob && lastJob && (.@eac & EAJL_UPPER)) {
			KR_RequireLevel(.KR_ReqSecond[0], .KR_ReqSecond[1]);
			KR_JobOptions(.@job_opt, lastJob + Job_Novice_High);
		} else {
			if (.@class1 > 0) {
				KR_RequireLevel(.KR_ReqSecond[0], .KR_ReqSecond[1]);
				KR_JobOptions(.@job_opt, .@class1);
			}
			if (.@class2 > 0) {
				KR_RequireLevel(.KR_ReqSecond[0], .KR_ReqSecond[1]);
				KR_JobOptions(.@job_opt, .@class2);
			}
		}
	}

	// ---------------------------------------------------------
	// Mostrar menu de clases disponibles
	// ---------------------------------------------------------
	function KR_JobMenu;
	KR_JobMenu(.@job_opt);
	close;

// Menu de seleccion de clase
function	KR_JobMenu	{
	function KR_ConfirmChange;
	while (true) {
		.@opt_cnt = getarraysize(getarg(0));
		if (.@opt_cnt <= 0) {
			mes "No hay mas clases disponibles para ti.";
			close;
		}

		.@selected = 0;
		if (.@opt_cnt > 1) {
			mes "Selecciona tu nueva clase.";
			.@menu$ = "";
			for (.@i = 0; .@i < .@opt_cnt; .@i++) {
				if (getelementofarray(getarg(0), .@i) == Job_Novice_High)
					.@jobname$ = "^0055FFRenacer (Transcender)^000000";
				else
					.@jobname$ = jobname(getelementofarray(getarg(0), .@i));
				.@menu$ = .@menu$ + " ~ " + .@jobname$ + ":";
			}
			.@menu$ = .@menu$ + " ~ ^777777Cancelar^000000";
			.@selected = select(.@menu$) - 1;
			if (.@selected < 0 || .@selected >= .@opt_cnt)
				close;
			next;
			mes .KR_Name$;
		}
		.@class = getelementofarray(getarg(0), .@selected);

		// Requisito especial para Super Novice
		if ((.@class == Job_Super_Novice || .@class == Job_Super_Baby) &&
				BaseLevel < .KR_SNovice) {
			mes "Se requiere nivel base ^FF0000" + .KR_SNovice +
				"^000000 para convertirte en " + jobname(.@class) + ".";
			return;
		}

		KR_ConfirmChange(.@class, .@opt_cnt > 1);
		next;
		mes .KR_Name$;
	}
	return;
}

// Ejecuta el cambio de clase
function	KR_DoJobChange	{
	.@previous_class = Class;
	.@to_cls = getarg(0);
	next;
	mes .KR_Name$;
	callfunc "KR_Separator";
	mes "Felicidades!";
	mes "Ahora eres ^4444FF" + jobname(.@to_cls) + "^000000.";
	mes "Buena suerte en tu aventura.";
	if (.@to_cls == Job_Novice_High && .KR_LastJob)
		lastJob = Class;
	jobchange .@to_cls;
	if (.@to_cls == Job_Novice_High)
		resetlvl(1);
	else if (.@to_cls == Job_Baby) {
		resetstatus;
		resetskill;
		set SkillPoint, 0;
	}
	specialeffect2 EF_ANGEL2;
	specialeffect2 EF_ELECTRIC;
	if (.@previous_class != Class) {
		if (.KR_Platinum)
			callfunc "F_GetPlatinumSkills";
		if (.KR_GetJobEquip)
			KR_GiveEquip;
	}
	close;
}

// Confirmacion de cambio
function	KR_ConfirmChange	{
	.@class = getarg(0, -1);
	.@back = getarg(1, false);
	if (.@class < 0 || eaclass(.@class) == -1) {
		mes "Error: clase desconocida.";
		close;
	}
	if (.@class == Job_Novice_High)
		mes "Deseas ^0055FFRenacer^000000 (Transcender)?";
	else
		mes "Deseas cambiar a ^4444FF" + jobname(.@class) + "^000000?";
	.@opt$ = (.@class == Job_Novice_High) ?
		" ~ ^0055FFRenacer^000000" :
		" ~ Cambiar a ^4444FF" + jobname(.@class) + "^000000";
	if (select(.@opt$ + ": ~ ^777777" +
			((.@back) ? "Volver" : "Cancelar") + "^000000") == 1) {
		KR_DoJobChange(.@class);
	}
	if (!.@back)
		close;
	return;
}

// ---------------------------------------------------------
// Configuracion inicial del NPC
// ---------------------------------------------------------
OnInit:
	.KR_Name$ = "[ ^4444FFJob Master^000000 ]";

	// Clases habilitadas
	.KR_FourthClass = true;
	.KR_FourthExpanded = true;
	.KR_ThirdClass = true;
	.KR_RebirthClass = true;
	.KR_SecondExpanded = true;
	.KR_BabyNovice = true;
	.KR_BabyClass = true;
	.KR_BabyThird = true;
	.KR_BabyExpanded = true;
	.KR_BabySummoner = true;

	// Comportamiento
	.KR_LastJob = true;          // Forzar cambio de clase lineal
	.KR_SkillPointCheck = true;  // Forzar uso de skill points
	.KR_Platinum = true;         // Dar platinum skills automaticamente
	.KR_GetJobEquip = false;     // Dar equipo al cambiar de clase

	// Requisitos de nivel [BaseLevel, JobLevel]
	setarray .KR_ReqFirst[0],    1,  10;   // 1ra clase
	setarray .KR_ReqSecond[0],   1,  40;   // 2da clase
	setarray .KR_ReqRebirth[0],  99, 50;   // Transcendencia
	setarray .KR_ReqThird[0],    99, 50;   // 3ra clase
	setarray .KR_ReqFourth[0],   200, 70;  // 4ta clase
	setarray .KR_ReqExpNJGS[0],  99, 70;   // Ninja/Gunslinger expandido
	setarray .KR_ReqExpSNovi[0], 99, 99;   // Super Novice expandido
	setarray .KR_ReqSHandler[0], 200, 60;  // Spirit Handler
	.KR_SNovice = 45;                      // Nivel base minimo para Super Novice

	// Ajustes por version del cliente
	if (PACKETVER < 20161207) {
		if (.KR_BabyExpanded)
			debugmes "KunaiRO JobMaster: BabyExpanded deshabilitado por PACKETVER.";
		if (.KR_BabySummoner)
			debugmes "KunaiRO JobMaster: BabySummoner deshabilitado por PACKETVER.";
		.KR_BabyExpanded = false;
		.KR_BabySummoner = false;
	}
	end;
}
