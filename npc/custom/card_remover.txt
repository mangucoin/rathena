//===== Description: =========================================
//= Removes cards from equipped items.
//===== Additional Comments: =================================
//= 1.0 First version. [TyrNemesis^]
//= 1.2 Optimized and fixed getequipname menu. [Kisuka]
//= 1.2a Added 'disable_items' command. [Euphy]
//============================================================

prt_in,28,73,4	script	Wise Old Woman#eAcustom	78,{

	set .zenycost,200000;    // base cost of the card remover services (in Zeny)
	set .percardcost,25000;  // cost per card of the card remover services (in Zeny)
	set .faildestroy,1;      // should the card remover have a chance of failure that destroys items? (1=yes, 0=no)

	disable_items;
	mes "[Wise Old Woman]";
	mes "Buen dia, joven. Tengo el poder de remover cartas que hayas compuesto en tu equipo. Te interesa?";
	next;
	switch(select("Si, me interesa.:Cuanto cobras?:No gracias.")) {
	case 1:
		mes "[Wise Old Woman]";
		mes "Muy bien. Que objeto quieres que examine?";
		next;

		setarray .@indices[1], EQI_HEAD_TOP, EQI_ARMOR, EQI_HAND_L, EQI_HAND_R, EQI_GARMENT, EQI_SHOES, EQI_ACC_L, EQI_ACC_R, EQI_HEAD_MID, EQI_HEAD_LOW;
		for( set .@i,1; .@i <= 10; set .@i,.@i+1 ) {
			if( getequipisequiped(.@indices[.@i]) )
				set .@menu$, .@menu$ + F_getpositionname(.@indices[.@i]) + "-[" + getequipname(.@indices[.@i]) + "]";
			set .@menu$, .@menu$ + ":";
		}
		set .@part, .@indices[ select(.@menu$) ];
		if(!getequipisequiped(.@part)) {
			mes "[Wise Old Woman]";
			mes "Joven... No llevas nada equipado ahi de lo que pueda remover cartas.";
			close;
		}
		if(getequipcardcnt(.@part) == 0) {
			mes "[Wise Old Woman]";
			mes "Joven... No hay cartas compuestas en este objeto. No puedo hacer nada con el, me temo.";
			close;
		}
		set .@cardcount,getequipcardcnt(.@part);
		
		if (!checkweight(1202,(.@cardcount+1))) { 
			mes "^3355FFUn momento!";
			mes "No puedo ofrecerte mis";
			mes "servicios porque llevas";
			mes "demasiadas cosas encima.";
			mes "Guarda tus objetos extra en";
			mes "el Almacen Kafra y vuelve~";
			close;
		}
		mes "[Wise Old Woman]";
		mes "Este objeto tiene " + .@cardcount + " cartas compuestas. Para realizar mi magia, necesitare " + (.zenycost+(.@cardcount * .percardcost)) + " zeny, un ^0000FFStar Crumb^000000, y una ^0000FFYellow Gemstone^000000.";
		next;
		if(select("Muy bien. Hazlo.:No importa.") == 2) {
			mes "[Wise Old Woman]";
			mes "Muy bien. Vuelve cuando busques mis servicios.";
			close;
		}
		if((zeny < (.zenycost+(.@cardcount * .percardcost))) || (countitem(1000) < 1) || (countitem(715) < 1)) {
			mes "[Wise Old Woman]";
			mes "No tienes todos los materiales que necesito para hacer mi magia. Vuelve cuando los tengas.";
			close;
		}
		mes "[Wise Old Woman]";
		mes "Antes de empezar, debo advertirte--puedo fallar. Si lo hago, podria destruir las cartas, el objeto, o ambos. No doy reembolsos. Dicho esto, que es mas importante para ti: Las cartas, o el objeto?";
		next;
		switch(select("Cambie de opinion.:El objeto.:Las cartas.")) {
		case 1:
			mes "[Wise Old Woman]";
			mes "Muy bien. Vuelve cuando busques mis servicios.";
			close;
		case 2:
			set .@failtype,1;
			break;
		case 3:
			set .@failtype,2;
		}
		mes "[Wise Old Woman]";
		mes "Muy bien. Comenzare.";
		set Zeny, Zeny - (.zenycost+(.@cardcount * .percardcost));
		delitem 1000,1; //Star_Crumb
		delitem 715,1; //Yellow_Gemstone
		
		// Replace the constants in the next 3 lines with failure chance values defined in refine_db.txt
		// First value = Total failure chance (item and cards destroyed)
		// Second value = Partial failure chance (one or the other is destroyed, player decides which one is safe)
		// Third value = Harmless failure chance (all that's lost is your investment)

		set .@failchance,rand(100);
		if (.faildestroy==1) {
			if(.@failchance < 2) {
				next;
				failedremovecards .@part,0;
				mes "[Wise Old Woman]";
				mes "El proceso fue un fracaso total. Me temo que el objeto y las cartas fueron destruidos.";
				close;
			}

			if(.@failchance < 8) {
				if (.@failtype == 1) {
					next;
					failedremovecards .@part,1;
					mes "[Wise Old Woman]";
					mes "Aunque logre remover las cartas del objeto, fueron destruidas en el proceso. El objeto, sin embargo, esta bien.";
					close;
				}

				if (.@failtype == 2) {
					next;
					failedremovecards .@part,2;
					mes "[Wise Old Woman]";
					mes "Que desafortunado. Logre remover las cartas, pero el objeto fue destruido en el proceso.";
					close;
				}
			}
		}

		if(.@failchance < 10) {
			next;
			failedremovecards .@part,3;
			mes "[Wise Old Woman]";
			mes "No pude remover las cartas. Por suerte, tanto el objeto como las cartas estan bien.";
			close;
		}
		next;
		successremovecards .@part;
		mes "[Wise Old Woman]";
		mes "El proceso fue un exito. Aqui estan tus cartas y tu objeto. Hasta luego.";
		close;
	case 2:
		mes "[Wise Old Woman]";
		mes "Cobro una tarifa fija de "+callfunc("F_InsertComma",.zenycost)+" zeny, mas "+callfunc("F_InsertComma",.percardcost)+" zeny por cada carta que remueva del objeto. Ademas, necesito un Star Crumb y una Yellow Gemstone para hacer mi magia.";
		close;
	case 3:
		mes "[Wise Old Woman]";
		mes "Muy bien. Vuelve cuando busques mis servicios.";
		close;
	}
}
