gef_fild10,242,202,0	script	Dimensional Gorge Piece	406,{

	set .@party_id,getcharid(1);
	set .@md_name$,"Orc's Memory";

	if (!instance_check_party(.@party_id,2,30,80)) {
		mes "Solo usuarios entre Niveles ^ff000030 ~ 80^000000 pueden entrar a este Calabozo.";
		close;
	}

	set .@orctime,checkquest(12059,PLAYTIME);
	if (.@orctime == -1) {
		if (is_party_leader() == true) {
			mes "Estado del grupo confirmado. Te gustaria reservar la entrada a "+.@md_name$+"?";
			next;
			switch(select("Reservar "+.@md_name$+":Entrar al Calabozo:Cancelar")) {
			case 1:
				if (instance_create(.@md_name$) < 0) {
					mes "Nombre del grupo: "+ getpartyname(.@party_id);
					mes "Lider del grupo: "+strcharinfo(0);
					mes "^0000ff"+.@md_name$+" ^000000 - Reserva fallida.";
					close;
				}
				mes "^0000ff"+.@md_name$+"^000000- Intentando reservar entrada";
				mes "Despues de hacer la reserva, debes seleccionar 'Entrar al Calabozo' del menu si deseas entrar a "+.@md_name$+".";
				close;
			case 2:
				callsub L_Enter,0;
			case 3:
				close;
			}
		}
		if(select(.@md_name$ + " Entrar al Calabozo Memorial:Cancelar") == 2)
			end;
		callsub L_Enter,1;
	} else if (.@orctime == 0 || .@orctime == 1) {
		mes "Puedes entrar al Calabozo si ya ha sido generado.";
		next;
		if(select("Entrar al Calabozo "+.@md_name$+":Cancelar") == 2)
			close;
		callsub L_Enter,0;
	} else if (.@orctime == 2) {
		mes "^0000ffTodos los registros y efectos residuales relacionados con el Calabozo Orc's Memory han sido eliminados. Ahora puedes regenerar o reingresar al calabozo.^000000";
		erasequest 12059;
		close;
	}
	mes "Para generar un calabozo debes ser el Lider del Grupo y tener al menos 2 miembros en el grupo.";
	close;
L_Enter:
	switch(instance_enter("Orc's Memory")) {
	case IE_OTHER:
		mes "Ha ocurrido un error desconocido.";
		close;
	case IE_NOINSTANCE:
		mes "El Calabozo Memorial Orc's Memory no existe.";
		mes "El Calabozo Memorial ha sido destruido por el Lider del Grupo, o por el limite de tiempo. Por favor intenta de nuevo despues de 2 horas.";
		close;
	case IE_NOMEMBER:
		mes "Solo un miembro del grupo puede entrar al Calabozo Memorial.";
		close;
	case IE_OK:
		mapannounce "gef_fild10", getpartyname( getcharid(1) ) +" miembro del grupo "+strcharinfo(0)+" ha entrado a Orc's Memory.",bc_map,"0x00ff99";
		if (checkquest(12059) == -1) setquest 12059;
		//warp "1@orcs",179,15;
		if (getarg(0) == 0) close;
		else end;
	}

}

		}
