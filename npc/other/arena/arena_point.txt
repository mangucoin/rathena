prt_are_in,103,11,3	script	Arena Point Manager	732,{
	mes "[Arena Point Manager]";
	mes "Espero que la estes pasando";
	mes "bien en la Arena. Si has";
	mes "ganado Puntos de Turbo Track en";
	mes "el Turbo Track de Al De Baran,";
	mes "puedo convertirlos en Puntos de Arena.";
	next;
	switch(select("Consultar Puntos:Convertir Puntos:^660000Info de Conversion^000000")) {
	case 1:
		mes "[Arena Point Manager]";
		mes "" + strcharinfo(0) + ",";
		mes "actualmente tienes";
		mes "" + arena_point + " Puntos de Arena";
		mes "y " + tt_point + " Puntos de Turbo Track.";
		close;
	case 2:
		mes "[Arena Point Manager]";
		mes "" + strcharinfo(0) + ",";
		mes "actualmente tienes";
		mes "" + arena_point + " Puntos de Arena";
		mes "y " + tt_point + " Puntos de Turbo Track.";
		next;
		mes "[Arena Point Manager]";
		mes "Por favor elige entre las";
		mes "conversiones de Puntos de Track";
		mes "a Puntos de Arena. Ten en cuenta que";
		mes "cuando conviertas mas de 10";
		mes "Puntos de Track a la vez, solo puedes convertir en ^4D4DFFmultiplos de 10^000000.";
		next;
		switch(select("2 PT -> 1 PA:4 PT -> 2 PA:6 PT -> 3 PA:8 PT -> 4 PA:10 PT y mas:Cancelar")) {
		case 1:
			callsub S_ExchangePoints,28999,2,1;
		case 2:
			callsub S_ExchangePoints,28998,4,2;
		case 3:
			callsub S_ExchangePoints,28997,6,2;
		case 4:
			callsub S_ExchangePoints,28996,8,4;
		case 5:
			mes "[Arena Point Manager]";
			mes "Por favor ingresa el numero";
			mes "de veces que deseas convertir";
			mes "10 Puntos de Turbo Track en";
			mes "Puntos de Arena. El valor maximo";
			mes "que puedes ingresar es 20.";
			mes "Para cancelar, ingresa ''^3355FF0^000000.''";
			next;
			input .@input;
			if (.@input <= 0) {
				mes "[Arena Point Manager]";
				mes "Has";
				mes "cancelado";
				mes "este servicio.";
				close;
			}
			else if (.@input > 20) {
				mes "[Arena Point Manager]";
				mes "Tu solicitud excede";
				mes "el limite maximo. Por favor";
				mes "ingresa un valor no mayor a 20.";
				close;
			}
			else {
				set .@want_point1,10 * .@input;
				set .@want_point,5 * .@input;
				set .@my_arena_all,arena_point + .@want_point;
				set .@my_turbo_all,tt_point - .@want_point1;
				if (.@my_arena_all > 28999) {
					mes "[Arena Point Manager]";
					mes "Excederas la cantidad";
					mes "maxima de Puntos de";
					mes "Arena si procedemos";
					mes "con esta conversion de tus";
					mes "Puntos de Turbo Track. No puedes";
					mes "tener mas de 29,000 Puntos de Arena.";
					next;
					mes "[Arena Point Manager]";
					mes "Por favor gasta algunos de";
					mes "tus Puntos de Arena antes de";
					mes "usar este servicio nuevamente.";
					mes "Gracias por tu preferencia.";
					close;
				}
				if (.@my_turbo_all < 0) {
					mes "[Arena Point Manager]";
					mes "Lo siento, pero";
					mes "no tienes suficientes";
					mes "Puntos de Turbo Track para";
					mes "realizar esta conversion";
					mes "de Puntos de Arena.";
					close;
				}
				else {
					mes "[Arena Point Manager]";
					mes "Has convertido";
					mes "10 Puntos de Turbo Track";
					mes "en 5 Puntos de Arena " + .@input + " veces.";
					mes "Un total de " + .@want_point1 + " Puntos de Turbo Track fueron convertidos en";
					mes "" + .@want_point +" Puntos de Arena.";
					set tt_point, .@my_turbo_all;
					set arena_point, .@my_arena_all;
					next;
					mes "[Arena Point Manager]";
					mes "" + strcharinfo(0) + ",";
					mes "ahora tienes";
					mes "^4682B4" + arena_point + "^000000 Puntos de Arena";
					mes "y ^00688B" + tt_point + "^000000 Puntos de Turbo Track.";
					mes "Gracias por tu preferencia.";
					close;
				}
			}
		case 6:
			mes "[Arena Point Manager]";
			mes "Has";
			mes "cancelado";
			mes "este servicio.";
			close;
		}
	case 3:
		// Again. no dialog for this option.
		close;
	}
	close;

S_ExchangePoints:
	if (arena_point > getarg(0) ) {
		mes "[Arena Point Manager]";
		mes "Excederas la cantidad";
		mes "maxima de Puntos de";
		mes "Arena si procedemos";
		mes "con esta conversion de tus";
		mes "Puntos de Turbo Track. No puedes";
		mes "tener mas de 29,000 Puntos de Arena.";
		next;
		mes "[Arena Point Manager]";
		mes "Por favor gasta algunos de";
		mes "tus Puntos de Arena antes de";
		mes "usar este servicio nuevamente.";
		mes "Gracias por tu preferencia.";
		close;
	}
	.@track_point = getarg(1);
	if (tt_point >= .@track_point) {
		.@arena_point = getarg(2);
		mes "[Arena Point Manager]";
		mes "" + strcharinfo(0) + ",";
		mes "has convertido "+ .@track_point +" Puntos de";
		mes "Track en "+ .@arena_point +" Puntos de Arena.";
		tt_point = tt_point - .@track_point;
		arena_point = arena_point + .@arena_point;
		next;
		mes "[Arena Point Manager]";
		mes "" + strcharinfo(0) + ",";
		mes "ahora tienes";
		mes "^4682B4" + arena_point + "^000000 Puntos de Arena";
		mes "y ^00688B" + tt_point + "^000000 Puntos de Turbo Track.";
		mes "Gracias por tu preferencia.";
		close;
	}
	else {
		mes "[Arena Point Manager]";
		mes "Lo siento, pero no tienes";
		mes "suficientes Puntos de Turbo Track.";
		mes "Necesitas al menos 2 Puntos de Turbo Track para este servicio de conversion.";
		close;
	}
}
