sec_in02,72,180,3	script	Arena Manager#arena	802,{
	set .@i, callfunc("F_GM_NPC",1357,0);
	if (.@i == -1) {
		mes "[Arena Manager]";
		mes "El comando ha sido cancelado.";
		close;
	} else if (.@i == 0) {
		mes "La contrasena es incorrecta.";
		close;
	} else {
		mes "[Arena Manager]";
		mes "Selecciona una opcion.";
		next;
		switch(select("Reiniciar arena:Reorganizar el Tiempo del Ranking")) {
		case 1:
			switch(select("Nv 50:Nv 60:Nv 70:Nv 80:Modo Grupal")) {
			case 1:
				set .@arena,50;
				break;
			case 2:
				set .@arena,60;
				break;
			case 3:
				set .@arena,70;
				break;
			case 4:
				set .@arena,80;
				break;
			case 5:
				mes "[Arena Manager]";
				mes "== Precaucion ==";
				mes "Has elegido reiniciar la etapa de arena grupal.";
				mes "Deseas continuar?";
				next;
				switch(select("Si:No")) {
				case 1:
					donpcevent "Ponox::OnStart";
					mes "[Arena Manager]";
					mes "La etapa de arena ha sido reactivada exitosamente.";
					close;

				case 2:
					mes "[Arena Manager]";
					mes "El comando ha sido cancelado.";
					close;
				}
			}
			mes "[Arena Manager]";
			mes "== Precaucion ==";
			mes "Has elegido reiniciar la etapa de arena Nv "+.@arena+".";
			mes "Deseas continuar?";
			next;
			switch(select("Si:No")) {
			case 1:
				donpcevent "Lv"+.@arena+" Waiting Room::OnStart";
				mes "[Arena Manager]";
				mes "La etapa de arena ha sido reactivada exitosamente.";
				close;
			case 2:
				mes "[Arena Manager]";
				mes "El comando ha sido cancelado.";
				close;
			}
		case 2:
			mes "[Arena Manager]";
			mes "== Precaucion ==";
			mes "^CE0000Has elegido reorganizar el tiempo del ranking. Asegurate de que esta no es una buena decision a menos que algo serio haya ocurrido!^000000";
			next;
			mes "[Arena Manager]";
			mes "Estas seguro de que quieres reorganizar el tiempo del ranking?";
			next;
			switch(select("No:Si")) {
			case 1:
				mes "[Arena Manager]";
				mes "Uff~ :)";
				close;
			case 2:
				switch(select("Nv 50:Nv 60:Nv 70:Nv 80:Modo Grupal")) {
				case 1:
					set .@min$,"$top50min";
					set .@sec$,"$top50sec";
					set .@mode$,"Lv 50";
					break;
				case 2:
					set .@min$,"$top60min";
					set .@sec$,"$top60sec";
					set .@mode$,"Lv 60";
					break;
				case 3:
					set .@min$,"$top70min";
					set .@sec$,"$top70sec";
					set .@mode$,"Lv 70";
					break;
				case 4:
					set .@min$,"$top80min";
					set .@sec$,"$top80sec";
					set .@mode$,"Lv 80";
					break;
				case 5:
					set .@min$,"$top_ptmin";
					set .@sec$,"$top_ptsec";
					set .@mode$,"party";
					break;
				}
				mes "[Arena Manager]";
				// Doesn't work with Athena's implementation.
				//mes "Please enter 0 to cancel.";
				//mes "If not, please enter a value for minutes first and then seconds.";
				mes "Por favor ingresa primero un valor para los minutos y luego los segundos.";
				next;
				input .@arenamin;
				setd .@min$,.@arenamin;
				input .@arenasec;
				setd .@sec$,.@arenasec;
				mes "[Arena Manager]";
				mes "Current "+.@mode$+" ranker's play time has been rearranged to ^FF0000"+getd(.@min$)+"^000000 minutes and ^FF0000"+getd(.@sec$)+"^000000 segundos.";
				close;
			}
		}
	}
}

function	script	Func_Are_Rew	{
	.@item_id = getarg(0);
	.@amount = getarg(1);
	.@arena_point = getarg(2);

	mes "[Givu]";
	mes "Te gustaria canjear tus puntos de arena por "+ getitemname(.@item_id) +"?";
	mes "Puedes canjear ^3131FF"+ .@arena_point +" puntos de arena por "+ .@amount +" "+ getitemname(.@item_id) +"^000000.";
	mes "Si deseas cancelar, ingresa 0. Si no, ingresa cuantas ^3131FFveces^000000 de puntos de arena deseas gastar.";
	next;
	input .@reward;
	if (.@reward <= 0) {
		mes "[Givu]";
		mes "Has cancelado tu solicitud.";
		close;
	} else if (.@reward > 1500) {
		mes "[Givu]";
		mes "Has excedido la capacidad maxima.";
		close;
	}
	if (arena_point < (.@reward * .@arena_point)) {
		mes "[Givu]";
		mes "No tienes suficientes puntos de arena.";
		mes "Por favor verifica la cantidad total de puntos de arena que tienes.";
		close;
	}
	arena_point = arena_point - (.@reward * .@arena_point);
	getitem .@item_id,.@reward * .@amount;
	mes "[Givu]";
	mes "Gracias, vuelve pronto.";
	close;
}
