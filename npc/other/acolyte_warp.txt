prontera,98,121,4	script	Acolyte#Prtclear	90,{
	mes "[Keiki]";
	mes "Hola, aventurero.";
	mes "He estado estudiando magia de todo Rune-Midgarts para mejorar lo que considero una de las mejores habilidades disponibles para la clase acolito.";
	next;

	if (is_function("VIP_iRO_Acolyte"))
		callfunc "VIP_iRO_Acolyte","[Keiki]","Juno";

	mes "[Keiki]";
	mes "Soy la unica Acolita que ha alcanzado el nivel 10 de la habilidad Warp Portal!";
	emotion ET_THINK, playerattached();
	next;
	mes "[Keiki]";
	mes "Asi es! Y...";
	mes "Te prometo que no olvido las ubicaciones que ya he memorizado.";
	mes "Un dia subire mis habilidades para teletransportarme a donde quiera~";
	next;
	mes "[Keiki]";
	mes "Estoy dispuesta a teletransportarte a las muchas ubicaciones que he memorizado por una pequena tarifa.";
	mes "Te gustaria usar este servicio?";
	next;
	switch(select("Si:No")) {
	case 1:
		mes "[Keiki]";
		mes "A donde te gustaria ir?";
		mes "Te deseo buena suerte en tu viaje.";
		next;

		setarray .@towns$[0], "Izlude", "Geffen", "Payon", "Morocc", "Alberta", "Al De Baran", "Comodo", "Umbala", "Juno";
		setarray .@cost[0],        600,     1200,    1200,     1200,      1800,          2200,     2200,     2200,   1800;

		set .@size, getarraysize(.@towns$);
		for(set .@i,0; .@i<.@size; set .@i,.@i+1)
			set .@menu$, .@menu$+.@towns$[.@i]+" -> "+.@cost[.@i]+"z:";
		set .@i, select(.@menu$+"Cancelar")-1;
		if (.@i == .@size)
			close;
		if (Zeny < .@cost[.@i]) {
			mes "[Keiki]";
			mes "Lo siento, pero no tienes";
			mes "suficiente zeny para el Servicio";
			mes "de Teletransporte. La tarifa para";
			mes "ir a "+.@towns$[.@i]+" es de "+.@cost[.@i]+" zeny.";
			close;
		}
		specialeffect2 EF_READYPORTAL;
		specialeffect2 EF_TELEPORTATION;
		specialeffect2 EF_PORTAL;
		next;
		set Zeny, Zeny - .@cost[.@i];
		switch(.@i) {
			case 0: if (checkre(0)) warp "izlude",128,98; else warp "izlude",91,105; break;
			case 1: warp "geffen",120,39; break;
			case 2: warp "payon",161,58; break;
			case 3: warp "morocc",156,46; break;
			case 4: warp "alberta",117,56; break;
			case 5: warp "aldebaran",168,112; break;
			case 6: warp "comodo",209,143; break;
			case 7: warp "umbala",100,154; break;
			case 8: warp "yuno",158,125; break;
		}
		close;
	case 2:
		close;
	}
}
