payon,173,141,4	script	Ripped Cabus#GymPass	899,{
	mes "[Ripped]";
	mes "Oye. La gente ya no es";
	mes "tan activa fisicamente como";
	mes "antes. Aunque pelees para";
	mes "vivir, tu cuerpo podria";
	mes "estar debil y flacido en algunas";
	mes "areas. Me entiendes?";
	next;
	mes "[Ripped]";
	mes "Oye, entrena conmigo, y te";
	mes "garantizo que podras";
	mes "cargar y llevar mas de tus";
	mes "cosas. Solo dame tu";
	mes "^FF0000Gym Pass^000000 cada vez,";
	mes "y estamos listos.";
	next;
	mes "[Ripped]";
	mes "Pero no te emociones demasiado:";
	mes "sin importar cuanto entrenamiento";
	mes "te haga pasar, puedes";
	mes "excederte. Alguna vez oiste de";
	mes "alguien que se puso demasiado musculoso?";
	mes "Es porque estan muertos. Ves?";
	next;
	mes "[Ripped]";
	mes "Yo diria que seria seguro";
	mes "para ti entrenar seriamente";
	mes "conmigo y aumentar tu";
	mes "capacidad de carga de items ^FF000010 veces^000000.";
	mes "Entonces... Estas listo para sudar?";
	next;
	switch(select("Si:No:Ehm, mi entrenamiento se desgasto.")) {
	case 1:
		if (gympassmemory < 10) {
			set .@add_carry,gympassmemory + 1;
			set .@remain_carry,10 - .@add_carry;
			if (countitem(7776) > 0) {
				mes "[Ripped]";
				mes "Oh, genial, veo que";
				mes "trajiste tu Gym Pass.";
				mes "Bien, solo haz lo que yo hago,";
				mes "e intenta sentir el ardor.";
				mes "Listo? Vamos a hacerlo.";
				next;
				specialeffect2 EF_EARTHSPIKE;
				next;
				specialeffect2 EF_DEVIL;
				next;
				specialeffect2 EF_COIN;
				next;
				specialeffect2 EF_SIGHTRASHER;
				next;
				mes "[Ripped]";
				mes "Listo, ahora deberias poder";
				mes "cargar mas cosas contigo.";
				mes "Veamos, podemos aumentar";
				mes "tu capacidad de carga de items";
				mes "^FF00000" + .@remain_carry + "^000000 veces mas si seguimos";
				mes "entrenando juntos asi.";
				delitem 7776,1; //Max_Weight_Up_Scroll
				set gympassmemory,.@add_carry;
				skill "ALL_INCCARRY",.@add_carry,SKILL_PERM_GRANT;
				close;
			}
			else {
				mes "[Ripped]";
				mes "Oye, que te dije?";
				mes "Tienes que traerme tu";
				mes "^FF0000Gym Pass^000000 si quieres";
				mes "entrenar y fortalecer tus";
				mes "musculos de carga de items.";
				close;
			}
		}
		else {
			mes "[Ripped]";
			mes "Oye, no creo que podamos";
			mes "fortalecer tus musculos de carga";
			mes "de items mas de eso.";
			mes "Es demasiado peligroso para tu";
			mes "cuerpo si lo intentaramos! Vamos,";
			mes "te dije sobre los limites.";
			close;
		}
	case 2:
		mes "[Ripped]";
		mes "Ay, que lastima.";
		mes "Bueno, vuelve si";
		mes "cambias de opinion. Dile";
		mes "a tus amigos sobre mi:";
		mes "si estan flacidos, los ayudare";
		mes "a ponerse en forma.";
		close;
	case 3:
		if (gympassmemory > 0) {
			mes "[Ripped]";
			mes "Que paso?";
			mes "Dejaste que tus musculos de carga";
			mes "de items se atrofiaran? Por suerte";
			mes "para ti, existe algo llamado";
			mes "memoria muscular. No tomara";
			mes "tanto tiempo reconstruirlos...";
			next;
			specialeffect2 EF_EARTHSPIKE;
			next;
			specialeffect2 EF_DEVIL;
			next;
			specialeffect2 EF_COIN;
			next;
			specialeffect2 EF_SIGHTRASHER;
			next;
			mes "[Ripped]";
			mes "Que tal eso?";
			mes "Tus musculos de carga";
			mes "de items volvieron a crecer,";
			mes "asi de facil! Intenta no";
			mes "debilitarte de nuevo, si?";
			skill "ALL_INCCARRY",gympassmemory,SKILL_PERM_GRANT;
			close;
		}
		else {
			mes "[Ripped]";
			mes "Ehh...";
			mes "No entrenamos";
			mes "juntos antes.";
			mes "Estoy seguro de eso.";
			close;
		}
	}
}
