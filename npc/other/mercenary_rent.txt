-	script	Mercenary Manager#main	-1,{

		if( compare( .@npc$,.@name$[.@type] ) ) break;

			mes "[Mercenary Manager]";
			mes "Quieres contratar un";
			mes "Mercenario " + .@name$[.@type] + "?";
			mes "Que Grado te";
			mes "interesa contratar?";
			next;
			for (.@i = 1; .@i <= 9; .@i++)
				.@menu$ = .@menu$ + callfunc("F_GetNumSuffix",.@i) + " Grade " + .@name$[.@type] + " Mercenary:";
			.@Grade = select(.@menu$);
			.@BaseLevel = 5 + (.@Grade * 10);
			.@BaseLevel = (.@val > 90)? 90 : .@BaseLevel;
			.@ZenyCost = 7 * .@Grade;
			setarray .@FaithCost[7], 50,100,300;
			.@FaithCost = .@FaithCost[.@Grade];
			mes "[Mercenary Manager]";
			mes "Asi que quieres contratar un Mercenario";
			mes "de " + callfunc("F_GetNumSuffix",.@Grade);
			mes "Grado " + .@name$[.@type] + "?";
			mes "Necesitas haber alcanzado";
			mes "Nivel Base " + .@BaseLevel + " o superior, y";
			mes "debes pagar la tarifa de " + .@ZenyCost + ",000 zeny.";
			next;
			if(select("Si:No") == 2) {
				mes "[Mercenary Manager]";
				mes "Oh, en serio? Bueno, puede que";
				mes "ahora no sea un buen momento";
				mes "para que consideres contratar";
				mes "un Mercenario, pero sientete";
				mes "libre de volver si tus";
				mes "necesidades cambian. Gracias~";
			}
			else if(.@FaithCost && .@faith_merc < .@FaithCost) {
				mes "[Mercenary Manager]";
				mes "Oh... Tu calificacion de Lealtad";
				mes "con el Gremio de Mercenarios " + .@name$[.@type];
				mes "no es lo suficientemente alta para";
				mes "contratar este Mercenario. Por favor";
				mes "vuelve despues de ganar";
				mes "" + .@FaithCost + " o mas de Lealtad con nosotros.";
			}
			else if (BaseLevel < .@BaseLevel) {
				mes "[Mercenary Manager]";
				mes "Lo siento, pero tu Nivel";
				mes "Base no es lo suficientemente alto";
				mes "para contratar este Mercenario.";
				mes "Por favor vuelve a verme";
				mes "una vez que alcances el Nivel Base " + .@BaseLevel + ".";
			}
			else if (Zeny < .@ZenyCost * 1000) {
				mes "[Mercenary Manager]";
				mes "Lo siento, pero";
				mes "no tienes suficiente zeny";
				mes "para contratar este Mercenario.";
				mes "La tarifa de contratacion es de " + .@ZenyCost + ",000 zeny.";
			}
			else {
				mes "[Mercenary Manager]";
				mes "Genial! Nuestros Mercenarios";
				mes "son sinceros y dedicados";
				mes "a proteger a sus clientes.";
				mes "Los Mercenarios invocados";
				mes "ofreceran su apoyo por";
				mes "30 minutos. Cuidate.";
				Zeny = Zeny - (.@ZenyCost * 1000);
				getitem .@item[.@type] - 10 + .@Grade, 1;
			}
			close;
		case 2:
			mes "[Mercenary Manager]";
			mes "Los Mercenarios son soldados";
			mes "que pelearan a tu lado";
			mes "en el campo de batalla, pero hay";
			mes "algunos terminos y condiciones";
			mes "que debes cumplir para contratarlos.";
			next;
			mes "[Mercenary Manager]";
			mes "Debes cumplir un requisito de nivel";
			mes "y pagar una tarifa en zeny";
			mes "para contratar un Mercenario.";
			mes "Los Mercenarios de grado superior";
			mes "tambien requieren que construyas";
			mes "una calificacion de Lealtad con nosotros.";
			next;
			mes "[Mercenary Manager]";
			mes "Los contratos de Mercenarios no pueden";
			mes "ser transferidos a otras personas,";
			mes "y solo permitimos una diferencia de 5 Niveles Base entre el Mercenario";
			mes "y el cliente, asi que no puedes contratar uno mucho mas fuerte que tu.";
			next;
			mes "[Mercenary Manager]";
			mes "Bueno, puedes averiguar los";
			mes "detalles cuando realmente formes";
			mes "un contrato con uno de nuestros";
			mes "Mercenarios, y recibas";
			mes "el Pergamino de Invocacion que";
			mes "llamara a un Mercenario a tu lado.";
			next;
			mes "[Mercenary Manager]";
			mes "No puedes darle este pergamino";
			mes "a nadie mas, y el";
			mes "Mercenario solo permanecera";
			mes "contigo por 30 minutos despues";
			mes "de que lo invoques. No olvides";
			mes "el limite de tiempo, si?";
			close;
		case 3:
			mes "[Mercenary Manager]";
			mes "No? No necesitabas";
			mes "ayuda? Bueno, sientete libre";
			mes "de preguntarme si tienes alguna";
			mes "pregunta sobre los Mercenarios.";
			close;
		case 4:
			mes "[Mercenary Manager]";
			mes "Los Mercenarios de Grado 10 son";
			mes "lo mejor que tenemos para ofrecer,";
			mes "y usamos criterios diferentes";
			mes "para que nuestros clientes los contraten.";
			mes "No hay tarifa en zeny, pero debes";
			mes "tener 500 de Lealtad.";
			next;
			mes "[Mercenary Manager]";
			mes "Una vez que hagas un contrato";
			mes "con un Mercenario de Grado 10,";
			mes "tu calificacion de Lealtad sera";
			mes "reducida en 400. En otras";
			mes "palabras, pagas 400 de Lealtad";
			mes "para contratar un Mercenario de Grado 10.";
			next;
			mes "[Mercenary Manager]";
			mes "Tambien debes ser de";
			mes "Nivel Base 90 o superior para";
			mes "contratar un Mercenario de Grado 10.";
			mes "Aun te interesa";
			mes "formar este contrato?";
			next;
			if (select("Si:No") == 2) {
				mes "[Mercenary Manager]";
				mes "Entiendo... Se necesita";
				mes "trabajo duro y sacrificio para";
				mes "incluso llegar al punto donde";
				mes "puedes contratar un Mercenario de";
				mes "Grado 10. Has considerado";
				mes "contratar un Mercenario de menor grado?";
			}
			else if (.@faith_merc < 500) {
				mes "[Mercenary Manager]";
				mes "Lo siento, pero tu";
				mes "calificacion de Lealtad es";
				mes "demasiado baja para contratar un";
				mes "Mercenario de Grado 10. Debes tener";
				mes "500 o mas de Lealtad para";
				mes "formar un contrato con uno.";
			}
			else if (BaseLevel < 90) {
				mes "[Mercenary Manager]";
				mes "Lo siento, pero debes";
				mes "ser de Nivel Base 90 o";
				mes "superior para formar un contrato";

			}

				mes "Felicidades! Parece";
				mes "que estas planeando enfrentar";

-	script	Mercenary Merchant#dummy::MercMerchant	-1,{
	mes "[Mercenary Goods Merchant]";
	mes "Hola, vendo productos";
	mes "que los Mercenarios pueden";
	mes "usar. Hay algo en";
	mes "particular que";
	mes "estes buscando?";
	next;
	setarray .@item, 12184,	12185,	12241,	12242,	12243;
	setarray .@cost, 2500,	5000,	800,	1500,	3000;
	for (.@size = getarraysize(.@item); .@i < .@size; .@i++)
		.@menu$ = .@menu$ + getitemname(.@item[.@i]) + ":";
	.@m = select(.@menu$)-1;
	.@str$ = .@cost[.@m] + "";
	.@len = getstrlen(.@str$);
	.@str_cost$ = (.@len <= 3)? .@str$ : insertchar(.@str$,",",.@len-3);
	mes "[Mercenary Goods Merchant]";
	mes getitemname(.@item[.@m]);
	mes "cada uno cuesta " + .@str_cost$ + " zeny.";
	mes "Cuantos te gustaria?";
	next;
	input .@input;
	if (!.@input) {
		mes "[Mercenary Goods Merchant]";
		mes "Cambiaste de opinion?";
		mes "Bien, sientete libre de volver";
		mes "a mi cuando quieras";
		mes "comprar Pociones de Mercenario.";
		close;
	}
	else if (.@input < 0 || .@input > 10000) {
		mes "[Mercenary Goods Merchant]";
		mes "Lo siento, pero solo";
		mes "puedes comprar hasta";
		mes "10,000 de estas pociones";
		mes "a la vez. Por favor ingresa";
		mes "un numero del 1 al 10,000.";
		close;
	}
	else if (set(.@total_po, .@input * .@cost[.@m]) > Zeny) {
		mes "[Mercenary Goods Merchant]";
		mes "Lo siento, pero no tienes";
		mes "suficiente zeny para tantas";
		mes "pociones. Bueno, estare";
		mes "aqui cuando estes listo";
		mes "para comprar algo";
		mes "para tus Mercenarios.";
		close;
	}
	else if (!checkweight(.@item[.@m], .@input)) {
		mes "[Mercenary Goods Merchant]";
		mes "Si te diera tantas";
		mes "pociones, no podrias";
		mes "cargarlas contigo.";
		mes "Por favor vuelve despues";
		mes "de liberar algo de espacio";
		mes "en tu Inventario.";
		close;
	}
	mes "[Mercenary Goods Merchant]";
	if (.@input == 1)
		mes "Aqui tienes tu " + getitemname(.@item[.@m]) + ".";
	else {
		mes "Aqui tienes, esto es exactamente";
		mes callfunc("F_InsertPlural",.@input,getitemname(.@item[.@m])) + ".";
	}
	mes "Gracias, y por favor vuelve";
	mes "cuando necesites mas";
	mes "pociones para tus Mercenarios.";
	Zeny = Zeny - .@total_po;
	getitem .@item[.@m], .@input;
	close;
}

-	script	Mercenary_Switch	-1,{

	setarray .@name$, "Spear","Sword","Bow";
	.@npc$ = strnpcinfo(2);
	for (.@size = getarraysize(.@name$); .@type < .@size; .@type++)
		if (.@npc$ == .@name$[.@type]) break;

	mes "[Checker]";
	mes "Por favor ingresa una contrasena.";
	next;
	.@i = callfunc("F_GM_NPC",1854,0,0,10000);
	mes "[Checker]";
	if (.@i == -2)
		mes "Error.";
	else if (.@i == 0)
		mes "Numero incorrecto";
	else if (.@i == 1) {
		mes "Por favor selecciona.";
		next;
		switch( select( "Desactivar NPC de Mercenarios","Activar NPC de Mercenarios" ) ) {
		case 1:
			mes "Los NPCs han sido desactivados.";
			disablenpc "Mercenary Manager#" + .@name$[.@type];
			disablenpc "Mercenary Merchant#" + .@name$[.@type];
			break;
		case 2:
			mes "Los NPCs han sido activados.";
			enablenpc "Mercenary Manager#" + .@name$[.@type];
			enablenpc "Mercenary Merchant#" + .@name$[.@type];
			break;
		}
	}
	close;
}
