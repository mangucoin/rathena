pvp_y_room,77,85,4	script	Guild Battle Guide	105,{

	setarray .@map1X[1],7,9,16,8,20;
	setarray .@map2X[1],91,90,83,91,79;
	setarray .@mapY[1],49,49,50,49,50;

	mes "[Guild Battle Guide]";
	mes "Como estas?";
	mes "Soy el Guia de Batalla de Gremios";
	mes "para los nuevos mapas de PvP.";
	mes "Dime a cual mapa";
	mes "quieres que te lleve.";
	next;
	set .@i, select("Mapa 1:Mapa 2:Mapa 3:Mapa 4:Mapa 5");
	mes "[Guild Battle Guide]";
	mes "Has elegido el Mapa "+.@i+".";
	mes "Ahora, en que equipo";
	mes "estas? Puedes elegir";
	mes "el Equipo 1 o el Equipo 2.";
	next;
	set .@j, select("Equipo 1:Equipo 2");
	mes "[Guild Battle Guide]";
	mes "Bien, te llevare";
	mes "al Mapa "+.@i+" como miembro";
	mes "del Equipo "+.@j+". Estas listo";
	mes "para ser transportado ahora?";
	next;
	if(select("Si:No") == 2) {
		mes "[Guild Battle Guide]";
		mes "Bien, espero verte";
		mes "de nuevo en los";
		mes "campos de PvP!";
		close;
	}
	mes "[Guild Battle Guide]";
	mes "Genial! Que la pases bien~";
	close2;
	warp "guild_vs"+.@i,getd(".@map"+.@j+"X["+.@i+"]"),.@mapY[.@i];
	end;
}
