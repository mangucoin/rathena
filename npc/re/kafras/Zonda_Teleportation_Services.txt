-	script	ZondaTeleportationServices	HIDDEN_NPC,{
	cutin "zonda_do01", 2;
	mes "[Zonda Employee]";
	mes "Bienvenido.";
	mes "Nuestro equipo siempre trabaja para superar tus expectativas en cuanto a calidad de servicio.";
	mes "Por favor elige tu destino.";
	next;
	switch( select( "Norte de Bifrost", "Eclage", "Splendide", "Campamento de la Expedicion de Midgard", "Manuk", "Alrededores de la Montana Kamidal", "Cancelar.") ) {
	case 1:
		callsub( S_Zone, "Norte de Bifrost", 17000,850, 7213 ); // Teleport value not confirmed
		warp "bif_fild02",160,230; // Unverified Coordinates
		end;
	case 2:
		callsub( S_Zone, "Eclage", 17000,850, 11311 );
		warp "eclage",111,39;
		end;
	case 3:
		callsub( S_Zone, "Splendide", 17000,850, 10078 );
		warp "splendide",171,115;
		end;
	case 4:
		callsub( S_Zone, "Campamento de la Expedicion de Midgard", 19000,950, 10078 );
		warp "mid_camp",210,288;
		end;
	case 5:
		callsub( S_Zone, "Manuk", 22000,1100, 10078 );
		warp "manuk",263,177;
		end;
	case 6:
		callsub( S_Zone, "Alrededores de la Montana Kamidal", 24000,1200, 7184 );
		warp "dic_fild01",188,187;
		end;
	case 7:
		mes "[Zonda Employee]";
		mes "Siempre nos esforzamos por garantizar los mejores servicios para nuestros clientes.";
		mes "Gracias por usar nuestros servicios.";
		close3;
	}

S_Zone:
	.@zeny_req = getarg(1);
	.@item_req = getarg(2);
	.@quest_req = getarg(3);

	if (isbegin_quest(.@quest_req) == 0) {
		mes "[Zonda Employee]";
		//mes getarg(0);
		mes "Debes completar la mision de acceso para teletransportarte a " + getarg(0) + " antes de usar nuestros servicios de Teletransporte.";
		close3;
	}
	mes "[Zonda Employee]";
	mes getarg(0) + " esta disponible por ^ff0000" + .@zeny_req + " Zenys^000000 o ^ff0000" + .@item_req + " Malang Sp Can^000000.";
	next;
	switch( select( .@zeny_req + " Zeny", .@item_req + " Malang Sp Can", "Cancelar" ) ) {
	case 1:
		if (Zeny < .@zeny_req)
			break;
		Zeny -= .@zeny_req;
		return;
	case 2:
		if (countitem(12636) < .@item_req)
			break;
		delitem 12636,.@item_req; // Malang_Sp_Can
		return;
	case 3:
		mes "[Zonda Employee]";
		mes "Puedo informarte sobre otras areas de servicio?";
		close3;
	}
	mes "[Zonda Employee]";
	mes "Senor! No tienes suficiente para lo que estamos pidiendo.";
	mes "Por favor revisa e intenta de nuevo.";
	close3;
}
