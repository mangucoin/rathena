izlude,127,175,3	script	Hypnotist#novice	124,{
	mes "[Hipnotista]";
	mes "Saludos, aventurero.";
	mes "Soy miembro de la Academia";
	mes "de Hipnotistas enviado aqui a Izlude";
	mes "para ofrecer servicios de Reinicio de Habilidades";
	mes "a ciertos personajes de Primera Clase a un muy buen precio: gratis!";
	next;
	mes "[Hipnotista]";
	mes "Aunque ofrezco reinicios";
	mes "ilimitados de stats/habilidades por ahora, tengo";
	mes "dos condiciones que deben";
	mes "cumplirse. Primero, debes ser";
	mes "menor que ^FF0000Nivel Base 50^000000.";
	next;
	mes "[Hipnotista]";
	mes "Segundo, debes ser un";
	mes "^FF0000Espadachin, Acolito, Mago,";
	mes "Ladron, Arquero, Mercader,";
	mes "Taekwon Boy, o Taekwon Girl,";
	mes "Pistolero y Ninja^000000";
	mes "para calificar.";
	mes "Ahora, tienes alguna pregunta?";
	next;
	switch(select("^FF0000Reinicio de Stats/Habilidades?:Reiniciar stats:Reiniciar habilidades^000000:Cancelar")) {
	case 1:
		mes "[Hipnotista]";
		mes "Los Reinicios de Stats/Habilidades permiten";
		mes "a los aventureros redistribuir sus";
		mes "Puntos de Habilidad si no estan";
		mes "contentos con sus habilidades actuales.";
		next;
		mes "[Hipnotista]";
		mes "Antes de proceder con";
		mes "un Reinicio de Stats/Habilidades, debes";
		mes "reducir todo el peso";
		mes "de todos los objetos que llevas";
		mes "a 0. Puedes poner";
		mes "objetos extra en el Almacen Kafra.";
		next;
		mes "[Hipnotista]";
		mes "Ah, tambien es importante";
		mes "remover tu ^FF0000Carreta^000000";
		mes "si tienes una equipada.";
		mes "De lo contrario, la hipnosis no";
		mes "funcionara, o saldra mal...";
		close;
	case 2:
		set .@str$,"Stat";
	case 3:
		if (.@str$ == "")
			set .@str$,"Skill";
		mes "[Hipnotista]";
		mes "Estas seguro de que";
		mes "quieres proceder con";
		mes "mi servicio de ^FF0000Reinicio de "+.@str$+"^000000?";
		next;
		if(select("Si:Cancelar") == 1)
			break;
	case 4:
		mes "[Hipnotista]";
		mes "Gracias, y buena";
		mes "suerte en tus aventuras.";
		mes "Por favor viaja con seguridad~";
		close;
	}
	if ((Class >= Job_Swordman && Class <= Job_Thief) || Class == Job_Taekwon || Class == Job_Gunslinger || Class == Job_Ninja) {
		if (checkcart()) {
			mes "[Hipnotista]";
			mes "Oh! Por favor remueve tu";
			mes "Carreta antes de proceder";
			mes "con el servicio de Reinicio de "+.@str$+".";
			mes "Gracias por cooperar~";
			close;
		}
		if (Weight != 0) {
			mes "[Hipnotista]";
			mes "Si estas aqui por mi servicio de";
			mes "Reinicio de "+.@str$+", por favor";
			mes "recuerda que no puedes";
			mes "reiniciar tus "+.@str$+"s hasta que el";
			mes "^FF0000peso de los objetos que llevas en";
			mes "tu Inventario se reduzca a 0^000000.";
			next;
			mes "[Hipnotista]";
			mes "Por que no colocas tus";
			mes "cosas en el Almacen Kafra";
			mes "por ahora? De esa manera, puedes";
			mes "guardar todos tus bienes de forma segura.";
			close;
		}
		if (BaseLevel >= 50 && (Class == Job_Taekwon || Class == Job_Gunslinger || Class == Job_Ninja)) {
			mes "[Hipnotista]";
			mes "Lo siento, pero los personajes";
			mes "con Niveles Base mayores";
			mes "a 50 no son elegibles para el";
			mes "servicio de Reinicio de "+.@str$+" que ofrezco.";
			close;
		}
		mes "[Hipnotista]";
		mes "Gracias por usar";
		mes "mis servicios de Redistribucion";
		mes "de "+.@str$+". Oh, y la mejor";
		mes "de las suertes en tus";
		mes "viajes, aventurero.";
		if (.@str$ == "Stat")
			resetstatus;
		else
			resetskill;
		close;
	} else {
		mes "[Hipnotista]";
		mes "Lo siento, pero tu";
		mes "Clase de Trabajo no califica";
		mes "para el servicio de Reinicio de "+.@str$;
		mes "que ofrezco. Solo puedo";
		mes "ofrecer Reinicios de "+.@str$+" a los";
		mes "siguientes Trabajos...";
		next;
		mes "[Hipnotista]";
		mes "^FF0000Espadachin, Acolito,";
		mes "Mago, Ladron, Arquero,";
		mes "Mercader, Taekwon";
		mes "Boy, Taekwon Girl,";
		mes "Pistolero y Ninja^000000.";
		close;
	}
}
