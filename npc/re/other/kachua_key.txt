//===== rAthena Script =======================================
//= Dynamic NPC: Kachua's Secret Box
//===== Description: =========================================
//- [Official conversion]
//= Kachua's Secret Box is a Gachapon NPC.
//= It lets the player exchange a Kachua's Secret Key for a random item.
//= It also gives a Kachua's Mileage Coupon for each pull.
//===== Changelogs: ==========================================
//= 1.0 First version. [secretdataz]
//============================================================

sec_in02,126,178,3	script	Kachua's Secret Box#bm	4_TREASURE_BOX,{
	.@key$ = "K_Secret_Key";
	.@keyname$ = getitemname( .@key$ );
	.@box$ = "Main_Lucky_Box";
	.@boxname$ = getitemname( .@box$ );
	mes "Una Caja Secreta donde nadie sabe que hay dentro.";
	mes "^4d4dffPuedes abrir esta caja consumiendo " + mesitemlink( .@key$ ) + ".^000000";
	next;
	switch(select("^4d4dffAbrir la caja 1 vez (1 " + .@keyname$ + ")^000000","^4d4dffAbrir la caja 10 veces (10 " + .@keyname$ + "s)^000000")) {
		case 1:
			mes "^FF0000[Aviso]^000000";
			mes "^FF0000Intercambiar un objeto aleatorio con una " + .@keyname$ + ".^000000";
			mes "^FF0000El objeto intercambiado no puede ser devuelto, ni se puede intercambiar por " + .@keyname$ + ".^000000";
			next;
			if(select("Continuar.","Detener conversacion.") == 2) {
				mes "Has decidido no abrir la Caja.";
				close;
			} else if (countitem(.@key$) < 1) {
				mes "No tienes suficientes " + mesitemlink( .@key$ ) + ".";
				close;
			} else {
				if (checkweight(1201,1) == 0 || ((MaxWeight - Weight) * 100 / MaxWeight) < 10) {
					mes "^4d4dffPor favor asegurate de tener suficiente espacio en tu inventario.^000000";
					close;
				}
				delitem(.@key$, 1);
				consumeitem(.@box$);
				mes .@boxname$ + " fue abierta!";
				mes "Tuviste suerte?";
				specialeffect2 EF_VALLENTINE;
				close;
			}
		case 2:
			mes "^FF0000[Aviso]^000000";
			mes "^FF0000Intercambiar 10 objetos aleatorios con 10 " + .@keyname$ + "s.^000000";
			mes "^FF0000Los objetos intercambiados no pueden ser devueltos, ni se pueden intercambiar por " + .@keyname$ + ".^000000";
			next;
			if(select("Continuar.","Detener conversacion.") == 2) {
				mes "Has decidido no abrir la Caja.";
				close;
			} else if (countitem(.@key$) < 10) {
				mes "No tienes suficientes " + mesitemlink( .@key$ ) + ".";
				close;
			} else {
				for (.@i = 1; .@i <= 10; ++.@i) {
					progressbar "4d4dff",2;
					if (checkweight(1201,1) == 0 || ((MaxWeight - Weight) * 100 / MaxWeight) < 10) {
						mes "^4d4dffPor favor asegurate de tener suficiente espacio en tu inventario.^000000";
						close;
					}
					if (countitem(.@key$) < 1) { // Custom check, just in case
						close;
					}
					delitem(.@key$, 1);
					consumeitem(.@box$);
					dispbottom .@boxname$ + " fue abierta " + .@i + " veces. Otra esta siendo abierta.",0xFFFFFF;
					specialeffect2 EF_VALLENTINE;
				}
				mes .@boxname$ + " fue abierta 10 veces!";
				mes "Tuviste suerte?";
				close;
			}
	}
}
