function	script	VIP_Third_Class	{
	if (!vip_status(VIP_STATUS_ACTIVE)) {
		mes "[Voz de la nada]";
		mes "El acceso al contenido Premium solo esta disponible para aquellos que posean cierta cantidad de Piedras de Reinicio.";
		next;
		if (countitem(6320) < 1) {
			mes "[Voz de la nada]";
			mes "Para cambiar a tercera clase, debes traer 1 ^0000CCPiedra de Reinicio^000000 o tener una cuenta Premium.";
			//mes "Por favor visita nuestra iShop para comprar la ^0000CCPiedra de Reinicio^000000.";
			close;
		} else {
			mes "[Voz de la nada]";
			mes "Te gustaria participar en esta mision por 1 ^0000CCPiedra de Reinicio^000000?";
			next;
			switch(select("Si:No")) {
			case 1:
				mes "[Voz de la nada]";
				mes "Buena suerte, aventurero.";
				delitem 6320,1; //Premium_Reset_Stone
				next;
				break;
			case 2:
				mes "[Voz de la nada]";
				mes "Muy bien.";
				close;
			}
		}
	}
	return;
}
