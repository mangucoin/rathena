function	script	F_AvenBoard	{
	.@mo_board = select( "Buscar monstruos por piso",
					"Sistema de Busqueda de Informacion de Monstruos",
					( countitem(909) == 30000 ? "Solicitar" : "" ),
					"Cancelar" );
	if (.@mo_board == 1)
		return 1;
	else if (.@mo_board == 2) {
		while(1) {
			mes "[Sistema de Busqueda de Informacion de Monstruos]";
			mes "Este es el Sistema de Busqueda de Informacion de Monstruos. Se puede buscar informacion de los monstruos que aparecen en esta mazmorra. Ingresa el Nombre del Monstruo, incluyendo espacios, al buscar.";
			next;
			input .@inputstr$;
			setarray .@element$[0],"Neutral","Agua","Tierra","Fuego","Viento","Veneno","Sagrado","Sombra","Fantasma","No-Muerto";
			setarray .@size$[0],"Mediano","Pequeno","Grande";
			setarray .@race$[0],"Amorfo","No-Muerto","Animal","Planta","Insecto","Pez","Demonio","Semihumano","Angel","Dragon";
			.@size = getargcount();
			.@i = 0;
			while( .@i < .@size && getarg(.@i+1,"") != .@inputstr$ ) .@i += 3;
			mes "[Sistema de Busqueda de Informacion de Monstruos]";
			mes "Buscando";
			next;
			mes "[Sistema de Busqueda de Informacion de Monstruos]";
			mes "La busqueda se ha completado.";
			mes "Mostrando los resultados de la busqueda.";
			next;
			mes "[Sistema de Busqueda de Informacion de Monstruos]";
			if (.@i >= .@size) {
				mes "No hay informacion disponible para ese monstruo.";
				mes "Por favor verifica el nombre del monstruo.";
			}
			else {
				.@mob_id = getarg(.@i);
				mes "- Nombre : "+ getarg(.@i+1);
				mes "- Elemento : "+ .@element$[ getmonsterinfo(.@mob_id,MOB_ELEMENT) ];
				mes "- Tamano : "+ .@size$[ getmonsterinfo(.@mob_id,MOB_SIZE) ];
				mes "- Raza : "+ .@race$[ getmonsterinfo(.@mob_id,MOB_RACE) ];
				mes "--------------------------------^FD8D02";
				mes getarg(.@i+2);
			}
			next;
			mes "[Sistema de Busqueda de Informacion de Monstruos]";
			mes "Te gustaria buscar informacion de otro monstruo?";
			next;
			if (select( "Search","Cancelar" ) == 1)
				continue;
			mes "[Sistema de Busqueda de Informacion de Monstruos]";
			mes "El sistema de busqueda de monstruos se apagara ahora.";
			next;
			return 2;
		}
	}
	else if (.@mo_board == 3) {
		mes "La informacion solicitada aparecera - No hay problema si aparece";
		next;
		return 3;
	}
	else if (.@mo_board == 4)
		close;
	else {
		mes "Error de Operacion.";
		close;
	}
}

function	script	F_InfoBoard	{
	next;
	mes "[Grupo de Aventureros]";
	mes "Para mas informacion sobre los monstruos";
	mes "Sistema de Busqueda de Informacion de Monstruos";
	mes "de Informacion de Monstruos.";
	next;
	return;
}
