dali,112,95,4	script	Logistics Manager	4_M_MERCAT2,{
	mes "[Gerente de Logistica]";
	mes "Bienvenido. Estoy a cargo de la logistica para cuidar tus objetos en este lugar extrano. Necesitas acceder a tu almacen?";
	next;
	switch(select("Abrir Almacen 200 Zeny.:No, gracias.")) {
	case 1:
		if (Zeny > 199) {
			if(!callfunc("F_CanOpenStorage")){
				mes "[Gerente de Logistica]";
				mes "Aun no estas calificado para usar el Almacen.";
			}
			else {
				Zeny -= 200;
				openstorage;
			}
		}
		else {
			mes "[Gerente de Logistica]";
			mes "Necesitas al menos 200 zeny para usar el Almacen.";
		}
		break;
	case 2:
		mes "[Gerente de Logistica]";
		mes "Avisame si cambias de opinion.";
	}
	close;
}
