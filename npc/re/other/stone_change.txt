				// break;
			// }
		// }
		.@count = countitem(.@item);
		delitem .@item, .@count;
		getitem .@item, .@count;
		mes "[Examinadora de Minerales Sorty]";
		mes "Esta hecho. Se ve genial! No lo crees?";
		close;
	case 2:
		mes "[Examinadora de Minerales Sorty]";
