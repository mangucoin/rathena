glast_01,195,273,3	script	Hugin's follower#1	2_F_MAGICMASTER,{
	mes "[Hugin's Follower]";
	mes "No puedo moverme entre el tiempo y las dimensiones como el maestro Hugin, pero puedo fingir que si.";
	next;
	mes "[Hugin's Follower]";
	mes "Que esta causando que este Glast Heim este imbuido con una fuerza oscura tan poderosa?";
	next;
	if( select( "No me importa:Eso es interesante, cuentame mas" ) == 2 ){
		mes "[Hugin's Follower]";
		mes "Gracias a Hugin, encontre la manera de acceder a este Antiguo Glast Heim, que esta imbuido por fuerzas oscuras tan poderosas.";
		next;
		mes "[Hugin's Follower]";
		mes "Que dices? Quieres explorar ahi?";
		next;
		if( select( "No quiero explorar:Realmente quiero explorar!" ) == 2 ){
			mes "[Hugin's Follower]";
			mes "Claro, pero este no es un servicio gratuito! Tengo que obtener algo a cambio. Joho...";
			next;
			mes "[Hugin's Follower]";
			mes "Puedo enviarte a dos lugares. A cual lugar te gustaria que te envie?";
			next;
			switch( select( "Cancelar:Cementerio del Monasterio (Nv100~130) - 5000z:Corredor del 2do piso (Mayor a Nv140) - 10000z" ) ){
				case 1:
					mes "[Hugin's Follower]";
					mes "De acuerdo, por favor regresa cuando cambies de opinion. Creo que tendras mucho de que hablar conmigo...";
					close;
				case 2:
					.@map$ = "gl_chyard_";
					.@x = 148;
					.@y = 144;
					.@zeny = 5000;
					break;
				case 3:
					.@map$ = "gl_cas02_";
					.@x = 104;
					.@y = 31;
					.@zeny = 10000;
					break;
			}
			if( Zeny < .@zeny ){
				mes "[Hugin's Follower]";
				mes "Que? Esperabas este servicio gratis?";
				close;
			}
			Zeny -= .@zeny;
			warp .@map$, .@x, .@y;
			end;
		}
	}
	mes "[Hugin's Follower]";
	mes "Bueno, si no te interesa, no importa.";
	close;
}
