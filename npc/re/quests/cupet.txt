function	script	cute_pet_manager	{
	// getarg(0) -> .@tame_id Array
	// getarg(1) -> .@tame_amount Array
	// getarg(2) -> .@hunt_id Array
	// getarg(3) -> .@hunt_amount Array
	// getarg(4) -> .@mob_id Array
	// getarg(5) ->  Attempt Variable
	// getarg(6) -> .@tame_gets Variable
	// getarg(7) -> .@hunt_id2 Array
	// getarg(8) -> .@hunt_amount2 Array

	mes "[Cute Pet Manager]";
	mes "Hola~! Soy un Administrador de Mascotas";
	mes "que esta a cargo de las relaciones publicas";
	mes "del Nuevo Sistema Mejorado de Mascotas.";
	next;

	mes "[Cute Pet Manager]";
	mes "Si tienes alguna duda sobre";
	mes "el sistema de Mascotas estoy aqui para guiarte.";
	mes "Dime que objeto de domesticacion";
	mes "quieres,";
	mes "y te dire el monstruo";
	mes "que puedes domesticar con el objeto.";
	next;

	mes "[Cute Pet Manager]";
	mes "Tambien te explicare que materiales";
	mes "necesitas traer para poder";
	mes "obtener el objeto de domesticacion.";
	next;

	mes "[Cute Pet Manager]";
	mes "Si ya trajiste los materiales";
	mes "para intercambiar por el objeto de domesticacion,";
	mes "puedo intercambiarlos por objetos de domesticacion de inmediato.";
	next;

	mes "[Cute Pet Manager]";
	mes "Sin embargo, hay un limite";
	mes "para intercambiar objetos de domesticacion.";
	mes "Solo puedes obtener objetos de domesticacion";
	mes getarg(6) + " veces. Asi que asegurate";
	mes "de cuantas veces has intercambiado.";
	next;

	mes "[Cute Pet Manager]";
	mes "Entonces, que objeto de domesticacion quieres?";
	mes "Por favor elige uno de la lista.";
	next;

	// Create Menu System
	set .@size, getarraysize(getarg(0));
	for (set .@a, 0; .@a < .@size; set .@a, .@a + 1)
		set .@menu$, .@menu$ + (.@menu$ == "" ? "" : ":") + getitemname(getelementofarray(getarg(0), .@a));

	// Query Player Choice
	set .@choice, select(.@menu$) - 1;

	// Store Variables (Less Lookup)
	set .@tame_id, getelementofarray(getarg(0), .@choice);
	set .@tame_amount, getelementofarray(getarg(1), .@choice);
	set .@hunt_id, getelementofarray(getarg(2), .@choice);
	set .@hunt_amount, getelementofarray(getarg(3), .@choice);
	set .@mob_id, getelementofarray(getarg(4), .@choice);
	set .@hunt_id2, getelementofarray(getarg(7), .@choice);
	set .@hunt_amount2, getelementofarray(getarg(8), .@choice);

	// dispbottom "Tame ID: " + .@tame_id;
	// dispbottom "Tame Amount: " + .@tame_amount;
	// dispbottom "Hunt ID: " + .@hunt_id;
	// dispbottom "Hunt Name: " + getitemname(.@hunt_id);
	// dispbottom "Hunt Amount: " + .@hunt_amount;
	// dispbottom "Mob ID: " + .@mob_id;
	if (.@hunt_id2) {
		// dispbottom "Hunt ID2: " + .@hunt_id2;
		// dispbottom "Hunt Amount2: " + .@hunt_amount2;
		set .@hunt2_count, countitem(.@hunt_id2);
	}
	if (countitem(.@hunt_id) >= .@hunt_amount && countitem(6083) > 0 && .@hunt2_count >= .@hunt_amount2) {
		mes "[Cute Pet Manager]";
		mes "Ah, has reunido todos los objetos";
		mes "con tus amigos. Ahora puedes";
		mes "intercambiar por un " + getitemname(.@tame_id) + ".";
		mes "Quieres intercambiar conmigo?";
		next;
		if  (select("Intercambiar:No intercambiar") == 2) {
			mes "[Cute Pet Manager]";
			mes "No quieres? Volveras de nuevo.";
			close;
		}
		delitem .@hunt_id, .@hunt_amount;
		delitem 6083, 1;
		if (.@hunt_id2)
			delitem .@hunt_id2, .@hunt_amount2;
		set getarg(5), getarg(5) + 1;
		getitem .@tame_id, .@tame_amount;

		mes "[Cute Pet Manager]";
		mes "Sabia eleccion.";
		mes "Espero que tu y tu mascota se lleven bien.";
		close;
	} else {
		mes "[Cute Pet Manager]";
		mes "Has elegido " + getitemname(.@tame_id) + ".";
		mes "Puedes usarlo para domesticar";
		mes "monstruos ^FF0000" + getmonsterinfo(.@mob_id, MOB_NAME) + "^000000.";
		next;
		mes "[Cute Pet Manager]";
		mes "Trae ^FF0000" + .@hunt_amount + " " + getitemname(.@hunt_id) + "^000000" + (.@hunt_id2 ? ", ^FF0000" + .@hunt_amount2 + " " + getitemname(.@hunt_id2) : "") + " ^000000 y ^FF00001 Dolly Capsule^000000";
		mes "y podras intercambiarlos";
		mes "por un " + getitemname(.@tame_id) + ".";
		close;
	}
}
