que_ng,21,72,6	script	Master Armor Craftsman	4_M_RASWORD,{
	mes "[Gyo]";
	mes "Como definirias la victoria?";
	mes "Derribar al enemigo...";
	mes "Sin embargo, lo mas importante es...";
	mes "Nunca caer tu mismo!";
	next;
	mes "[Gyo]";
	mes "Para eso, debes protegerte mejor.";
	mes "No puedes proteger a 'otros' mientras descuidas tu 'propia' proteccion.";
	next;
	if(select("Combinar armadura.:Terminar conversacion.") == 2) {
		mes "[Gyo]";
		mes "La prudencia tambien es una virtud importante en la batalla.";
		close;
	}
	switch(select("Wolf Armguard[1]:Crescent Armguard[1]:Ninja Scale Armor[1]:Tenebris Latitantes[1]:Salir.")) {
	case 1:
		mes "[Gyo]";
		mes "Wolf Armguard... Una pieza de armadura bestial.";
		next;
		callsub L_Select,2172;
		if (countitem(2171) > 0 && countitem(6252) > 9) {  //custom translation
			delitem 2171,1; //Fox_Armguard
			delitem 6252,10; //Wolf_Blood
			getitem 2172,1; //Wolf_Armguard
			mes "[Gyo]";
			mes "Sientes la naturaleza del lobo?";
			mes "Tanto salvaje como encantador...";
			close;
		}
		break;
	case 2:
		mes "[Gyo]";
		mes "Crescent Armguard... Algo de belleza y fuerza... Sin que ninguna sea menor.";
		next;
		callsub L_Select,2173;
		if (countitem(2171) > 0 && countitem(7321) > 99) {  //custom translation
			delitem 2171,1; //Fox_Armguard
			delitem 7321,100; //Fragment_Of_Crystal
			getitem 2173,1; //Crescent_Armguard
			mes "[Gyo]";
			mes "Que belleza...";
			mes "Como si todo el cuerpo desbordara de magia.";
			close;
		}
		break;
	case 3:
		mes "[Gyo]";
		mes "Ninja Scale Armor... Se dice que un Ninja legendario la uso alguna vez.";
		next;
		callsub L_Select,15054;
		if (countitem(7451) > 29 && countitem(7562) > 29 && countitem(6091) > 29) {  //custom translation
			delitem 7451,30; //Scale_Of_Red_Dragon
			delitem 7562,30; //Ice_Scale
			delitem 6091,30; //Dark_Red_Scale
			getitem 15054,1; //Ninja_Scale_Armor
			mes "[Gyo]";
			mes "Esta es verdaderamente una armadura legendaria...";
			mes "Aunque no creo en leyendas...";
			close;
		}
		break;
	case 4:
		mes "[Gyo]";
		mes "Armadura del Rey de las Sombras... Fue hecha para proteger a otros.";
		next;
		callsub L_Select,15055;
		if (countitem(15056) > 0 && countitem(6089) > 9) {  //custom translation
			delitem 15056,1; //Special_Ninja_Suit_
			delitem 6089,10; //Piece_Of_Darkness
			getitem 15055,1; //Tenebris_Latitantes
			mes "[Gyo]";
			mes "Que tal?";
			mes "No parece como si la armadura fueran dos?";
			close;
		}
		break;
	case 5:
		close;
	}
	mes "[Gyo]";
	mes "No tienes suficientes ingredientes.";
	mes "Revisa de nuevo...";
	close;

L_Select:
	.@string1$ = "Revisar Ingredientes.";
	.@string2$ = "Revisar Estadisticas.";
	while(1) {
		switch(select(.@string1$,.@string2$,"Combinar.","Salir.")) {
		case 1:
			.@string1$ = "";
			callsub L_Ingredients, getarg(0);
			break;
		case 2:
			.@string2$ = "";
			callsub L_Stats, getarg(0);
			break;
		case 3:
			callsub L_Combine;
			return;
		case 4:
			mes "[Gyo]";
			mes "Parece que necesitas mas tiempo para decidir.";
			close;
		}
	}

L_Ingredients:
	mes "[Gyo]";
	switch (getarg(0)) {
	case 2172:	//Wolf_Armguard
		mes "Para hacer un ^ff0000Wolf Armguard, necesitas 1 Fox Armguard y 10 Blood of Wolf^000000.";
		mes "Puedes sentirlo? El latido de la sangre de lobo...";
		break;
	case 2173:	//Crescent_Armguard
		mes "Para hacer un ^ff0000Crescent Armguard, necesitas 1 Fox Armguard y 100 Crystal Fragments^000000.";
		mes "Una media luna tan hermosa y clara como un cristal sera grabada en el.";
		break;
	case 15054:	//Ninja_Scale_Armor
		mes "Para hacer una ^ff0000Ninja Scale Armor, necesitas 30 Fire Dragon Scales, 30 Ice Scales, y 30 Darkred Scale Pieces^000000.";
		mes "Puedes sentir sus poderes legendarios.";
		break;
	case 15055:	//Tenebris_Latitantes
		mes "Para hacer una ^ff0000Armadura del Rey de las Sombras, necesitaras un Special Ninja Suit con ranura y 10 Dark Pieces^000000.";
		mes "Con la oscuridad borra la existencia misma...";
		break;
	}
	next;
	return;

L_Stats:
	mes "[" + getitemname(getarg(0)) + "]";
	switch (getarg(0)) {
	case 2172:	//Wolf_Armguard
		mes "Probabilidad de tener 5 seg. de ATK+100, FLEE -50 al atacar cuerpo a cuerpo.";
		mes "Tipo Escudo, Defensa 45, Ranura 1";
		mes "Nivel Requerido 100, Solo Tipo Ninja";
		break;
	case 2173:	//Crescent_Armguard
		mes "Disminuye el retraso post habilidad en 2% por cada mejora.";
		mes "Tipo Escudo, Defensa 70, Ranura 1";
		mes "Nivel Requerido 100, Solo Tipo Ninja";
		break;
	case 15054:	//Ninja_Scale_Armor
		mes "MHP+15%, MSP-30%";
		mes "Tipo Armadura, Defensa 90, Ranura 1";
		mes "Nivel Requerido 100, Solo Tipo Ninja";
		break;
	case 15055:	//Tenebris_Latitantes
		mes "Probabilidad de activar Nv. 1 Ilusion - Sombra al ser atacado cuerpo a cuerpo.";
		mes "Tipo Armadura, Defensa 60, Ranura 1";
		mes "Nivel Requerido 100, Solo Tipo Ninja";
		break;
	}
	next;
	return;

L_Combine:
	mes "[Gyo]";
	mes "Asegurate de que ^ff0000tengas los ingredientes y equipo correctos a la mano^000000.";
	mes "No sirve lamentarse despues.";
	next;
	switch(select("Combinar.:Salir.")) {
	case 1:
		if (checkweight(1201,1) == 0 || MaxWeight - Weight < 2000) {
			mes "- Espera!! -";
			mes "- No puedes recibir objetos -";
			mes "- porque cargas demasiado. -";
			mes "- Por favor intenta de nuevo -";
			mes "- despues de aligerar tu carga. -";
			close;
		}
		return;
	case 2:
		mes "[Gyo]";
		mes "Parece que necesitas mas tiempo para decidir.";
		close;
	}
	end;
}
