yuno,179,174,4	script	Gomer	851,{
	if (checkweight(1201,1) == 0 || MaxWeight - Weight < 1000) {
		mes "[Gomer]";
		mes "Estas cargando demasiado. Deja algo de tu equipaje y vuelve conmigo.";
		close;
	}
	if (countitem(1549)) {
		mes "[Gomer]";
		mes "Y como te va con ese Pilebuncker?";
		close;
	}
	if (Class != Job_Mechanic && Class != Job_Mechanic_T && Class != Job_Baby_Mechanic) {
		mes "[Gomer]";
		mes "Soy un Mecanico retirado y no creo que tenga asuntos contigo.";
		close;
	}
	mes "[Gomer]";
	mes "Y? Que se siente estar en un Madogear?";
	mes "Es algo que no esta permitido para aquellos que fallaron en encontrar su propio camino.";
	next;
	mes "[Gomer]";
	mes "Yo? Me retire hace mucho tiempo. Porque tuve un accidente... me lastime gravemente y no podre montar de nuevo. Estoy molesto por eso.";
	next;
	mes "[Gomer]";
	mes "Asi que decidi quedarme aqui forjando engranajes para Madogear.";
	mes "Eh? No estoy diciendo que tenga quejas sobre este trabajo.";
	next;
	mes "[Gomer]";
	mes "Ser obligado a hacer algo que no te gusta, no hay nada mas miserable que eso.";
	next;
	mes "[Gomer]";
	mes "Hay personas que son felices solo siendo un apoyo, ya sabes, como ser el muro en el que alguien se recarga.";
	next;
	mes "[Gomer]";
	mes "Eso me hace sentir bien y estoy por hacer un nuevo engranaje. Quiero que me ayudes. Que dices?";
	next;
	switch(select("No estoy tan interesado.:Que vas a hacer?:Donde consigues los materiales?:Traje los materiales.")) {
	case 1:
		mes "[Gomer]";
		mes "No puedo hacer nada";
		mes "al respecto entonces.";
		mes "Nos vemos la proxima!";
		close;
	case 2:
		mes "[Gomer]";
		mes "Um, un enorme palo de metal que podria aplastar a un oponente de un solo golpe!";
		next;
		mes "[Gomer]";
		mes "No se ve elegante, pero disene un dispositivo con resorte para atacar con gran fuerza.";
		next;
		mes "[Gomer]";
		mes "Esta oculto, asi que cuando encuentres una oportunidad, un solo disparo podria poner a un oponente en condicion critica!";
		mes "No puedo esperar a verlo en accion!";
		next;
		mes "[Gomer]";
		mes "Si pudieras imaginar lo buena que podria ser este tipo de arma, la hare realidad.";
		next;
		mes "[Gomer]";
		mes "Traeme 200 Steel, 30 Flexible Tubes, y 1 Brocca. Podria hacer lo que acabo de describir con esos materiales.";
		close;
	case 3:
		mes "[Gomer]";
		mes "Deberias poder obtener steel de muchos monstruos como Kobolds.";
		next;
		mes "[Gomer]";
		mes "Los Flexible Tubes son mas raros, los sueltan los Metalings cerca de Einbroch y Lighthalzen.";
		next;
		mes "[Gomer]";
		mes "Por ultimo, encontrar una Brocca es muy dificil, pero deberias poder hacer algo al respecto. Jaja, ve a buscarla!";
		close;
	case 4:
		if (countitem(999) < 200 || countitem(7325) < 30 || countitem(1415) < 1) {
			mes "[Gomer]";
			mes "No has traido todos los materiales que te pedi.";
			next;
			mes "[Gomer]";
			mes "Tengo la costumbre de abandonar lo que estoy haciendo e intentar hacer cosas nuevas cuando llega la inspiracion.";
			mes "Mejor date prisa!";
			close;
		} else {
			mes "[Gomer]";
			mes "Oh, estuve esperando. No viniste a tiempo, asi que me preguntaba si lo habias olvidado. Por favor espera un momento.";
			specialeffect EF_REPAIRWEAPON;
			next;
			mes "[Gomer]";
			mes "Esta listo. Que te parece? Nada mal, eh? Aunque lo hice yo mismo, debo decir que quedo bastante bien.";
			next;
			mes "[Gomer]";
			mes "Combinacion de peso, volumen y mira su color y brillo. Me cautiva el corazon!";
			next;
			mes "[Gomer]";
			mes "...........";
			next;
			mes "[Gomer]";
			mes "Eh, con una pequena mejora, podria producirse en masa y entregarse. Estoy seguro de que puedo hacer mas de estos asi que te dare este.";
			next;
			mes "[Gomer]";
			mes "Oh, y resulta que despues de todo no necesite todo ese Steel. Asi que toma, te devuelvo la mayoria.";
			delitem 999,50; //Steel
			delitem 7325,30; //Tube
			delitem 1415,1; //Skewer
			getitem 1549,1; //Pilebuncker
			getitem 1360,1; //Two_Handed_Axe
			close;
		}
	}
}
