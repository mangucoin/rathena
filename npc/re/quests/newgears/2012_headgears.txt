L_CheckHunting:
	for (set .@quest,getarg(0)+1; .@quest<getarg(1)+getarg(2); set .@quest,.@quest+1) {
		if (checkquest(.@quest,HUNTING) == 2) {
			mes "[Reno]";
			mes "Mision de caza de recoleccion "+getarg(3)+" completada.";
			next;
			setquest getarg(0);
			erasequest .@quest;
			getitem getarg(4),getarg(5);
			mes "Recibes "+getarg(5)+" "+getitemname(getarg(4))+"s como recompensa de la recoleccion "+getarg(3)+".";
			close;
		}
	}
	return;
