c_tower1,232,222,5	script	Belljamin Button#ct	968,{
	mes "[Belljamin Button]";
	mes "Los monstruos aqui son capaces de controlar el tiempo.";
	if( BaseLevel < 150 ){
		mes "Regresa cuando hayas alcanzado el ^0000FFNivel 150^000000.";
		close;
	}
	mes "El tiempo, los recuerdos, los entornos... todo puede distorsionarse aqui.";
	next;
	mes "[Belljamin Button]";
	mes "Te ruego que no preguntes mas al respecto. Podria poner tu vida en peligro.";
	next;
	.@menu = select("No preguntar:Ir al 2do piso de la torre distorsionada:Ir al 3er piso de la torre distorsionada");
	switch(.@menu){
		case 1:
			break;
		case 2:
		case 3:
			if( .@menu == 2 ){
				.@map$ = "c_tower2_";
				.@x = 268;
				.@y = 26;
			}else{
				.@map$ = "c_tower3_";
				.@x = 65;
				.@y = 147;
			}
			mes "[Belljamin Button]";
			mes "Ten cuidado con tu eleccion.";
			next;
			mes "[Belljamin Button]";
			mes "Necesito ^0000FF10,000z^000000 para dejarte entrar. Es un lugar bastante peligroso.";
			next;
			switch( select( "Salir:Pagar 10,000z:Por que necesitas dinero?" ) ){
				case 1:
					break;
				case 2:
					if( Zeny < 10000 ){
						mes "[Belljamin Button]";
						mes "En serio? No tienes el dinero y me pediste que te lleve ahi?";
						close;
					}
					Zeny -= 10000;
					mes "[Belljamin Button]";
					mes "No estoy seguro de como llevaras la cuenta del tiempo ahi dentro, pero ten todo el cuidado que puedas.";
					close2;
					warp .@map$, .@x, .@y;
					end;
				case 3:
					mes "[Belljamin Button]";
					mes "Estoy aqui todo el dia por la seguridad de los aventureros. Siempre les advierto sobre el peligro. Por eso necesito alguna forma de pago.";
					emotion ET_SWEAT;
					close;
			}
			break;
	}
	mes "[Belljamin Button]";
	mes "Uf...Hiciste una buena eleccion. Este lugar es brutal asi que probablemente deberias regresar a un area segura.";
	close;
}
