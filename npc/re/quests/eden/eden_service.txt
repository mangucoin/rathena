-	script	Goods Cabinet#00::pggc	-1,{
	mes "Es un Gabinete de Almacenamiento de Bienes.";
	mes "Un mensaje esta escrito en un pedazo de papel.";
	next;
	mes "+ El gabinete es exclusivo +";
	mes "+ para el Grupo Eden +";
	mes "Si quieres usar este";
	mes "gabinete por favor revisa";
	mes "lo siguiente:";
	next;
	mes "1.Eres miembro";
	mes "   del Grupo Eden?";
	mes "2.Has aprendido";
	mes "   suficientes habilidades basicas?";
	mes "3.La tarifa del gabinete es";
	mes "  ^4d4dff500 zeny^000000!";
	next;
	if (callfunc("F_HasEdenGroupMark")) {
		mes "Necesitas insertar zeny para usar el gabinete.";
		mes "Costo : 500 Zeny ";
		mes "Te gustaria usarlo?";
		next;
		switch(select("Usar el Gabinete.:Cancelar.")) {
			case 1:
				if (Zeny > 499) {
					if(!callfunc("F_CanOpenStorage")){
						mes "El gabinete no funciona para mi.";
						mes "Tal vez aun no estoy calificado para usar el Gabinete.";
						close;
					}
					set Zeny, Zeny - 500;
					close2;
					openstorage;
					end;
				}
				mes "No tengo suficiente zeny.";
				mes "Necesito al menos 500 zeny para usar el Gabinete.";
				close;
			case 2:
				mes "Lo usare la proxima vez.";
				close;
		}
	}
	mes "Necesito una Insignia del Grupo Eden para usar este Gabinete.";
	close;
}
