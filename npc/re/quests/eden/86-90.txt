		mes "[Tablero de Misiones 86-90]";
		mes "Necesitas esperar 3 horas antes de poder tomar esta mision de nuevo.";
		close;
	}
	set .@hunting, checkquest(.@quest1,HUNTING);

		mes "[Tablero de Misiones 86-90]";
		mes "Ya terminaste de cazar los 30 "+getarg(1)+getarg(5,"")+"?";

		mes "Has completado la caza.";
		mes "Por favor acepta esta recompensa como compensacion.";
		erasequest .@quest1;
		if (.@playtime > -1) erasequest .@quest2;
		setquest .@quest2;
		getexp getarg(2),0;
		close;

		mes "Caza 30 monstruos "+getarg(1)+" ahi y publica tu exito en este tablero.";
