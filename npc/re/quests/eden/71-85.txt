moc_para01,44,38,3	script	Mission [71 - 85]	857,{
	if (!callfunc("F_HasEdenGroupMark")) {
		mes "- Solo miembros del -";
		mes "- Grupo Eden pueden leer -";
		mes "- este tablero de misiones. -";
		close;
	}

	// Clear the original quest data (no longer used).
	if (slv_quest) set slv_quest,0;
	if (checkquest(10102) > -1 || checkquest(10103) > -1 || checkquest(10104) > -1 || checkquest(10105) > -1 || checkquest(10106) > -1) {
		for(set .@quest,10102; .@quest<=10106; set .@quest,.@quest+1) {
			if (checkquest(.@quest) > -1)
				erasequest .@quest;
		}
		mes "- Como el cliente se ha mudado -";
		mes "- a otro lugar, -";
		mes "- el formulario de solicitud ha sido eliminado. -";
		next;
	}

	if (BaseLevel < 71) {
		mes "- Solo miembros del -";
		mes "- nivel adecuado pueden leer -";
		mes "- este tablero de misiones. -";
		mes "- Intenta encontrar un tablero -";
		mes "- que se adapte a tu nivel. -";
		close;
	} else if (BaseLevel > 85) {
		// Quest IDs: 10107-10123, 5055-5057
		setarray .@quests[0],
			10107,10108,10109,
			10110,10111,10112,10113,
			10114,10115,10116,10117,
			10118,10119,10120,
			10121,10122,10123,5055,5056;
		for(set .@i,0; .@i<getarraysize(.@quests); set .@i,.@i+1) {
			if (checkquest(.@quests[.@i],HUNTING) == 2)
				set .@complete[getarraysize(.@complete)], .@quests[.@i];
		}
		if (checkquest(5057) > -1 && countitem(7187) >= 30)
			set .@complete[getarraysize(.@complete)],5057;
		if (getarraysize(.@complete)) {
			mes "Tienes misiones en progreso.";
			mes "Quieres entregarlas?";
			next;
			if(select("Por supuesto.:No.") == 2)
				close;
			for(set .@i,0; .@i<getarraysize(.@complete); set .@i,.@i+1)
				callsub L_Quest,.@complete[.@i];
			close;
		}
		mes "- Solo miembros del -";
		mes "- nivel adecuado pueden leer -";
		mes "- este tablero de misiones. -";
		mes "- Intenta encontrar un tablero -";
		mes "- que se adapte a tu nivel. -";
		close;
	}
	mes "- Elige donde -";
	mes "- te gustaria cazar. -";
	next;
	switch(select("Calabozo Sphinx:Glast Heim:Area de Juno:Torre del Reloj:Islas Localizadas")) {
	case 1:
		mes "- Hay varias solicitudes -";
		mes "- relacionadas al Calabozo Sphinx -";
		mes "- en este Tablero de Misiones. -";
		next;
		set .@str1$,"Los aldeanos de Morocc estan preocupados por el creciente numero de monstruos en el Calabozo Sphinx. ";
		set .@str2$,"- Meidi de Morocc -";
		switch(select("Cazar Requiem:Cazar Marduk:Cazar Pasana")) {
			case 1: callsub L_Quest,10107,.@str1$+"Caza 10 Requiems.",.@str2$;
			case 2: callsub L_Quest,10108,.@str1$+"Caza 10 Marduks.",.@str2$;
			case 3: callsub L_Quest,10109,.@str1$+"Caza 10 Pasanas.",.@str2$;
		}
	case 2:
		mes "- Hay varias solicitudes -";
		mes "- relacionadas a Glast Heim -";
		mes "- en este Tablero de Misiones. -";
		next;
		set .@str1$,"Tengo mucho miedo de que los monstruos de Glast Heim nos ataquen. ";
		set .@str2$,"- Ancellia de Geffen -";
		switch(select("Cazar Dark Frame:Cazar Evil Druid:Cazar Wraith:Cazar Raydric Archer")) {
			case 1: callsub L_Quest,10110,.@str1$+"Caza 10 Dark Frames.",.@str2$;
			case 2: callsub L_Quest,10111,.@str1$+"Caza 10 Evil Druids.",.@str2$;
			case 3: callsub L_Quest,10112,.@str1$+"Caza 10 Wraiths.",.@str2$;
			case 4: callsub L_Quest,10113,.@str1$+"Caza 10 Raydric Archers.",.@str2$;
		}
	case 3:
		mes "- Hay varias solicitudes -";
		mes "- relacionadas al area -";
		mes "- alrededor de Juno. -";
		next;
		set .@str1$,"Los que no podemos pelear, como yo, encontramos mucho mas dificil viajar debido al gran numero de monstruos. ";
		set .@str2$,"- Jeanbai, viajero -";
		switch(select("Cazar Grand Peco:Cazar Sleeper:Cazar Goat:Cazar Harpy")) {
			case 1: callsub L_Quest,10114,.@str1$+"Caza 20 Grand Pecos.",.@str2$;
			case 2: callsub L_Quest,10115,.@str1$+"Caza 20 Sleepers.",.@str2$;
			case 3: callsub L_Quest,10116,.@str1$+"Caza 20 Goats.",.@str2$;
			case 4: callsub L_Quest,10117,.@str1$+"Caza 20 Harpies.",.@str2$;
		}
	case 4:
		mes "- Hay varias solicitudes -";
		mes "- relacionadas a la Torre del Reloj -";
		mes "- en este Tablero de Misiones. -";
		next;
		set .@str1$,"Me encantaba escuchar la dulce melodia de la Torre del Reloj de Al De Baran pero ya no puedo escuchar nada por todo el ruido que hacen los monstruos. ";
		set .@str2$,"- Rizingsetter, Presidente de la comunidad Amor al Reloj -";
		switch(select("Cazar Clock:Cazar Punk:Cazar Rideword")) {
			case 1: callsub L_Quest,10118,.@str1$+"Caza 15 Clocks antes de que se rompa.",.@str2$;
			case 2: callsub L_Quest,10119,.@str1$+"Caza 15 Punks antes de que se rompa.",.@str2$;
			case 3: callsub L_Quest,10120,.@str1$+"Caza 15 Ridewords antes de que se rompa.",.@str2$;
		}
	case 5:
		mes "- Hay varias solicitudes -";
		mes "- relacionadas a las Islas Localizadas -";
		mes "- en este Tablero de Misiones. -";
		next;
		set .@str1$,"Quiero ir de aventura pero hay tantas cosas que tengo que hacer. Si yo no puedo, los demas tampoco! ";
		set .@str2$,"- Funfy, quien quiere ir de aventura -";
		switch(select("Cazar Kikimora:Cazar Miyabi Doll:Cazar Mi Gao:Cazar Headless Mule:Cazar Tamruan:Recolectar Mascaras de Festival para el Festival")) {
			case 1: callsub L_Quest,10121,.@str1$+"Caza 15 Kikimoras en Moscovia!",.@str2$;
			case 2: callsub L_Quest,10122,.@str1$+"Caza 15 Miyabi Dolls en Amatsu!",.@str2$;
			case 3: callsub L_Quest,10123,.@str1$+"Caza 15 Mi Gaos en Luoyang!",.@str2$;
			case 4: callsub L_Quest,5055,"Se que hay muchos monstruos aterradores fuera de la ciudad, pero lo que me aterroriza son estas Headless Mules merodeando. Espero que nunca se aventuren en el pueblo. Caza 15 Headless Mules!","- Mariana de Brasilis -";
			case 5: callsub L_Quest,5056,"Los Tamruans hacen mucho ruido. No me importa en la manana pero no me dejan dormir en la noche. Quiero dormir. Caza 15 Tamruans!","- Shaman Pumo de Ayothaya -";
			case 6: callsub L_Quest,5057,"Los Dumpling Childs robaron todas las mascaras que habia preparado para el festival. Intenta conseguir 30 mascaras! Te lo agradecere~","- SaYumMoon Jefe de Kunlun -";
		}
	}
	end;

//callsub L_Quest,<quest ID>{,"<description1>","<description2>"};
L_Quest:
	set .@quest, getarg(0);
	switch(.@quest) {
		case 10107: set .@name$,"Cazar Requiem"; set .@exp,15000; break;
		case 10108: set .@name$,"Cazar Marduk"; set .@exp,15000; break;
		case 10109: set .@name$,"Cazar Pasana"; set .@exp,15000; break;
		case 10110: set .@name$,"Cazar Dark Frame"; set .@exp,20000; break;
		case 10111: set .@name$,"Cazar Evil Druid"; set .@exp,20000; break;
		case 10112: set .@name$,"Cazar Wraith"; set .@exp,20000; break;
		case 10113: set .@name$,"Cazar Raydric Archer"; set .@exp,20000; break;
		case 10114: set .@name$,"Cazar Grand Peco"; set .@exp,30000; break;
		case 10115: set .@name$,"Cazar Sleeper"; set .@exp,30000; break;
		case 10116: set .@name$,"Cazar Goat"; set .@exp,30000; break;
		case 10117: set .@name$,"Cazar Harpy"; set .@exp,30000; break;
		case 10118: set .@name$,"Cazar Clock"; set .@exp,25000; break;
		case 10119: set .@name$,"Cazar Punk"; set .@exp,25000; break;
		case 10120: set .@name$,"Cazar Rideword"; set .@exp,25000; break;
		case 10121: set .@name$,"Cazar Kikimora"; set .@exp,30000; break;
		case 10122: set .@name$,"Cazar Miyabi Doll"; set .@exp,30000; break;
		case 10123: set .@name$,"Cazar Mi Gao"; set .@exp,30000; break;
		case 5055: set .@name$,"Cazar Headless Mule"; set .@exp,30000; break;
		case 5056: set .@name$,"Cazar Tamruan"; set .@exp,30000; break;
		case 5057: set .@name$,"Recolectar Mascaras de Festival para el Festival"; set .@exp,30000; break;
	}
	if (getargcount() == 1) {
		mes "'"+.@name$+"' completada. Quieres entregarla y recibir tu recompensa?";
		next;
		if(select("Entregar mision.:Cancelar.") == 1) {
			if (.@quest == 5057) delitem 7187,30; //Festival_Mask
			erasequest .@quest;
			getexp .@exp,0;
			mes "Has recibido EXP por completar la mision.";
			next;
			return;
		}
		close;
	} else {
		if (.@quest == 5057) {
			// Quest 5057 is the only collection quest, so separate conditions are unnecessary.
			if (checkquest(5057) == -1)
				set .@hunting,-1;
			else if (countitem(7187) >= 30)
				set .@hunting,2;
		} else
			set .@hunting, checkquest(.@quest,HUNTING);
		if (.@hunting == 0 || .@hunting == 1) {
			mes getarg(1);
			mes getarg(2);
			next;
			mes "Regresa aqui cuando hayas terminado la mision.";
		} else if (.@hunting == 2) {
			mes "Has completado la mision.";
			next;
			if(select("Entregar mision.:Cancelar.") == 1) {
				if (.@quest == 5057) delitem 7187,30; //Festival_Mask
				erasequest .@quest;
				getexp .@exp,0;
				mes "Has recibido EXP por completar la mision.";
			}
		}/* else if (.@hunting == 3) {
//			recall_completequest .@quest
			erasequest .@quest;
			mes "Has re-aceptado la mision '"+.@name$+"'.";
		}*/ else {
			mes getarg(1);
			mes getarg(2);
			next;
			mes "Quieres aceptar la mision?";
			next;
			if(select("Si.:No.") == 1) {
				mes "Has recibido la mision '"+.@name$+"'. Revisa tu informacion de misiones para mas detalles.";
				setquest .@quest;
			}
		}
		close;
	}
}
