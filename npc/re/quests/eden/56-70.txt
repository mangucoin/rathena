			next;
			mes "Esta mision no tiene limite de tiempo.";
		}
		close;

		if (countitem(getarg(5)) < getarg(6))
			set .@items,1; //incomplete
		else
			set .@items,2; //complete

		mes "Tienes una mision en progreso. Te gustaria revisar los detalles?";
		next;
		if(select("Revisar los detalles.:Cancelar.") == 1)
			callsub L_Details, .@quest_id;
		close;
	} else {
		mes "Lo he hecho bastante bien en la mision. Deberia reportarla ahora?";
		next;
		if(select("Reportar la mision.:No reportar aun.") == 1) {
			delitem getarg(3),getarg(4);
			if (.@items)
				delitem getarg(5),getarg(6);
			getexp getarg(1),getarg(2);
			erasequest .@quest_id;
			mes "Has completado la mision. Recibe tus recompensas.";
		}
		close;
	}
