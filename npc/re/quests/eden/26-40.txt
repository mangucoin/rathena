moc_para01,38,38,3	script	Mission [26 - 40]	857,{
	if (!callfunc("F_HasEdenGroupMark")) {
		mes "No eres un miembro calificado. No puedes usar el tablero de misiones.";
		close;
	}
	mes "[Tablero de Misiones Nv. 26 ~ 40]";
	mes "- Muchas misiones estan -";
	mes "- en el tablero de anuncios. -";
	next;
	if ((BaseLevel < 26) || (BaseLevel > 40)) {
		mes "Tu nivel no califica. No puedes usar el tablero de misiones.";
		close;
	}
	switch(select("Mision de Caza:Mision de Entrega:Mision de Recoleccion:Mision de Escolta")) {
	case 1:
		mes "[Tablero de Misiones de Caza]";
		mes "- Los detalles de la Mision de Caza -";
		mes "- estan en el tablero. -";
		next;
		switch(select("Cazar Poison Spore:Cazar Smokie:Cazar Elder Willow(Nv.30+):Cazar Coco(Nv.35+):Cazar Thara Frog(Nv.38+)")) {
			case 1: callsub L_Quest,8266,0,"Cazar Poison Spore","Sponiac","Geffen","Caza 10 Poison Spores";
			case 2: callsub L_Quest,8267,0,"Cazar Smokie","Sponiac","Geffen","Caza 10 Smokies";
			case 3: callsub L_Quest,8268,30,"Cazar Elder Willow","Sponiac","Geffen","Caza 10 Elder Willows";
			case 4: callsub L_Quest,8269,35,"Cazar Coco","Sponiac","Geffen","Caza 10 Cocos";
			case 5: callsub L_Quest,8270,38,"Cazar Thara Frog","Sponiac","Geffen","Caza 10 Thara Frogs";
		}
	case 2:
		mes "[Tablero de Misiones de Entrega]";
		mes "- Los detalles de la Mision de Entrega -";
		mes "- estan en el tablero. -";
		next;
		switch(select("Entregar un Insecticida:Entregar Miel:Entregar Cobijas(Nv.35+)")) {
			case 1: callsub L_Quest,8271,0,"Entregar un Insecticida","Dieshin Man","Ruinas de Morocc","Consigue un 'Insecticida' de Dieshin Man y entregalo al sureste de Payon";
			case 2: callsub L_Quest,8272,0,"Entregar Miel","Dieshin Man","Ruinas de Morocc","Consigue 'Miel' de Dieshin Man y entregala al oeste de la Abadia St. Capitolina";
			case 3: callsub L_Quest,8273,35,"Entregar Cobijas","Dieshin Man","Ruinas de Morocc","Consigue Cobijas de Dieshin Man y entregalas al campo al sur de Lutie";
		}
	case 3:
		mes "[Tablero de Misiones de Recoleccion]";
		mes "- Los detalles de la Mision de Recoleccion -";
		mes "- estan en el tablero. -";
		next;
		switch(select("Recolectar Huesos:Recolectar Pieles de Animal:Recolectar Cascaras(Nv.30+):Recolectar Colas(Nv.32+):Recolectar Galletas(Nv.35+):Recolectar Bigotes(Nv.35+)")) {
			case 1: callsub L_Quest,8274,0,"Recolectar Huesos","Dashia","Payon","Recolecta '10 Skel-Bones' y daselos a Dashia";
			case 2: callsub L_Quest,8275,0,"Recolectar Pieles de Animal","Dashia","Payon","Recolecta '10 Bear's Footskins' y '5 Animal Skins' y daselos a Dashia";
			case 3: callsub L_Quest,8276,30,"Recolectar Cascaras","Dashia","Payon","Recolecta '20 Worm Peelings' y '20 Garlets' y daselos a Dashia";
			case 4: callsub L_Quest,8277,32,"Recolectar Colas","Dashia","Payon","Recolecta '10 Yoyo Tails', '10 Acorns' y '10 Raccon Leafs' y daselos a Dashia";
			case 5: callsub L_Quest,8278,35,"Recolectar Galletas","Dashia","Payon","Recolecta '5 Well-baked Cookies' y '1 Piece of Cake' y daselos a Dashia";
			case 6: callsub L_Quest,8279,35,"Recolectar Bigotes","Dashia","Payon","Recolecta '5 Mole Whiskers' y '3 Mole Claws' y daselos a Dashia";
		}
	case 4:
		mes "[Tablero de Misiones de Escolta]";
		mes "Actualmente no hay Misiones de Escolta registradas en el tablero.";
		close;
	}
	end;
L_Quest:
	//QuestID,ReqLvl,Quest,Client,Loc,Desc
	.@string$ = getarg(2);
	if (BaseLevel < getarg(1,26)) {
		mes "Tu nivel no cumple los requisitos. No puedes tomar la mision '"+ .@string$ +"'.";
		close;
	}
	mes "["+ .@string$ +" !!]";
	mes "Cliente : "+getarg(3);
	mes "Ubicacion del Cliente : "+getarg(4);
	mes "Mision Solicitada : "+getarg(5);
	mes " ";
	mes "Aceptas esta mision?";
	next;
	if(select("No, no la acepto.:Si, la acepto.") == 1) {
		mes "Has rechazado la mision "+ .@string$ +".";
		close;
	}
	if (checkquest(getarg(0)) > -1) mes "Ya estas haciendo esta mision.";
	else {
		setquest getarg(0);
		mes "Has aceptado la mision "+ .@string$ +".";
	}
	mes "Puedes obtener una recompensa del cliente despues de terminar la mision.";
	next;
	mes "Para mas detalles, presiona 'Alt + U' para obtener mas informacion en la Ventana de Misiones.";
	close;
}
