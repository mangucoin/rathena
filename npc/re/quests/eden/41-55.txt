moc_para01,40,38,3	script	Mission [41 - 55]	857,{
	if (!callfunc("F_HasEdenGroupMark")) {
		mes "No eres miembro del Grupo Eden. No estas calificado para acceder al tablero de misiones.";
		close;
	}
	if (checkweight(1201,1) == 0 || (MaxWeight - Weight) < 1000) {
		mes "^008800Espera un momento!!";
		mes "No puedes recibir mas objetos porque llevas demasiados. Por favor intenta de nuevo despues de aligerar tu carga.^000000";
		close;
	}
	if (checkquest(12088) == -1) {
		if (BaseLevel < 41 || BaseLevel > 55) {
			mes "Tablero de misiones para aventureros Nv.41~55.";
			close;
		}
		mes "[Tablero de Misiones]";
		mes "Tablero de misiones.";
		next;
		if(select("Revisar las Misiones.:Cancelar.") == 2) {
			mes "^000077Ok. Revisemos las misiones la proxima vez.^000000.";
			close;
		}
		switch(select("Caza de Thara Frog:Caza de Cruiser:Caza de Kukre:Caza de Orc Baby:Caza de Orc Warrior:Caza de Vadon:Caza de Megalodon:Caza de Marse:Caza de Cornutus:Caza de Myst:Caza de Shellfish:Caza de Marine Sphere")) {
			case 1: callsub L_Quest,12072,"Has visto una rana roja? Se llama Thara Frog y es muy rara. Estamos tratando de verificar donde habita. Asi que por favor caza 20 Thara Frogs.";
			case 2: callsub L_Quest,12073,"Escuche que algunos juguetes atacan a la gente. Es ridiculo! Vamos a mostrarles nuestro poder! Tenemos que cazar 15 Cruisers!";
			case 3: callsub L_Quest,12074,"Alguien fue robado cerca de la playa! El posible sospechoso es un monstruo criminal llamado Kukre. Por favor caza 30 Kukre.";
			case 4: callsub L_Quest,12075,"Los Orc Babies se estan volviendo mas violentos estos dias. Por favor caza 15 Orc Babies.";
			case 5: callsub L_Quest,12076,"Los Orc Warriors estan aumentando su agresividad. Por favor caza 20 Orc Warriors.";
			case 6: callsub L_Quest,12077,"Has visto cangrejos con caparazones gruesos? Son Vadons. Por favor caza 15 Vadons.";
			case 7: callsub L_Quest,12078,"Hay peces muertos por toda la playa y esta empeorando. Por favor caza 30 Megalodon.";
			case 8: callsub L_Quest,12079,"Puedes cazar Marse? Necesito 15 Marse para mi estudio. Por favor ayudame.";
			case 9: callsub L_Quest,12080,"Un tonto Cornutus mordio el pie del Alcalde. El Alcalde esta molesto asi que necesitamos cazar 20 Cornutus para hacerlo feliz!";
			case 10: callsub L_Quest,12081,"La gente esta asustada por los Myst. Parecen fantasmas y asustan a la gente. Por favor caza 15 Myst.";
			case 11: callsub L_Quest,12082,"No preguntes por que, pero por favor caza 30 Shellfish. Es una mision secreta!";
			case 12: callsub L_Quest,12083,"Marine Sphere? Sabes que es una Marine Sphere? Tengo una mision para que caces 15 Marine Sphere, asi que por favor hazlo.";
			case 13: callsub L_Quest,12084,"El alcalde quiere comer un delicioso platillo de pescado. Phen sera el ingrediente perfecto. Por favor caza 20 Phen!";
			case 14: callsub L_Quest,12085,"Monstruos Orc muertos regresaron a nuestro pueblo y revivieron como Orc Skeletons. Estamos en caos! Por favor caza 15 Orc Skeleton.";
			case 15: callsub L_Quest,12086,"Unos monstruos extranos llamados Zenorc nos estan atacando. Por favor caza 30 Zenorc.";
			case 16: callsub L_Quest,12087,"Preparate! Es una mision cruel! Caza 15 Mummies!! Lo intente una vez, pero no pude. Es demasiado aterrador!";
		}
		end;
	}
	setarray .@Exp[0],9600,7200,15750,7875,11100,8550,17100,9000,12300,9900,20250,10350,13800,11025,22500,11250;
	for(set .@i,0; .@i<16; set .@i,.@i+1)
		if (checkquest(12072+.@i,HUNTING) == 2) {
			mes "[Tablero de Misiones]";
			mes "Has completado la mision.";
			mes "Aqui esta tu recompensa.";
			erasequest 12088;
			erasequest 12072+.@i;
			specialeffect2 EF_STEAL;
			getexp .@Exp[.@i],0;
			close;
		}
	if ((checkquest(12072,HUNTING) == 0) && (checkquest(12073,HUNTING) == 0) && (checkquest(12074,HUNTING) == 0) && (checkquest(12075,HUNTING) == 0) && (checkquest(12076,HUNTING) == 0) && (checkquest(12077,HUNTING) == 0) && (checkquest(12078,HUNTING) == 0) && (checkquest(12079,HUNTING) == 0) && (checkquest(12080,HUNTING) == 0) && (checkquest(12081,HUNTING) == 0) && (checkquest(12082,HUNTING) == 0) && (checkquest(12083,HUNTING) == 0) && (checkquest(12084,HUNTING) == 0) && (checkquest(12085,HUNTING) == 0) && (checkquest(12086,HUNTING) == 0) && (checkquest(12087,HUNTING) == 0)) {
		mes "Ahora puedes ver el tablero de anuncios.";
		erasequest 12088;
		close;
	}
	mes "No hay nada especial que hacer.";
	close;
L_Quest:
	mes "[Memo del Tablero de Misiones]";
	mes getarg(1);
	next;
	if(select("Aceptar la mision.:Cancelar.") == 1) {
		mes "[Tablero de Misiones]";
		mes "Has aceptado la mision. Para mas detalles, presiona 'alt + u', puedes obtener informacion de la Ventana de Misiones.";
		setquest 12088;
		setquest getarg(0);
	}
	close;
}
