moc_para01,36,38,3	script	Mission [11 - 25]#Tuto	4_BOARD3,{
	if (!callfunc("F_HasEdenGroupMark")) {
		mes "- Necesitas tener una -";
		mes "- ^4d4dff'Insignia del Grupo Eden'^000000 -";
		mes "- para recibir estas misiones. -";
		close;
	}
	for ( .@i = 11124; .@i < 11134; .@i++ )
		if (checkquest(.@i,PLAYTIME) == 2) erasequest .@i;
	for( .@i = 11114; .@i < 11124; .@i++ ) {
		.@j = checkquest(.@i,HUNTING);
		if (.@j == 2) {
			mes " - Debes recoger tu - ";
			mes " - recompensa antes de iniciar - ";
			mes " - una nueva mision. - ";
			mes " ";
			mes " - Encargada : Spike - ";
			close;
		}
		else if (.@j >= 0) {
			mes " - Solo puedes solicitar - ";
			mes " - una mision a la vez. - ";
			mes " ";
			mes " - Encargada : Spike - ";
			close;
		}
	}
	if (BaseLevel < 11) {
		mes " - No hay tareas - ";
		mes " - disponibles para ti ahora. - ";
		close;
	}
	if (BaseLevel > 25) {
		mes " - Estas misiones son muy - ";
		mes " - faciles para tu nivel. - ";
		mes " - Busca una mision mas - ";
		mes " - apropiada para tu nivel. - ";
		close;
	}
	mes " - Tablero de misiones -";
	mes " - para principiantes. -";
	mes " ";
	mes " - Encargada : Spike -";
	next;
	switch(select("Caza de Hornets:Caza de Condors:Pata de Grasshopper:Caza de Worm Tails:Caza de Spores:Control de Plagas:Caza de Mukas:Caza de Familiars:Recolectar Plumas:Recolectar Poison Spores:Cancelar")) {
		case 1: callsub L_Quest,11114,"Caza de Hornets"," - Caza 10 ^4d4dff'Hornets'^000000 - "," - al noroeste de Prontera. - ";
		case 2: callsub L_Quest,11115,"Caza de Condors"," - Caza 20 ^4d4dff'Condors'^000000 - "," - al sur de Morocc. - ";
		case 3: callsub L_Quest,11116,"Pata de Grasshopper"," - Caza 10 ^4d4dff'Rockers'^000000 - "," - al suroeste de Prontera y - "," - trae 10 ^4d4dffGrasshopper's Legs^000000. - ";
		case 4: callsub L_Quest,11117,"Caza de Worm Tails"," - Caza 20 ^4d4dff'Worm Tails'^000000 - "," - al sureste de Payon.- ";
		case 5: callsub L_Quest,11118,"Caza de Spores"," - Caza 30 ^4d4dff'Spores'^000000 - "," - alrededor de Payon. - ";
		case 6: callsub L_Quest,11119,"Control de Plagas"," - Caza 20 ^4d4dffThief Bug Eggs^000000 - "," - en la Alcantarilla de Prontera - "," - y trae 10 ^4d4dffChrysalis^000000. - ";
		case 7: callsub L_Quest,11120,"Caza de Mukas"," - Caza 20 ^4d4dff'Muka'^000000 - "," - al sur de Prontera. - ";
		case 8: callsub L_Quest,11121,"Caza de Familiars"," - Caza 20 ^4d4dffFamiliar^000000 - "," - en las Alcantarillas - "," - al oeste de Prontera. - ";
		case 9: callsub L_Quest,11122,"Recolectar Plumas"," - Caza 30 ^4d4dff'Peco Pecos'^000000 - "," - al sur de Prontera. - ";
		case 10: callsub L_Quest,11123,"Recolectar Poison Spores"," - Caza 30 ^4d4dffPoison Spores^000000 - "," - dentro de la Torre de Geffen y - "," - trae 5 ^4d4dffPoison Spores^000000. - ";
		case 11:
			mes " - Dejar de leer el tablero de misiones. - ";
			close;
	}

L_Quest:
	.@quest_status = checkquest(getarg(0)+10,PLAYTIME);
	if (.@quest_status == 0 || .@quest_status == 1) {
		mes " - Mision ^4d4dff"+ getarg(1) +"^000000 - ";
		mes " - ya fue completada por hoy. - ";
		mes " ";
		mes " - Encargada : Spike - ";
		close;
	}
	mes " - Mision '"+ getarg(1) +"' - ";
	for ( .@i = 2; .@i < getargcount(); .@i++ )
		mes getarg(.@i);
	mes " ";
	mes " - Encargada : Spike - ";
	next;
	if (select("Hare esta mision.:Buscar otras misiones.") == 2) {
		mes " - Busquemos -";
		mes " - otras misiones. - ";
		close;
	}
	for ( .@i = 2; .@i < getargcount(); .@i++ )
		mes getarg(.@i);
	setquest getarg(0);
	close;
}

moc_para01,32,30,6	script	Spike	4_F_CHILD,{
	for( .@i = 11114; .@i < 11124; .@i++ ) {
		.@j = checkquest(.@i,HUNTING);
		if (.@j == 2) {
			switch(.@i) {
				case 11114: callsub L_Quest,.@i,"Caza de Hornets",900;
				case 11115: callsub L_Quest,.@i,"Caza de Condors",1800;
				case 11116: callsub L_Quest,.@i,"Pata de Grasshopper",1400,940,10;
				case 11117: callsub L_Quest,.@i,"Caza de Worm Tails",2600;
				case 11118: callsub L_Quest,.@i,"Caza de Spores",3900;
				case 11119: callsub L_Quest,.@i,"Exterminar Insectos",2900,915,10;
				case 11120: callsub L_Quest,.@i,"Caza de Mukas",3200;
				case 11121: callsub L_Quest,.@i,"Caza de Familiars",3200;
				case 11122: callsub L_Quest,.@i,"Recolectar Plumas",5400;
				case 11123: callsub L_Quest,.@i,"Recolectar Poison Spores",6000,7033,5;
			}
		}
		else if (.@j >= 0) {
			mes "[Spike]";
			mes "Aun no has terminado tu mision.";
			mes "Sigue asi. Ya casi terminas.";
			close;
		}
	}
	mes "[Spike]";
	mes "Estas buscando trabajo?";
	mes "Siempre tenemos mucho que hacer.";
	next;
	mes "[Spike]";
	mes "Por que no echas un vistazo y encuentras algunas tareas interesantes?";
	close;

L_Quest:
	.@quest_id = getarg(0);
	if (checkquest(.@quest_id,HUNTING) == 2) {
		.@item_req = getarg(3,0);
		.@req_amount = getarg(4,0);
		if (.@item_req) {
			if (countitem(.@item_req) < .@req_amount) {
				mes "[Spike]";
				mes "No tienes suficientes "+ getitemname(.@item_req) +".";
				mes "Necesitas traer "+ .@req_amount +" "+ getitemname(.@item_req) +".";
				close;
			}
		}
		mes "[Spike]";
		mes "Hmm... dejame ver~";
		mes "Has completado la mision '"+ getarg(1) +"'.";
		next;
		mes "[Spike]";
		mes "Buen trabajo~!!";
		mes "Nuestro cliente esta muy contento con tu trabajo. Apreciara lo que has hecho.";
		mes "Por favor ayudanos de nuevo. Gracias!";
		if (.@item_req)
			delitem .@item_req,.@req_amount;
		erasequest .@quest_id;
		setquest .@quest_id+10;
		getexp getarg(2),0;
		close;
	}
}
