	end;

	.@map$ = strnpcinfo(4);
	if (compare( .@map$,"te_alde" ))
		callsub S_Event, "alde0"+ charat( .@map$,getstrlen(.@map$)-1 ) + "::OnCommandOn";// #popa_alde0[1-5]::OnCommandOn

		callsub S_Event, "prt0"+ charat( .@map$,getstrlen(.@map$)-1 ) + "::OnCommandOn";
	end;

OnReset:
	.@map$ = strnpcinfo(4);
	if (compare( .@map$,"te_alde" ))
		callsub S_Event, "alde0"+ charat( .@map$,getstrlen(.@map$)-1 ) + "::OnReset";

		callsub S_Event, "prt0"+ charat( .@map$,getstrlen(.@map$)-1 ) + "::OnReset";

	.@string$ = getarg(0);
	donpcevent "#popa_"+ .@string$;
	donpcevent "#popb_"+ .@string$;
	donpcevent "#popc_"+ .@string$;
	donpcevent "#popd_"+ .@string$;
	end;
}

			mes "Es una solicitud para encontrar ^4d4dff"+ .@num_items +" o mas de "+ .@string$[2] +"^000000.";
			callsub( S_Acceptquest, .@main_quest, .@npc_name$ );
		}
	case 2:
		mes "["+ .@npc_name$ +"]";
		mes "Ya que la Region TE tiene un ambiente mas casual,";
		mes "no es divertido solo hacer batallas de asedio, verdad?";
		mes "Asi que traje ^4d4dffalgunos trabajos que puedes hacer durante las horas de asedio^000000~";
		next;
		mes "["+ .@npc_name$ +"]";
		mes "Puedes aceptar misiones ^4d4dffdesde 1 hora antes del inicio del asedio y hasta el final de la batalla de asedio TE^000000,";
		mes "y cada mision tiene ^4d4dffun limite de tiempo de 4 horas^000000 para completarse.";
		next;
		mes "["+ .@npc_name$ +"]";
		mes "Simplemente, aceptas una mision mia durante las horas de asedio y";
		mes "^4d4dffregresas a darme un reporte dentro del limite de tiempo^000000.";
		next;
		mes "["+ .@npc_name$ +"]";
		mes "Intentalo por diversion.";
		close;
	case 3:
		mes "["+ .@npc_name$ +"]";
		mes "Ya veo.";
		mes "Regresa cuando quieras si cambias de opinion.";
		mes "Puede que no este disponible, sin embargo. Jejeje.";
		close;
	}

S_Cancelquest:
	.@npc_name$ = getarg(1);
	if (WoeTETimeStart(3600) == 1 || agitcheck3()) {
		mes "Parece que aun no has completado la mision?";
		mes getarg(2);
		next;
		mes "["+ .@npc_name$ +"]";
		mes "Quieres cancelar por casualidad?";
		next;
		switch( select( "No.","Si." ) ) {
		case 1:
			mes "["+ .@npc_name$ +"]";
			mes "Buena suerte!";
			close;
		case 2:
			mes "["+ .@npc_name$ +"]";
			mes "Como deseaste, esta mision ha sido cancelada.";
			erasequest getarg(0);
			close;
		}
	}
	else {
		mes "Hmm aun te queda algo de tiempo, pero no es hora de asedio";
		mes "No creo que debas continuar.";
		mes "Deberias cancelar esta mision. Yo diria que fue fallida.";
		next;
		switch( select( "Me gustaria continuar.","Si, la cancelare." ) ) {
		case 1:
			mes "["+ .@npc_name$ +"]";
			mes "No creo que valga la pena. Bueno, buena suerte!";
			close;
		case 2:
			mes "["+ .@npc_name$ +"]";
			mes "Como deseaste, esta mision ha sido cancelada.";
			erasequest getarg(0);
			close;
		}
	}

S_Acceptquest:
	.@npc_name$ = getarg(1);
	next;
	switch( select( "Aceptarla.","No aceptarla." ) ) {
	case 1:
		mes "["+ .@npc_name$ +"]";
		mes "^4d4dffEl limite de tiempo es de 4 horas.^000000";
		mes "Completa la mision dentro de ese tiempo y reportame.";
		setquest getarg(0);
		close;
	case 2:
		mes "["+ .@npc_name$ +"]";
		mes "Intenta otra mision entonces.";
		close;
	}

S_Rentbox:
	setarray .@item_id[0], 2950,15067,20709,22012;// Rune_Ring, Rune_Suit, Mana_Manteau, Mana_Boots
	for ( .@count = 0; .@count < getarg(0); .@count++ ) {
		.@miss_item01 = rand(100);
		if (.@miss_item01 > 3)
			mes "No encontraste nada.";
		else {
			rentitem .@item_id[ .@miss_item01 ],86400;
			mes "Encontraste '"+ getitemname(.@item_id[ .@miss_item01 ]) +" !'";
		}
	}
	close;

S_Menu:
	for ( .@i = 1; .@i < getargcount(); .@i += 2 ) {
		if ( isbegin_quest(getarg(.@i)) )
			.@menu$ = .@menu$ + getarg(.@i+1) + " ^4d4dffVerificar resultado^000000:";
		else
			.@menu$ = .@menu$ + getarg(.@i+1) + ":";
		set getelementofarray( getarg(0),.@index ), getarg(.@i);
		.@index++;
	}
	return .@menu$;
}
