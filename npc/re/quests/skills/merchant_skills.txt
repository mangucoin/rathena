alberta,47,37,5	script	Cart Remodel Expert	4_M_REPAIR,{
	.@max_skill_lvl = 1;
	if (cart_remodel == 0) {
		mes "[Jason]";
		mes "El carrito original es hermoso!";
		next;
		mes "[Jason]";
		mes "El carrito de color gris solido.";
		next;
		mes "[Jason]";
		mes "Puedo remodelarlo en nuevos disenos lindos.";
		next;
		switch( select( "Que es Decoracion de Carrito?", "Aprender Decoracion de Carrito", "Terminar Dialogo" ) ) {
		case 1:
			mes "[Jason]";
			mes "Me gustaria remodelar completamente el pobre carrito.";
			next;
			mes "[Jason]";
			mes "Una pequena remodelacion no es suficiente, no crees?";
			next;
			mes "[Jason]";
			mes "Por eso he inventado algunos disenos nuevos.";
			next;
			mes "[Jason]";
			mes "Es lindo y refrescante!";
			close;
		case 2:
			if (BaseClass != Job_Merchant) {
				mes "[Jason]";
				mes "Por que quieres aprender Decoracion de Carrito? No me molestes!";
				close;
			}
			if (countitem(6606) == 0) {
				mes "[Jason]";
				mes "Es dificil aprender la Decoracion de Carrito.";
				next;
				mes "[Jason]";
				mes "Te ensenare si me traes el ^3131FFCupon de Remodelacion de Carrito Lindo^000000.";
				close;
			}
			mes "[Jason]";
			mes "Has traido el Cupon de Remodelacion de Carrito Lindo. Estas seguro de que quieres aprender Decoracion de Carrito?";
			next;
			if (select("SI", "NO") == 2) {
				mes "[Jason]";
				mes "Vuelve despues de que hayas cambiado de opinion!";
				close;
			}
			mes "[Jason]";
			mes "De acuerdo! Sigueme.";
			next;
			mes "[Jason]";
			mes "Este diseno";
			mes "necesita agarrar el carrito...";
			next;
			mes "[Jason]";
			mes "Este diseno";
			mes "necesita agarrar el carrito.";
			mes "Luego agregar esto....";
			mes "...";
			next;
			mes "[Jason]";
			mes "Eres un genio! Lo has aprendido!";
			next;
			mes "[Jason]";
			mes "Ahora es tu turno de hacerlo.";
			mes "Presiona Alt+S para cambiar tu carrito!";
			next;
			mes "[Jason]";
			mes "Este es el nuevo diseno lindo del carrito. Ahora eres parte de nuestra familia...!";
			delitem 6606,1;	// P_Cart_C
			cart_remodel = 1;
			skill "MC_CARTDECORATE",1,SKILL_PERM;
			close;
		case 3:
			close;
		}
	}
	mes "[Jason]";
	mes "Has decorado el carrito?";
	next;
	switch( select( "Hay mas que aprender?", "Olvide como decorar mi carrito" ) ) {
	case 1:
		if (cart_remodel == .@max_skill_lvl) {
			mes "[Jason]";
			mes "No. Aun estoy explorando nuevos disenos. Te ensenare cuando lo haya aprendido.";
			close;
		}
		mes "[Jason]";
		mes "Tengo algo que ensenar, quieres aprender?";
		next;
		mes "[Jason]";
		mes "Necesito un ^3131FFCupon de Remodelacion de Carrito Lindo^000000.";
		next;
		if (select("Quiero aprender", "No") == 2) {
			mes "[Jason]";
			mes "Puedes volver cuando quieras.";
			close;
		}
		if (countitem(6606) < 0) {
			mes "[Jason]";
			mes "No tienes ningun Cupon de Remodelacion de Carrito Lindo?";
			close;
		}
		mes "[Jason]";
		mes "Ok, comencemos!";
		next;
		mes "[Jason]";
		mes "Hacerlo asi es mejor.";
		mes "Haz eso asi tambien...";
		next;
		mes "[Jason]";
		mes "....";
		mes "Hagamoslo!";
		next;
		mes "[Jason]";
		mes "Te he ensenado la habilidad, puedes probarla ahora.";
		delitem 6606,1;	// P_Cart_C
		cart_remodel++;
		skill "MC_CARTDECORATE", cart_remodel ,SKILL_PERM;
		close;
	case 2:
		mes "[Jason]";
		mes "Oh? Dejame ver.";
		next;
		if (BaseClass != Job_Merchant) {
			mes "[Jason]";
			mes "Por favor vuelve cuando puedas usar el carrito!";
			close;
		}
		mes "[Jason]";
		mes "No recuerdas lo que te ensene?";
		mes "Te ensene este metodo.";
		next;
		mes "[Jason]";
		mes "Lo has olvidado?";
		mes "Recordemos ahora!";
		next;
		mes "[Jason]";
		mes "Piensa con cuidado, lo recordaras de nuevo.";
		skill "MC_CARTDECORATE",1,SKILL_PERM;
		close;
	}
}
