function	script	mora_remains	{
	if (MaxWeight - Weight < 1000) {
		mes "Tienes que hacer espacio en tu inventario.";
		close;
	}
	if (checkquest(getarg(0)) > -1) {
		specialeffect2 EF_BLIND;
		specialeffect2 EF_BEGINSPELL;
		progressbar "ffff00",4;
		erasequest getarg(0);
		mes "Estos deben ser los Restos de [PROT000]. Asi que reuna con cuidado sus restos.";
		specialeffect2 EF_STEAL;
		getitem getarg(3), rand(1,getarg(4));
		if (rand(10)) {
			next;
			mes "Recogiste '+getarg(2)+' Mora Coins junto a los restos.";
			specialeffect2 EF_STEAL;
			getitem 6380,getarg(2); //Mora_Coin
		}
		close;
	}
	mes "Ves rastros de excavaciones recientes.";
	close;
}
