function	script	F_Brilliant_Soda	{
	next;
	switch( select( "Nivel 70-79", "Nivel 80-89", "Nivel 90+", "Cancelar" ) ) {
	case 1:
		callsub( S_Menu,
			70, "Campo Noreste de Einbroch", "ein_fild06", 253,351,
			71, "Campo Noroeste de Geffen", "gef_fild06",214,31,
			71, "Campo Norte de Lighthalzen", "lhz_fild01", 240,103,
			75, "Isla Tortuga", "tur_dun01",155,46,
			75, "Frente a la Torre Thanatos", "hu_fild01", 137,157,
			75, "Zona de Coral Estelar", "mal_dun01",146,126,
			76, "Campo Noroeste de Einbroch", "ein_fild03",135,50,
			77, "Campo Este de Lighthalzen", "lhz_fild03", 167,340
		);
		return;
	case 2:
		// Official menu text seems wrong
		callsub( S_Menu,
			// 80, "Turtle Island Dungeon", "ra_fild08",164,47,
			80, "Noreste de Rachel", "ra_fild08",164,47,
			80, "Mazmorra de Hielo", "ice_dun01",158,18,
			// 80, "Rachel East Ida Plain", "ra_fild03",363,183,
			80, "Llanura Ida Norte de Rachel", "ra_fild03",363,183,
			85, "Noreste de la llanura de Rachel", "ra_fild04", 269,69,
			// 85, "Rachel Ice Cave 1st Floor", "ve_fild07",158,350,
			85, "Sur de Veins", "ve_fild07",158,350,
			86, "Llanura Ida Norte de Rachel", "ra_fild05", 53,356,
			86, "Pradera Audumrah Noreste Norte de Rachel", "ra_fild01", 229,319,
			// 86, "Veins South Field", "ra_fild12", 54,219,
			86, "Campo Este de Rachel", "ra_fild12", 54,219,
			// 87, "Rachel North North East Audumla Grassland", "tur_dun03", 130,186,
			// 88, "Rachel North East North Audumla Grassland", "tur_dun02", 146,256
			87, "Mazmorra Isla Tortuga 2", "tur_dun03", 130,186,
			88, "Mazmorra Isla Tortuga 3", "tur_dun02", 146,256
		);
		return;
	case 3:
		callsub( S_Menu,
			// 90, "Rachel Southwest Field", "ve_fild01", 354,276,
			90, "Campo Norte Norte de Veins", "ve_fild01", 354,276,
			90, "Campo Norte de Veins", "ve_fild04", 52,247,
			93, "Porto Luna Norte de Lighthalzen", "ra_fild06", 287,39,
			95, "Frente al Volcan Thor", "ve_fild03", 348,226
		);
		return;
	case 4:
		return;
	}
	end;

S_Menu:
	.@count = getargcount();
	.@cancel_index = .@count / 5;

	for ( .@i = 0; .@i < .@count; .@i += 5 )
		.@menu$ += "[" + getarg(.@i) + "] " + getarg(.@i+1) + ":";
	.@menu$ += "Cancelar";

	.@s = select(.@menu$) - 1;
	if (.@s == .@cancel_index)	// cancel
		return;
	.@s = .@s * 5;
	if (BaseLevel < getarg(.@s)) {
		mes " - Disponible solo para niveles superiores a " + getarg(.@s) + " - ";
		close;
	}
	delitem 1000374,1;	// Brilliant_Soda
	warp getarg(.@s+2), getarg(.@s+3),getarg(.@s+4);
	end;
}
