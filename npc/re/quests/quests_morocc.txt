	next;
	mes "[Suspicious Cat]";

		close;

	mes "De ninguna manera! Pase tanto tiempo encontrando ese camino!!";
	next;

	mes "Hmm... pero si insistes, solo dame 5000 Zeny. Te guiare.";
	mes "Que~? No me mires asi! Tienes idea de lo dificil que es para un gato sobrevivir en Midgard!";

	mes "^FF0000Pero pensandolo bien, un debilucho como tu no sobrevivira ahi. Quizas un segundo si tienes suerte?^000000";
	mes "Bueno, no es mi problema. Pero te lo adverti!";
	next;
	switch(select("Pagar 5,000 zeny y partir.:Olvidalo.")) {
	case 1:
		if (Zeny < 5000) {
			mes "[Suspicious Cat]";
			mes "A donde crees que vas! Estas en la ruina!";
			close;
		}
		set Zeny, Zeny-5000;
		mes "[Suspicious Cat]";
		mes "Bah, que pasa con este dinero viejo... no tienes nada brillante o reluciente?!";
		mes "Ahh, solo sigueme!";
		close2;
		warp "moc_prydn1",93,96;
		end;
	case 2:
		mes "[Suspicious Cat]";
		mes "Cielos, ni siquiera tienes 5000 Zeny, verdad?";

	}
}
