function brief;
function mob_brief;
function calcexp;
function calcjobexp;

	if (!checkweight(1201,3)){
		mes "Tienes demasiados objetos para continuar con la mision.";
		mes "Por favor aligera tu carga e intenta de nuevo.";
		close;
	}

	.@zone$ = strnpcinfo(2);
	if(.@zone$ == "gqaldeg"){
		.@zoneid = 3;
		.@edition = 1;
	} else if(.@zone$ == "gqaru"){
		.@zoneid = 4;
		.@edition = 2;
	} else if(.@zone$ == "gqgefg"){
		.@zoneid = 2;
		.@edition = 1;
	} else if(.@zone$ == "gqpayg"){
		.@zoneid = 1;
		.@edition = 1;
	} else if(.@zone$ == "gqprtg"){
		.@zoneid = 0;
		.@edition = 1;
	} else if(.@zone$ == "gqsch"){
		.@zoneid = 5;
		.@edition = 2;
	} else if(.@zone$ == "gqtealde"){
		.@zoneid = 6;
		.@edition = 3;
	} else if(.@zone$ == "gqteprt"){
		.@zoneid = 7;
		.@edition = 3;
	} else {
		// Custom error message.
		debugmes "Guild mission script called from an unknown agit region [" + .@zone$ + "].";
		end;
	}
	.@npcname$ = .npcname$[.@zoneid];

	mes .@npcname$;
	mes "Proporcionamos misiones y misiones diarias que puedes completar en la region de " + .region$[.@zoneid] + ".";
	mes "Hay misiones diarias simples asi como misiones que debes completar solo durante las horas de asedio. Asi que, piensalo~";
	next;
	switch(select("Conquistar " + .region$[.@zoneid], "Misiones Diarias de Mazmorra de Gremio", "Sobre las misiones", "No lo necesito.")){
		case 1: // Conquering.
			.@gid = getcharid(2);
			if(.@gid && is_guild_leader() == true){
				.@time_check1 = checkquest(.conquer_delay_questid[.@zoneid], PLAYTIME);
				if(.@time_check1 == 0){
					mes .@npcname$;
					mes "Ha pasado menos de una hora desde la ultima vez que verificamos!";
					mes "Vuelve cuando el tiempo de espera haya terminado.";
					next;
					mes .@npcname$;
					mes "Si tienes suerte de tu lado, tendras buenos resultados.";
					close;
				} else if(.@time_check1 == 2){
					erasequest .conquer_delay_questid[.@zoneid];
					mes .@npcname$;
					mes "He confirmado que el tiempo de espera para las misiones de conquista ha terminado.";
					mes "Eliminare los registros innecesarios por ti.";
					close;
				} else {
					.@siege_check = (.@edition == 1 ? agitcheck() : .@edition == 2 ? agitcheck2() : agitcheck3());
					.@time = atoi(gettimestr("%H%M",4));
					.@after_time = ((.weekday[.@zoneid] == gettime(4)) && ((.@time > .start_time[.@zoneid]) && (.@time < .end_time[.@zoneid])));

					if(.@siege_check && .@after_time){
						mes .@npcname$;
						mes "Segun ^4d4dffla cantidad de fortalezas que tu gremio ha conquistado en este preciso momento,^000000 te dare una recompensa.";
						mes "Quieres verificar ahora mismo? Sabes que ^4d4dffla cantidad puede cambiar incluso mientras hablamos,^000000 verdad?";
						next;
						if(select("Volvere mas tarde.","Verifiquemos ahora!") == 1){
							mes .@npcname$;
							mes "Bien. La vida se trata de elegir el momento adecuado.";
							mes "Vuelve cuando este brillando al maximo.";
							close;
						} else {
							.@total = 0;
							for(.@i = 0; .@i < 5; ++.@i){
								.@total += (.@gid == getcastledata(.castle$[.@i+.@zoneid*5],CD_GUILD_ID));
							}

							if(!.@total){
								mes .@npcname$;
								mes "Desafortunadamente, no hay ninguna fortaleza que tu gremio haya conquistado.";
								mes "Intenta de nuevo!";
								close;
							} else {
								mes .@npcname$;
								if(.@total == 5)
									mes "Increible.";
								else
									mes "Actualmente, el numero total de fortalezas que tu gremio ha conquistado";
								for(.@i = 0; .@i < 5; ++.@i){
									mes getcastlename(.castle$[.@i+.@zoneid*5]) + (.@gid == getcastledata(.castle$[.@i+.@zoneid*5],CD_GUILD_ID) ? ": ^4d4dffOcupada^000000" : "");
								}
								if(.@total == 5)
									mes "Tu gremio ha conquistado " + .region$[.@zoneid] + "!";
								else
									mes "es " + .@total + "!";
								next;
								mes .@npcname$;
								mes "Te dare una recompensa acorde. Espero que le des buen uso.";
								mes "Tienes una hora de tiempo de espera hasta el proximo reporte. Tenlo en cuenta e intenta de nuevo.";
								setquest .conquer_delay_questid[.@zoneid];
								getitem 6489,.@total;
								getitem 6615,.@total;
								getitem 12888,.@total;
								close;
							}
						}
					} else {
						mes .@npcname$;
						mes "Lo siento pero puedes completar esta mision solo durante las horas de asedio. Me temo que no puedo hacer nada aqui.";
						close;
					}
				}
			} else {
				mes .@npcname$;
				mes "Lo siento pero las misiones de conquista son solo para maestros de gremio.";
				mes "Los miembros deberian intentar las misiones diarias de mazmorra de gremio.";
				close;
			}
			break;
		case 2: // Daily Quests
			mes .@npcname$;
			mes "Estas son misiones que puedes completar en las mazmorras de gremio del area de " + .region$[.@zoneid] + ".";
			mes "Puedes cancelarlas si quieres. Sientete libre de desafiarte.";
			next;
			.@cancel_index = (.@edition == 1 ? 3 : 2);
			for(.@i = 0; .@i < 3; ++.@i){
				if(.daily_hunting_questid[.@zoneid*3 + .@i])
					.@menu$ = .@menu$ + .region$[.@zoneid] + " Mision " + (.@i+1) + (isbegin_quest(.daily_hunting_questid[.@zoneid*3 + .@i]) ? " ^4d4dffVerificar resultado^000000" : "") + ":";
			}
			.@sel = select(.@menu$) - 1;
			.@index = .@zoneid*3 + .@sel;
			if(.@sel == .@cancel_index){
				mes .@npcname$;
				mes "No necesitas apurarte ya que estas misiones se realizan en mazmorras de gremio.";
				mes "Sientete libre de intentarlo cuando quieras.";
				close;
			}
			if(BaseLevel < .level_req[.@index]){
				mes .@npcname$;
				mes "Hmm, desafortunadamente, esta mision esta limitada a usuarios con ^4d4dffnivel " + .level_req[.@index] + "^000000 o superior.";
				mes "Tu nivel no es lo suficientemente alto.";
				close;
			}
			.@hunting = checkquest(.daily_hunting_questid[.@index], HUNTING);
			.@playtime = checkquest(.daily_delay_questid[.@index], PLAYTIME);
			if(.@hunting > -1){
				if(.@hunting == 2){
					mes .@npcname$;
					mes "Oh, hiciste un buen trabajo.";
					mes "No es mucho, pero espero que te sea util.";
					setquest .daily_delay_questid[.@index];
					erasequest .daily_hunting_questid[.@index];
					getexp calcexp(.@zoneid,.@index),calcjobexp(.@edition);
					getitem 6615,1;
				} else {
					mes .@npcname$;
					mes "Hmm? Parece que aun no has completado la mision..";
					mes "Hay algun problema?";
					next;
					if(select("No.","Abandonar la mision.") == 1){
						mes .@npcname$;
						mes "No estoy seguro de cuanto tiempo puedes usar la mazmorra de gremio, pero buena suerte.";
						close;
					} else {
						mes .@npcname$;
						mes "Bueno, nadie te esta obligando.";
						mes "Haz lo que quieras.";
						mes "Cancelare la mision entonces.";
						erasequest .daily_hunting_questid[.@index];
						close;
					}
				}
			} else {
				if(.@playtime == 0 || .@playtime == 1){
					mes .@npcname$;
					mes "La regla aqui establece que solo puedes completar una mision por dia.";
					mes "Me gustaria que volvieras cuando el tiempo de espera haya terminado.";
					close;
				} else if(.@playtime == 2){
					erasequest .daily_delay_questid[.@index];
					mes .@npcname$;
					mes "He confirmado que el tiempo de espera para esta mision ha terminado.";
					mes "Eliminare los registros usados para la confirmacion.";
					mes "Ahora, intenta completar la mision!";
					close;
				} else {
					mes .@npcname$;
					mes "Esta es la " + .ordinal$[.@sel] + " mision en " + .region$[.@zoneid] + ".";
					brief(.daily_hunting_questid[.@index]);
					next;
					mes .@npcname$;
					if(mob_brief(.daily_hunting_questid[.@index]))
						mes "Te recompensare acorde al completar esta mision.";
					else
						mes "Te dare puntos de experiencia a cambio.";
					next;
					if(select("Aceptar.","Rechazar.") == 1){
						mes .@npcname$;
						mes "La entrada a la mazmorra de gremio esta restringida. Pero no es que no puedas encontrar una forma de ir ahi tampoco.";
						mes "Rezare por tu exito luchando contra esos monstruos.";
						setquest .daily_hunting_questid[.@index];
						close;
					} else {
						mes .@npcname$;
						mes "Bueno, haz lo que desees.";
						close;
					}
				}
			}
			break;
		case 3: // About quests
			mes .@npcname$;
			mes "No es dificil en absoluto.";
			mes "Doy una recompensa ^4d4dffdependiendo de cuantas fortalezas hayas conquistado.^000000";
			mes "^4d4dffEl tipo y calidad de la recompensa dependen del numero de fortalezas conquistadas en el momento exacto en que verifico.^000000";
			next;
			mes .@npcname$;
			mes "Y por supuesto, verifico solo durante las horas de asedio.";
			mes "^4d4dffRecibir una recompensa va seguido de una hora de tiempo de espera. Asi que sera mejor que calcules bien el momento.^000000";
			next;
			mes .@npcname$;
			mes "Si eres maestro de un gremio que participa actualmente en batallas de asedio, deberias intentarlo.";
			next;
			mes .@npcname$;
			mes "Las misiones diarias de mazmorra de gremio consisten en misiones de combate que puedes completar en mazmorras de gremio.";
			mes "Es simple.";
			close;
		case 4: // I don't need it.
			mes .@npcname$;
			mes "Eres realmente alguien especial~!";
			close;
	}
	end;

	function calcexp {
		.@zoneid = getarg(0);
		.@index = getarg(1);
		.@maxlevel = .max_level[.@index];
		.@isboss = (.@zoneid < 4 && !((.@index+1)%3));

		if(!.@isboss && BaseLevel > .@maxlevel){
			return 1000 * (100 + ((.@maxlevel - 100) * 3));
		} else {
			return 1000 * (100 + ((BaseLevel - 100) * 3)) * (1 + .@isboss);
		}
	}

	function calcjobexp {
		.@edition = getarg(0);

		if(.@edition == 3){
			if(jobcanentermap("te_prtcas01")){
				if (JobLevel > 50)
					return 1000 * (50 + ((JobLevel - 50) * 3));
				else
					return 5000;
			} else {
				debugmes "Unexpected case for Job: " + Class + "(" + jobname(Class) + ") for gld_mission#core::calcjobexp";
				return 0;
			}
		} else if(.@edition == 2 ) {
			if(JobLevel > 50)
				return 1000 * (50 + ((JobLevel - 50) * 3));
			else
				return 3000;
		} else {
			return 5000 * JobLevel;
		}
		return 0;
	}
