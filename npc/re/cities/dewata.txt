alberta,212,202,4	script	Dewata Sailor#alberta	536,{
	mes "[Dewata Sailor]";
	mes "Quieres visitar la ^8B4513Isla Dewata^000000?";
	mes "Con sus deslumbrantes olas y encantadoras vistas es un gran lugar para relajarse.";
	if (VIP_SCRIPT) {
		set .@cost, vip_status(VIP_STATUS_ACTIVE)?1000:10000;
		mes "La tarifa de transito es 10,000 Zeny y 1,000 para VIP.";
	} else {
		set .@cost,10000;
		mes "La tarifa de transito es 10,000 Zeny.";
	}
	next;
	switch(select("Si!:No.")) {
	case 1:
		if (Zeny >= .@cost) {
			mes "[Dewata Sailor]";
			mes "^8B4513Dewata^000000 es una hermosa y pacifica isla.";
			mes "Buen viaje~";
			close2;
			set Zeny, Zeny - .@cost;
			warp "dewata",232,53;
			end;
		} else {
			mes "[Dewata Sailor]";
			mes "No tienes la plata, chico.";
			mes "Vuelve cuando tengas mas dinero.";
			close;
		}
	case 2:
		mes "[Dewata Sailor]";
		mes "Avisame";
		mes "cuando estes listo para viajar~";
		close;
	}
}
