		mes "Eres un heroe de la flota de gatos. Tu actividad es leyenda entre nosotros.";
		next;
		mes "[Odgnalam]";
		mes "Como eres un heroe de la flota de gatos dejame enviarte a ^A2314BMalangdo^000000 gratis.";
		next;
		if(select("Enviame a Malangdo!:No ir.") == 1) {
			mes "[Odgnalam]";
			mes "Ten un viaje comodo... Muy bien, vamos~";
			warp "malangdo",217,85;
		}

	mes .@s1$;

		mes "[Odgnalam]";

	close;
}

		set .@yes_money$, "Buen viaje y vuelve pronto.";

		mes "Wow~ Contribuiste mucho a nuestra flota. Muy bien!! Precio especial con descuento 500 zeny. A donde quieres ir?";
		next;
	} else {
		set .@price,0;
		set .@s1$,"Gratis para el heroe de nuestra flota!!";
		set .@s2$, "Nadie se opone a que el heroe de nuestra flota se quede en Malangdo...";
		mes "[Kong]";
		mes "Ah... Eres el legendario heroe de nuestra flota de gatos.";
		next;
		mes "[Kong]";
		mes "No podemos cobrarle al heroe de nuestra flota. Cual es tu destino?";
		next;
	}
	set .@i, select("- Destino Izlude --- "+.@s1$+":- Destino Alberta --- "+.@s1$+":- Quedarse en Malangdo");
	mes "[Kong]";
	if (.@i == 3) {
		mes .@s2$;
		close;
	}
	if (ep13_yong1 > 80) {
		mes "Ten un viaje comodo.";
		if (.@i == 1)
			warp "izlude",195,212; // Old coordinates: (194,180)
		else
			warp "alberta",192,150;
		close;
	}
	if (Zeny < .@price) {
		mes .@no_money$;
		close;
	}
	set Zeny, Zeny - .@price;
	mes .@yes_money$;
	if (.@i == 1)
		warp "izlude",195,212;
	else
		warp "alberta",192,150;
	close;
}
