// Channel Warper :: channel
//============================================================
function	script	F_IzludeChannel	{
	mes "[Channel Warp Official]";
	mes "Izlude es un lugar muy animado, no crees?";
	mes "Estoy aqui para ayudar a que Izlude se sienta mas fluido!";
	mes "Puedo enviarte a otra copia de Izlude.";
	mes "Quieres ir a otra copia?";
	next;
	set .@i, select(getarg(0)+":No, gracias");
	if (.@i < 6) {
		setarray .@maps$[1],"izlude","izlude_a","izlude_b","izlude_c","izlude_d";
		savepoint .@maps$[.@i],128,98,1,1;
		warp .@maps$[.@i],128,98;
		end;
	}
	close;
}
