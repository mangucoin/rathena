ecl_in02,164,56,3	script	Receptionist#Laphine	835,{
	mes "[Receptionist]";
	mes "Bienvenido a la Clinica de Eclage.";
	next;
	switch (select("Guardar ubicacion:Descansar:Terminar la conversacion")) {
	case 1:
		mes "[Receptionist]";
		mes "La clinica ha sido establecida como tu punto de guardado. Ten cuidado~";
		savepoint "ecl_in02",162,50,1,1;
		close;
	case 2:
		mes "[Receptionist]";
		mes "Restaura tu mente y cuerpo en nuestra clinica. Esta llena de la energia pura de Eclage.";
		mes "Normalmente cobramos 1 Moneda de Splendide, pero ofrecemos un descuento especial para viajeros. Tu total es 5,000 zeny. ";
		next;
		switch (select("Pagar con Moneda de Splendide:Pagar con Zeny:Olvidalo")) {
		case 1:
			if (!countitem(6081)) {
				mes "[Receptionist]";
				mes "Parece que no tienes suficientes monedas. Podrias revisar tu cuenta de nuevo, por favor?";
				close;
			}
			mes "[Receptionist]";
			mes "Sientete como en casa~";
			close2;
			delitem 6081,1; //6081
			percentheal 100,100;
			warp "ecl_in02",167,49;
			end;
		case 2:
			if (Zeny < 5000) {
				mes "[Receptionist]";
				mes "Parece que no tienes suficientes recursos para continuar. Te importaria revisar tu cuenta de nuevo?";
				close;
			}
			mes "[Receptionist]";
			mes "Sientete como en casa~";
			close2;
			Zeny = Zeny - 5000;
			percentheal 100,100;
			warp "ecl_in02",167,49;
			end;
		case 3:
			close;
		}
	case 3:
		close;
	}
}
