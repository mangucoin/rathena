moscovia,161,76,3	script	Moscovia Guide#01mosc	959,{
	mes "[Moscovia Guide]";
	mes "Un paraiso en el mar,";
	mes "hermosas olas y calido sol.";
	mes "^8B4513Moscovia^000000 te esta esperando.";
	mes "Estoy aqui para ofrecer informacion sobre este lugar.";
	mes "Si quieres preguntar algo, no dudes en decirme.";
	F_Navi("[Moscovia Guide]");
	set .@str$,"Hay algo mas en lo que pueda ayudarte?";
	while (1) {
		next;
		switch(select("[ Empleada Kafra ]:[ Palacio ]:[ Capilla ]:[ Tienda de Armas ]:[ Tienda de Armaduras ]:[ Tienda de Herramientas ]:[ Posada ]:[ Regresar a Alberta ]:Quitar Marcas del Mini-Mapa:Cancelar")) {
		case 1:
			mes "[Moscovia Guide]";
			mes "La ^0000FFEmpleada Kafra^000000 ha";
			mes "sido marcada en tu mini-mapa.";
			mes "Revisa tu mini-mapa.";
			mes "Necesitas algo mas?";
			viewpoint 1,223,191,0,0x0A82FF;
			break;
		case 2:
			mes "[Moscovia Guide]";
			mes "En el "+F_Navi("Palacio","moscovia,255,140","^006400")+" encontraras a nuestro Senor el Zar y otros sirvientes.";
			mes .@str$;
			viewpoint 1,255,140,1,0x0A82FF;
			break;
		case 3:
			mes "[Moscovia Guide]";
			mes "La "+F_Navi("Capilla","moscovia,253,179","^008080")+" se encuentra aqui.";
			mes .@str$;
			viewpoint 1,253,179,2,0xAAFF00;
			break;
		case 4:
			mes "[Moscovia Guide]";
			mes "Puedes comprar armas de los mejores artesanos de Moscovia si vas a la "+F_Navi("Tienda de Armas","moscovia,186,191","^FF1493")+".";
			mes .@str$;
			viewpoint 1,186,191,3,0x008080;
			break;
		case 5:
			mes "[Moscovia Guide]";
			mes "Solo encontraras el mejor equipo en la "+F_Navi("Tienda de Armaduras","moscovia,204,169","^8B4513")+" de Moscovia.";
			mes .@str$;
			viewpoint 1,204,169,4,0xB200FF;
			break;
		case 6:
			mes "[Moscovia Guide]";
			mes "La "+F_Navi("Tienda de Herramientas","moscovia,221,176","^FF1493")+" esta ubicada justo al sur del centro del pueblo.";
			mes "Puedes encontrar todo tipo de cosas que necesites para tus viajes.";
			mes .@str$;
			viewpoint 1,221,176,5,0xFF1493;
			break;
		case 7:
			mes "[Moscovia Guide]";
			mes "La Posada "+F_Navi("Sticky Herb Tree","moscovia,227,206")+" esta justo al norte del centro del pueblo.";
			mes "Si necesitas descansar, no hay mejor lugar donde quedarse.";
			mes .@str$;
			viewpoint 1,227,206,6,0x8B4513;
			break;
		case 8:
			mes "[Moscovia Guide]";
			mes "Puedes encontrar al";
			mes F_Navi("Oficial de R.P. de Moscovia","moscovia,166,53")+" en los muelles si";
			mes "quieres regresar a Alberta.";
			viewpoint 1,166,53,7,0x00BFFF;
			break;
		case 9:
			mes "[Moscovia Guide]";
			mes "Quitare todas las marcas de tu mini-mapa.";
			mes .@str$;
			viewpoint 2,1,1,0,0xFFFFFF;
			viewpoint 2,1,1,1,0xFFFFFF;
			viewpoint 2,1,1,2,0xFFFFFF;
			viewpoint 2,1,1,3,0xFFFFFF;
			viewpoint 2,1,1,4,0xFFFFFF;
			viewpoint 2,1,1,5,0xFFFFFF;
			viewpoint 2,1,1,6,0xFFFFFF;
			viewpoint 2,1,1,7,0xFFFFFF;
			break;
		case 10:
			mes "[Moscovia Guide]";
			mes "Disfruta tu viaje~!";
			close;
		}
	}
}
moscovia,226,191,4	duplicate(Moscovia Guide#01mosc)	Moscovia Guide#02mosc	959
