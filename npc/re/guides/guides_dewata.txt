dewata,202,106,4	script	Dewata Guide#dewata01	4_M_DEWWOMAN,{
	mes "[Dewata Guide]";
	mes "Bienvenido a la ^8B4513Isla Dewata^000000,";
	mes "un lugar con olas deslumbrantes y vistas encantadoras, es un gran lugar para relajarse.";
	mes "Si tienes alguna pregunta, por favor consultame.";
	next;
	mes "[Dewata Guide]";
	mes "A donde puedo guiarte?";
	F_Navi("[Dewata Guide]");
	while (1) {
		next;
		switch(select("[ Empleada Kafra ]:[ Vendedor de Armas ]:[ Vendedor de Armaduras ]:[ Vendedor de Herramientas ]:[ Restaurante ]:[ Guardian de Krakatau ]:[ Marinero de Alberta ]:Quitar Marcas del Mini-Mapa:Cancelar")) {
		case 1:
			callsub L_Mark, "^0000FFEmpleada Kafra^000000";
			viewpoint 1,202,184,0,0x000FFF;
			break;
		case 2:
			callsub L_Mark, F_Navi("Vendedor de Armas","dewata,218,164","^008080");
			viewpoint 1,218,163,1,0x008080;
			break;
		case 3:
			callsub L_Mark, F_Navi("Vendedor de Armaduras","dewata,158,182","^008080");
			viewpoint 1,158,182,2,0x008080;
			break;
		case 4:
			callsub L_Mark, F_Navi("Vendedor de Herramientas","dewata,182,164","^FF1493");
			viewpoint 1,182,164,3,0xFF1493;
			break;
		case 5:
			callsub L_Mark, F_Navi("Restaurante","dewata,146,109","^FF1493");
			viewpoint 1,146,109,4,0xFF1493;
			break;
		case 6:
			callsub L_Mark, F_Navi("Guardian de Krakau","dewata,235,56","^8A2BE2");
			viewpoint 1,235,56,5,0x8A2BE2;
			break;
		case 7:
			callsub L_Mark, F_Navi("Marinero de Alberta","dewata,229,49","^00BFFF");
			viewpoint 1,229,49,6,0x00BFFF;
			break;
		case 8:
			mes "[Dewata Guide]";
			mes "Quitare todas las marcas de tu mini-mapa.";
			mes "Hay algo mas en lo que pueda ayudarte?";
			viewpoint 2,202,184,0,0xFFFFFF;
			viewpoint 2,218,163,1,0xFFFFFF;
			viewpoint 2,158,182,2,0xFFFFFF;
			viewpoint 2,182,164,3,0xFFFFFF;
			viewpoint 2,232,59,4,0xFFFFFF;
			viewpoint 2,195,235,5,0xFFFFFF;
			viewpoint 2,230,52,6,0xFFFFFF;
			break;
		case 9:
			mes "[Dewata Guide]";
			mes "Disfruta tu viaje!!";
			close;
		}
	}

L_Mark:
	mes "[Dewata Guide]";
	mes "La ubicacion de "+getarg(0)+" esta";
	mes "marcada en tu mini-mapa.";
	mes "Revisa tu mini-mapa.";
	mes "Hay algo mas en lo que pueda ayudarte?";
	return;
}
