harboro1,356,211,5	script	Guide#rockridge01	4_F_ANYA,{
	mes "[Sheriff Assistant]";
	mes "Bienvenido a Rock Ridge, el caluroso y arido desierto del Oeste donde la aventura te espera!";
	mes "Soy la asistente del sheriff, y estoy aqui para guiar a los visitantes. Hay algun lugar al que quieras ir?";
	next;
	mes "[Sheriff Assistant]";
	mes "Dime a donde quieres ir y te mostrare donde esta.";
	mes "Ademas, puedes usar el modo de ^B9062FNavegacion^000000 haciendo ^B9062Fclic en el nombre del lugar^000000 que te muestre.";
	while(true) {
		next;
		.@s = select( "[Empleada Kafra]", "[Posada]", "[Tienda de Armas/Armaduras]", "[Tienda de Herramientas]", "[Oficina del Sheriff]", "Limpiar mini-mapa.", "Cancelar." );
		switch(.@s) {
		case 1:
			mes "[Sheriff Assistant]";
			mes "La ^006400Empleada Kafra^000000, entendido. Por favor haz clic en el ^0000FFicono de Caja^000000 en tu mini-mapa.";
			viewpoint 1,291,207,1,0xFFFF33;
			break;
		case 2:
			mes "[Sheriff Assistant]";
			mes "He marcado la ubicacion de la <NAVI>Posada<INFO>harboro1,156,215,0,000,0</INFO></NAVI> en tu mini-mapa.";
			mes "Haz clic en el nombre para activar el modo de Navegacion.";
			viewpoint 1,156,215,2,0xAAFF00;
			break;
		case 3:
			mes "[Sheriff Assistant]";
			mes "He marcado la ubicacion de la <NAVI>Tienda de Armas/Armaduras<INFO>harboro1,312,193,0,000,0</INFO></NAVI> en tu mini-mapa.";
			mes "Haz clic en el nombre para activar el modo de Navegacion.";
			viewpoint 1,312,193,3,0x008080;
			break;
		case 4:
			mes "[Sheriff Assistant]";
			mes "He marcado la ubicacion de la <NAVI>Tienda de Herramientas<INFO>harboro1,312,193,0,000,0</INFO></NAVI> en tu mini-mapa.";
			mes "Haz clic en el nombre para activar el modo de Navegacion.";
			viewpoint 1,312,193,4,0xFF1493;
			break;
		case 5:
			mes "[Sheriff Assistant]";
			mes "He marcado la ubicacion de la <NAVI>Oficina del Sheriff<INFO>harboro1,241,218,0,000,0</INFO></NAVI> en tu mini-mapa.";
			mes "Haz clic en el nombre para activar el modo de Navegacion.";
			viewpoint 1,241,218,5,0xFFCCFF;
			break;
		case 6:
			mes "[Sheriff Assistant]";
			mes "Dejame quitar todas las marcas de tu mini-mapa.";
			viewpoint 2,291,207,1,0xFFFF33;
			viewpoint 2,156,215,2,0xAAFF00;
			viewpoint 2,312,193,3,0x008080;
			viewpoint 2,312,193,4,0xFF1493;
			viewpoint 2,241,218,5,0xFFCCFF;
			break;
		case 7:
			mes "[Sheriff Assistant]";
			mes "Disfruta tu estancia en Rock Ridge!";
			close;
		}
		if (.@s == 7)
			mes "Necesitabas algo mas?";
		else
			mes "Hay algun otro lugar que quieras revisar?";
	}
}
