aldebaran,139,63,4	script	Guide#01aldebaran::GuideAldearan	105,{
	cutin "prt_soldier",2;
	mes "[Al De Baran Guide]";
	mes "Bienvenido a ^8B4513Al De Baran^000000,";
	mes "la Ciudad Fronteriza.";
	mes "Necesitas ayuda para navegar por la ciudad?";
	F_Navi("[Al De Baran Guide]");
	next;
	set .@str$,"Te gustaria revisar alguna otra ubicacion?";
	while (1) {
		switch(select("[ Instalaciones Principales ]:[ Comerciantes y Ayudantes ]:Quitar Marcas del Mini-Mapa:Cancelar")) {
		case 1:
			set .@loop,1;
			while (.@loop) {
				switch(select("[ Sede Kafra ]:[ Gremio de Alquimistas ]:[ Torre del Reloj ]:[ Posada y Tienda de Herramientas ]:[ Reparar y Comprar Equipo ]:[ Taberna ]:Menu Anterior")) {
				case 1:
					callsub L_Mark, F_Navi("Sede Kafra","aldebaran,61,229","^B9062F");
					mes .@str$;
					viewpoint 1,61,229,0,0xFF0000;
					next;
					break;
				case 2:
					callsub L_Mark, F_Navi("Gremio de Alquimistas","aldebaran,60,60","^0000FF");
					mes .@str$;
					viewpoint 1,60,60,1,0x0A82FF;
					next;
					break;
				case 3:
					callsub L_Mark, F_Navi("Torre del Reloj","aldebaran,139,133","^FF5400");
					mes .@str$;
					viewpoint 1,139,133,2,0xFFB400;
					next;
					break;
				case 4:
					mes "[Al De Baran Guide]";
					mes "Dejame marcar la ubicacion del";
					mes "edificio de la "+F_Navi("Posada y Tienda de Herramientas","aldebaran,197,70","^006400");
					mes "en tu mini-mapa.";
					mes .@str$;
					viewpoint 1,197,70,3,0xAAFF00;
					next;
					break;
				case 5:
					mes "[Al De Baran Guide]";
					mes "Si quieres";
					mes F_Navi("Comprar o Reparar Equipo","aldebaran,197,70","^7F3300")+",";
					mes "entonces este es el lugar que querras visitar.";
					mes .@str$;
					viewpoint 1,72,197,4,0xD2691E;
					next;
					break;
				case 6:
					callsub L_Mark, F_Navi("Taberna","aldebaran,231,106","^800080");
					viewpoint 1,231,106,5,0xDA70D6;
					next;
					break;
				case 7:
					set .@loop,0;
					break;
				}
			}
			break;
		case 2:
			set .@loop,1;
			while (.@loop) {
				switch(select("[ Oficial de Teletransporte Eden ]:[ Empleada Kafra ]:[ Sr. Claus ]:[ Asistente de Chef ]:Menu Anterior")) {
				case 1:
					callsub L_Mark, F_Navi("Oficial de Teletransporte Eden","aldebaran,133,119","^B9062F");
					mes .@str$;
					viewpoint 1,133,119,6,0xFF0000;
					next;
					break;
				case 2:
					callsub L_Mark, "^0000FFEmpleada Kafra^000000";
					viewpoint 1,143,119,7,0x0A82FF;
					next;
					break;
				case 3:
					mes "[Al De Baran Guide]";
					mes "El Sr. Claus puede teletransportarte";
					mes "a "+F_Navi("Lutie","aldebaran,168,168","^006400")+".";
					mes "Dejame marcar su ubicacion";
					mes "en tu mini-mapa.";
					mes .@str$;
					viewpoint 1,168,168,8,0xAAFF00;
					next;
					break;
				case 4:
					callsub L_Mark, F_Navi("Asistente de Chef","aldebaran,165,107","^7F3300");
					mes .@str$;
					viewpoint 1,165,107,9,0xD2691E;
					next;
					break;
				case 5:
					set .@loop,0;
					break;
				}
			}
			break;
		case 3:
			mes "[Al De Baran Guide]";
			mes "Claro, quitare todas las marcas de tu mini-mapa.";
			mes "Hay algo mas en lo que pueda ayudarte?";
			viewpoint 2,1,1,0,0xFFFF00;
			viewpoint 2,1,1,1,0x000000;
			viewpoint 2,1,1,2,0xFF0000;
			viewpoint 2,1,1,3,0xFFFF00;
			viewpoint 2,1,1,4,0xFFFF00;
			viewpoint 2,1,1,5,0xFFFF00;
			viewpoint 2,1,1,6,0xFFFF00;
			viewpoint 2,1,1,7,0xFFFF00;
			viewpoint 2,1,1,8,0xFFFF00;
			viewpoint 2,1,1,9,0xFFFF00;
			next;
			break;
		case 4:
			mes "[Al De Baran Guide]";
			mes "Muy bien entonces,";
			mes "disfruta tu estancia!";
			close2;
			cutin "prt_soldier",255;
			end;
		}
	}
	end;
L_Mark:
	mes "[Al De Baran Guide]";
	mes "Dejame marcar la ubicacion de";
	mes getarg(0);
	mes "en tu mini-mapa.";
	return;
}
