gonryun,163,60,4	script	Kunlun Guide#01gonryun	780,{
	mes "[He Yuen Zhe]";
	mes "Ni Hao!";
	mes "Bienvenido a ^8B4513Kunlun^000000~";
	mes "Da un paseo y experimenta";
	mes "la antigua historia y tradicion";
	mes "de nuestra impresionante ciudad.";
	next;
	mes "[He Yuen Zhe]";
	mes "Soy el responsable de ayudarte";
	mes "con cualquier pregunta que tengas.";
	mes "No dudes en preguntarme lo que sea.";
	F_Navi("[He Yuen Zhe]");
	while (1) {
		next;
		switch(select("[ Empleada Kafra ]:[ Residencia del Jefe ]:[ Vendedor de Herramientas ]:[ Vendedor de Armas ]:[ Vendedor de Armaduras ]:[ Productor de Vinos ]:[ Enviado de Kunlun ]:Quitar Marcas del Mini-Mapa:Cancelar")) {
		case 1:
			callsub L_Mark, "^0000FFEmpleada Kafra^000000";
			viewpoint 1,159,122,0,0x0A82FF;
			break;
		case 2:
			callsub L_Mark, F_Navi("Residencia del Jefe","gonryun,110,131","^006400");
			viewpoint 1,110,131,1,0xAAFF00;
			break;
		case 3:
			callsub L_Mark, F_Navi("Vendedor de Herramientas","gonryun,147,84","^D8BFD8");
			viewpoint 1,147,84,2,0xD8BFD8;
			break;
		case 4:
			callsub L_Mark, F_Navi("Vendedor de Armas","gonryun,174,101","^FF1493");
			viewpoint 1,174,101,3,0xFF1493;
			break;
		case 5:
			callsub L_Mark, F_Navi("Vendedor de Armaduras","gonryun,173,84","^8B4513");
			viewpoint 1,173,84,4,0x8B4513;
			break;
		case 6:
			callsub L_Mark, F_Navi("Productor de Vinos","gonryun,213,115","^9400D3");
			viewpoint 1,213,115,5,0x9400D3;
			break;
		case 7:
			callsub L_Mark, F_Navi("Enviado de Kunlun","gonryun,161,11","^00BFFF");
			viewpoint 1,161,11,6,0x00BFFF;
			break;
		case 8:
			mes "[He Yuen Zhe]";
			mes "Quitare todas las marcas de tu mini-mapa.";
			mes "Hay algo mas en lo que pueda ayudarte?";
			viewpoint 2,1,1,0,0xFFFFFF;
			viewpoint 2,1,1,1,0xFFFFFF;
			viewpoint 2,1,1,2,0xFFFFFF;
			viewpoint 2,1,1,3,0xFFFFFF;
			viewpoint 2,1,1,4,0xFFFFFF;
			viewpoint 2,1,1,5,0xFFFFFF;
			viewpoint 2,1,1,6,0xFFFFFF;
			break;
		case 9:
			mes "[He Yuen Zhe]";
			mes "Buen viaje!";
			close;
		}
	}
	end;
L_Mark:
	mes "[He Yuen Zhe]";
	mes "La ubicacion de "+getarg(0)+" esta";
	mes "marcada en tu mini-mapa.";
	mes "Revisa tu mini-mapa.";
	mes "Hay algo mas en lo que pueda ayudarte?";
	return;
}
