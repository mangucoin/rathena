xmas,140,137,3	script	Lutie Guide#xmas	717,{
	mes "[Lutie Guide]";
	mes "Bienvenido a ^8B4513Lutie^000000,";
	mes "la Aldea de la Nieve.";
	mes "Necesitas ayuda para navegar por la aldea?";
	F_Navi("[Lutie Guide]");
	while (1) {
		next;
		switch(select("[ Tienda de Regalos ]:[ Iglesia ]:[ Tienda de Armas y Armaduras ]:[ Tienda de Herramientas ]:[ Casa de Santa ]:[ Fabrica de Juguetes ]:Quitar Marcas del Mini-Mapa:Cancelar")) {
		case 1:
			callsub L_Mark, F_Navi("Tienda de Regalos","xmas,172,131","^008080");
			viewpoint 1,172,131,2,0x008080;
			break;
		case 2:
			callsub L_Mark, F_Navi("Iglesia","xmas,104,287","^006400");
			viewpoint 1,104,287,1,0xAAFF00;
			break;
		case 3:
			callsub L_Mark, F_Navi("Tienda de Armas y Armaduras","xmas,171,158","^FF1493");
			viewpoint 1,171,158,3,0xFF1493;
			break;
		case 4:
			callsub L_Mark, F_Navi("Tienda de Herramientas","xmas,122,131","^8B4513");
			viewpoint 1,122,131,4,0x8B4513;
			break;
		case 5:
			callsub L_Mark, F_Navi("Casa de Santa","xmas,149,237","^B9062F"),1;
			viewpoint 1,149,237,5,0x9400D3;
			break;
		case 6:
			callsub L_Mark, F_Navi("Fabrica de Juguetes","xmas,143,312","^9400D3");
			viewpoint 1,143,312,6,0xFF0000;
			break;
		case 7:
			mes "[Lutie Guide]";
			mes "Claro, quitare todas las marcas de tu mini-mapa.";
			mes "Hay algo mas en lo que pueda ayudarte?";
			viewpoint 2,1,1,0,0xFFFFFF;
			viewpoint 2,1,1,1,0xFFFFFF;
			viewpoint 2,1,1,2,0xFFFFFF;
			viewpoint 2,1,1,3,0xFFFFFF;
			viewpoint 2,1,1,4,0xFFFFFF;
			viewpoint 2,1,1,5,0xFFFFFF;
			viewpoint 2,1,1,6,0xFFFFFF;
			break;
		case 8:
			mes "[Lutie Guide]";
			mes "Cuidate en tus viajes.";
			close;
		}
	}
	end;
L_Mark:
	mes "[Lutie Guide]";
	mes "Dejame marcar la ubicacion de "+((getarg(1,0))?"":"la");
	mes getarg(0)+"^000000";
	mes "en tu mini-mapa.";
	mes "Te gustaria revisar alguna otra ubicacion?";
	return;
}
