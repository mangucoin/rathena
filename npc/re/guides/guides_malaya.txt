malaya,71,72,4	script	Port Malaya Guide#01::GuideMalaya	579,{
	mes "[Port Malaya Guide]";
	mes "Bienvenido a ^8B4513Port Malaya^000000, la Aldea Viviente.";
	mes "Donde las culturas y leyendas cobran vida!";
	mes "Necesitas ayuda para navegar por la aldea?";
	F_Navi("[Port Malaya Guide]");
	while(1) {
		next;
		switch(select("[ Empleadas Kafra ]:[ Posada y Taberna ]:[ Tienda de Armas y Armaduras ]:[ Tienda de Herramientas ]:[ Hospital ]:[ Parque ]:[ Puerto ]:Quitar Marcas del Mini-Mapa:Cancelar")) {
		case 1:
			callsub L_Mark, "^0000FFEmpleadas Kafra^000000",1;
			viewpoint 1,71,79,0,0x0A82FF;
			viewpoint 1,234,204,1,0x0A82FF;
			break;
		case 2:
			callsub L_Mark, F_Navi("Posada y Taberna","malaya,193,215","^006400");
			viewpoint 1,193,215,2,0xAAFF00;
			break;
		case 3:
			callsub L_Mark, F_Navi("Tienda de Armas y Armaduras","malaya,114,212","^008080");
			viewpoint 1,114,212,3,0x4F543C;
			break;
		case 4:
			callsub L_Mark, F_Navi("Tienda de Herramientas","malaya,297,167","^FF1493");
			viewpoint 1,297,167,4,0xFF1493;
			break;
		case 5:
			callsub L_Mark, F_Navi("Hospital","malaya,50,75","^8B4513");
			viewpoint 1,50,75,5,0x8B4513;
			break;
		case 6:
			callsub L_Mark, F_Navi("Parque","malaya,280,371","^8B4513");
			viewpoint 1,280,371,6,0x8B4513;
			break;
		case 7:
			callsub L_Mark, F_Navi("Puerto","malaya,276,55","^8A2BE2");
			viewpoint 1,276,55,7,0x8A2BE2;
			break;
		case 8:
			mes "[Port Malaya Guide]";
			mes "Quitare todas las marcas de tu mini-mapa.";
			mes "Hay algo mas en lo que pueda ayudarte?";
			viewpoint 2,1,1,0,0xFFFFFF;

			viewpoint 2,1,1,5,0xFFFFFF;
			viewpoint 2,1,1,6,0xFFFFFF;
			viewpoint 2,1,1,7,0xFFFFFF;
			break;
		case 9:
			mes "[Port Malaya Guide]";
			mes "Espero que disfrutes tu viaje.";
			close;
		}
	}
	end;
L_Mark:
	mes "[Port Malaya Guide]";
	mes "La ubicacion de "+getarg(0)+" "+((getarg(1,0))?"estan":"esta");

	mes "Revisa tu mini-mapa.";
	mes "Hay algo mas en lo que pueda ayudarte?";
	return;
}
