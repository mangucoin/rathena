niflheim,201,187,3	script	Niflheim Guide#01nif	849,{
	mes "[Niflheim Guide]";
	mes "Bienvenido a.. ^8B4513Niflheim^000000...";
	mes "el Reino de los Muertos...";
	mes "Necesitas ayuda para navegar por el reino...?";
	if (F_Navi() == 0) {
		next;
		mes "[Niflheim Guide]";
		mes "Cuando hagas clic en el.. ^B9062F[nombre de la ubicacion]^000000... recibiras los servicios de ^B9062FNavegacion^000000 mas avanzados...!";
	}
	while (1) {
		next;
		switch(select("[ Tienda de Herramientas ]:[ Taberna ]:[ Tienda de Armas ]:[ Castillo de la Bruja ]:Quitar Marcas del Mini-Mapa:Cancelar")) {
		case 1:
			callsub L_Mark, F_Navi("Tienda de Herramientas","niflheim,218,196","^0000FF");
			viewpoint 1,218,196,0,0x0A82FF;
			break;
		case 2:
			callsub L_Mark, F_Navi("Taberna","niflheim,189,208","^006400");
			viewpoint 1,189,208,1,0xAAFF00;
			break;
		case 3:
			callsub L_Mark, F_Navi("Tienda de Armas","niflheim,218,170","^800080");
			viewpoint 1,218,170,2,0xDA70D6;
			break;
		case 4:
			callsub L_Mark, F_Navi("Castillo de la Bruja","niflheim,255,195","^FF1493");
			viewpoint 1,255,195,3,0xFF1493;
			break;
		case 5:
			mes "[Niflheim Guide]";
			mes "Claro, quitare todas las marcas de tu mini-mapa...";
			mes "Hay algo mas en lo que pueda ayudarte...?";
			viewpoint 2,1,1,0,0xFFFFFF;
			viewpoint 2,1,1,1,0xFFFFFF;
			viewpoint 2,1,1,2,0xFFFFFF;
			viewpoint 2,1,1,3,0xFFFFFF;
			break;
		case 6:
			mes "[Niflheim Guide]";
			mes "......Adios~!..";
			close;
		}
	}
	end;
L_Mark:
	mes "[Niflheim Guide]";
	mes "Dejame marcar la ubicacion de";
	mes getarg(0)+"^000000...";
	mes "en tu mini-mapa.";
	mes "Te gustaria revisar alguna otra ubicacion?";
	return;
}
