amatsu,202,91,3	script	Amatsu Guide#amatsu	758,{
	mes "[Amachang]";
	mes "Bienvenido a ^8B4513Amatsu^000000,";
	mes "el pueblo de habitantes amables";
	mes "y hermosas flores de cerezo.";
	next;
	mes "[Amachang]";
	mes "Soy Amachang,";
	mes "la 13va Miss Amatsu.";
	mes "Te guiare por el pueblo";
	mes "como Miss Amatsu.";
	mes "Por favor dime";
	mes "si quieres saber algo.";
	F_Navi("[Amachang]");
	while (1) {
		next;
		switch(select("[ Empleada Kafra ]:[ Bar ]:[ Vendedor de Armas ]:[ Vendedor de Herramientas ]:[ Palacio de Amatsu ]:[ Asistente de Chef ]:[ Edificio del Gremio Ninja ]:[ Capitan del Mar ]:Quitar Marcas del Mini-Mapa:Cancelar")) {
		case 1:
			callsub L_Mark, "^0000FFEmpleada Kafra^000000";
			viewpoint 1,102,149,0,0x0A82FF;
			break;
		case 2:
			callsub L_Mark, F_Navi("Bar","amatsu,215,116","^006400");
			viewpoint 1,215,116,1,0xAAFF00;
			break;
		case 3:
			callsub L_Mark, F_Navi("Vendedor de Armas","amatsu,129,117","^008080");
			viewpoint 1,129,117,2,0x008080;
			break;
		case 4:
			callsub L_Mark, F_Navi("Vendedor de Herramientas","amatsu,97,117","^FF1493");
			viewpoint 1,97,117,3,0xFF1493;
			break;
		case 5:
			callsub L_Mark, F_Navi("Palacio de Amatsu","amatsu,87,235","^8B4513");
			viewpoint 1,87,235,4,0x8B4513;
			break;
		case 6:
			callsub L_Mark, F_Navi("Asistente de Chef","amatsu,206,150","^8A2BE2");
			viewpoint 1,206,150,5,0x8A2BE2;
			break;
		case 7:
			callsub L_Mark, F_Navi("Edificio del Gremio Ninja","amatsu,148,137","^4B0082");
			viewpoint 1,148,137,6,0x4B0082;
			break;
		case 8:
			callsub L_Mark, F_Navi("Capitan del Mar","amatsu,195,79","^00BFFF");
			viewpoint 1,195,79,7,0x00BFFF;
			break;
		case 9:
			mes "[Amachang]";
			mes "Quitare todas las marcas de tu mini-mapa.";
			mes "Hay algo mas en lo que pueda ayudarte?";
			viewpoint 2,1,1,0,0xFFFFFF;
			viewpoint 2,1,1,1,0xFFFFFF;
			viewpoint 2,1,1,2,0xFFFFFF;
			viewpoint 2,1,1,3,0xFFFFFF;
			viewpoint 2,1,1,4,0xFFFFFF;
			viewpoint 2,1,1,5,0xFFFFFF;
			viewpoint 2,1,1,6,0xFFFFFF;
			viewpoint 2,1,1,7,0xFFFFFF;
			break;
		case 10:
			mes "[Amachang]";
			mes "Disfruta tu viaje~!!";
			close;
		}
	}
	end;
L_Mark:
	mes "[Amachang]";
	mes "La ubicacion de "+getarg(0)+" esta";
	mes "marcada en tu mini-mapa.";
	mes "Hay algo mas en lo que pueda ayudarte?";
	return;
}
