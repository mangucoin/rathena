morocc,153,286,6	script	Guide#01morocc::GuideMorocc	707,{
	cutin "moc_soldier",2;
	mes "[Morocc Guide]";
	mes "Bienvenido a ^8B4513Morocc^000000,";
	mes "el Pueblo del Desierto.";
	mes "Necesitas ayuda para navegar por el pueblo?";
	F_Navi("[Morocc Guide]");
	next;
	while (1) {
		switch(select("[ Instalaciones Principales ]:[ Comerciantes y Ayudantes ]:Quitar Marcas del Mini-Mapa:Cancelar")) {
		case 1:
			set .@loop,1;
			while (.@loop) {
				switch(select("[ Gremio de Ladrones ]:[ Mazmorra de la Esfinge ]:[ Entrada al Oasis ]:[ Forja ]:[ Tienda de Armas y Armaduras ]:[ Posada ]:[ Taberna ]:Menu Anterior")) {
				case 1:
					mes "[Morocc Guide]";
					mes "El "+F_Navi("Gremio de Ladrones","morocc,26,294","^B9062F")+" se encuentra";
					mes "dentro de las ^B9062FPiramides^000000";
					mes "que estan cerca del ^B9062FOasis^000000.";
					mes "Te gustaria revisar alguna otra ubicacion?";
					viewpoint 1,26,294,0,0xFF0000;
					next;
					break;
				case 2:
					mes "[Morocc Guide]";
					mes "La "+F_Navi("Mazmorra de la Esfinge","morocc,26,164","^0000FF")+" esta";
					mes "justo afuera de la puerta oeste.";
					mes "Te gustaria revisar alguna otra ubicacion?";
					viewpoint 1,26,164,1,0x0A82FF;
					next;
					break;
				case 3:
					mes "[Morocc Guide]";
					mes "Dejame marcar la entrada al "+F_Navi("Oasis","morocc,26,294","^FF5400");
					mes "en tu mini-mapa.";
					mes "Te gustaria revisar alguna otra ubicacion?";
					viewpoint 1,26,294,2,0xFFB400;
					next;
					break;
				case 4:
					callsub L_Mark, F_Navi("Forja","morocc,47,47","^006400");
					viewpoint 1,47,47,3,0xAAFF00;
					next;
					break;
				case 5:
					callsub L_Mark, F_Navi("Tienda de Armas y Armaduras","morocc,253,56","^7F3300");
					viewpoint 1,253,56,4,0xD2691E;
					next;
					break;
				case 6:
					mes "[Morocc Guide]";
					mes "Actualmente la "+F_Navi("Posada","morocc,273,269","^800080");
					mes "ha sido cerrada hasta nuevo aviso,";
					mes "pero marcare su ubicacion";
					mes "en tu mini-mapa.";
					mes "Te gustaria revisar alguna otra ubicacion?";
					viewpoint 1,273,269,5,0xDA70D6;
					next;
					break;
				case 7:
					callsub L_Mark, F_Navi("Taberna","morocc,52,259","^B9062F");
					viewpoint 1,52,259,6,0xFF0000;
					next;
					break;
				case 8:
					set .@loop,0;
					break;
				}
			}
			break;
		case 2:
			set .@loop,1;
			while (.@loop) {
				switch(select("[ Empleada Kafra ]:[ Vendedor de Hierbas Venenosas ]:[ Fabricante de Tintes ]:[ Seiyablem ]:Menu Anterior")) {
				case 1:
					callsub L_Mark,"^0000FFEmpleadas Kafra^000000";
					viewpoint 1,156,97,7,0x0A82FF;
					viewpoint 1,160,258,8,0x0A82FF;
					next;
					break;
				case 2:
					callsub L_Mark, F_Navi("Vendedor de Hierbas Venenosas","morocc,191,94","^006400");
					viewpoint 1,190,96,9,0xAAFF00;
					next;
					break;
				case 3:
					mes "[Morocc Guide]";
					mes "El "+F_Navi("Fabricante de Tintes","morocc,273,269","^7F3300")+" se";
					mes "encuentra actualmente";
					mes "en el 1er piso de la Posada.";
					mes "Dejame marcar su ubicacion";
					mes "en tu mini-mapa.";
					mes "Te gustaria revisar alguna otra ubicacion?";
					viewpoint 1,273,269,10,0xD2691E;
					next;
					break;
				case 4:
					mes "[Morocc Guide]";
					mes F_Navi("Seiyablem","morocc,51,41","^800080");
					mes "es un tecnico";
					mes "que puede crear ^0000FFranuras^000000 en armas y armaduras,";
					mes "pero no todo el equipo";
					mes "puede recibir ranuras. Bueno, eso es lo que el dice.";
					mes "Te gustaria revisar alguna otra ubicacion?";
					viewpoint 1,51,41,11,0xDA70D6;
					next;
					break;
				case 5:
					set .@loop,0;
					break;
				}
			}
			break;
		case 3:
			mes "[Morocc Guide]";
			mes "Claro, quitare todas las marcas de tu mini-mapa.";
			mes "Hay algo mas en lo que pueda ayudarte?";
			viewpoint 2,1,1,0,0xFFFF00;
			viewpoint 2,1,1,1,0x000000;
			viewpoint 2,1,1,2,0xFF0000;
			viewpoint 2,1,1,3,0xFFFF00;

			viewpoint 2,1,1,5,0xFFFF00;
			viewpoint 2,1,1,6,0xFFFF00;
			viewpoint 2,1,1,7,0xFFFF00;
			viewpoint 2,1,1,8,0xFFFF00;
			viewpoint 2,1,1,9,0xFFFF00;
			viewpoint 2,1,1,10,0xFFFF00;
			viewpoint 2,1,1,11,0xFFFF00;
			next;
			break;
		case 4:
			mes "[Morocc Guide]";
			mes "Cuidate en tus viajes.";
			close2;
			cutin "moc_soldier",255;
			end;
		}
	}
	end;
L_Mark:
	mes "[Morocc Guide]";
	mes "Dejame marcar la ubicacion de";
	mes getarg(0);
	mes "en tu mini-mapa.";
	mes "Te gustaria revisar alguna otra ubicacion?";
	return;
}
