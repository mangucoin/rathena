ayothaya,203,169,4	script	Ayothaya Guide Noi#01ay	839,{
	if (strnpcinfo(0) == "Ayothaya Guide Noi#01ay")
		set .@n$, "[Noi]";
	else
		set .@n$, "[Noa]";
	mes .@n$;
	mes "Hola~";
	mes "Bienvenido a ^8B4513Ayothaya^000000.";
	mes "Este pueblo esta conectado a un rio a traves del bosque salvaje.";
	mes "Es un pais realmente tranquilo.";
	mes "Tomate tu tiempo para explorar.";
	F_Navi(.@n$);
	while (1) {
		next;
		switch(select("[ Tienda de Armas ]:[ Tienda de Herramientas ]:[ Taberna ]:[ Santuario ]:[ Lugar de Pesca ]:[ Aibakthing ]:Quitar Marcas del Mini-Mapa:Cancelar")) {
		case 1:
			mes .@n$;
			mes "La "+F_Navi("Tienda de Armas","ayothaya,163,90","^0000FF");
			mes "en Ayothaya tiene armas especiales.";
			mes "Puedes encontrar grandes armas ahi.";
			mes "Hay algo mas en lo que pueda ayudarte?";
			viewpoint 1,163,90,0,0x0A82FF;
			break;
		case 2:
			mes .@n$;
			mes "Nuestros ciudadanos compran muchos productos en la "+F_Navi("Tienda de Herramientas","ayothaya,131,86","^006400")+".";
			mes "Asegurate de conseguir todo lo que necesites ahi.";
			mes "Necesitas algo mas?";
			viewpoint 1,131,86,1,0xAAFF00;
			break;
		case 3:
			mes .@n$;
			mes "Parte de la aventura es recopilar informacion.";
			mes "La "+F_Navi("Taberna","ayothaya,229,72","^008080")+" esta llena de viajeros donde puedes obtener informacion util.";
			mes "Hay algo mas en lo que pueda ayudarte?";
			viewpoint 1,229,72,2,0x008080;
			break;
		case 4:
			mes .@n$;
			mes "Para calmar tu corazon, ve al "+F_Navi("Santuario","ayothaya,208,280","^FF1493")+" y haz una oracion.";
			mes "Hay algo mas en lo que pueda ayudarte?";
			viewpoint 1,208,280,3,0xFF1493;
			break;
		case 5:
			mes .@n$;
			mes "Ayothaya esta cerca de la playa, asi que la mayoria disfruta pescar.";
			mes "Realmente les encanta pescar.";
			mes "Ve a "+F_Navi("pescar","ayothaya,255,99","^8B4513")+" e intenta atrapar un pez grande.";
			mes "Hay algo mas en lo que pueda ayudarte?";
			viewpoint 1,255,99,4,0x8B4513;
			break;
		case 6:
			mes .@n$;
			mes "Si quieres regresar a casa";
			mes "busca a "+F_Navi("Aibakthing","ayothaya,151,68","^00BFFF");
			mes "y pidele que te envie de vuelta.";
			mes "Hay algo mas en lo que pueda ayudarte?";
			viewpoint 1,151,68,5,0x00BFFF;
			break;
		case 7:
			mes .@n$;
			mes "Quitare todas las marcas de tu mini-mapa.";
			mes "Necesitas algo mas??";
			viewpoint 2,1,1,0,0xFFFFFF;
			viewpoint 2,1,1,1,0xFFFFFF;
			viewpoint 2,1,1,2,0xFFFFFF;
			viewpoint 2,1,1,3,0xFFFFFF;
			viewpoint 2,1,1,4,0xFFFFFF;
			viewpoint 2,1,1,5,0xFFFFFF;
			break;
		case 8:
			mes .@n$;
			mes "Disfruta tu viaje~!";
			close;
		}
	}
}
