malangdo,220,167,5	script	Coin Exchanger CX-1	564,{
	if (checkweight(1201,1) == 0) {
		mes "Tienes demasiados tipos de objetos. Intentemos continuar despues de reducirlos.";
		close;
	}
	if (MaxWeight - Weight < 10000) {
		mes "No puedes continuar porque tienes objetos muy pesados. Intentemos continuar despues de reducir el peso.";
		close;
	}
	mes "[Coin Exchanger CX-1]";
	mes "Bienvenido a CX-1. Si tienes monedas, puedes intercambiarlas por monedas de mayor o menor grado.";
	next;
	setarray .@coins[1],
		6417, //Silvervine
		6418, //Agrade_Coin
		6419, //Bgrade_Coin
		6420, //Cgrade_Coin
		6421, //Dgrade_Coin
		6422; //Egrade_Coin
	.@menu$ = "Detener:";
	for(.@i = 1; .@i<=6; .@i++) {
		if (countitem(.@coins[.@i]))
			.@menu$ = .@menu$ + getitemname(.@coins[.@i]) + ":";
		else
			.@menu$ = .@menu$ + "^aaaaaa"+getitemname(.@coins[.@i])+" (None)^000000:";
	}
	.@i = select(.@menu$);
	switch(.@i) {
	case 1:
		mes "[Coin Exchanger CX-1]";
		mes "Gracias por venir.";
		close;
	default:
		.@coin = .@coins[.@i-1];
		.@coin_select = .@i-1;
		break;
	}
	if (countitem(.@coin) == 0) {
		mes "[Coin Exchanger CX-1]";
		mes "Elegiste una moneda que no tienes.";
		mes "Gracias por venir.";
		close;
	}
	mes "[Coin Exchanger CX-1]";
	mes "----Tasa de cambio----";
	mes "Cambio superior - 30 a 10";
	mes "Cambio inferior - 10 a 30";
	mes "------------------------";
	mes "^ff3333Al intercambiar, cobramos una tarifa segun el precio actual del mercado.^000000";
	next;
	setarray .@exchange_name$[0],"higher","lower";
	setarray .@exchange_rate[0],30,10;
	setarray .@exchange_loss[0],rand(1,4),rand(1,2); // Amount deducted per exchange.
	setarray .@exchange_id[0],.@coins[.@coin_select-1],.@coins[.@coin_select+1];
	if (.@exchange_id[0] == .@coins[1]) .@exchange_id[0] = 0; // Cannot exchange for Silvervine.
	.@menu$ = "Detener:";
	for(.@i = 0; .@i<2; .@i++) {
		if (.@exchange_id[.@i] == 0)
			.@menu$ = .@menu$ + "^ff3333Intercambio no disponible a moneda "+.@exchange_name$[.@i]+"^000000:";
		else if (countitem(.@coin) < .@exchange_rate[.@i])
			.@menu$ = .@menu$ + "^aaaaaaCambiar a moneda "+.@exchange_name$[.@i]+" (nulo)^000000:";
		else
			.@menu$ = .@menu$ + "Cambiar a moneda "+.@exchange_name$[.@i]+" - "+getitemname(.@coin)+" ("+.@exchange_rate[.@i]+" necesarias):";
	}
	.@i = select(.@menu$)-2;
	if (.@i == -1) {
		mes "[Coin Exchanger CX-1]";
		mes "Gracias por venir.";
		close;
	}
	if (.@exchange_id[.@i] == 0) {
		mes "[Coin Exchanger CX-1]";
		mes "Esta moneda no esta disponible para cambiar a moneda "+.@exchange_name$[.@i]+".";
		mes "Gracias por venir.";
		close;
	}
	if (countitem(.@coin) < .@exchange_rate[.@i]) {
		mes "[Coin Exchanger CX-1]";
		mes "No tienes suficientes monedas para cambiar a moneda "+.@exchange_name$[.@i]+".";
		mes "Gracias por venir.";
		close;
	}
	.@exchange_total = .@exchange_rate[(!.@i)] - .@exchange_loss[(!.@i)];
	delitem .@coin, .@exchange_rate[.@i];
	getitem .@exchange_id[.@i], .@exchange_total;
	mes "[Coin Exchanger CX-1]";
	mes "^003399"+.@exchange_rate[.@i]+"^000000 unidad(es) de ^003399 "+getitemname(.@coin)+"^000000 tomadas.";
	mes "Las intercambie por ^003399monedas "+.@exchange_name$[.@i]+"^000000, despues de deducir ^003399"+.@exchange_loss[(!.@i)]+"^000000 unidad(es), ^003399"+.@exchange_total+"^000000 unidad(es) restantes.";
	close;
}
