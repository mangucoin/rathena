dali02,51,130,4	script	Dimension Traveler	4_F_SHABBY,{
	if (checkweight(28100,1) == 0) {// Thanos Axe
		mes "- No puedes continuar porque tienes demasiados objetos contigo. -";
		close;
	}
	.@item_name_req$ = getitemname(6672);// Gray_Shard
	mes "[Dimension Traveler]";
	mes "Amigo aventurero, " + mesitemlink( 6672 ) + " es lo que necesito. ¿Que tal si lo intercambias por algo que tengo?";
	next;
	.@type = select( "Muestrame tus armas.", "Muestrame tu armadura.", "¿Algo especial?", "No necesito nada." ) - 1;
	mes "[Dimension Traveler]";
	switch(.@type) {
	case 0:
		mes "¡Oh! ¿Te interesan las armas? Tenemos todas las armas que se pueden obtener del Ghost Palace. ¿Por que no eliges una?";
		break;
	case 1:
		mes "¿Armadura? Tenemos todas las armaduras que se pueden obtener del Ghost Palace. ¿Por que no eliges una?";
		break;
	case 2:
		mes "¡Uhh! Parece que muchos ya escucharon que tengo las mejores cosas. Sientete libre de revisar mis colecciones.";
		break;
	case 3:
		mes "¡Adios amigo~ Vuelve pronto!";
		close;
	}
	next;
	mes "[Dimension Traveler]";
	mes "¡Ah! El numero que esta escrito junto a " + mesitemlink( 6672 ) + " sera necesario.";
	mes "Por favor tenlo en cuenta.";

	switch(.@type) {
	case 0:
		setarray .@items[0],
			13441, 200,		// Thanos Sword
			21009, 200, 	// Thanos Great Sword
			1438,  200,		// Thanos Spear
			1496,  200,		// Thanos Long Spear
			1669,  200,		// Thanos Staff
			2023,  200,		// Thanos Rod
			18119, 200,		// Thanos Bow
			13093, 200,		// Thanos Dagger
			28000, 200,		// Thanos Katar
			1836,  200,		// Thanos Knuckle
			16028, 200,		// Thanos Hammer
			28100, 200,		// Thanos Axe
			1933,  200,		// Thanos Violin
			1988,  200;		// Thanos Whip
		break;
	case 1:
		setarray .@items[0],
			2187,  100,		// Shield Of Gray
			15090, 100, 	// Armor Of Gray
			15091, 100,		// Gray Robe
			20721, 100,		// Cloak Of Gray
			22033, 100,		// Boots Of Gray
			18820, 100;		// Gray Helmet
		break;
	case 2:
		setarray .@items[0],
			27184, 10000,		// Knight Sakray Card
			23676, 50, 			// Gray Charcoal Melee (Melee)
			23677, 50,			// Gray Charcoal Magic (Magic)
			23678, 50;			// Gray Charcoal Range (Range)
		break;
	}
	.@size = getarraysize(.@items);
	for ( .@i = 0; .@i < .@size; .@i += 2 )
		.@menu$ += getitemname(.@items[.@i]) + " - " + .@items[.@i+1] + ":";

	while(true) {
		next;
		.@s = (select(.@menu$) - 1) * 2;
		mes "[Dimension Traveler]";
		if (.@type != 1)	// armor type doesn't display item info
			mes "¿Te refieres a " + mesitemlink( .@items[.@s] ) + "?";
		mes "Se requieren " + .@items[.@s+1] + " ^006400" + .@item_name_req$ + "^000000 a cambio.";
		next;
		switch( select( "Hagamos el intercambio.", "Dejame ver otras cosas.", "Volvere despues." ) ) {
		case 1:
			break;
		case 2:
			mes "[Dimension Traveler]";
			switch(.@type) {
			case 0:
				mes "Tengo mas que suficientes armas.";
				break;
			case 1:
				mes "Tenemos muchas armaduras para ti. ¿Por que no eliges una?";
				break;
			case 2:
				mes "Si me traes " + .@item_name_req$ + ", lo intercambiare por cualquier cosa que quieras.";
				break;
			}
			continue;
		case 3:
			mes "[Dimension Traveler]";
			mes "¡Cuidate amigo~ Nos vemos la proxima vez!";
			close;
		}
		if (countitem(6672) < .@items[.@s+1]) {
			mes "[Dimension Traveler]";
			mes "Me pones en una situacion dificil, amigo.";
			mes "Necesito " + .@item_name_req$ + ". Lo intercambiare por ^006400" + .@items[.@s+1] + " " + .@item_name_req$ + "^000000.";
			close;
		}
		mes "[Dimension Traveler]";
		mes "Gracias. Vuelve si necesitas algo mas.";
		delitem 6672, .@items[.@s+1];
		getitem .@items[.@s], 1;
		close;
	}
}
