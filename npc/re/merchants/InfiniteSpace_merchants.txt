cmd_fild07,375,167,1	script	Ruins Black Trader#pa0829_01	4_F_JOB_HUNTER,{
	mes "[Ruins Black Trader]";
	mes "Oye, ¿no es dificil caminar dentro y fuera de aqui cada vez? Por solo ^0000ff20,000^000000 Zeny, puedes tener un pergamino que te llevara directo a la entrada de las ruinas, ¿que opinas?";
Purchase:
	next;
	switch( select( "Dame uno", "No lo creo" )) {
		case 1:
			if (!checkweight(22980,1) || (MaxWeight - Weight) < 1000) {
				mes "No puedes continuar la conversacion porque tienes una gran cantidad de articulos.";
				mes "Por favor organiza tus articulos e intenta de nuevo.";
				close;
			}
			else if (Zeny < 20000) {
				mes "[Ruins Black Trader]";
				mes "Parece que no tienes mas dinero. Un pergamino cuesta 20,000 Zeny.";
				close;
			}
			Zeny -= 20000;
			getitem 22980,1;
			mes "[Ruins Black Trader]";
			mes "Fue un buen trato. ¿Necesitas mas?";
			goto Purchase;
		case 2:
			mes "[Ruins Black Trader]";
			mes "Vuelve cuando quieras";
			close;
	}
}

// Equipment Shop
cmd_fild07,57,275,5	script	Artifact Appraiser#pa0829_01	1_F_02,{
	if (!checkweight(1201,1) || (MaxWeight - Weight) < 1000) {
		mes "No puedes continuar la conversacion porque tienes una gran cantidad de articulos.";
		mes "Por favor organiza tus articulos e intenta de nuevo.";
		close;
	}
	.@stone_id = 6905;
	mes "[ Artifact Appraiser ]";
	mes "Elige el tipo de equipo que deseas comprar. Puedes comprar tantos como quieras, siempre que tengas " + getitemname(.@stone_id) + ".";
	next;
	switch (select("Cancelar:Armas:Armaduras")) {
		case 1:
			mes "[ Artifact Appraiser ]";
			mes "Por favor vuelve cuando quieras~";
			close;

		case 2:
			setarray .@equip_id,1994,1938,13323,13126,28703,2024,16038,21014,28105,18128;
			.@price = 50;
			break;

		case 3:
			setarray .@equip_id,15141,22075,20779,19033;
			.@price = 50;
			break;
	}
	.@menu$ = "Cancel:";
	for (.@i = 0; .@i < getarraysize(.@equip_id); .@i++)
		.@menu$ += getitemname(.@equip_id[.@i]) + ":";
	.@s = select(.@menu$) - 1;
	switch (.@s) {
		case 0:
			mes "[ Artifact Appraiser ]";
			mes "Por favor vuelve cuando quieras~";
			close;

		default:
			.@s--;
			mes "[ Artifact Appraiser ]";
			mes "Necesitas ^0000FF" + .@price + "^000000 " + getitemname(.@stone_id) + " para comprar el " + getitemname(.@equip_id[.@s]) + "~";
			next;
			if (select("Cancelar:Comprar") == 1) {
				mes "[ Artifact Appraiser ]";
				mes "Por favor vuelve cuando quieras~";
				close;
			}
			if (countitem(.@stone_id) < .@price) {
				mes "[ Artifact Appraiser ]";
				mes "No tienes suficientes " + getitemname(.@stone_id) + " para comprar este articulo.";
				close;
			}
			mes "[ Artifact Appraiser ]";
			mes "Gracias por tu confianza. Por favor vuelve de nuevo la proxima vez.";
			delitem .@stone_id,.@price;
			getitem .@equip_id[.@s],1;
			close;
	}
}

// Equipment Enchanter
cmd_fild07,60,275,3	script	Artifact Enhancer#pa0829_01	4_F_JOB_BLACKSMITH,{
	if (!checkweight(1201,1) || (MaxWeight - Weight) < 1000) {
		mes "No puedes continuar la conversacion porque tienes una gran cantidad de articulos.";
		mes "Por favor organiza tus articulos e intenta de nuevo.";
		close;
	}
	disable_items;
	.@stone_id = 6905;
	function equip_check;
	mes "[ Artifact Enhancer ]";
	mes "Si buscas fortalecer el equipo que obtuviste en el Infinite Space, has venido al lugar correcto.";
	mes "¿Tienes algun " + getitemname(.@stone_id) + "?";
	next;
	switch (select("¿Como encanto mi equipo?:Encantar equipo.:Inicializar encantamiento del equipo.")) {
		case 1:
			mes "[ Artifact Enhancer ]";
			mes "Podras obtener " + getitemname(.@stone_id) + ", si exploras el espacio bajo el Faro de Paros.";
			next;
			mes "[ Artifact Enhancer ]";
			mes "Con ese material, puedo reforzar el equipo que obtuviste del Infinite Space.";
			next;
			mes "[ Artifact Enhancer ]";
			mes "Solo puedo encantar el 3er y 4to slot.";
			next;
			mes "[ Artifact Enhancer ]";
			mes "No hay posibilidad de que el equipo se destruya durante el proceso de encantamiento, pero hay una alta probabilidad de que se destruya al inicializar los encantamientos.";
			next;
			mes "[ Artifact Enhancer ]";
			mes "Entonces visitame cuando quieras encantar el equipo.";
			break;

		case 2:
			.@stone_id = 6905;
			.@fee = 20;
			mes "[ Artifact Enhancer ]";
			mes "Por favor selecciona el equipo que deseas encantar.";
			next;
			switch (select("Cancel:Weapon:Armor:Shoes:Garment:Helm")) {
				case 1:
					mes "[ Artifact Enhancer ]";
					mes "Entonces visitame cuando quieras encantar el equipo.";
					close;

				case 2:
					.@part = EQI_HAND_R;
					break;

				case 3:
					.@part = EQI_ARMOR;
					break;

				case 4:
					.@part = EQI_SHOES;
					break;

				case 5:
					.@part = EQI_GARMENT;
					break;

				case 6:
					.@part = EQI_HEAD_TOP;
					break;
			}
			.@equip_id = getequipid(.@part);
			.@refine = getequiprefinerycnt(.@part);
			equip_check(.@equip_id);
			for (.@i = 0;.@i < 4;.@i++) {
				.@card[.@i] = getequipcardid(.@part,.@i);
				.@check[.@i] = getequipcardid(.@part,.@i);
			}
			if (.@card[2] > 0) {
				mes "[ Artifact Enhancer ]";
				mes "Solo puedo encantar hasta el tercer slot. Tu equipo no puede ser encantado mas.";
				close;
			}
			switch (.@part) {
				case EQI_HAND_R:
					setarray .@enchant$,
					"4700,4701:4710,4711:4720,4721",
					"4811,4810,4809,4808,4820,4821,4822,4823:4815,4814,4813,4812,4826,4827,4828,4829:4832,4833,4834,4835,4836,4837,4838,4839";
					break;

				case EQI_ARMOR:
				case EQI_SHOES:
					setarray .@enchant$,
					"4700,4701,4702,4703:4710,4711,4712,4713:4720,4721,4722,4723",
					"4795,4796,4797:4870,4871,4800:4870,4871,4800";
					break;

				case EQI_GARMENT:
				case EQI_HEAD_TOP:
					setarray .@enchant$,
					"4700,4701,4702,4703:4710,4711,4712,4713:4720,4721,4722,4723",
					"4861,4862,4867,4868,4900:4861,4862,4867,4868,4900:4861,4862,4867,4868,4900";
					break;
			}
			.@slot = (.@card[3] > 0 ? 2 : 3);
			.@index = (.@slot == 3 ? 0 : 1);
			mes "[ Artifact Enhancer ]";
			mes "Puedes elegir 3 tipos de encantamiento, y la tarifa de encantamiento es " + .@fee + " " + getitemname(.@stone_id) + ".";
			mes "Me asegurare de que se encante sin problemas.";
			next;
			.@type = select("Salir:Fisico:Magico:Rango") - 2;
			mes "[ Artifact Enhancer ]";
			if (.@slot == 3)
				mes "Bien, comencemos el primer encantamiento.";
			else
				mes "Bien, procedamos con el segundo encantamiento.";
			next;
			if (select("Regresare despues.:Por favor procede.") == 1) {
				mes "[ Artifact Enhancer ]";
				mes "Si cambias de opinion, solo vuelve a mi.";
				close;
			}
			if (countitem(.@stone_id) < .@fee) {
				mes "[ Artifact Enhancer ]";
				mes "Hm. Por cierto, creo que no estabas escuchando, no tienes suficientes " + getitemname(.@stone_id) + ".";
				close;
			}
			explode(.@T$,.@enchant$[.@index],":");
			explode(.@TT$,.@T$[.@type],",");
			.@enchant = atoi(.@TT$[rand(getarraysize(.@TT$))]);
			.@card[.@slot] = .@enchant;
			delitem .@stone_id,.@fee;
			if (callfunc("F_IsEquipIDHack", .@part, .@equip_id) || callfunc("F_IsEquipRefineHack", .@part, .@refine) || callfunc("F_IsEquipCardHack", .@part, .@check[0], .@check[1], .@check[2], .@check[3]))
				close;
			delequip .@part;
			getitem2 .@equip_id,1,1,.@refine,0,.@card[0],.@card[1],.@card[2],.@card[3];
			specialeffect2 EF_REPAIRWEAPON;
			mes "[ Artifact Enhancer ]";
			mes "Hm. Quedo bien hecho. Revisa tu equipo de inmediato.";
			break;
		
		case 3:
			.@stone_id = 6905;
			.@fee = 30;
			.@break_chance = 30;
			mes "[ Artifact Enhancer ]";
			mes "Por favor selecciona el equipo que deseas encantar.";
			next;
			switch (select("Cancel:Weapon:Armor:Shoes:Garment:Helm")) {
				case 1:
					mes "[ Artifact Enhancer ]";
					mes "Entonces visitame cuando quieras encantar el equipo.";
					close;

				case 2:
					.@part = EQI_HAND_R;
					break;

				case 3:
					.@part = EQI_ARMOR;
					break;

				case 4:
					.@part = EQI_SHOES;
					break;

				case 5:
					.@part = EQI_GARMENT;
					break;

				case 6:
					.@part = EQI_HEAD_TOP;
					break;
			}
			.@equip_id = getequipid(.@part);
			.@refine = getequiprefinerycnt(.@part);
			equip_check(.@equip_id);
			for (.@i = 0; .@i < 4; .@i++) {
				.@card[.@i] = getequipcardid(.@part,.@i);
				.@check[.@i] = .@card[.@i];
			}
			if (.@card[3] == 0) {
				mes "[ Artifact Enhancer ]";
				mes "Tu equipo no tiene ningun encantamiento.";
				close;
			}
			mes "[ Artifact Enhancer ]";
			mes "Existe la posibilidad de que tu equipo de infinite space se destruya en el proceso de inicializacion. ¿Aun estas dispuesto a proceder?";
			next;
			if (select("Salir:Proceder") == 1) {
				mes "[ Artifact Enhancer ]";
				mes "Si cambias de opinion, solo vuelve a mi.";
				close;
			}
			if (countitem(.@stone_id) < .@fee) {
				mes "[ Artifact Enhancer ]";
				mes "Hm. Por cierto, creo que no estabas escuchando, no tienes suficientes " + getitemname(.@stone_id) + ".";
				close;
			}
			delitem .@stone_id,.@fee;
			if (callfunc("F_IsEquipIDHack", .@part, .@equip_id) || callfunc("F_IsEquipRefineHack", .@part, .@refine) || callfunc("F_IsEquipCardHack", .@part, .@check[0], .@check[1], .@check[2], .@check[3]))
				close;
			delequip .@part;
			if (rand(1,100) > .@chance) {
				getitem2 .@equip_id,1,1,.@refine,0,.@card[0],.@card[1],0,0;
				specialeffect2 EF_REPAIRWEAPON;
				mes "[ Artifact Enhancer ]";
				mes "Hm. Quedo bien hecho. Revisa tu equipo de inmediato.";
			} else {
				specialeffect2 EF_REFINEFAIL;
				mes "[ Artifact Enhancer ]";
				mes "Bueno, te lo adverti. No tienes suerte, ¿eh?";
			}
			break;
	}
	close;

	function	equip_check	{
		setarray .@equip_id,1994,1938,13323,13126,28703,2024,16038,21014,28105,18128,15141,22075,20779,19033;
		if (!getarg(0)) {
			mes "[ Artifact Enhancer ]";
			mes "¿Te quitaste el equipo?";
			close;
		}
		if (inarray(.@equip_id,getarg(0)) == -1) {
			mes "[ Artifact Enhancer ]";
			mes "Este equipo no es adecuado para encantar. No olvides que solo la serie de equipo infinite puede ser encantada.";
			close;
		}
		return;
	}
}
