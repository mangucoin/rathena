//= Description: =========================================
//= [Official Conversion]
//= Refiner that uses Enriched ores to increase upgrade success.
//===== Changelog: ===========================================
//= 1.0 Added Malangdo Refiner "Holink". [Euphy]
//= 1.1 Removed re-roll behavior. [Secret]
//= 1.2 Added db-based material ID [Secret]

// Main NPC :: mal_jerun
//============================================================
malangdo,221,174,6	script	Holink#mal_cash	559,{
	disable_items;
	mes "[Holink]";
	mes "Soy el miau~ Herrero Holink~";
	mes "Maestro del refinamiento, Holink~";
	mes "Soy el gato especial Holink que aprendio de Morocc~";
	mes "Mi hija esta muy orgullosa de mi, Holink~";
	mes "Que deberia refinar Holink~ hoy?";
	next;

	setarray .@indices[1], EQI_HEAD_TOP, EQI_ARMOR, EQI_HAND_L, EQI_HAND_R, EQI_GARMENT, EQI_SHOES, EQI_ACC_L, EQI_ACC_R, EQI_HEAD_MID, EQI_HEAD_LOW;
	for (set .@i,1; .@i<=10; set .@i,.@i+1)
		set .@menu$, .@menu$ + ( getequipisequiped(.@indices[.@i]) ? getequipname(.@indices[.@i]) : F_getpositionname(.@indices[.@i]) +"-[Vacio]" ) +":";
	set .@part, .@indices[ select(.@menu$) ];

	if (!getequipisequiped(.@part)) {
		mes "[Holink]";
		switch(.@part) {
		case EQI_HEAD_TOP:
			mes "Mi maestro Aragam dijo miau~";
			mes "No hay cura para la estupidez...";
			break;
		case EQI_ARMOR:
			mes "No hay nada que ver aqui, miau!";
			break;
		case EQI_HAND_L:
			mes "Miau? Que quieres que haga con esta mano izquierda...?";
			break;
		case EQI_HAND_R:
			mes "Miau? Que quieres que haga con esta mano derecha...?";
			break;
		case EQI_GARMENT:
			mes "Miau? No llevas nada puesto.";
			break;
		case EQI_SHOES:
			mes "Kyang~! No molestes mi sensible olfato, miau~.";
			break;
		case EQI_ACC_L:
		case EQI_ACC_R:
			mes "Miau? Donde esta el accesorio?";
			break;
		case EQI_HEAD_MID:
		case EQI_HEAD_LOW:
			mes "Miau? Hablas de las otras partes de la cabeza, miau?~";
			break;
		}
		close;
	}
	if (!getequipisenableref(.@part)) {
		mes "[Holink]";
		mes "Ni siquiera Aragam puede refinar algo asi, miau.";
		close;
	}
	if (getequiprefinerycnt(.@part) >= 10) {
		mes "[Holink]";
		mes "Miau~ Refinamiento perfecto. Esto lo hizo Aragam, miau?~";
		close;
	}
	mes "[Holink]";
	.@refineitemid = getequipid(.@part); // save id of the item
	.@itemtype = getiteminfo( .@refineitemid, ITEMINFO_TYPE );
	.@refinerycnt = getequiprefinerycnt(.@part); //save refinery count
	setarray .@card[0], getequipcardid(.@part,0), getequipcardid(.@part,1), getequipcardid(.@part,2), getequipcardid(.@part,3);
// Holink has different hardcoded prices
//	.@price = getequiprefinecost(.@part, REFINE_COST_NORMAL, REFINE_ZENY_COST);
	.@material = getequiprefinecost(.@part, REFINE_COST_NORMAL, REFINE_MATERIAL_ID);

	if( .@itemtype == IT_ARMOR ){
		.@equip_lv = getequiparmorlv( .@part );
		.@type$ = "armor";
		mes "Has elegido una armadura, miau~";

		switch(.@equip_lv) {
			case 1:
				.@price = 15000;
				break;
			default:
				// TODO:
				close;
		}
	}else if( .@itemtype == IT_WEAPON ){
		.@equip_lv = getequipweaponlv( .@part );
		.@type$ = "weapon";

		switch( .@equip_lv ){
			case 1: // Level 1 Weapon
				.@price = 500;
				mes "Un arma de nivel 1...?";
				break;
			case 2: // Level 2 Weapon
				.@price = 2000;
				mes "Miau, un arma de nivel 2...?";
				break;
			case 3: // Level 3 Weapon
				.@price = 20000;
				mes "Miau Miau~~ Un arma de nivel 3~~";
				break;
			case 4: // Level 4 Weapon
				.@price = 50000;
				mes "Mi-Miau!... Un arma de nivel 4...!";
				mes "Solo lo he visto dos veces mientras";
				mes "aprendia de Aragam... Mi-Miau!!";
				break;
			default:
				// TODO:
				close;
		}
	}else{
		// TODO:
		close;
	}

	mes "Necesitas ^ff9999"+getitemname(.@material)+"^000000 y ^ff9999"+.@price+"^000000 Zeny para este refinamiento, miau~";
	mes "Quieres continuar, miau?~";
	next;
	if(select("Si!!:No!!") == 2) {
		mes "[Holink]";
		mes "Kyack!!";
		mes "No confias en el maestro refinador Holink, miau?~";
		close;
	}
	if (getequippercentrefinery(.@part, true) < 100) {
		mes "[Holink]";
		mes "Miau!!";
		if (.@type$ == "armor")
			mes "Esta armadura ya fue refinada muchas veces, miau.";
		else {
			mes "Peligro. Peligro~";
			mes "Esta arma fue refinada mucho, miau~";
			next;
			mes "[Holink]";
		}
		mes "Podrias romperla si continuas";
		mes "intentando refinar este objeto, miau.";
		next;
		mes "[Holink]";
		mes "Una vez que el "+.@type$+" se rompa, no podras";
		mes "usarlo nunca mas, miau. Sin mencionar... todas las";
		mes "^ff0000cartas y encantamientos actuales desapareceran seguro^000000.";
		mes "Aun quieres intentarlo, miau~?";
		next;
		if(select("Si, quiero!!:Olvidalo!!") == 2) {
			mes "[Holink]";
			mes "Miau! Sabia decision, miau.";
			mes "Pero!!";
			mes "No estoy contento de verte dudando del maestro refinador Holink, miau~";
			close;
		}
	}
	if (countitem(.@material) == 0 || Zeny < .@price) {
		mes "[Holink]";
		mes "No tienes los ingredientes.";
		mes "Necesitas ^ff9999"+getitemname(.@material)+"^000000 y ^ff9999"+.@price+"^000000 Zeny, miau~";
		mes "Ve a conseguirlos, miau~";
		close;
	}
	delitem .@material,1;
	set Zeny, Zeny-.@price;

	// anti-hack
	if (callfunc("F_IsEquipIDHack", .@part, .@refineitemid) || callfunc("F_IsEquipCardHack", .@part, .@card[0], .@card[1], .@card[2], .@card[3]) ||
		callfunc("F_IsEquipRefineHack", .@part, .@refinerycnt)) {
		mes "[Holink]";
		emotion ET_FRET;
		mes "Espera un segundo...";
		mes "Crees que soy tonto?!";
		mes "Cambiaste el objeto mientras no miraba! Fuera de aqui!";
		close;
	}

	if (getequippercentrefinery(.@part, true) > rand(100)) {
		successrefitem .@part;
		mes "[Holink]";
		mes "Mi~ Mi~ Miau! Divertido divertido refinar~";
		next;
		emotion ET_CHUP;
		mes "[Holink]";
		mes "Perfecto!! Perfecto, miau!!";
		mes "Soy el discipulo del mago del refinamiento Aragam~";
		mes "Holink!!";
		mes "Otro dia de refinamiento exitoso, miau!!";
		close;
	}
	failedrefitem .@part;
	mes "[Holink]";
	mes "Miau~ Miau~ Kyaaak!!";
	next;
	switch(rand(1,5)) {
		case 1: emotion ET_CRY; break;
		case 2: emotion ET_PROFUSELY_SWEAT; break;
		case 3: emotion ET_KEK; break;
		case 4: emotion ET_SCRATCH; break;
		case 5: emotion ET_BIGTHROB; break;
	}
	mes "[Holink]";
	mes "Miau!! Aaaaakk~~!!!!";
	mes "Kyaaak!! He fallado, miau!!";
	next;
	mes "[Holink]";
	mes "......";
	mes "......";
	mes "Todo~ Todito~ Roto, miau...";
	next;
	mes "[Holink]";
	mes "Miau.... El maestro Aragam dijo una vez,";
	mes "aprende de tus fracasos...";
	mes "Humano, este fracaso sera el comienzo de tu exito en el futuro.";
	close;
}
