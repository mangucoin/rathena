			mes "Sin embargo, para Rangers, ¡tengo un precio especial de 10,000 zeny!";

			mes "[Falcon Flute Trader]";

		}

			if (countitem(12848) >= 1) {
				mes "[Falcon Flute Trader]";
				mes "¡Puedo ver que ya tienes una Flauta de Halcon!";
				mes "No te preocupes, joven amigo, una vez que hayas comprado la Flauta de Halcon no necesitaras hacerlo de nuevo";

			mes "Aqui tienes, toma esta flauta.";
			mes "Con ella puedes silbar para llamar a tu Halcon desde cualquier lugar.";

			getitem 12848,1; //Falcon_Flute
			next;
			mes "[Falcon Flute Trader]";
			mes "Solo un recordatorio.";
			mes "¡Necesitas la Habilidad ^ff0000Falcon Mastery^000000 para poder usarla!";
			close;
		}
		mes "[Falcon Flute Trader]";
		mes "Lo siento pero no tienes suficiente dinero para comprar una Flauta de Halcon, ¡por favor vuelve cuando tengas mas dinero!";
		close;
	}
	if (BaseJob == Job_Archer) {
		mes "¡Hola, joven!";
		mes "Actualmente no puedo ayudarte,";
		mes "pero ¿por que no vuelves a visitarme cuando te hayas convertido en un gran Hunter y hayas aprendido a manejar Halcones correctamente?";
		close;
	}
	mes "¿No es un dia hermoso hoy?";
	close;
}
