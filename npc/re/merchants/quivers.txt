// Mora :: ep14_1_tre
//============================================================
mora,106,117,3	script	Quiver Maker#mora	4_M_RAFLE_GR,{
	if (checkweight(1201,1) == 0) {
		mes "[Quiver Maker]";
		mes "Tienes demasiadas cosas contigo.";
		mes "Haz espacio en tu inventario y vuelve. Te dire algo interesante.";
		close;
	}
	if (MaxWeight - Weight < 2000) {
		mes "[Quiver Maker]";
		mes "Pareces agotado con todas esas cosas.";
		mes "Haz espacio en tu inventario y vuelve. Te dire algo interesante.";
		close;
	}
	mes "[Quiver Maker]";
	mes "Los aldeanos de Mora preguntan para que sirven los carcajes. Simplemente no saben como funciona el mundo.";
	mes "No es de extrañar que no sepan nada de carcajes - untando mermelada sobre hojas todo el dia.";
	next;
	if (getbattleflag("feature.barter")) {
		.@menu$ = "Quiero flechas";
	}
	switch( select("Por favor hazme un carcaj.", .@menu$, "¿Que es un carcaj?") ) {
	case 1:
		if (getbattleflag("feature.barter")) {
			mes "[Quiver Maker]";
			mes "Finalmente, alguien ha aparecido que conoce el verdadero valor de un carcaj. Lo hare para quienes conocen su valor.";
			close2;
			callshop "Quiver_Archer", 1;
			end;
		}
		mes "[Quiver Maker]";
		mes "¡Por fin alguien aprecia un carcaj!";
		mes "Puedo hacer Elven Quivers y Hunting Quivers.";
		mes "¿Cual necesitas?";
		next;
		switch(select("Un Elven Quiver:Un Hunting Quiver:No necesito un carcaj.")) {
		case 1: callsub S_BuyQuiver,1773,500,500,12575; //Arrow_Of_Elf_Cntr
		case 2: callsub S_BuyQuiver,1774,500,500,12576; //Hunting_Arrow_Cntr
		case 3:
			mes "[Quiver Maker]";
			mes "Puedes comprar flechas del mercader de herramientas a mi lado.";
			close;
		}
		end;
	case 2:
		if (getbattleflag("feature.barter")) {
			mes "[Quiver Maker]";
			mes "Yo no vendo flechas comunes. Estas son flechas artesanales que solo se pueden comprar aqui. Reconocerias su valor, ¿verdad?";
			close2;
			callshop "Catch_Arrows", 1;
		}
		end;
	case 3:
		mes "[Quiver Maker]";
		mes "Una flecha puede ser delgada y ligera, pero cargar cientos y miles de flechas es como cargar un tronco entero.";
		next;
		mes "[Quiver Maker]";
		mes "Pero cuando tienes carcajes, puedes poner flechas en ellos y ahorrar peso y espacio.";
		mes "Si te interesa uno, te coso uno al instante.";
		close;
	}
	end;

S_BuyQuiver:
	.@item_req = getarg(0);
	.@req_amount = getarg(1);
	if (countitem(getarg(0)) < .@req_amount) {
		mes "[Quiver Maker]";
		mes "Trae mas de "+ .@req_amount +" "+ getitemname(.@item_req) +" y te hare un carcaj.";
		close;
	}
	mes "[Quiver Maker]";
	mes "Oh, ¡veo que tienes "+ getitemname(.@item_req) +" en tu mano!";
	mes "¿Te interesa usar un carcaj? ¡Es realmente conveniente!";
	mes "Si te interesa, puedo intercambiar "+ .@req_amount +" de esas flechas por uno de estos carcajes por ^FF3131"+getarg(2)+" zeny^000000.";
	next;
	switch(select("Intercambiar todas las flechas que tengas:Obtener solo un carcaj:No intercambiar")) {
	case 1:
		set .@arrows, countitem(.@item_req);
		set .@quiver, .@arrows / .@req_amount;
		set .@arrows_used, .@quiver * .@req_amount;
		set .@arrow_zeny01, .@quiver * getarg(2);
		mes "La cantidad de flechas que tienes : ^3131FF"+.@arrows+"^000000";
		mes "La cantidad de carcajes disponibles : ^3131FF"+.@quiver+"^000000";
		mes "Zeny necesarios para el intercambio : ^3131FF"+.@arrow_zeny01+" zeny^000000";
		next;
		mes "¿Intercambiar?";
		next;
		if(select("Intercambiar:No intercambiar") == 2) {
			mes "[Quiver Maker]";
			mes "Oye, ¿no dudaras de mis habilidades, verdad?";
			close;
		}
		break;
	case 2:
		set .@quiver, 1;
		set .@arrows_used, .@req_amount;
		set .@arrow_zeny01, getarg(2);
		set .@zeny_mes,1;
		break;
	case 3:
		mes "[Quiver Maker]";
		mes "Oye, ¿no dudaras de mis habilidades, verdad?";
		close;
	}
	if (Zeny < .@arrow_zeny01) {
		mes "[Quiver Maker]";
		if (.@zeny_mes == 1)
			mes "Dije que aceptaria monedas humanas solo por ti, ¿y aun asi no quieres gastar el dinero?";
		else
			mes "Realmente no esperas conseguir una obra maestra gratis, ¿verdad?";
		close;
	}
	mes "[Quiver Maker]";
	mes "Oye, aqui tienes.";
	mes "Hay ^3131FFalgo que necesitas saber^000000 - intenta recordarlo.";
	mes "^FF0000No puedes usar carcajes cuando tu carga supera el 70%.^000000";
	mes "Mejor tenlo en cuenta, o podrias tener problemas despues.";
	set Zeny, Zeny-.@arrow_zeny01;
	delitem .@item_req,.@arrows_used;
	getitem getarg(3),.@quiver;
	close;
}
