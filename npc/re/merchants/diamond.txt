-	script	RareDiamondMerchant	58,{

	set .@npc$, "[Rare Diamond Merchant]";

	mes .@npc$;
	mes "¡¡¡Damas y Caballeros!!!";
	mes "Han escuchado los rumores pero nunca lo han visto con sus propios ojos";
	mes "¡el diamante mas raro del mundo!";
	mes "¡El ^FF82FF'Diamante de 17 Quilates'^000000 esta actualmente en venta a un precio barato!!";
	mes "¡Si los diamantes valiosos te interesan entonces presta atencion!";
	next;

	mes .@npc$;
	mes "Puedes regalar el diamante a tu ser amado.";
	mes "Quien reciba el ^FF82FF'Diamante de 17 Quilates'^000000 lo atesorara";
	mes "y te guardara en su corazon.";
	emotion ET_BLABLA;
	next;

	mes .@npc$;
	mes "¿Que? ¿No tienes pareja?";
	mes "Aun puedes quedartelo para ti.";
	mes "El aura encantadora del ^FF82FF'Diamante de 17 Quilates'^000000 probablemente";
	mes "traera amor a tu vida.";
	next;

	mes .@npc$;
	mes "Conocido como el cautivador diamante de diamantes...";
	mes "¡Es el ^FF82FF'Diamante de 17 Quilates'^000000!";
	next;

	set .@menu$, "¿^FF82FF'Diamante de 17 Quilates'^000000?";
	if (countitem(6024))
		set .@menu$, .@menu$ + ":Cambiar mi Diamante de 17 Quilates por zeny...";

	switch(select(.@menu$)) {
	case 1:
		mes .@npc$;
		mes "¡Si!";
		mes "Jeje, el ^FF82FF'Diamante de 17 Quilates'^000000 que ves ahora";
		mes "es uno de los diamantes de 17 quilates de la mas alta calidad.";
		mes "¡Compararlo con cualquier piedra incolora del tamaño de un guisante que hayas";
		mes "desenterrado en Morocc es inaceptable!";
		emotion ET_HUK;
		next;

		mes .@npc$;
		mes "¡Algunos dicen que este es el mismo diamante";
		mes "que fue visto por Kachua en Comodo!";
		next;

		if (select("Terminar Conversacion:¿Cuanto cuesta?") == 1) {
			mes .@npc$;
			mes "¡Es bastante economico!";
			mes "Si te interesa, ¡habla conmigo cuando quieras!";
			close;
		}

		mes .@npc$;
		mes "¡Por supuesto! ¡Lo mas importante es el precio!";
		mes "Como dije antes, el ^FF82FF'Diamante de 17 Quilates'^000000 ha sido";
		mes "tasado como un diamante de 17 quilates.";
		mes "No hay muchos como este en el mundo.";
		next;

		mes .@npc$;
		mes "¡El precio es de tan solo 500 millones de zeny!";
		mes "Tambien hay una tarifa de servicio de 1,000,000z.";
		mes "Comparado con la calidad del diamante";
		mes "es un precio pequeño a pagar.";
		next;

		if (select("^828282Es demasiado caro.^000000:Lo comprare.") == 1) {
			mes .@npc$;
			mes "Decir que un diamante como el";
			mes "^FF82FF'Diamante de 17 Quilates'^000000 es demasiado caro,";
			mes "me dejas sin palabras.";
			close;
		}

		mes .@npc$;
		mes "¡OOOH! Supe desde el momento en que te vi que eras el dueño legitimo";
		mes "de este maravilloso diamante.";
		mes "¡Me alegra no haberme equivocado!";
		mes "Creo que este diamante te quedara muy bien.";
		mes "Todo lo que necesito de ti es completar este contrato.";
		next;

		while(1) {
			if (select("^B9062FFirmare el contrato.^000000:^828282No importa, cambie de opinion.^000000") == 2) {
				mes .@npc$;
				mes "Ya veo, que lastima..";
				close;
			}

			mes .@npc$;
			mes "¿Cual es tu nombre?";
			next;

			mes "[" + strcharinfo(0) + "]";
			mes "Mi nombre es " + strcharinfo(0) + ".";
			next;

			mes .@npc$;
			mes "Muy bien, por favor firma aqui.";
			next;

			input .@charname$;
			if (.@charname$ == strcharinfo(0)) {
				mes .@npc$;
				mes "Bien, perfecto.";
				mes "Todo lo que queda ahora es el pago.";
				next;

				if (Zeny >= 501000000) {
					mes .@npc$;
					mes ". . . . . .";
					next;

					mes .@npc$;
					mes "Calculando el precio y la tarifa de servicio..";
					mes "501,000,000 zeny.";
					mes "Monto confirmado.";
					emotion ET_BLABLA;
					set Zeny, Zeny - 501000000;
					getitem 6024, 1;
					next;

					mes .@npc$;
					mes "Puedes cambiar el diamante de vuelta por zeny en cualquier momento.";
					mes "Sin embargo, habra una tarifa de servicio de 1,000,000z.";
					mes "Gracias por tu compra.";
					emotion ET_THANKS;
					close;
				}
				else {
					mes .@npc$;
					mes "¿Hm?";
					mes "Lo siento, pero no tienes fondos suficientes.";
					mes "Incluyendo la tarifa de servicio, se requiere un total de 501,000,000z.";
					mes "Por favor verifica tu saldo de zeny e intenta de nuevo.";
					emotion ET_SCRATCH;
					close;
				}
			}
			else {
				mes .@npc$;
				mes "¿Esta es realmente tu firma?";
				mes "¿Firmaras de nuevo para confirmar?";
				next;
			}
		}
	case 2:
		mes .@npc$;
		mes "AAH. ¿Te gustaria cambiar tu ^FF82FF'Diamante de 17 Quilates'^000000 por zeny?";
		emotion ET_SURPRISE;
		next;
		if (select("Si:No") == 2) {
			mes .@npc$;
			mes "Entonces, ¿que es lo que deseas?..";
			mes ". . . . . ....";
			emotion ET_THINK;
			close;
		}

		mes .@npc$;
		mes "Ya veo. Despues de recibir tu firma,";
		mes "deberas entregar tu ^FF82FF'Diamante de 17 Quilates'^000000.";
		mes "¡Tambien no olvides que hay una tarifa de servicio de 1,000,000z!";
		next;

		mes .@npc$;
		mes "¿Cual es tu nombre?";
		next;

		mes "[" + strcharinfo(0) + "]";
		mes "Mi nombre es " + strcharinfo(0) + ".";
		next;

		mes .@npc$;
		mes "Muy bien, por favor firma aqui.";
		next;

		while (1) {
			if (select("Firmar:No Firmar") == 2) {
				mes .@npc$;
				mes "El intercambio no puede completarse sin tu firma.";
				close;
			}

			input .@charname$;
			if (.@charname$ == strcharinfo(0)) {
				mes .@npc$;
				mes "Tu firma ha sido recibida.";
				mes "Ahora tomare tu ^FF82FF'Diamante de 17 Quilates'^000000..";
				mes "Recibiras ^0000FF499,000,000z^000000 despues.";
				next;

				mes .@npc$;
				mes "Gracias por tu compra.";
				mes "Cuando necesites un ^FF82FF'Diamante de 17 Quilates'^000000,";
				mes "vuelve cuando quieras con 500 millones de zeny.";
				emotion ET_THANKS;
				delitem 6024, 1;
				set Zeny, Zeny + 499000000;
				close;
			}
			else {
				mes .@npc$;
				mes "¿Esta es realmente tu firma?";
				mes "¿Firmaras de nuevo para confirmar?";
				next;
			}
		}
	}
}
