moc_para01,22,16,5	script	Catalog Magician#catal01	704,{
	if (!checkweight(1301,3)) {
		mes "- Tienes demasiados objetos. Por favor haz espacio. -";
		close;
	}
	set .@ticket_cost, 200;
	mes "[Catalog Magician]";
	mes "Mira... la Academia de Magia";
	mes "en Geffen ahora esta vendiendo";
	mes "directamente el nuevo";
	mes "^007777Universal Catalog Silver^000000!";
	next;
	mes "Esta mirando las palabras";
	mes "escritas en su mano.";
	mes "Parece demasiado ocupado";
	mes "como para mirarte.";
	next;
	mes "[Catalog Magician]";
	mes "¡Un catalogo completamente nuevo";
	mes "acaba de salir!";
	mes "Puedes verificar";
	mes "que objetos puedes comerciar";
	mes "y que objetos estan en venta";
	mes "del vendedor.";
	next;
	mes "[Catalog Magician]";
	mes "El ^007777Universal Catalog Silver^000000";
	mes "¡cuesta solo 200z!";
	mes "¡Puedes comprar hasta 50 piezas a la vez!";
	next;
	switch(select("Comprar un ^007777Universal Catalog Silver^000000:¡No lo necesito!:¡No pareces un vendedor!")) {
	case 1:
		mes "[Catalog Magician]";
		mes "¿Quieres las 50 piezas?";
		mes "¿O cuantas necesitas?";
		next;
		while (1) {
			input .@input;
			mes "[Catalog Magician]";
			if (.@input == 0) {
				mes "¡El intercambio ha sido detenido!";
				mes "No se que hacer ahora...";
				mes "¿Que deberia hacer?";
				emotion ET_PROFUSELY_SWEAT, playerattached();
				close;
			} else if (.@input < 1 || .@input > 50) {
				mes "Deben ser menos de 50 piezas.";
				next;
			} else
				break;
		}
		set .@sell, .@ticket_cost * .@input;
		mes "La cantidad total de catalogo(s) que intentas comprar es de " + .@input + " piezas.";
		mes "Cuesta " + .@sell + " z.";
		if (Zeny < .@sell) {
			mes "Parece que no tienes suficiente dinero.";
			mes "Puedes vender tu equipo";
			mes "para ganar mas dinero... si quieres.";
			close;
		}
		mes "Ok, aqui tienes.";
		set Zeny, Zeny - .@sell;
		getitem 12580,.@input; //Vending_Search_Scroll
		close;
	case 2:
		mes "[Catalog Magician]";
		mes "Argh... realmente necesitas este";
		mes "objeto... ¿Quieres que te";
		mes "explique de nuevo? No se";
		mes "que hacer ahora...";
		mes "¿Que deberia hacer?";
		emotion ET_PROFUSELY_SWEAT, playerattached();
		close;
	case 3:
		mes "[Catalog Magician]";
		mes "Mi verdadero trabajo es mago.";
		mes "Iba a pedirle al gremio de";
		mes "mercaderes que los vendiera, pero";
		mes "estaban intentando usarlos como";
		mes "complemento para venta de escobas. Por eso";
		mes "los estoy vendiendo yo mismo.";
		next;
		mes "[Catalog Magician]";
		mes "Hay muchos aventureros";
		mes "aqui que van y vienen.";
		mes "Estoy bastante seguro de que podre venderle a todos ellos pronto.";
		mes "¿No crees?";
		mes "Asi que... ¡empecemos!! ¡Compra mas, por favor! ¡¡POR FAVOR...!!";
		close;
	}
}
