-	script	GuildWarehouse	-1,{
	mes "[Warehouse Manager]";
	mes "¿Como estas? Somos especialistas en almacenes de gremio. Esta es nuestra ^0000cdhistoria^000000:";
	mes "¿Por que los miembros del gremio no pueden compartir un almacen? Empezamos con esa simple pregunta.";
	next;
	mes "[Warehouse Manager]";
	mes "Puedes abrir el almacen por 1000 Zeny o consultar el historial de uso del gremio.";
	mes "¿En que puedo ayudarte?";
	next;
	if( select( "Abrir almacen del gremio:Ver historial de uso del almacen" ) == 1 ){
		.@guildid = getcharid( 2 );

		if( .@guildid == 0 ){
			mes "[Warehouse Manager]";
			mes "El almacen del gremio solo esta disponible para miembros del gremio.";
			close;
		}

		if( getgdskilllv( .@guildid, "GD_GUILD_STORAGE" ) == 0 || !guild_has_permission( GUILD_PERM_STORAGE ) ){
			mes "[Warehouse Manager]";
			mes "Parece que aun no es posible que tu gremio use el almacen del gremio o no tienes permisos de acceso al almacen.";
			mes "Por favor vuelve despues de verificar la habilidad del gremio y el permiso de acceso al almacen.";
			close;
		}

		if( Zeny < 1000 ){
			mes "[Warehouse Manager]";
			mes "Me temo que no tienes suficiente dinero para pagar la tarifa.";
			mes "La tarifa es de 1000 Zeny.";
			close;
		}

		mes "[Warehouse Manager]";
		mes "Entonces te abrire el almacen del gremio. ¡Que tengas un buen momento!";
		close2;

		if( Zeny < 1000 ){
			// Cheat prevention
			end;
		}

		if( guildopenstorage() == GSTORAGE_OPEN ){
			Zeny -= 1000;
			end;
		}else{
			mes "[Warehouse Manager]";
			mes "Me temo que alguien mas esta usando el almacen en este momento.";
			mes "Por favor intenta de nuevo despues de un momento.";
			close;
		}
	}else{
		.@guildid = getcharid( 2 );

		if( .@guildid == 0 ){
			mes "[Warehouse Manager]";
			mes "El almacen del gremio solo esta disponible para miembros del gremio.";
			close;
		}

		if( getgdskilllv( .@guildid, "GD_GUILD_STORAGE" ) == 0 || !guild_has_permission( GUILD_PERM_STORAGE ) ){
			mes "[Warehouse Manager]";
			mes "Parece que aun no es posible que tu gremio use el almacen del gremio o no tienes permisos de acceso al almacen.";
			mes "Por favor vuelve despues de verificar la habilidad del gremio y el permiso de acceso al almacen.";
			close;
		}

		mes "[Warehouse Manager]";
		mes "Te mostrare el historial de uso del almacen. El historial de uso se conserva hasta por 3 meses.";
		mes "¡Que tengas un buen momento!";
		close2;
		guildopenstorage_log();
		end;
	}
}
