// Rare Herb Merchants (Guillotine Cross) :: rare_herb_3rd_trader
//============================================================
job3_guil01,91,93,3	script	Rare Herb Collector	1_M_03,{
	if (checkweight(1201,1) == 0 || MaxWeight - Weight < 2000) {
		mes "- ¡Espera un momento! -";
		mes "- Actualmente estas cargando -";
		mes "- demasiados objetos contigo. -";
		mes "- Por favor intenta de nuevo -";
		mes "- despues de reducir tu peso. -";
		close;
	}
	if (Class == Job_Guillotine_Cross || Class == Job_Guillotine_Cross_T || Class == Job_Baby_Guillotine_Cross) {
		mes "[Rare Poison Herb Collector]";
		mes "Viajo por el mundo recolectando hierbas venenosas raras. Recientemente, empece a comerciar con la hierba llamada Izidor. Si te interesa, puedes comprarlas.";
		next;
		.@i = select("¿Como puedo comprarlas?:Cambiar por Animal Blood:Cambiar por Bitter Herb:Cambiar por Deadly Noxious Herb:Cambiar por Frozen Rose:Cambiar por Ment:Cambiar por Hinalle")-2;
		if (.@i == -1) {
			mes "[Rare Poison Herb Collector]";
			mes "Puedes comprarlas por 10,000 zeny y uno de estos objetos raros: Animal Blood, Bitter Herb, Deadly Noxious Herb, Frozen Rose, Ment o Hinalle.";
			next;
			mes "[Rare Poison Herb Collector]";
			mes "¿Por que pido dinero? Es solo para mostrar un poco de respeto a un coleccionista como yo. Ja ja ja...";
			close;
		} else {
			setarray .@exchange[0],702,621,631,749,605,703;
			.@item = .@exchange[.@i];
			.@price = 10000;

			mes "[Rare Poison Herb Collector]";
			mes "¿Cuantas quieres?";
			mes "Solo puedes comprar un maximo de '2000' unidades.";
			mes "Ingresa '0' si quieres cancelar.";
			next;
			input .@amount;
			if (.@amount == 0) {
				mes "[Rare Poison Herb Collector]";
				mes "Has cancelado el intercambio.";
				close;
			}
			if (.@amount > 2000) {
				mes "[Rare Poison Herb Collector]";
				mes "¡La cantidad debe ser menor a 2,000!";
				close;
			}
			.@total = .@amount * .@price;
			if (countitem(.@item) < .@amount || Zeny < .@total) {
				mes "[Rare Poison Herb Collector]";
				mes "Oye, vuelve cuando tengas todos los requisitos para el intercambio.";
				close;
			}
			mes "[Rare Poison Herb Collector]";
			mes "Bien. He recibido el dinero y el objeto especial.";
			delitem .@item, .@amount;
			Zeny = Zeny - .@total;
			getitem 709, .@amount; //Izidor
			close;
		}
	}
	mes "[Rare Poison Herb Collector]";
	mes "Viajo por el mundo y recolecto hierbas venenosas raras. Pero no tengo ganas de venderle mis hierbas a alguien como tu... ja ja ja...";
	close;
}
