einbech,149,257,4	script	Site manager Elmen#extdq0	4_TOWER_16,{
	if (checkweight(25814,100) == 0 || checkweight(1201,1) == 0 || (MaxWeight - Weight) < 10000) {
		mes "- No puedes continuar porque tienes demasiados articulos. -";
		close;
	}
	mes "[Elmen]";
	mes "Hola. Aventurero.";
	mes "Si estas aqui para comprar equipo de mineria Dynite o refinar mineral en Dynite, has venido al lugar correcto.";
	next;
	switch( select( "Refinamiento de Dynite", "Tienda de Dynite", "Intercambio de Equipo Dynite" ) ) {
	case 1:
		npctalk "Elmen: Puedes refinar hasta 100 Dynite a la vez. La maquina es pequeña.", "", bc_self;
		mes "[Elmen]";
		mes "" + mesitemlink(25814) + " Mineral Refinado.";
		mes "" + mesitemlink(25809) + " ^777777(actualmente " + countitem(25809) + ")^000000";
		mes "" + mesitemlink(25810) + " ^777777(actualmente " + countitem(25810) + ")^000000";
		mes "" + mesitemlink(25811) + " ^777777(actualmente " + countitem(25811) + ")^000000";
		mes "" + mesitemlink(25812) + " ^777777(actualmente " + countitem(25812) + ")^000000";
		mes "" + mesitemlink(25813) + " ^777777(actualmente " + countitem(25813) + ")^000000";
		mes "Necesitas uno de cada uno de los cinco tipos de mineral.";
		next;
		input .@val;
		if (.@val > 100) {
			mes "[Elmen]";
			mes "Lo siento. Aventurero.";
			mes "El Refinamiento de Dynite solo puede refinar hasta 100 a la vez. La maquina es pequeña, asi que no se puede hacer nada. Por favor entiende.";
			close;
		}
		if (.@val == 0) {
			mes "[Elmen]";
			mes "¿Cancelar?";
			mes "Si necesitas refinamiento de Dynite, buscame cuando quieras. No hay nadie que pueda hacerlo sin mi.";
			close;
		}
		if (countitem(25809) < .@val || countitem(25810) < .@val || countitem(25811) < .@val || countitem(25812) < .@val || countitem(25813) < .@val) {
			mes "[Elmen]";
			mes "¡Oye! Esto nos lo pone dificil a ambos, ¿no te quedaste sin materiales necesarios para el refinamiento? ¡Los materiales!";
			close;
		}
		specialeffect EF_REPAIRWEAPON;
		progressbar_npc "FFFF00",1;
		delitem 25809,.@val;	// Ein_REDIUM
		delitem 25810,.@val;	// Ein_RINDIUM
		delitem 25811,.@val;	// Ein_ODIUM
		delitem 25812,.@val;	// Ein_PURDIUM
		delitem 25813,.@val;	// Ein_WHIDIUM
		getitem 25814,.@val;	// Ein_DYNITE
		mes "[Elmen]";
		mes "" + .@val + " Dynite han sido refinados. ¡Hoy quedo perfecto!";
		close;
	case 2:
		mes "[Elmen]";
		mes "Hay varios accesorios y permisos de modificacion. Mira bien lo que quieres antes de comprar.";
		close2;
		callshop "barter_ein_dun03";
		end;
	case 3:
		mes "[Elmen]";
		mes "Puedo intercambiar el equipo con el Dynite, pero ya esta en una caja sellada firmemente.";
		mes "Dame 50 ^4d4dffDynites^000000 y abrire una ^4d4dffcaja de armas^000000 para ti.";
		next;
		mes "[Elmen]";
		mes "No se que saldra, pero la calidad es buena. ¿Te gustaria abrirla?";
		next;
		if (select( "Cerrar los ojos e intercambiar.", "No intercambiar." ) == 2) {
			mes "[Elmen]";
			mes "Kee, que desperdicio, era un buen trato.";
			close;
		}
		if (countitem(25814) < 50) {
			mes "[Elmen]";
			mes "No tienes suficientes Dynite, amigo.";
			mes "No se que va a salir, pero es dificil ir a otro lugar y conseguir este tipo de equipo a este precio.";
			close;
		}
		delitem 25814,50;
		// unknown text
		getgroupitem(IG_EIN_1HITEMBOX);
		end;
	}
}
