		mes "Hay un problema, ¡por favor vuelve de nuevo!";// custom
		close;
	}
	if (.@card[3] == 0 && .@sc_type < 4)
		.@index_slot = 3;
	else if (.@card[2] == 0 && .@sc_type < 3)
		.@index_slot = 2;
	else if (.@card[1] == 0 && .@sc_type < 2)
		.@index_slot = 1;
	else {
		mes "No se que es esto, pero creo que no es algo que pueda manejar.";// custom
		close;
	}
	if (countitem(7642) > 14) {// Bloody_Coin
		delitem 7642,15;// Bloody_Coin

		// anti-hack
		if (callfunc("F_IsEquipIDHack", .@eq_num, .@item_id) || callfunc("F_IsEquipCardHack", .@eq_num, .@card[0], .@card[1], .@card[2], .@card[3]) || callfunc("F_IsEquipRefineHack", .@eq_num, .@refine))
			close;

		delequip .@eq_num;
		if (.@enchant == 0) {
			mes "¡Ouch! Pobre de ti, el encantamiento ha fallado y tu equipo se rompio, ¡cielos!";
			close2;
			specialeffect2 EF_LORD;
			end;
		}
		.@card[ .@index_slot ] = .@enchant;
		getitem2 .@item_id,1,1,.@refine,0,.@card[0],.@card[1],.@card[2],.@card[3];
		mes "Hmm.. Fue exitoso. Echale un vistazo.";
		specialeffect2 EF_REPAIRWEAPON;
		close;
	}
	mes "Hmm.. Por cierto, necesitas juntar Bloody Coins. Ni siquiera puedes pagar la tarifa.";
	close;
}
