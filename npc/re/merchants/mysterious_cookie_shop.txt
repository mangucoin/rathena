malangdo,124,152,3	script	Mysterious Cookie Shop#stone	4_COOKIEHOUSE,{
	mes "[Mysterious Cookie Shop]";
	mes "" + mesitemlink(7000) + " se puede intercambiar por cualquier cosa que desees.";
	next;
	switch( select( "Piedras de ^4d4dffCabeza^000000", "Piedras de ^1C8008Medio^000000", "Piedras ^ff0000Inferiores^000000", "Piedras de ^9905d8Capa^000000", "Piedras ^8B4513Duales y de efecto^000000", "^9905d8Costumes^000000 en el tendedero" ) ) {
	case 1:
		mes "[Mysterious Cookie Shop]";
		mes "Veamos que piedras puedes obtener con tu millaje!";
		close2;
		callshop( "barter_stone_top" );
		end;
	case 2:
		mes "[Mysterious Cookie Shop]";
		mes "Veamos que piedras puedes obtener con tu millaje!";
		close2;
		callshop( "barter_stone_mid" );
		end;
	case 3:
		mes "[Mysterious Cookie Shop]";
		mes "Veamos que piedras puedes obtener con tu millaje!";
		close2;
		callshop( "barter_stone_low" );
		end;
	case 4:
		mes "[Mysterious Cookie Shop]";
		mes "Veamos que piedras puedes obtener con tu millaje!";
		close2;
		callshop( "barter_stone_garment" );
		end;
	case 5:
		mes "[Mysterious Cookie Shop]";
		mes "Veamos que piedras puedes obtener con tu millaje!";
		close2;
		callshop( "barter_stone_dual" );
		end;
	case 6:
		mes "[Mysterious Cookie Shop]";
		mes "Usa el millaje que tienes para sacudir el tendedero donde cuelgan todo tipo de ropas!";
		next;
		switch( select( "Primer tendedero", "Segundo tendedero" ) ) {
		case 1:
			callshop( "barter_costume_1" );
			end;
		case 2:
			callshop( "barter_costume_2" );
			end;
		}
	}
}
