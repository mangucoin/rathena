//============================================================
// +11 and above Refiner Function
//============================================================
//= To allow auto safe refining/multiple refining set the
//= second argument to '1' in the function call.
//= If you enable this function, be sure to edit the value of
//= .@safe to the max safe refine in refine.yml as well.
//=
//= On official servers, if an item is unsuccessfully refined
//= it will break at a 20% rate and downgrade at an 80% rate.
//============================================================
function	script	refinenew	{
	.@npc_name$ = getarg(0);
	disable_items;
	mes "["+ .@npc_name$ +"]";
	mes "Soy el mejor Herrero de todos!";
	mes "No refino articulos normales y aburridos.";
	mes "Solo refino articulos que son Nivel 10 o mayor.";
	next;
	mes "["+ .@npc_name$ +"]";
	mes "De todos modos, puedes usar mis servicios si tu articulo es Nivel 10 o mayor.";
	mes "Que quieres que refine?";
	next;

	setarray .@indices[1], EQI_HEAD_TOP, EQI_ARMOR, EQI_HAND_L, EQI_HAND_R, EQI_GARMENT, EQI_SHOES, EQI_ACC_L, EQI_ACC_R, EQI_HEAD_MID, EQI_HEAD_LOW;
	for(.@i = 1; .@i<=10; ++.@i) {
		if (getequipisequiped(.@indices[.@i])) {
			.@menu$ = .@menu$ + F_getpositionname(.@indices[.@i]) + "-[" + getequipname(.@indices[.@i]) + "]";
			.@equipped = 1;
		}
		.@menu$ = .@menu$ + ":";
	}
	if (.@equipped == 0) {
		mes "["+ .@npc_name$ +"]";
		mes "No creo que pueda refinar ningun articulo que tengas...";
		close;
	}
	.@part = .@indices[ select(.@menu$) ];

	if (!getequipisequiped(.@part)) { //custom check
		mes "["+ .@npc_name$ +"]";
		mes "No llevas puesto";
		mes "nada ahi que";
		mes "pueda refinar.";
		emotion ET_FRET;
		close;
	}
	if (!getequipisenableref(.@part)) {
		mes "["+ .@npc_name$ +"]";
		mes "No creo que pueda";
		mes "refinar este articulo en absoluto...";
		close;
	}
	.@refinerycnt = getequiprefinerycnt(.@part); //save refinery count
	if (.@refinerycnt < 10) {
		mes "["+ .@npc_name$ +"]";
		mes "Dije que no trabajo con articulos menores a Nivel 10.";
		close;
	}
	if (.@refinerycnt >= 20) { //custom check
		mes "["+ .@npc_name$ +"]";
		mes "No puedo refinar esto";
		mes "mas. Esto esta tan";
		mes "refinado como se puede!";
		close;
	}
	.@refineitemid = getequipid(.@part); // save id of the item
	setarray .@card[0], getequipcardid(.@part,0), getequipcardid(.@part,1), getequipcardid(.@part,2), getequipcardid(.@part,3);
	.@price = getequiprefinecost(.@part, REFINE_COST_NORMAL, REFINE_ZENY_COST);
	.@material = getequiprefinecost(.@part, REFINE_COST_NORMAL, REFINE_MATERIAL_ID);
	.@safe = 10;
	.@itemtype = getiteminfo( .@refineitemid, ITEMINFO_TYPE );
	
	if( .@itemtype == IT_WEAPON ){
		.@article$ = "a";
		.@type$ = "weapon";
	}else if( .@itemtype == IT_ARMOR ){
		.@article$ = "an";
		.@type$ = "armor";
	}else{
		// TODO:
		close;
	}

	mes "["+ .@npc_name$ +"]";
	mes "Hmm " + .@article$ + " " + .@type$ + ", esta bien?";
	mes "Si quieres refinar este " + .@type$ + ",";
	mes "Necesitare 1 ^003366" + getitemname(.@material) + "^000000 y " + callfunc("F_InsertComma",.@price) + " zeny.";
	mes "Estas seguro de que quieres continuar?";
	next;
	if (select("Si:No") == 2) {
		mes "["+ .@npc_name$ +"]";
		mes "Hm... si te importa... olvidalo...";
		close;
	}
	if (getarg(1) != 1) {
		if (getequippercentrefinery(.@part) < 100) {
			mes "["+ .@npc_name$ +"]";
			mes "Este "+.@type$+" ya ha sido refinado varias veces.";
			mes "Podria destruirse si lo intento de nuevo.";
			mes "No se rompera seguro, pero hay una pequena posibilidad.";
			next;
			mes "["+ .@npc_name$ +"]";
			mes "Podrias estar ^FF0000bajando el nivel de mejora^000000 del "+.@type$+",";
			mes "o si se rompe, perderas ^FF0000cualquier carta^000000 o propiedad especial anadida.";
			next;
			mes "["+ .@npc_name$ +"]";
			mes "Aun quieres que lo refine?";
			mes "Creo que te he dado suficiente advertencia.";
			next;
			if(select("Si.:No.") == 2) {
				mes "["+ .@npc_name$ +"]";
				mes "Bueno, no arriesgarse es una opcion...";
				mes "Sin riesgo... eso podria ser sabio.";
				close;
			}
		}
		if (countitem(.@material) < 1 || Zeny < .@price) {
			mes "["+ .@npc_name$ +"]";
			mes "Hm. You don't seem to have enough money or "+getitemname(.@material)+".";
			mes "Please come back when you have them.";
			close;
		}
		Zeny = Zeny - .@price;
		delitem .@material,1;

		// anti-hack
		if (callfunc("F_IsEquipIDHack", .@part, .@refineitemid) || callfunc("F_IsEquipCardHack", .@part, .@card[0], .@card[1], .@card[2], .@card[3]) ||
		    callfunc("F_IsEquipRefineHack", .@part, .@refinerycnt)) {
			mes "["+ .@npc_name$ +"]";
			emotion ET_FRET;
			mes "Espera un segundo...";
			mes "Crees que soy estupido?!";
			mes "Cambiaste el articulo mientras no miraba! Fuera de aqui!";
			close;
		}

		if (getequippercentrefinery(.@part) > rand(100)) {
			mes "Clang! Clang! Clang! Clang!";
			successrefitem .@part;
			next;
			emotion ET_BEST;
			mes "["+ .@npc_name$ +"]";
			mes "Bien! Exito!!!";
			mes "Soy el mejor Herrero.";
			close;
		} else {
			if (rand(100) < 80) {
				mes "["+ .@npc_name$ +"]";
				mes "Clang! Clang! Clang! Clang!";
				downrefitem .@part,3; // Failed refine attempts decrease the item's refine level by 3
				next;
				emotion (!rand(5))?ET_MONEY:ET_HUK;
				mes "["+ .@npc_name$ +"]";
				mes "Ahhh!!!";
				next;
				mes "["+ .@npc_name$ +"]";
				mes "Dios mio!";
				mes "El nivel de mejora ha bajado...";
			} else {
				mes "["+ .@npc_name$ +"]";
				mes "Clang! Clang! Clang!";
				failedrefitem .@part;
				next;
				emotion (!rand(5))?ET_MONEY:ET_HUK;
				mes "["+ .@npc_name$ +"]";
				mes "Hmmm!";
				next;
				mes "["+ .@npc_name$ +"]";
				mes "Dios mio! No pude refinar el articulo...";
				mes "No fue mi intencion!";
			}
			mes "Pude haber cometido un error aunque soy el mejor Herrero de todos.";
			mes "Simplemente no estaba destinado a ser.";
			next;
			mes "["+ .@npc_name$ +"]";
			mes "Lo hare mejor la proxima vez! No te preocupes!";
			close;
		}
	}
// New +11 and above Refining Functions ========================
	if (.@refinerycnt < .@safe) {
		mes "["+ .@npc_name$ +"]";
		mes "Puedo refinar esto al limite seguro o un numero deseado de veces. Tu decides.";
		next;
		.@menu2 = select("Al limite seguro, por favor.","Yo decidire cuantas veces.","Cambie de opinion...");
	} else
		.@menu2 = 2;
	switch(.@menu2){
	case 1: 
		.@refinecnt = .@safe - .@refinerycnt;
		break;
	case 2:
		mes "["+ .@npc_name$ +"]";
		mes "Cuantas veces te gustaria que refinara tu articulo?";
		next;
		input .@refinecnt;
		.@refinecheck = .@refinecnt + .@refinerycnt;
		if (.@refinecnt < 1 || .@refinecheck > 20) {
			mes "["+ .@npc_name$ +"]";
			mes "No puedo refinar este articulo tantas veces.";
			close;
		}
		if (.@refinecheck > .@safe) {
			.@refinecheck = .@refinecheck - .@safe;
			mes "["+ .@npc_name$ +"]";
			mes "Esto intentara refinar el equipo " + .@refinecheck + " veces mas alla del limite seguro. Tu equipo puede ser destruido... esta bien?";
			next;
			if(select("Si...","No...") == 2){
				mes "["+ .@npc_name$ +"]";
				mes "Tu lo dijiste... Que asi sea.";
				close;
			}
		}
		break;
	case 3:
		mes "["+ .@npc_name$ +"]";
		mes "Tu lo dijiste... Que asi sea.";
		close;
	}
	.@fullprice = .@price * .@refinecnt;
	mes "["+ .@npc_name$ +"]";
	mes "Eso te costara " + .@refinecnt + " " + getitemname(.@material) + " y " + .@fullprice + " Zeny. Esta bien?";
	next;
	if(select("Si:No...") == 2){
		mes "["+ .@npc_name$ +"]";
		mes "Tu lo dijiste... Que asi sea.";
		close;
	}
	if (countitem(.@material) < .@refinecnt || Zeny < .@fullprice) {
		mes "["+ .@npc_name$ +"]";
		mes "Hm. Parece que no tienes suficiente dinero o "+getitemname(.@material)+".";
		mes "Por favor vuelve cuando los tengas.";
		close;
	}
	Zeny = Zeny - .@fullprice;
	delitem .@material,.@refinecnt;
	while(.@refinecnt){
		if (getequipisequiped(.@part) == 0) {
			mes "["+ .@npc_name$ +"]";
			mes "Mira aqui... no tienes ningun articulo puesto...";
			close;
		}
		// anti-hack
		if (callfunc("F_IsEquipIDHack", .@part, .@refineitemid) || callfunc("F_IsEquipCardHack", .@part, .@card[0], .@card[1], .@card[2], .@card[3]) ||
		    callfunc("F_IsEquipRefineHack", .@part, .@refinerycnt))
			close;

		if (getequipid(.@part) != .@refineitemid || (.@menu2 == 1 && getequippercentrefinery(.@part) < 100)) {
			mes "["+ .@npc_name$ +"]";
			mes "Clang... No, pero te imaginaste que podia ser tan estupido?!";
			mes "Lo cambiaste...";
			mes "Fuera antes de que te aturda con mi Martillo!!";
			close;
		}
		if (getequippercentrefinery(.@part) > rand(100)) {
			mes "Clang! Clang! Clang! Clang!";
			successrefitem .@part;
			.@refinecnt = .@refinecnt - 1;
			next;
		} else {
			if (rand(100) < 80) {
				mes "["+ .@npc_name$ +"]";
				mes "Clang! Clang! Clang! Clang!";
				downrefitem .@part,3; // Failed refine attempts decrease the item's refine level by 3
				next;
				emotion (!rand(5))?ET_MONEY:ET_HUK;
				mes "["+ .@npc_name$ +"]";
				mes "Ahhh!!!";
				next;
				mes "["+ .@npc_name$ +"]";
				mes "Dios mio!";
				mes "El nivel de mejora ha bajado...";
			} else {
				mes "["+ .@npc_name$ +"]";
				mes "Clang! Clang! Clang!";
				failedrefitem .@part;
				next;
				emotion (!rand(5))?ET_MONEY:ET_HUK;
				mes "["+ .@npc_name$ +"]";
				mes "Hmmm!";
				next;
				mes "["+ .@npc_name$ +"]";
				mes "Dios mio! No pude refinar el articulo...";
				mes "No fue mi intencion!";
			}
			mes "Pude haber cometido un error aunque soy el mejor Herrero de todos.";
			mes "Simplemente no estaba destinado a ser.";
			next;
			mes "["+ .@npc_name$ +"]";
			mes "Lo hare mejor la proxima vez! No te preocupes!";
			close;
		}
		.@refinerycnt = getequiprefinerycnt(.@part);
	}
	emotion ET_BEST;
	mes "["+ .@npc_name$ +"]";
	mes "Bien! Exito!!!";
	mes "Soy el mejor Herrero.";
	close;
}

		emotion ET_FRET;
		mes "Wait a second...";
		mes "Do you think I'm stupid?!";
		mes "You switched the item while I wasn't looking! Get out of here!";
		close;

	if (getequippercentrefinery(.@part) <= rand(100)) {

		mes "[Clink]";
		mes "Llora Martillo!! Llora!!!";

		switch(rand(1,5)) {
			case 1: emotion ET_CRY; break;
			case 2: emotion ET_PROFUSELY_SWEAT; break;
			case 3: emotion ET_KEK; break;
			case 4: emotion ET_SCRATCH; break;
