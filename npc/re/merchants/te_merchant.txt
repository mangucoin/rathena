					20704,"TE Siege Magic Cape" );	// TE_Woe_Magic_Manteau
				break;
			case 3:
				callsub( S_Rent, "cascos, accesorios y zapatos",
					2496,"TE Siege Shoes",			// TE_Woe_Shoes
					2497,"TE Siege Boots",			// TE_Woe_Boots

S_Rent:
	mes "[Rental Items Manager]";
	if (checkweight(1304,3) == 0) {
		mes "Ya tienes demasiados objetos para recibir un nuevo equipo de alquiler.";
		mes "Por favor organiza tus objetos e intenta de nuevo.";
		close;
	}
	mes "Tenemos "+ getarg(0) +" listos para ti.";
	mes "Selecciona el objeto deseado.";
	next;
	for ( .@i = 1; .@i < getargcount(); .@i += 2 ) {
		if (rentalcountitem(getarg(.@i)) > isequippedcnt(getarg(.@i)))// don't count item equipped
			.@menu$ = .@menu$ + "^4d4dff"+ getarg(.@i+1) +" - Ya alquilado^000000:";
		else
			.@menu$ = .@menu$ + getarg(.@i+1) + ":";
	}
	.@total_item = (getargcount() -1) /2;
	.@s = select( .@menu$ + ":Ver otros objetos" ) -1;
	if (.@s <= .@total_item) {
		.@index = .@s *2 +1;
		mes "[Rental Items Manager]";
		if (rentalcountitem(getarg(.@index)) > isequippedcnt(getarg(.@index)))
			mes "Ya tienes "+ getarg(.@index+1) +".";
		else {
			mes "Aqui tienes "+ getarg(.@index+1) +".";
			mes "^4d4dffEste objeto es efectivo por las proximas 24 horas^000000 y despues de eso expira.";
			rentitem getarg(.@index),86400;
		}
		mes "Necesitas algo mas?";
		next;
	}
	return;
