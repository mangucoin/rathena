alde_alche,31,186,3	script	Craft Book Merchant#alde	883,{
	mes "[Craft Book Merchant]";
	mes "Bienvenido.";
	mes "Estoy aqui para vender";
	mes "^0000FFlibros de fabricacion especiales^000000 para Geneticists.";
	next;
	mes "[Craft Book Merchant]";
	mes "Los Geneticists tienen algunas habilidades";
	mes "que requieren libros de fabricacion para activarlas.";
	next;
	mes "[Craft Book Merchant]";
	mes "Mientras mas libros de fabricacion tengas,";
	mes "mas habilidades podras usar.";
	next;
	mes "[Craft Book Merchant]";
	mes "Por favor entiende que";
	mes "me especializo en la venta de estos libros especiales y";
	mes "^FF0000no vendo otros objetos^000000.";
	next;
	mes "[Craft Book Merchant]";
	mes "¿Te gustaria echar un vistazo a mis libros de fabricacion?";
	next;
	if(select("Claro.:No.") == 2) {
		mes "[Craft Book Merchant]";
		mes "Gracias por visitar mi tienda.";
		mes "Por favor cuidate en tus viajes.";
		close;
	}
	if (checkweight(1201,1) == 0) {
		mes "[Craft Book Merchant]";
		mes "Lo siento, pero tu inventario esta casi lleno.";
		mes "Por favor vacia tu inventario primero.";
		close;
	}
	if (MaxWeight - Weight < 2500) {
		mes "[Craft Book Merchant]";
		mes "Por favor vacia tu inventario primero.";
		close;
	}
	mes "[Craft Book Merchant]";
	mes "Muy bien, aqui esta la lista de libros de fabricacion disponibles.";
	next;
	switch(select("[Apple Bomb Craft Book] 100,000 zeny:[Pineapple Bomb Craft Book] 100,000 zeny:[Coconut Bomb Craft Book] 100,000 zeny:[Melon Bomb Craft Book] 100,000 zeny:[Banana Bomb Craft Book] 100,000 zeny:[Plant Gene Cultivation Method] 100,000 zeny:[Superior Potion Craft Manual] 100,000 zeny:[Mix Cooking Book] 100,000 zeny:[Health Improvement Research Book] 100,000 zeny:[Vigor Drink Recipe] 100,000 zeny:Cerrar")){
	case 1: set .@item,6279; break;
	case 2: set .@item,6280; break;
	case 3: set .@item,6281; break;
	case 4: set .@item,6282; break;
	case 5: set .@item,6283; break;
	case 6: set .@item,6284; break;
	case 7: set .@item,6285; break;
	case 8: set .@item,11022; break;
	case 9: set .@item,11023; break;
	case 10: set .@item,11024; break;
	case 11:
		mes "[Craft Book Merchant]";
		mes "Gracias por tu compra.";
		mes "Por favor vuelve pronto.";
		close;
	}
	mes "[Craft Book Merchant]";
	mes "^0000FF["+getitemname(.@item)+"] cuesta";
	mes "100,000 zeny.";
	mes "¿Cuantos te gustaria comprar?";
	next;
	input .@amount;
	if (.@amount == 0) {
		mes "[Craft Book Merchant]";
		mes "¿Te gustaria ver otros libros diferentes?";
		close;
	}
	if (.@amount < 1 || .@amount > 99 ){
		mes "[Craft Book Merchant]";
		mes "No puedes comprar mas de 100 a la vez.";
		close;
	}
	mes "[Craft Book Merchant]";
	mes "Has ingresado "+.@amount+"x ^0000FF["+getitemname(.@item)+"]^000000 para comprar.";
	mes "¿Te gustaria continuar?";
	next;
	if (select("Si:No") == 1) {
		set .@total, .@amount * 100000;
		if (Zeny < .@total) {
			mes "[Craft Book Merchant]";
			mes "Lo siento, pero no tienes suficiente dinero.";
			close;
		}
		if (checkweight(.@item,.@amount) == 0){
			mes "[Craft Book Merchant]";
			mes "Parece que no puedes cargar todo.";
			mes "Por favor revisa el espacio en tu inventario.";
			close;
		}
		mes "[Craft Book Merchant]";
		mes "Gracias por tu compra.";
		set Zeny,Zeny - .@total;
		getitem .@item,.@amount;
		close;
	}
	mes "[Craft Book Merchant]";
	mes "Por favor tomate tu tiempo";
	mes "antes de tomar tu decision.";
	close;
}
