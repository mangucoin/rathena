-	script	::CardSeparation_mal	-1,{
	disable_items;
	if (checkweight(1201,1) == 0) {
		mes "Tienes demasiados tipos de objetos. Intenta continuar despues de reducir esos objetos.";
		close;
	}
	if (MaxWeight - Weight < 10000) {
		mes "No puedes continuar porque tienes objetos muy pesados. Intenta continuar despues de reducir el peso.";
		close;
	}
	if (strnpcinfo(1) == "Jeremy") {
		set .@Jeremy,1;
		set .@n$, "[Jeremy]";
		setarray .@equip_name$[0],  "Armor",  "Shoes",  "Garment", "Upper Hat";
		setarray .@equip_slot[0], EQI_ARMOR,EQI_SHOES,EQI_GARMENT,EQI_HEAD_TOP;

		mes .@n$;
		mes "¡Cuanto tiempo sin verte~!";
		mes "He aprendido una nueva habilidad que separa cartas de Armadura, Zapatos, Capa y Sombrero. ¿Quieres probarla?";
	} else {
		set .@Jeremy,0;
		set .@n$, "[Richard]";
		setarray .@equip_name$[0], "Left hand", "Right hand";
		setarray .@equip_slot[0],   EQI_HAND_L,   EQI_HAND_R;

		mes .@n$;
		mes "El tonto de Jeremy no quiere tocar armas y escudos porque ha dañado varias cartas. Por eso yo prepare una habilidad de separacion de cartas para armas y escudos...";
	}
	next;
	mes .@n$;
	mes "Generalmente la tarifa es 1,000,000 Zeny. Durante la separacion de cartas, puedes usar ^990000objetos especiales que reducen la probabilidad de destruir el equipo o las cartas^000000. No cobramos zeny adicional por esto.";
	next;
	mes .@n$;
	mes "Existe la posibilidad de destruirlos incluso usando un objeto especial. Ademas, ^ff0000el nivel de refinamiento podria perderse^000000. ¿Tienes algun equipo para separar?";
	next;

	for(set .@i,0; .@i<getarraysize(.@equip_slot); set .@i,.@i+1) {
		if (getequipisequiped(.@equip_slot[.@i]))
			set .@menu$, .@menu$+getequipname(.@equip_slot[.@i])+":";
		else
			set .@menu$, .@menu$+"^999999"+.@equip_name$[.@i]+" (vacio)^000000:";
	}

	set .@i, select("Detener el trabajo:"+((.@Jeremy)?"¿Como es posible?":"")+":"+.@menu$);
	switch(.@i) {
	case 1:
		mes .@n$;
		mes "Cuando necesites el trabajo, visitame aqui.";
		close;
	case 2:
		mes .@n$;
		mes "Te preguntas que tiene de especial. Bueno, odio dar solo una guia, asi que dejame contarte la historia...";
		next;
		mes "^000099Jeremy esta estirando sus hombros y manos. Tal vez esta esperando a alguien con quien hablar.^000000";
		next;
		mes .@n$;
		mes "¿Sabias que la especialidad de Malangdo es la comida enlatada?";
		next;
		select("Lo sabia bien. ¿Es tan limitado?");
		mes .@n$;
		mes "Jeje... A todos les encanta. Pero hubo algunos problemas antes.";
		next;
		select("¿Problemas? ¿Hay algun pescado defectuoso?");
		mes .@n$;
		mes "No, el pescado no tiene ningun problema. El problema es la gran cantidad de aceite de pescado producido despues del procesamiento. Es un residuo industrial.";
		next;
		mes .@n$;
		mes "Sin embargo, despues de un proceso de refinamiento revitalizante, este aceite de pescado se volvio valioso para el cuidado de equipo viejo y como lubricante industrial.";
		next;
		mes .@n$;
		mes "Ademas, este aceite es tan util para separar reliquias de armas equipadas que no podemos comprar Premium Lubricant y Ordinary Lubricant con Zeny.";
		next;
		mes .@n$;
		mes "Bueno, no te preocupes por el dinero. Sin duda el Premium Lubricant es caro. Si pagas algo de zeny, puedo darte un lubricante mas barato.";
		next;
		mes .@n$;
		mes "No estoy seguro de la calidad del exito. De todas formas, es muy barato, ¿no?";
		close;
	default:
		set .@equip_num, .@equip_slot[.@i-3];
		if (!getequipisequiped(.@equip_num)) {
			mes .@n$;
			if (.@Jeremy)
				mes "¿En esta parte no hay nada?";
			else
				mes "¿No hay nada en esa parte?";
			close;
		}
		break;
	}

	setarray .@check_equip_card[0], getequipcardid(.@equip_num,0),getequipcardid(.@equip_num,1),getequipcardid(.@equip_num,2),getequipcardid(.@equip_num,3);
	setarray .@equip_card[0], .@check_equip_card[0], .@check_equip_card[1], .@check_equip_card[2], .@check_equip_card[3];
	set .@mvp_list$,
		"|4408|4128|4456|4168|4142"+  //Gloom_Under_Night_Card, Golden_Bug_Card, Nidhogg_Shadow_Card, Dark_Lord_Card, Doppelganger_Card
		"|4134|4137|4386|4407|4357"+  //Dracula_Card, Drake_Card, Detale_Card, Randgris_Card, B_Seyren_Card
		"|4146|4132|4147|4372|4145"+  //Maya_Card, Mistress_Card, Baphomet_Card, Bacsojin_Card, Berzebub_Card
		"|4374|4352|4367|4236|4425"+  //Apocalips_H_Card, B_Ygnizem_Card, B_Shecil_Card, Amon_Ra_Card, Atroce_Card
		"|4359|4123|4144|4135|4143"+  //B_Eremes_Card, Eddga_Card, Osiris_Card, Orc_Load_Card, Orc_Hero_Card
		"|4263|4131|4430|4276|4419"+  //Incant_Samurai_Card, Moonlight_Flower_Card, Ifrit_Card, Lord_Of_Death_Card, Ktullanux_Card
		"|4403|4399|4376|4441|4302"+  //Kiel_Card, Thanatos_Card, Lady_Tanee_Card, Fallen_Bishop_Card, Tao_Gunka_Card
		"|4305|4148|4318|4121|4365"+  //Turtle_General_Card, Pharaoh_Card, Knight_Windstorm_Card, Phreeoni_Card, B_Katrinn_Card
		"|4363|4324|4361|4330|4342|"; //B_Magaleta_Card, Garm_Card, B_Harword_Card, Dark_Snake_Lord_Card, Rsx_0806_Card

	if (.@Jeremy) {
		for ( .@i = 0; .@i < MAX_SLOTS; .@i++ ) {
			if (getiteminfo(.@equip_card[.@i], ITEMINFO_SUBTYPE) == CARD_ENCHANT)
				.@equip_card[.@i] = 0;// Armor Enchant System
		}
		if (!getarraysize(.@equip_card)) {
			mes .@n$;
			mes "La carta no esta equipada. ¿Quieres verificar de nuevo?";
			close;
		}
		if ((.@equip_card[0] && compare(.@mvp_list$,"|"+.@equip_card[0]+"|")) ||
			(.@equip_card[1] && compare(.@mvp_list$,"|"+.@equip_card[1]+"|")) ||
			(.@equip_card[2] && compare(.@mvp_list$,"|"+.@equip_card[2]+"|")) ||
			(.@equip_card[3] && compare(.@mvp_list$,"|"+.@equip_card[3]+"|")))
			set .@boss_chk,1;
	} else {
		// Official "Richard" script uses a hardcoded list including every possible item.
		//if (!getequipisequiped(.@equip_num)) {
		//	mes "[Richard]";
		//	mes "Lo siento. Aun no ofrecemos ese equipo.";
		//	close;
		//}

		mes "[Richard]";
		mes "¿De que numero de ranura quieres separar la carta? Desde la ranura izquierda, estan ordenadas 1,2,3,4.";
		next;
		set .@menu$,"";
		for ( .@i = 0; .@i < MAX_SLOTS; .@i++ ) {
			if (.@equip_card[.@i] && getiteminfo(.@equip_card[.@i], ITEMINFO_SUBTYPE) == CARD_NORMAL) // Armor Enchant System
				.@menu$ = .@menu$ + "Ranura " + (.@i+1) + " - " + getitemname(.@equip_card[.@i])+":";
			else {
				.@menu$ = .@menu$ + "^777777Ranura " + (.@i+1) + " - Sin carta^000000:";
			}
		}
		set .@i, select("Detener el trabajo:"+.@menu$);
		switch(.@i) {
		case 1:
			mes .@n$;
			mes "Cuando necesites trabajo, por favor ven a verme.";
			close;
		default:
			set .@slot, .@i-2;
			if (.@equip_card[.@slot] == 0 || getiteminfo(.@equip_card[.@slot], ITEMINFO_SUBTYPE) == CARD_ENCHANT) {
				mes .@n$;
				mes "Esta ranura no tiene ninguna carta equipada. ¿Por que no verificas de nuevo?";
				close;
			}
			break;
		}
		if (compare(.@mvp_list$,"|"+.@equip_card[.@slot]+"|"))
			set .@boss_chk,1;
	}
	if (.@boss_chk == 0) {
		mes .@n$;
		if (.@Jeremy)
			mes "Excepto las cartas, ^ff0000todos los efectos encantados desapareceran.^000000 Si estas de acuerdo con esto, por favor elige el tipo de trabajo:";
		else
			mes "Por favor elige la tarifa del trabajo.";
		next;
		set .@menu$,
			"La proxima vez...:"+
			((Zeny >= 1000000)?"Usar 1,000,000z (No usar objeto especial):":"^999999Usar 1,000,000z (Insuficiente)^000000:")+
			((countitem(6441))?"Usar Premium Lubricant:":"^999999Premium Lubricant (Insuficiente)^000000:")+
			((countitem(6440))?"Usar Ordinary Lubricant":"^999999Ordinary Lubricant (Insuficiente)^000000");
		switch(select(.@menu$)) {
		case 1:
			mes .@n$;
			mes "Cuando necesites el trabajo, visitame aqui.";
			close;
		case 2:
			if (Zeny < 1000000) {
				mes .@n$;
				mes "No tienes suficiente zeny. Por favor vuelve con la tarifa suficiente.";
				close;
			}
			mes .@n$;
			mes "Este es un equipo bastante viejo. Hay una alta probabilidad de destruir las cartas o el equipo durante el trabajo. ¿Estas seguro de que quieres continuar?";
			next;
			if(select("La proxima vez...:Continuar") == 1) {
				mes .@n$;
				mes "Cuando necesites el trabajo, visitame aqui.";
				close;
			}
			set .@sf_c_num,150;
			set .@sf_r_num,150;
			set .@sf_w_num,150;
			set Zeny, Zeny - 1000000;
			break;
		case 3:
			if (countitem(6441) == 0) {
				mes .@n$;
				mes "No tienes Premium Lubricant.";
				close;
			}
			mes .@n$;
			mes "Si usas el Premium Lubricant, la probabilidad de destruccion se reducira bastante, pero no puedo darte una garantia del 100%. ¿Estas seguro de que quieres continuar?";
			next;
			if(select("La proxima vez...:Continuar") == 1) {
				mes .@n$;
				mes "Cuando necesites el trabajo, visitame aqui.";
				close;
			}
			set .@sf_c_num,75;
			set .@sf_r_num,75;
			set .@sf_w_num,75;
			delitem 6441,1; //High_RankLubricant
			break;
		case 4:
			if (countitem(6440) == 0) {
				mes .@n$;
				mes "No tienes Ordinary Lubricant.";
				close;
			}
			mes .@n$;
			mes "Si usas el Ordinary Lubricant, la probabilidad de destruccion se reducira bastante, pero no puedo darte una garantia del 100%. ¿Estas seguro de que quieres continuar?";
			next;
			if(select("La proxima vez...:Continuar") == 1) {
				mes .@n$;
				mes "Cuando necesites el trabajo, visitame aqui.";
				close;
			}
			set .@sf_c_num,75;
			set .@sf_r_num,150;
			set .@sf_w_num,150;
			delitem 6440,1; //General_Lubricant
			break;
		}
	} else if (.@boss_chk == 1) {
		mes .@n$;
		mes "Este equipo contiene una preciosa carta MVP. Esta carta no puede ser separada con lubricante. Si traes el super surfactante ^0000ffSillit Pong^000000, podre trabajar.";
		next;
		switch(select("La proxima vez...:Tengo un Sillit Pong.")) {
		case 1:
			mes .@n$;
			mes "Cuando necesites el trabajo, visitame aqui.";
			close;
		case 2:
			if (countitem(6443) == 0) {
				mes .@n$;
				mes "No tienes Sillit Pong.";
				close;
			}
			break;
		}
		mes .@n$;
		if (.@Jeremy) {
			mes "Excepto las cartas, ^ff0000todos los efectos encantados desapareceran.^000000 Si estas de acuerdo con esto, por favor elige el tipo de trabajo:";
			set .@menu$,"¡De acuerdo, hagamoslo!";
		} else {
			mes "¿Puedo continuar?";
			set .@menu$,"¡Entendido. Solo hazlo rapido!";
		}
		next;
		switch(select("La proxima vez...:"+.@menu$)) {
		case 1:
			mes .@n$;
			mes "Cuando necesites el trabajo, visitame aqui.";
			close;
		case 2:
			set .@sf_c_num,60;
			set .@sf_r_num,60;
			set .@sf_w_num,60;
			delitem 6443,1; //Sillit_Pong_Bottle
			break;
		}
	}

	set .@equip_id, getequipid(.@equip_num);
	set .@equip_refine, getequiprefinerycnt(.@equip_num);

	// anti-hack
	if (callfunc("F_IsEquipCardHack", .@equip_num, .@check_equip_card[0], .@check_equip_card[1], .@check_equip_card[2], .@check_equip_card[3]))
		close;

	delequip .@equip_num;

	// Chance of retaining refine level.
	if (rand(1,.@sf_r_num) >= 61)
		set .@equip_refine,0;

	if (.@Jeremy) {
		// Chance of retaining equipment.
		if (rand(1,.@sf_w_num) < 61) {
			set .@equip_safe,1;
			getitem2 .@equip_id,1,1,.@equip_refine,0,0,0,0,0;
		}

		// Chance of retaining cards.
		for(set .@i,0; .@i<4; set .@i,.@i+1) {
			if (.@equip_card[.@i]) {
				if (rand(1,.@sf_c_num) < 61)
					getitem .@equip_card[.@i],1;
				else
					set .@card_break,1;
			}
		}
	} else {
		set .@card, .@equip_card[.@slot];
		set .@check_equip_card[.@slot],0;

		// Chance of retaining equipment (all enchantments are preserved).
		if (rand(1,.@sf_w_num) < 61) {
			set .@equip_safe,1;
			getitem2 .@equip_id,1,1,.@equip_refine,0,.@check_equip_card[0],.@check_equip_card[1],.@check_equip_card[2],.@check_equip_card[3];
		}

		// Chance of retaining card.
		if (rand(1,.@sf_c_num) < 61)
			getitem .@card,1;
		else
			set .@card_break,1;
	}

	// Display corresponding effect.
	if (!.@equip_safe && .@card_break)
		specialeffect2 EF_LORD;
	else if (.@equip_safe && .@card_break)
		specialeffect2 EF_SUI_EXPLOSION;
	else if (!.@equip_safe && !.@card_break)
		specialeffect2 EF_FIREPILLAR;
	else
		specialeffect2 EF_MAXPOWER;

	// Output results.
	mes "-- Resultado de la Separacion de Cartas --";
	if (.@equip_safe) {
		mes "No se produjeron grietas durante el proceso de separacion de cartas.";
		mes "^0000FFLa separacion del equipo es normal.^000000";
	} else {
		mes "Se produjeron grietas durante el proceso de separacion de cartas.";
		mes "El equipo ha sido dañado. ^ff0000Irrecuperable.^000000";
	}
	mes "-------------------";
	if (!.@card_break) {
		mes "No se produjo erosion de superficie durante el proceso de separacion de cartas.";
		mes "^0000ffLa separacion de la carta fue exitosa.^000000";
	} else {
		mes "Se produjo erosion de superficie durante el proceso de separacion de cartas.";
		mes "La carta ha sido dañada. ^ff0000Irrecuperable.^000000";
	}
	next;
	mes .@n$;
	mes "Esos son todos los resultados de la separacion de cartas. Por favor vuelve pronto.";
	close;
}
malangdo,215,166,4	duplicate(CardSeparation_mal)	Jeremy#pa0829	553	// Armors :: mal_yong
