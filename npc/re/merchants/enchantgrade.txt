grademk,34,184,4	script	Sratos#sratos	4_JP_GARM_H,{
	mes "[Sratos]";
	mes "Hola, querido cliente que entro, esperemos un milagro hoy.";
	mes "¿En que puedo ayudarte?";
	next;
	switch( select( "Mejorar el grado del equipo.", "Intercambiar objetos Etel", "¿Que es la Mejora de Grado?" ) ){
		case 1:
			mes "[Sratos]";
			mes "Nuestro cliente. Quieres mejorar el grado de tu equipo.";
			mes "No es facil mover el poder magico de las joyas.";
			next;
			mes "[Sratos]";
			mes "El poder magico puede explotar en el acto.";
			mes "¡Entonces el arma que heredara el poder magico tambien explota!";
			next;
			switch( select( "¡Aun asi lo hare!", "Lo pensare." ) ){
				case 1:
					mes "[Sratos]";
					mes "¡Les deseo buena suerte a nuestros valientes clientes!";
					close2;
					enchantgradeui();
					end;
				case 2:
					mes "[Sratos]";
					mes "Cuando tengas el coraje para el desafio, por favor vuelve.";
					close;
			}
		case 2:
			mes "[Sratos]";
			mes "Estas son joyas usadas para mejorar grados.";
			mes "Etel Dust, las joyas que seran la base, y si me das una pequeña tarifa las intercambiare por joyas Etel.";
			close2;
			callshop "EnchantGradeExchange";
			end;

		case 3:
			mes "[Sratos]";
			mes "Ocasionalmente, se encuentran joyas inestables con mucho poder magico puro.";
			mes "Las llamamos gemas etericas, ¿verdad?";
			next;
			mes "[Sratos]";
			mes "Puedes transferir el poder magico de la gema eterica a otras armas, pero por supuesto hay algo de riesgo.";
			mes "Es delicado, asi que si no tienes cuidado, ¡pum! Explotara.";
			next;
			mes "[Sratos]";
			mes "De todas formas, si usas esta gema eterica, puedes mejorar el arma que estas usando.";
			mes "El rendimiento mejora. A esto lo llamamos mejora de grado.";
			next;
			mes "[Sratos]";
			mes "Tu tambien, ¿no serias feliz si las armas que amas crecen un paso mas?";
			mes "Nosotros somos las personas que asisten en ello. Aunque es un poco arriesgado. Ajajaja.";
			close;
	}

OnInit:
	setunittitle(getnpcid(0), "<Grade Enhancer>");
	end;
}
