		next;
		mes "[Esseray]";
		mes "Conozco una forma de ser mas fuerte. Estas dispuesto a intentarlo?";
		next;
		if(select("No, estoy bien asi.:Quiero ser mas fuerte!") == 1) {
			mes "[Esseray]";
			mes "Si estas satisfecho con la mediocridad, no puedo ayudarte.";
			mes "Vive una vida normal entonces.";
			close;

		mes "Bien!";
		mes "Una leyenda del mundo Novice ha desaparecido para hacer algo en Comodo.";
		mes "Deberias encontrarlo.";

		mes "[Esseray]";
		mes "El puede guiarte por";
		mes "el camino correcto...";
