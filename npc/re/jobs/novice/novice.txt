	// Calculate player's job.
	// In the official script, higher .@who_job value always wins in a tie.
	setarray .@points[0],
		.@swordman_p, .@magician_p, .@archer_p, .@merchant_p, .@thief_p, .@acolyte_p, .@taekwon_p, .@gunslinger_p, .@ninja_p;
	set .@max, .@points[0];
	for(set .@i,1; .@i<9; set .@i,.@i+1) {
		if (.@points[.@i] >= .@max) {
			set .@max, .@points[.@i];
			set .@ele, .@i;
		}
	}
	set .@who_job, .@ele+1; // To match the official script's numbering.

	mes "Gracias, que tengas una larga vida.";
	mes "La evaluacion de aptitud vocacional ha terminado.";

	mes "[Lark]";
	mes "... ... el resultado es un agregado. ... ...";
	next;
	mes "[Lark]";
	mes "La tabulacion esta completa.";
	mes "'^0000FFAsique^000000', si presionas este boton";
	mes "puedes ver los resultados.";
	next;
	switch(.@who_job) {

		set .@job$, "Espadachin";
		mes "[Lark]";

		next;
		break;

		mes "...tipo: observa todo...";
		next;
		break;

		set .@job$, "Arquero";
		mes "[Lark]";

		next;

	case 4:

		next;
		break;

		set .@job$, "Ladron";
		mes "[Lark]";

		mes "...siempre intentando encontrar un nuevo trabajo...";
		mes "...contribucion al mundo,...";
		mes "...ese tipo. Aventurero,...";
		mes "...optimista.";
		next;
		break;
	case 6:
		set .@job$, "Acolito";

		mes "Se que he amado a alguien,...";
		mes "...ser de ayuda para otros y por otros...";
		mes "...tipo de buen cuidado.";
		mes "...siempre quiere ayudar a los demas.";
		next;
		break;
	case 7:
		set .@job$, "Taekwon Boy/Girl";
		mes "[Lark]";
		mes "...creer con una conviccion obstinada...";
		mes "...corazon y espiritu de la pradera es consistente: eso...";
		mes "...dar pasos para avanzar.";
		mes "...tipo de comentarios o querer entender las creencias...";
		mes "...ese tipo de esfuerzo por corregir a traves del entrenamiento,...";
		mes "...para tener una mente sana...";
		next;
		break;
	case 8:
		set .@job$, "Pistolero";
		mes "[Lark]";
		mes "La gente a menudo elogio la Justicia.";
		mes "A veces el miedo y la confusion pueden existir...";
		mes "...en ese momento, la definicion de confusion...";
		mes "...para mantener el equilibrio, estas cargando la responsabilidad...";
		mes "...para ti...";
		next;
		break;
	case 9:
		set .@job$, "Ninja";
		mes "[Lark]";
		mes "...para evitar el tipo de autoaspiracion...";
		mes "...casado con una meta prominente y evitando silenciosamente...";

	mes "Quieres seleccionar un trabajo?";

		mes "El curso del Centro de Entrenamiento para Novatos ahora se cierra, "+.@job$+". Te gustaria hacer esta ocupacion tu profesion?";

	callfunc "F_NvErase",1;
	set .@str$, "^A62A2A'"+strcharinfo(0)+"^000000'! Cuidate y que Freya te bendiga en tu viaje.";
	switch(.@who_job) {
	case 1:
		mes "[Lark]";

	case 2:

		warp "geffen_in",163,98;
		end;
	case 3:
		mes "[Lark]";
		mes "Te enviaremos";
		mes "al Gremio de Arqueros en Payon, la ciudad de montanas.";
		mes .@str$;
		mes "Adios.";
		close2;
		savepoint "payon",256,242,1,1;
		warp "payon_in02",64,65;
		end;
	case 4:
		mes "[Lark]";
		mes "Te enviaremos";
		mes "al gremio de comerciantes en Alberta, la Ciudad Portuaria.";
		mes .@str$;
		mes "Adios.";
		close2;
		savepoint "alberta",30,232,1,1;
		warp "alberta_in",62,44;
		end;
	case 5:
		mes "[Lark]";
		mes "Te enviaremos";
		mes "al gremio de ladrones en Morocc, la ciudad del desierto.";
		mes .@str$;
		mes "Adios.";
		close2;
		savepoint "morocc",150,99,1,1;
		warp "moc_prydb1",99,185;
		end;
	case 6:
		mes "[Lark]";
		mes "Te enviaremos";
		mes "a Prontera, la ciudad capital del Reino Rune-Midgarts.";
		mes .@str$;
		mes "Adios.";
		close2;
		savepoint "prontera",117,72,1,1;
		warp "prt_church",172,19;
		end;
	case 7:
		mes "[Lark]";
		mes "Te enviaremos";
		mes "al Phoenix en Payon, la ciudad de montanas.";
		mes .@str$;
		mes "Adios.";
		close2;
		savepoint "payon",164,58,1,1;
		warp "payon",163,141;
		end;
	case 8:
		mes "[Lark]";
		mes "Seras enviado al aeronave";
		mes "en Izlude, la ciudad satelite.";
		mes "Puedes subir a la aeronave para ir a Einbroch, la ciudad de industrias.";

		warp "izlude",128,67;
		end;
	case 9:
		mes "[Lark]";
		mes "Te enviaremos";
		mes "con el capitan del barco de contrabando.";
		mes "Puedes moverte a Amatsu";
		mes "con la ayuda del capitan.";
		mes .@str$;

		savepoint "alberta",117,56,1,1;
		warp "alberta",171,132;
		end;
	default:
		mes "[Lark]";
		mes "Ha ocurrido un error. Si sigues teniendo el mismo problema, por favor contacta al equipo de GM local.";

	}
