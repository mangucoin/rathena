//Return: 1:win, 0:draw, -1:lose
function	script	F_KO_Survival_rps	{
	setarray .@rps$[1], "Tijeras", "Piedra", "Papel";
	setarray .@emote[1], ET_SCISSOR, ET_ROCK, ET_WRAP;  //might be iRO custom(?)

	mes "Piedra! Papel! Tijeras!";
	next;
	set .@pc, select("Dar tijeras.:Dar piedra.:Dar papel.");
	set .@npc, rand(1,3);
	emotion .@emote[.@pc], playerattached();
	emotion .@emote[.@npc];
	mes "=Resultados del Juego=";
	mes "Maquina: " + .@rps$[.@npc] + ".";
	mes "Tu: " + .@rps$[.@pc] + ".";
	next;
	if ((.@npc == 1 && .@pc == 2) || (.@npc == 2 && .@pc == 3) || (.@npc == 3 && .@pc == 1)) {
		mes "Ganaste";
		return 1;
	} else if (.@pc == .@npc) {
		mes "Empate";
		return 0;
	} else if ((.@npc == 1 && .@pc == 3) || (.@npc == 2 && .@pc == 1) || (.@npc == 3 && .@pc == 2)) {
		mes "Perdiste";
		return -1;
	} else {
		mes "Ha ocurrido un error desconocido.";
		mes "Por favor contacta al servicio al cliente.";
		close;
	}
}
