prt_church,184,41,4	script	Cleric#aco	60,{
	if (Upper == 1) {
		if (ADVJOB == Job_High_Priest || ADVJOB == Job_Champion) {
			if (Class == Job_Novice_High) {
				mes "[Father Mareusis]";
				mes "Ah, siento que has soportado";
				mes "una experiencia de vida pasada. Debes haber aprendido muchas cosas antes de entrar al Valhalla.";
				next;
				if (!callfunc("F_CanChangeJob")) {
					mes "[Father Mareusis]";
					mes "Desafortunadamente, no creo que estes listo para convertirte en Acolyte todavia. Por favor, termina de aprender todas las Basic Skills primero.";
					next;
					mes "[Father Mareusis]";
					mes "Mientras tanto,";
					mes "esperare hasta";
					mes "que estes listo.";
					mes "Que Dios este";
					mes "contigo.";
					close;
				}
				mes "[Father Mareusis]";
				mes "Bien, te doy la bienvenida";
				mes "de vuelta del Valhalla y";
				mes "te deseo suerte en tu";
				mes "nuevo viaje de vida.";
				next;
				skill "NV_TRICKDEAD",0,SKILL_PERM;
				jobchange Job_Acolyte_High;
				skill "AL_HOLYLIGHT",1,SKILL_PERM;
				mes "[Father Mareusis]";
				mes "Ahora, aventurate y busca a quienes necesiten tu ayuda. Que Dios ilumine tu camino.";
				close;
			}else{
				mes "[Father Mareusis]";
				mes "Ahora, aventurate a buscar personas que necesiten tu ayuda. Que Dios ilumine tu camino.";
				close;
			}
		}else{
			mes "[Father Mareusis]";
			mes "Siento que has soportado una experiencia de vida pasada. Debes haber aprendido muchas cosas antes de entrar al Valhalla.";
			next;
			mes "[Father Mareusis]";
			mes "Sin embargo, puedo notar que no eres apto para ser un Acolyte. Por favor, recuerda quien fuiste en tu vida pasada y encuentra tu camino.";
			close;
		}
	}
	mes "[Father Mareusis]";
	mes "Que es lo que buscas?";
	next;
	switch(select("Cambiar tu trabajo a Acolyte.:Preguntar los requisitos para ser Acolyte.:Dejarlo.")) {
	case 1:
		mes "[Father Mareusis]";
		if(BaseJob != Job_Novice) {
			if(BaseJob == Job_Acolyte)
				mes "Te sientes bien hoy? Puedo ver por tu atuendo que ya eres un Acolyte. No estas bromeando, verdad?";
			else
				mes "Lo siento, pero parece que ya tienes tu propio trabajo, no es asi?";
			close;
		}else{
			mes "Realmente deseas convertirte en un servidor de Dios?";
			mes "Veamos si estas listo para ello o no... Hmm...";
			next;
			if (!callfunc("F_CanChangeJob")) {
				mes "[Father Mareusis]";
				mes "Oh! No has completado la practica basica todavia?! Te falta mucho camino! Vuelve despues de aumentar tu nivel de trabajo!";
				close;
			}
			mes "[Father Mareusis]";
			mes "Hmm... tu nivel de trabajo es suficiente...";
			mes "Bien. Ahora te dare la cualificacion para convertirte en Acolyte.";
			next;
			mes "[Father Mareusis]";
			mes "Recuerda siempre estar agradecido con Dios, quien cuida de nosotros todo el tiempo. En el caos y los momentos dificiles, enfrenta tus dificultades con fe inquebrantable.";
			callfunc "Job_Change",Job_Acolyte;
			getitem 1545,1;	// N_Mace
			next;
			mes "[Father Mareusis]";
			mes "Por ultimo, quiero felicitarte sinceramente por perseverar a traves de tu prueba de penitencia.";
			close;
		}
	case 2:
		mes "[Father Mareusis]";
		mes "Deseas convertirte en un Acolyte?";
		mes "Entonces, debes cumplir los siguientes requisitos pensando que son las practicas dadas por Dios.";
		next;
		mes "[Father Mareusis]";
		mes "Primero, debes alcanzar al menos Novice Job Level 9 y aprender todas las Basic Skills.";
		mes "Esta es la cosa mas basica que hacer, asi que debes considerarlo como un camino de entrenamiento.";
		next;
		mes "[Father Mareusis]";
		mes "Cuando creas que has cumplido este requisito, vuelve a verme. Entonces tendras un trabajo sagrado en el que puedes difundir la voluntad de Dios.";
		close;
	case 3:
		close;
	}
}

				next;

				close2;

				set job_acolyte_q,6;
				end;
			}
			else {
				mes "Oh...";
				mes "Eres uno de los";
				mes "aspirantes a Acolyte...?";
				mes "Veamos...";
				next;
				mes "[Father Rubalkabara]";
				mes "Tu nombre es " + strcharinfo(0) + "?";
				mes "No creo que tu nombre";
				mes "este en mi lista. Hmmm...";
				next;
				mes "[Father Rubalkabara]";
				mes "Por que no regresas al Santuario de Prontera y verificas de nuevo?";
				close;
			}
		}
		else {
			mes "Eh? Que te trae por aqui? Este es un lugar muy peligroso para un Novice como tu!";
			close;
		}
	}
	else if (BaseJob == Job_Acolyte) callfunc "F_FatherRub";
	else {
		if (BaseJob == Job_Priest) {
			mes "Saludos.";
			next;
			mes "[Father Rubalkabara]";
			mes "Bienvenido a las Profundidades. Sientete libre de sentarte y contemplar el mensaje de Dios conmigo. Este lugar es hermoso, aunque el peligro acompana su sensacion de serenidad...";
			close;
		}
		else {
			mes "Oh jo...";
			mes "Has venido a las Profundidades aqui para entrenar? O solo eres un Wanderer?";
			next;
			mes "[Father Rubalkabara]";
			mes "Quienquiera que seas, por favor cuidate. Los monstruos aqui son increiblemente fuertes, contrario a su apariencia tierna.";
			close;
		}
	}
}
