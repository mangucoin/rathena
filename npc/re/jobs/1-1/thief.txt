moc_prydb1,39,129,2	script	Thief Guide#thief	69,{
	if (Upper == 1) {
		if (ADVJOB == Job_Assassin_Cross || ADVJOB == Job_Stalker) {
			if (Class == Job_Novice_High) {
				mes "[Thief Guide]";
				mes "Eh? Te conozco? Es raro que me parezcas tan familiar. No tienes un gemelo, verdad?";
				next;
				if (!callfunc("F_CanChangeJob")) {
					mes "[Thief Guide]";
					mes "Que, quieres ser un Thief? Lo siento, pero parece que necesitas mas entrenamiento.";
					next;
					mes "[Thief Guide]";
					mes "Tomate tu tiempo y aprende todas las Basic Skills, si? Bueno, nos vemos luego~!";
					close;
				}
				mes "[Thief Guide]";
				mes "Bueno, tengo la sensacion de que has pasado por toda una vida de peleas, asi que te ascendere a Thief ahora mismo. Mejor les doy a los tipos rudos lo que quieren...";
				next;
				skill "NV_TRICKDEAD",0,SKILL_PERM;
				jobchange Job_Thief_High;
				skill "TF_SPRINKLESAND",1,SKILL_PERM;
				skill "TF_BACKSLIDING",1,SKILL_PERM;
				skill "TF_PICKSTONE",1,SKILL_PERM;
				skill "TF_THROWSTONE",1,SKILL_PERM;
				mes "[Thief Guide]";
				mes "Ya que te has convertido en Thief, vive como un Thief. Ahora, adelante! Siguiente~";
				close;
			}else{
				mes "[Thief Guide]";
				if (Sex == SEX_MALE)
					mes "Hey, amigo.";
				else
					mes "Hey, preciosa~";
				close;
			}
		}else{
			mes "[Thief Guide]";
			if (Sex == SEX_MALE)
				mes "Hey, amigo.";
			else
				mes "Hey, preciosa.";
			mes "...Hey! Te ves demasiado buenito para querer ser un Thief!! Ahora largate, estoy ocupado. Siguiente!";
			close;
		}
	}
	if(q_job_thief == 1) {
		mes "[Thief Guildsman]";
		mes "Ehh... Pareces el "+(Sex == SEX_MALE?"hijo":"hija")+" querido de una familia rica, pero por que estas en este lugar sucio?";
		next;
		mes "[Thief Guildsman]";
		mes "Bueno, yo no estoy a cargo de hacerte Thief. Solo acepto solicitudes, entiendes?";
		mes "Si quieres ser un Thief, preguntale al tipo de ojos afilados junto a mi.";
		close;
	}
	mes "[Thief Guildsman]";
	if(BaseJob != Job_Novice) {
		if(BaseJob == Job_Thief) {
			mes "Hey~ si tienes algun problema, cuentamelo cuando quieras, eh?";
			close;
		}else{
			mes "Que demonios...?";
			if(Sex == SEX_MALE) {
				mes "Eh... eres un "+jobname(Class)+", cabeza dura!";
				next;
				mes "[Thief Guildsman]";
				mes "Hey, hermano.";
			}else{
				next;
				mes "[Thief Guildsman]";
				mes "Hey, seniorita.";
			}
			mes "Por que estas aqui? Regresa a tu lugar~ vuelve~~";
			close;
		}
	}
	mes "Ehh... Pareces el "+(Sex == SEX_MALE?"hijo":"hija")+" querido de una familia rica, pero por que estas en este lugar sucio?";
	next;
	if(select("Quiero ser un Thief.:Nada.") == 2) {
		mes "[Thief Guildsman]";
		mes "Sabes que no puedes ser un Thief sin una solicitud...";
		mes "Que tienes en mente...?";
		close;
	}
	mes "[Thief Guildsman]";
	mes "Bueno, tan orgulloso estas de eso?";
	mes "Me dices tan orgullosamente que quieres ser un Thief! Por que no vas a todas las aldeas y te anuncias como ladron?";
	mes "'Ja ja ja! Ve y pon un cartel que diga 'Sere un orgulloso ladron que roba las cosas de otros.'";
	next;
	mes "[Thief Guildsman]";
	mes "Quieres ser un Thief tan desesperadamente?";
	next;
	switch(select("Si.:No.:Y tu?")) {
	case 1:
		mes "[Thief Guildsman]";
		mes "Oh, en serio...? Eh.. bueno... uno vive y aprende al ver gente extrana como tu.";
		next;
		break;
	case 2:
		mes "[Thief Guildsman]";
		mes "Entonces por que estas aqui? Crees que puedes ser un Thief tan facilmente?";
		next;
		break;
	case 3:
		mes "[Thief Guildsman]";
		mes "Eh..? yo? yo?";
		mes "Bueno... simplemente encajo bien siendo un Thief... por mi caracter... No me importa este asunto tonto.";
		next;
		break;
	}
	mes "[Thief Guildsman]";
	mes "De todos modos, en el mundo exterior, nunca digas que quieres ser un Thief!!";
	next;
	mes "[Thief Guildsman]";
	mes "Entonces, quieres solicitar ser un Thief?";
	next;
	if(select("Si, quiero.:No.") == 2) {
		mes "[Thief Guildsman]";
		mes "Bueno... haz lo que quieras~ Sigue tu camino~";
		close;
	}
	mes "[Thief Guildsman]";
	mes "Bien, dime tu nombre. Eh? Que?";
	mes ""+strcharinfo(0)+". Entonces "+strcharinfo(0)+", verdad...?";
	next;
	if (!callfunc("F_CanChangeJob")) {
		mes "[Thief Guildsman]";
		mes "Puedo ver tu fuerte voluntad de convertirte en Thief......";
		mes "Pero solo con tu voluntad, no puedes lograrlo en una pelea real, verdad?";
		mes "Asi que ve y alcanza al menos Basic Skill Level 9.";
		close;
	}
	mes "[Thief Guildsman]";
	mes "Hmm.... encontre tus interesantes antecedentes criminales en los datos de la agencia de detectives. Bueno, parece que tienes lo que se necesita para ser un buen Thief.";
	mes ""+strcharinfo(0)+". Ahora declaro que has pasado la entrevista de trabajo!";
	next;
	mes "[Thief Guildsman]";
	mes "Bueno, yo no estoy a cargo de hacerte Thief. Solo acepto solicitudes, entiendes?";
	mes "Si quieres ser un Thief, preguntale al tipo junto a mi.";
	set q_job_thief,1;
	close;
}
