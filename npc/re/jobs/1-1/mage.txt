geffen_in,164,124,4	script	Mage Guildsman#mage	123,{
	if (Upper == 1) {
		if (ADVJOB == Job_High_Wizard || ADVJOB == Job_Professor) {
			if (Class == Job_Novice_High) {
				mes "[Mage Guildsman]";
				mes "Whoa, cuanto tiempo sin verte! Pero no se suponia que estabas muerto?";
				next;
				mes "[Mage Guildsman]";
				mes "Ah, debes haber renacido. Bueno, me alegra tenerte de vuelta.";
				next;
				if (!callfunc("F_CanChangeJob")) {
					mes "[Mage Guildsman]";
					mes "Lo siento, pero no creo que estes listo para aprender magia todavia. Por que no vas y terminas de aprender las Basic Skills primero?";
					next;
					mes "[Mage Guildsman]";
					mes "Tomate tu tiempo. Mientras mas aprendas, mas listo estaras para aprender magia de nuevo.";
					close;
				}
				mes "[Mage Guildsman]";
				mes "Bueno, ya que pasaste la prueba de Mage una vez, no cuestionare tu cualificacion. Quieres tener tus habilidades magicas de vuelta inmediatamente, verdad?";
				next;
				skill "NV_TRICKDEAD",0,SKILL_PERM;
				jobchange Job_Mage_High;
				skill "MG_ENERGYCOAT",1,SKILL_PERM;
				mes "[Mage Guildsman]";
				mes "Wow, por alguna razon, te ves mucho mejor que antes. De todos modos, creo que haras un mejor trabajo siendo Mage tambien.";
				close;
			}else{
				mes "[Mage Guildsman]";
				mes "Hay algo mas en que pueda ayudarte? Si no, por que no vas a probar tus habilidades? El mundo te espera~!";
				close;
			}
		}else{
			mes "[Mage Guildsman]";
			mes "Que, te interesa el gremio de Mage? No queria decirtelo, pero no perteneces aqui.";
			next;
			mes "[Mage Guildsman]";
			mes "No estoy seguro de por que sigues parado frente a mi, pero puedo notar que no estas destinado a ser un Mage.";
			close;
		}
	}
	mes "[Mage Guildsman]";
	if(BaseJob != Job_Novice) {
		if(BaseJob == Job_Mage) {
			mes "Hey, no te has dado cuenta? Ya eres un Mage, tonto!";
			mes "Uno de estos dias te daras cuenta del poder dentro de ti cuando puedas hacer Fuego con tu mente!";
		} else
			mes "Hey~ Vamos. Deja de jugar. No puedes ser un Mage porque ya tienes otro trabajo.";
		close;
	}else{
		mes "Hey?";
		next;
		switch(select("Quiero ser un Mage:Cuales son los requisitos para ser Mage?:Nada, gracias.")) {
		case 1:
			mes "[Mage Guildsman]";
			mes "Quieres ser un Mage? Eh...";
			next;
			mes "[Mage Guildsman]";
			if (Sex == SEX_MALE) {
				mes "Hey, mirate! Eres algo lindo~! Aunque no eres mi tipo...";
			}else{
				mes "Oooh, eres una chica muy linda~!";
				mes "Me gustan las chicas como tu~";
			}
			mes "Cierto, dijiste que querias ser un Mage?";
			next;
			if(select("Quiero ser un Mage.:Nada, gracias.") == 2) {
				mes "[Mage Guildsman]";
				mes "Queeee~?! Justo despues de decirme que quieres ser un Mage, cambias de opinion?! Se un poco mas decidido!";
				close;
			}
			mes "[Mage Guildsman]";
			if (!callfunc("F_CanChangeJob")) {
				mes "Oh, amigo, tu nivel de Basic Skill no alcanza lo suficiente para ser un Mage.";
				mes "Regresa y sube tu nivel de Basic Skill.";
				close;
			}
			mes "Bien.. firma el formulario de solicitud.... Wow~ tienes buena letra! Ahh, esta bien. Eso es suficiente.";
			mes "Tu eres... "+strcharinfo(0)+".";
			next;
			mes "[Mage Guildsman]";
			mes "Hmm puedo ver que te has esforzado a tu manera. Aunque parece un poco torpe, creo que esta bien!";
			mes "Bien! Siempre apegarse a lo basico es lo mejor! Te transformare de inmediato.";
			next;
			mes "[Mage Guildsman]";
			mes "Hahh..! Ahora eres un Mage, uno de nuestros colegas!";
			mes "Te damos la bienvenida al Mage Guild, nuestro nuevo amigo!";
			next;
			mes "[Mage Guildsman]";
			mes "'Bienvenido al Mage Guild~'";
			mes "Felicidades por convertirte en miembro del Mage Guild! Dale con todo!";
			callfunc "Job_Change",Job_Mage;
			getitem 1639,1;	// N_Rod
			close;
		case 2:
			mes "[Mage Guildsman]";
			mes "Quieres ser un Mage, eh?";
			mes "Estare encantado de explicar los requisitos para una chica bonita como tu!";
			next;
			mes "[Mage Guildsman]";
			mes "Primero que nada, debes alcanzar Novice Job Level 10 y aprender todas las Basic Skills.";
			next;
			mes "[Mage Guildsman]";
			mes "En el pasado, habia una prueba complicada de creacion de pociones. Por eso, perdiamos aspirantes lentamente.";
			mes "Asi que decidimos aceptar a todos los aspirantes que cumplan los requisitos basicos.";
			next;
			mes "[Mage Guildsman]";
			mes "No dudes. Solo se un mago!";
			close;
		case 3:
			mes "[Mage Guildsman]";
			mes "Nada...?";
			close;
		}
	}
}
