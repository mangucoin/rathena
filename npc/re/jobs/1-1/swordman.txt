izlude_in,74,172,4	script	Swordman#swd	119,{
	if (Upper == 1) {
		if (Class == Job_Novice_High && (ADVJOB == Job_Lord_Knight || ADVJOB == Job_Paladin)) {
			mes "[Swordman]";
			mes "No...";
			mes "Puede ser...";
			mes "Has renacido, verdad?";
			next;
			mes "[Swordman]";
			mes "Veo que estas recorriendo el camino del Swordman de nuevo! Una vez que te acostumbras a blandir una espada, no hay vuelta atras!!";
			next;
			if (!callfunc("F_CanChangeJob")) {
				mes "[Swordman]";
				mes "Hmm? Ah, primero debes dominar las Basic Skills antes de estar listo para convertirte en Swordman.";
				next;
				mes "[Swordman]";
				mes "Vuelve a verme cuando hayas terminado de aprender las Basic Novice Skills.";
				close;
			}
			mes "[Swordman]";
			mes "Excelente! Dejame promoverte a Swordman de inmediato!";
			next;
			skill "NV_TRICKDEAD",0,SKILL_PERM;
			jobchange Job_Swordman_High;
			skill "SM_MOVINGRECOVERY",1,SKILL_PERM;
			skill "SM_FATALBLOW",1,SKILL_PERM;
			skill "SM_AUTOBERSERK",1,SKILL_PERM;
			mes "[Swordman]";
			mes "Hmm... Pareces un Swordman con mucha experiencia. Aun asi, estoy seguro de que debes entrenar para mejorar tus habilidades y ganar fuerza!";
			close;
		}else{
			mes "[Swordman]";
			mes "Hm...?";
			mes "Eres un guerrero";
			mes "renacido, verdad?";
			next;
			mes "[Swordman]";
			mes "Hmmm...";
			mes "Parece que ser";
			mes "un Swordman no es parte";
			mes "de tu destino. Lo siento,";
			mes "pero parece que no hay nada";
			mes "que pueda hacer por ti.";
			close;
		}
	}
	mes "[Swordman Guildsman]";
	mes "Este es el Swordman Guild.";
	mes "Por que estas aqui?";
	next;
	switch(select("Cuentame sobre ser un Swordman.:Quiero ser un Swordman.:Nada.")) {
	case 1:
		mes "[Swordman Guildsman]";
		mes "Asi que deseas saber mas sobre el poderoso Swordman? De acuerdo!";
		next;
		mes "[Swordman Guildsman]";
		mes "La caracteristica mas distintiva del Swordman es que puede mostrar sus verdaderas habilidades en combate cuerpo a cuerpo.";
		mes "Hay tres razones!";
		next;
		mes "[Swordman Guildsman]";
		mes "Primero, el Swordman tiene HP mas alto que otros trabajos.";
		mes "Segundo, excepto Arcos y Varas, el Swordman puede usar todas las demas armas para poder pelear a su capacidad optima.";
		mes "Y tercero, la mayoria de las habilidades del Swordman dan poderosos ataques fisicos.";
		next;
		mes "[Swordman Guildsman]";
		mes "Aunque te di una explicacion simple, creo que entiendes el significado central de lo que es ser un Swordman.";
		mes "En mi opinion, Swordman es el mejor trabajo de todos!";
		close;
	case 2:
		mes "[Swordman Guildsman]";
		if(BaseJob != Job_Novice) {
			if(BaseJob == Job_Swordman) {
				mes "Ya eres un excelente Swordman, no es asi?";
				mes "Solo dedicadte a ser un gran Swordman.";
			}else{
				mes "Ya tienes uno de los otros trabajos, no?";
				mes "Te has pasado con esa broma.";
			}
			close;
		}else{
			if (!callfunc("F_CanChangeJob")) {
				mes "Lamento decirte esto pero para ser un Swordman, debes alcanzar al menos ^4A4AFFJob Level 10^000000.";
				mes "y ^4A4AFFBasic Skill Level 9^000000.";
				next;
				mes "[Swordman Guildsman]";
				mes "Quieres ser un Swordman sin tener el requisito minimo?";
				mes "Crees que ser un Swordman es tan facil?";
				close;
			}
			mes "Hmm, tanto tu Job Level como tu Basic Skill Level estan en orden.";
			mes "Bien. Quieres ser un Swordman ahora mismo?";
			next;
			if(select("Si, quiero.:Lo pensare de nuevo.") == 2) {
				mes "[Swordman Guildsman]";
				mes "Si. Se necesita una decision prudente para elegir un trabajo.";
				mes "Pero me da pena... que lo reconsideres despues de superar todas las dificultades....";
				close;
			}
			mes "[Swordman Guildsman]";
			mes "Felicidades! De ahora en adelante, vas a vivir la vida de un Swordman!";
			mes "Hagamoslo ahora mismo!";
			next;
			mes "[Swordman Guildsman]";
			mes "Felicidades de nuevo por ser un Swordman y espero que participes en muchas actividades para el resurgimiento de nuestro gremio.";
			callfunc "Job_Change",Job_Swordman;
			getitem 13415,1; // N_Falchion
			close;
		}
	case 3:
		close;
	}
}
