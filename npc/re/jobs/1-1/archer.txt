payon_in02,64,71,4	script	Archer Guildsman#archer	85,{
	if (Upper == 1) {
		if (Class == Job_Novice_High && (ADVJOB == Job_Sniper || ADVJOB == Job_Clown || ADVJOB == Job_Gypsy)) {
			mes "[Archer Guildsman]";
			mes "Hey, te conozco.";
			mes "Ya hiciste esta prueba";
			mes "antes, verdad?";
			next;
			mes "[Archer Guildsman]";
			mes "Ah, debes haber ido";
			mes "al Valhalla y renacido.";
			mes "Wow, eso es impresionante!";
			next;
			if (!callfunc("F_CanChangeJob")) {
				mes "[Archer Guildsman]";
				mes "Err...";
				mes "Mejor aprende todas las Basic Skills primero antes de poder convertirte en Archer.";
				next;
				mes "[Archer Guildsman]";
				mes "Bien, nos vemos luego.";
				close;
			}
			mes "[Archer Guildsman]";
			mes "Bueno entonces. No necesito";
			mes "decir nada mas.";
			mes "Se que seras un gran Archer...";
			next;
			skill "NV_TRICKDEAD",0,SKILL_PERM;
			jobchange Job_Archer_high;
			skill "AC_MAKINGARROW",1,SKILL_PERM;
			skill "AC_CHARGEARROW",1,SKILL_PERM;
			mes "[Archer Guildsman]";
			mes "Aunque no hay recompensa especial";
			mes "para ti esta vez, espero que entiendas. Cuidate mucho.";
			close;
		}else{
			mes "[Archer Guildsman]";
			mes "Oh...?";
			mes "Hey, que estas";
			mes "haciendo aqui...?";
			next;
			mes "[Archer Guildsman]";
			mes "Puedo notar que no estas hecho para ser un Archer. Se siente como que estas destinado a hacer";
			mes "algo mas...";
			close;
		}
	}
	mes "[Archer Guildsman]";
	mes "Gusto en conocerte. En que puedo ayudarte?";
	next;
	switch(select("Quiero ser un Archer.:Necesito los requisitos, por favor.:Nada, gracias.")) {
	case 1:
		mes "[Archer Guildsman]";
		if(BaseJob != Job_Novice) {
			if(BaseJob == Job_Archer)
				mes "Jaja, me estas tomando el pelo...";
			else {
				mes "Lo siento pero solo los Novices pueden cambiar de trabajo.";
				mes "Tu ya tienes tu propio trabajo decente, no?";
			}
			close;
		}else{
			mes "Tu eres... "+strcharinfo(0)+", verdad?";
			next;
			if (!callfunc("F_CanChangeJob")) {
				mes "[Archer Guildsman]";
				mes "Bueno, no estas en el nivel de habilidad correcto.";
				mes "Tu nivel de trabajo debe ser al menos ^4A4AFF10^000000 y tu nivel de Basic Skill debe alcanzar ^4A4AFFnivel 9";
				next;
				mes "[Archer Guildsman]";
				mes "Porque un Archer necesita una concentracion extremadamente alta, asi que no aceptamos a quienes tienen poca paciencia.";
				close;
			}
			mes "[Archer Guildsman]";
			mes "Tu Basic Skill ya es suficiente..";
			mes "....Hm~~ asi que ya estas listo para ser un Archer. Procedere de inmediato.";
			next;
			mes "[Archer Guildsman]";
			mes "Felicidades! Ahora eres un Archer! Tambien, esperamos que participes activamente en muchos programas para el resurgimiento del Archer Guild.";
			mes "Ah, llegaron items del Departamento de Produccion. Toma, son todos tuyos!";
			callfunc "Job_Change",Job_Archer;
			getitem 1742,1;	// N_Composite_Bow
			getitem 12004,1;	// Arrow_Container
			getitem 12009,1;	// Silver_Arrow_Container
			getitem 12008,1;	// Fire_Arrow_Container
			next;
			mes "[Archer Guildsman]";
			mes "Teniendo un arco y flechas, ahora te has convertido en un verdadero Archer.";
			mes "Si abres el contenedor de flechas, hay flechas dentro y luego puedes equiparlas.";
			mes "Bueno, espero escuchar mejores noticias tuyas. Es hora de despedirnos.";
			mes "Adios.";
			close;
		}
	case 2:
		mes "[Archer Guildsman]";
		mes "Un Archer tiene habilidades con el arco y posee varios talentos.";
		mes "La mayor habilidad de un Archer es atacar enemigos desde larga distancia.";
		next;
		mes "[Archer Guildsman]";
		mes "Aunque un Archer tiene HP mas bajo, el o ella puede disparar a enemigos desde lejos,";
		mes "asi que un Archer esta mas seguro en una batalla real.";
		next;
		mes "[Archer Guildsman]";
		mes "Aunque un Archer en Ragnarok tiene HP mas bajo, el o ella tiene alta precision y velocidad de ataque para poder matar monstruos antes de que se acerquen a un Archer.";
		next;
		mes "[Archer Guildsman]";
		mes "^8C2121Un Archer puede cambiar de trabajo a Hunter.^000000";
		mes "^8C2121Aparte de Hunter, si eres hombre, puedes cambiar tu trabajo a Bard y si eres mujer, puedes cambiar tu trabajo a Dancer.^000000";
		close;
	case 3:
		mes "[Archer Guildsman]";
		mes "Si tienes alguna pregunta, sientete libre de venir y preguntarme.";
		close;
	}
}
