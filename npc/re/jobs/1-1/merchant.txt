alberta_in,53,43,6	script	Merchant#mer	86,{
	if (Upper == 1) {
		if (Class == Job_Novice_High && (ADVJOB == Job_Whitesmith || ADVJOB == Job_Creator)) {
			mes "[Chief Mahnsoo]";
			mes "Cuanto tiempo sin verte!";
			mes "Hey, no dejaste tu";
			mes "negocio, verdad?";
			mes "Que paso?";
			next;
			mes "[Chief Mahnsoo]";
			mes "Whoa...";
			mes "Realmente has estado en el Valhalla?! Wow, has recorrido un largo camino...";
			next;
			if (!callfunc("F_CanChangeJob")) {
				mes "[Chief Mahnsoo]";
				mes "Hmmm...";
				mes "Parece que no estas listo para ser Merchant de nuevo. Ve y termina de aprender las Basic Novice Skills primero.";
				next;
				mes "[Chief Mahnsoo]";
				mes "No te preocupes, siempre tendremos una posicion de Merchant abierta para ti. Solo vuelve cuando estes listo, si?";
				close;
			}
			mes "[Chief Mahnsoo]";
			mes "Supongo que es el destino que nos encontremos asi una vez mas. De acuerdo. Una vez mas, dejame cambiarte a Merchant!";
			next;
			skill "NV_TRICKDEAD",0,SKILL_PERM;
			jobchange Job_Merchant_High;
			skill "MC_CARTREVOLUTION",1,SKILL_PERM;
			skill "MC_CHANGECART",1,SKILL_PERM;
			skill "MC_LOUD",1,SKILL_PERM;
			mes "[Chief Mahnsoo]";
			mes "Ah~ Que nostalgico. Igual que en los viejos tiempos! Bien, da lo mejor de ti!";
			close;
		}else{
			mes "[Chief Mahnsoo]";
			mes "^333333*Suspiro*^000000";
			mes "Estoy tan aburrido...";
			mes "Cuando sabra de mi querida Blossom?";
			close;
		}
	}
	mes "[Guildsman Mahnsoo]";
	mes "Hey, por que estas aqui?";
	next;
	switch(select("Quiero ser un Merchant.:Quiero saber mas sobre los Merchants.:Preguntarle los requisitos para ser Merchant.:Nada.")) {
	case 1:
		mes "[Guildsman Mahnsoo]";
		if(BaseJob != Job_Novice) {
			if(BaseJob == Job_Merchant) {
				mes ".....? Perdon? Que estas diciendo?";
				mes "Ya eres un Merchant. Ay..";
				mes "Eh?! ...Necesito reirme ahora?!";
			}else{
				mes "Ahh? Estas intentando tener lo mejor de ambos mundos?";
				mes "Que tal si simplemente te dedicas a tu trabajo original?";
				next;
				mes "[Guildsman Mahnsoo]";
				mes "Tenemos etica de negocios, sabes.";
			}
			close;
		}else{
			mes "Quieres ser un Merchant? Hmm...";
			next;
			if (!callfunc("F_CanChangeJob")) {
				mes "[Guildsman Mahnsoo]";
				mes "Pero si quieres ser un Merchant, tu nivel de Basic Skill debe alcanzar el Nivel 9 o debes gastar todos tus puntos de habilidad.";
				next;
				mes "[Guildsman Mahnsoo]";
				mes "No crees que necesitamos aprender algunas habilidades basicas aunque solo tratemos con dinero?";
				close;
			}
			mes "[Guildsman Mahnsoo]";
			mes "Bien, creo que estas completamente listo viendo que tu nivel de Basic Skill esta cumplido.";
			mes "Ahora te permito ser un Merchant.";
			next;
			mes "[Guildsman Mahnsoo]";
			mes "Felicidades por convertirte en Merchant!";
			mes "Felicidades de nuevo por ser miembro del Merchant Guild y uno de nuestros colegas. Espero tu participacion activa de ahora en adelante!";
			callfunc "Job_Change",Job_Merchant;
			getitem 1381,1;	// N_Battle_Axe
			next;
			mes "[Guildsman Mahnsoo]";
			mes "Absolutamente, necesitamos jovenes con pasion para lograr nuestro gran objetivo de asegurar el 20% del volumen de moneda mundial. Lo entiendes? Eh?";
			mes "Bueno, solo digo... significa que ganemos mucho dinero al final. Ya lo saben~";
			close;
		}
	case 2:
		mes "[Guildsman Mahnsoo]";
		mes "Merchant? Para decirlo simplemente, la persona que vende bienes y gana dinero es un Merchant.";
		mes "No es bueno peleando y no tiene habilidades especiales de ataque/recuperacion... pero un Merchant puede comprar cosas a bajo precio y luego venderlas y ganar dinero.";
		next;
		mes "[Guildsman Mahnsoo]";
		mes "Bueno, un Merchant tiene una habilidad definitiva llamada Mammonite que golpea a un enemigo con su dinero... Podemos equipar todo excepto Arcos, Varas y Espadas a Dos Manos. Pero siempre podemos vender y comprar eso.";
		mes "Si... nosotros los Merchants siempre tenemos el dinero en mente, entiendes?";
		close;
	case 3:
		mes "[Guildsman Mahnsoo]";
		mes "Para convertirte en Merchant, aunque solo vender y recibir dinero es nuestro trabajo, debes alcanzar al menos el nivel 9 de Basic Skill.";
		next;
		mes "[Guildsman Mahnsoo]";
		mes "Bueno, antes soliamos cobrar una tarifa de inicio. No era dificil reunir ese dinero?";
		mes "Habia demasiada gente que no podia reunir ese dinero y seguia llorando.";
		next;
		mes "[Guildsman Mahnsoo]";
		mes "Asi que recientemente, decidimos permitir que cualquier Novice que quiera ser Merchant pueda serlo.";
		mes "Porque estar vivo o no despues es cuestion de la habilidad de cada uno.";
		close;
	case 4:
		close;
	}
}
