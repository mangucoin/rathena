// callfunc "F_Mora_Mist",<warp to enable>,<success message>,<fail message>{,<end flag>}
function	script	F_Mora_Mist	{
	set .@map$, instance_mapname("1@mist");
	if (mobcount(.@map$,instance_npcname(strnpcinfo(0))+"::OnMyMobDead") == 0) {
		mes "Obviamente nadie lo está cuidando.";
		mes "Parece que puedes talar el árbol del jardín.";
		next;
		if(select("Talar el árbol del jardín.:Rendirse.") == 2)
			close;
		mes "Talas el árbol que bloqueaba el camino del laberinto, "+((getarg(3,0))?"despejando la salida del bosque.":"así que ahora puedes continuar.");
		mapannounce .@map$,getarg(1),bc_map,"0xccffcc"; //FW_NORMAL 12 0 0
		enablenpc instance_npcname(getarg(0));
		disablenpc();
		if (getarg(4,0))
			initnpctimer;
		close;
	} else
		mapannounce .@map$,((getarg(3,0))?getarg(2):getarg(2)+"'s Grito: ¿Eh? ¡¿Quién le está haciendo cosas malas a mi árbol?!"),bc_map,"0xccffcc"; //FW_NORMAL 12 0 0
	end;
}
