moro_cav,50,64,5	script	Yellow Seed#bios	CLEAR_NPC,{
	if (BaseLevel < 160) {
		mes "^ff0000Los personajes por debajo del Nivel 160 no pueden entrar a esta mazmorra.^000000";
		close;
	}
	if (isbegin_quest(15006) == 0) {
		mes "^ff0000Primero debes hablar con el Anciano Errante para acceder a esta mazmorra.^000000";
		close;
	}
	.@party_id = getcharid(1);
	if (.@party_id < 1) {
		mes "^ff0000Primero debes formar/unirte a un grupo de al menos 1 persona para entrar a esta mazmorra.^000000";
		close;
	}
	switch( checkquest(15005,PLAYTIME) ) {
	case -1:
		break;
	case 0:
	case 1:
		if (checkquest(15007,PLAYTIME) == 2) {// 5 mins to re-enter
			mes "^ff0000Esta Mazmorra Memorial no puede ser accedida durante 23 horas despues de tu ultima visita.^000000";
			close;
		}
		switch( checkquest(15008,PLAYTIME) ) {
		case -1:
			break;
		case 0:
		case 1:
			mes "^0000ffEsta Mazmorra Memorial no puede ser accedida durante 23 horas despues de tu ultima visita.^000000";
			close;
		case 2:
			break;
		}
		break;
	case 2:
		erasequest 15005;// Memorial Dungeon: Isle of Bios
		erasequest 15007;// Memorial Dungeon: Isle of Bios
		if (isbegin_quest(15008) > 0)
			erasequest 15008;// Memorial Dungeon: Isle of Bios
		break;
	}
	if (is_party_leader() == true)
		.@create$ = "Crear Mazmorra Memorial.";
	switch( select( .@create$, "Entrar Isle of Bios.", "Cancelar." ) ) {
	case 1:
		instance_create("Isle of Bios");
		mes "^ff0000Se ha creado la Mazmorra Memorial.^000000";
		mes "^ff0000Por favor haz clic y presiona Entrar de nuevo.^000000";
		close;
	case 2:
		switch( instance_enter("Isle of Bios") ) {
		case IE_NOMEMBER:
		case IE_NOINSTANCE:
		case IE_OTHER:
			if (is_party_leader(.@party_id) == false) {
				mes "^ff0000Ten en cuenta que esta Mazmorra Memorial no puede ser accedida durante 23 horas despues de tu ultima visita.^000000";
				next;
			}
			mes "^ff0000El lider de tu grupo aun no ha creado una Grieta del Tiempo.^000000";
			close;
		case IE_OK:
			mapannounce "moro_cav", "" + strcharinfo(0) + " del grupo " + getpartyname( getcharid(1) ) + " esta entrando a Isle of Bios.",bc_map,"0x00FF99";
			if (isbegin_quest(15005) == 0) {
				setquest 15005;// Memorial Dungeon: Isle of Bios
				setquest 15007;// Memorial Dungeon: Isle of Bios
			}
			end;
		}
	case 3:
		end;
	}
	end;

OnInit:
	while(true) {
		sleep 500;
		specialeffect EF_ENHANCE;
	}
	end;
}
