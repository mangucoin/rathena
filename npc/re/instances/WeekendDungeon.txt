pay_arche,44,124,5	script	Marry Jae	4_F_JOB_BLACKSMITH,{
	mes "[Marry Jae]";
	if (BaseLevel < 60) {
		mes "Tu nivel base es muy bajo, vuelve cuando alcances el nivel 60+.";
		close;
	}
	mes "^ff0000Si intentas domar monstruos, no contara como eliminacion. Por favor ten cuidado.^000000";
	next;
	.@day = gettime(DT_DAYOFWEEK);
	if (.@day != SATURDAY && .@day != SUNDAY) {
		mes "[Marry Jae]";
		mes "Oh, un curioso? Soy Marry Jae, la encargada de los fines de semana entre los dias de la semana.";
		next;
		mes "[Marry Jae]";
		mes "Que es la mazmorra de fin de semana? En realidad, es un espacio especialmente disenado para que los nuevos aventureros entrenen.";
		next;
		mes "^0000FFMirando alrededor, es obvio que Marry Jae te esta hablando.^000000";
		next;
		mes "[Marry Jae]";
		mes "Para los que no tienen grupo los fines de semana, este es un terreno de caza y experiencia para aventureros solitarios!";
		next;
		mes "[Marry Jae]";
		mes "Ademas, la ganancia de experiencia aqui es muy generosa.";
		next;
		select( "Cual es el horario de apertura?" );
		mes "[Marry Jae]";
		mes "La mazmorra estara abierta durante:";
		mes "^0000FFSabado 00:00 ~ Domingo 23:59^000000";
		next;
		mes "[Marry Jae]";
		mes "Espero verte cuando llegue el fin de semana~";
		close;
	}
	.@md_name$ = "Weekend Dungeon";

	if (getcharid(1) < 1) {
		mes "[Marry Jae]";
		mes "Necesitas formar un grupo de uno o mas miembros.";
		close;
	}
	switch( checkquest(12378,PLAYTIME) ) {
	case -1:
		break;
	case 0:
	case 1:
		switch( checkquest(12377,PLAYTIME) ) {
		case -1:
		case 2:
			mes "[Marry Jae]";
			mes "Lo siento, pero el periodo de entrada a la " + .@md_name$ + " has expired.";
			mes "Descansa mientras esperas a que la mazmorra este disponible de nuevo.";
			close;
		case 0:
		case 1:
			break;
		}
		break;
	case 2:
		erasequest 12377;
		erasequest 12378;
		mes "[Marry Jae]";
		mes "El tiempo de espera ha expirado.";
		mes "Puedes volver a entrar a la Mazmorra Memorial.";
		close;
	}

	if (is_party_leader() == true)
		.@menu$ = "Preparar Mazmorra Memorial";

	mes "[Marry Jae]";
	mes "Creo que ya estas casi listo.";
	mes "Te gustaria entrar ahora?";
	next;
	switch( select( .@menu$, "Entrar a Mazmorra de Fin de Semana", "Cancelar" ) ) {
	case 1:
		instance_create(.@md_name$);
		mes "[Marry Jae]";
		mes "La Mazmorra Memorial ha sido creada.";
		mes "Ya puedes entrar.";
		close;
	case 2:
		switch( instance_enter(.@md_name$) ) {
		case IE_NOMEMBER:
			end;
		case IE_NOINSTANCE:
			mes "No hay Mazmorra Memorial registrada.";
			close;
		case IE_OTHER:
			mes "Ha ocurrido un error desconocido.";
			close;
		case IE_OK:
			mes "[Marry Jae]";
			mes "Adelante!";
			mapannounce "pay_arche", "" + strcharinfo(0) + " of the party, " + getpartyname( getcharid(1) ) + ", is entering the " + .@md_name$ + ".", bc_map, 0xFF99;
			if (isbegin_quest(12377) == 0)
				setquest 12377;
			if (isbegin_quest(12378) == 0)
				setquest 12378;
			end;
		}
		end;
	case 3:
		mes "[Marry Jae]";
		mes "No te has preparado?";
		mes "Supongo que no se puede evitar.";
		close;
	}
}

		mes "Por favor vuelve cuando tengas suficientes materiales.";
		close;

		mes F_InsertComma(.@zeny) + "z to be specific.";
		close;
	}
	if (checkweight(23144,1) == 0) {
		mes "Tienes demasiados tipos de objetos. Limpia tu inventario.";
		close;
	}
	if (MaxWeight - Weight < 10000) {
		mes "No puedes continuar porque tienes objetos muy pesados. Reduce tu peso total limpiando tu inventario.";
		close;
	}
	mes "Confirm to spend " + F_InsertComma(.@zeny) + "z to package 5 <ITEM>" + getitemname(.@item) + "<INFO>" + .@item + "</INFO></ITEM> into <ITEM>" + getitemname(.@box) + "<INFO>" + .@box + "</INFO></ITEM>?";
	next;
	if (select( "Yes", "No" ) == 2) {
		mes "[Gift Supplies Clerk]";
		mes "Vuelve cuando necesites servicios de empaquetado!";
		close;
	}
	Zeny -= .@zeny;
	delitem .@item,5;
	getitem .@box,1;
	end;
}
