ma_scene01,174,179,4	script	Taho	541,{
	disable_items;
	if (checkweight(1201,1) == 0) {
		mes "Estas cargando demasiados objetos. Regresa despues de quitarte algunos.";
		close;
	}
	if (MaxWeight - Weight < 10000) {
		mes "Estas cargando demasiado peso. Regresa despues de reducir el peso.";
		close;
	}
	if (BaseLevel < 140) {
		mes "[Taho]";
		mes "Oye. Este es un lugar demasiado peligroso para ti. No puedes mantenerte a salvo. Sal de aqui y ve a la aldea ahora mismo!";
		close;
	}
	if (malaya_bakona1 < 7 && malaya_bakona2 < 15) {
		mes "[Taho]";
		mes "Ese Bakonawa nos va a matar a todos. Ah, debemos matarlo pero desafortunadamente no estamos preparados.";
		close;
	}

	set .@party_id,getcharid(1);
	set .@md_name$, "Bakonawa Lake";

	set .@baku_time, checkquest(12278,PLAYTIME);
	if (.@baku_time == -1) {
		if (checkquest(12279) == -1) setquest 12279;
	}
	if (!.@party_id) {
		mes "[Taho]";
		mes "No hay nadie que pueda ir contigo? O incluso si quieres ir solo, ven despues de organizar un grupo.";
		close;
	}
	if (.@baku_time == -1) {
		if (is_party_leader() == true) {
			mes "[Taho]";
			mes "Eres el lider de los cazadores de Bakonawa? El camino no esta bien asi que tenemos que bajar por el acantilado con una cuerda, esta bien?";
			next;
			set .@i, select("Por favor teje una cuerda.:Ahora bajare.:Cancelar.");
		} else {
			mes "[Taho]";
			mes "Viniste a matar a Bakonawa. El lider de tu grupo tiene que ayudarme a tejer la cuerda para bajar.";
			next;
			set .@i, select(":Ahora bajare.:Cancelar.");
		}
		switch(.@i) {
		case 1:
			if (instance_create(.@md_name$) < 0) {
				mes "[Taho]";
				mes "Oh, la cuerda se cayo. Tengo que hacer una nueva.";
				close;
			}
			mes "[Taho]";
			mes "Ahora estoy tejiendo, asi que podras bajar cuando termine.";
			close;
		case 2:
			switch(instance_enter(.@md_name$)) {
			case IE_OTHER:
				mes "[Taho]";
				mes "Ocurrio un error desconocido.";
				close;
			case IE_NOINSTANCE:
				mes "[Taho]";
				mes "La cuerda aun no ha sido tejida. Espera un poco.";
				close;
			case IE_NOMEMBER:
				mes "[Taho]";
				mes "Los civiles no pueden entrar.";
				close;
			case IE_OK:
				mapannounce "ma_scene01","Un miembro del grupo, "+strcharinfo(0)+" del grupo "+getpartyname(.@party_id)+" esta entrando a la mazmorra, "+.@md_name$+".",bc_map,"0x00ff99"; //FW_NORMAL 12 0 0
				setquest 12278;
				//warp "1@ma_b",64,51;
				close;
			}
		case 3:
			close;
		}
	} else if (.@baku_time == 0 || .@baku_time == 1) {
		mes "[Taho]";
		mes "La cuerda usada para llegar al Lago ya esta rota. Estamos limpiando los alrededores asi que regresa despues de un tiempo.";
		close;
	} else if (.@baku_time == 2) {
		mes "[Taho]";
		mes "Ya esta todo limpio por aqui. Ahora puedes bajar de nuevo despues de conseguir una cuerda.";
		erasequest 12278;
		close;
	}
}
