moc_para01,29,95,5	script	Runane#sara	4_F_RUNAIN,{
	cutin "lunain01",0;
	if (BaseLevel < 120) {
		mes "[Runane]";
		mes "^ff0000Debes ser Nivel 120.^000000";
		close3;
	}
	if (isbegin_quest(13181) != 1) {
		mes "[Runane]";
		mes "^ff0000Si quieres usar mi portal dimensional, primero tienes que hablar con Roel. El portal no esta abierto para todos.^000000";
		close3;
	}
	.@party_id = getcharid(1);
	.@md_name$ = "Nightmarish Jitterbug";
	if (.@party_id < 1) {
		mes "[Runane]";
		mes "^ff0000Primero debes formar/unirte a un grupo de al menos 1 persona para entrar a esta mazmorra.^000000";
		close3;
	}
	switch( checkquest(13182,PLAYTIME) ) {
	case -1:
		break;
	case 0:
	case 1:
		mes "[Runane]";
		mes "^ff0000Este portal dimensional no puede ser accedido por 20 horas despues de tu ultimo acceso.^000000";
		close3;
	case 2:
		mes "[Runane]";
		mes "El portal ya esta disponible para usar.";
		erasequest 13182;// Nightmarish Jitterbug: Waiting
		close3;
	}
	if (is_party_leader() == true)
		.@menu$ = "Abrir el portal dimensional.";
	switch( select( .@menu$, "Entrar a " + .@md_name$ + ".", "Cancelar." ) ) {
	case 1:
		if (instance_create(.@md_name$) < 0) {
			mes "Party Name: " + getpartyname( getcharid(1) );
			mes "Party Leader: " + strcharinfo(0);
			mes "^0000ff" + .@md_name$ + " ^000000- Reservation Failed!";
			close3;
		}
		mes "[Runane]";
		mes "^ff0000Preparando el portal dimensional...^000000";
		mes "^ff0000Por favor espera.^000000";
		close3;
	case 2:
		switch( instance_enter(.@md_name$) ) {
		case IE_NOMEMBER:
			close3;
		case IE_NOINSTANCE:
			if (is_party_leader() == false) {
				mes "[Runane]";
				mes "Teletransportando...";
				next;
				mes "[Runane]";
				mes "^ff0000Ten en cuenta que este portal no puede ser accedido por 23 horas despues de tu ultimo acceso.^000000";
				next;
			}
			mes "[Runane]";
			mes "^ff0000Tu lider de grupo aun no ha creado una Grieta Temporal.^000000";
			close3;
		case IE_OTHER:
			mes "Ha ocurrido un error desconocido.";
			close3;
		case IE_OK:
			setquest 13182;// Nightmarish Jitterbug: Waiting
			mapannounce "moc_para01", "" + strcharinfo(0) + " del grupo, " + getpartyname( getcharid(1) ) + ", esta entrando a " + .@md_name$ + ".", bc_map, 0xFF99;
			specialeffect2 EF_READYPORTAL2;
			specialeffect2 EF_PORTAL2;
			// warp "1@jtb",16,17;
			end;
		}
	case 3:
		cutin "",255;
		end;
	}
}
