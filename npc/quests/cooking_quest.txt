S_SellSets:
	.@item_id = getarg(0);
	.@item_cost = getiteminfo(.@item_id, ITEMINFO_BUY);
	.@item_weight = getiteminfo(.@item_id, ITEMINFO_WEIGHT);
	mes "[Madeleine Chu]";
	mes "Cuantos "+ (.@item_id == 12125 ? "Outdoor":"Indoor");
	mes "Cooking Kits quieres";
	mes "comprar? Si quieres";
	mes "cancelar, por favor";
	mes "ingresa el numero 0.";
	next;
	while(1) {
		input .@sell;
		if (.@sell == 0) {
			mes "[Madeleine Chu]";
			mes "Cambiaste de opinion?";
			mes "Bueno, si necesitas";
			mes "comprar "+ getitemname(.@item_id);
			mes "Kits despues, solo vuelve";
			mes "conmigo cuando quieras, si?";
			close;
		}
		else if (.@sell > 100) {
			mes "[Madeleine Chu]";
			mes "Oh, lo siento, pero";
			mes "no vendo mas de";
			mes "100 "+ getitemname(.@item_id);
			mes "a la vez, por seguridad.";
			next;
		}
		else break;
	}
	set .@total_cost,.@sell * .@item_cost;
	set .@total_weight,.@sell * .@item_weight;
	if (Zeny < .@total_cost) {
		mes "[Madeleine Chu]";
		mes "Oh, lo siento, pero no";
		mes "puedes pagar tantos";
		mes getitemname(.@item_id);
		mes "Por favor revisa tus zeny";
		mes "antes de comprar mis kits~";
		close;
	}
	if (!checkweight(.@item_id,.@sell)) {
		mes "[Madeleine Chu]";
		mes "Lo siento, pero no tienes";
		mes "suficiente espacio en tu";
		mes "Inventario para tantos";
		mes getitemname(.@item_id) +"...";
		close;
	}
	set Zeny, Zeny-.@total_cost;
	getitem .@item_id,.@sell;
	mes "[Madeleine Chu]";
	mes "Aqui tienes~";
	mes "Mucha suerte con";
	mes "tu entrenamiento culinario!";
	return;
}
