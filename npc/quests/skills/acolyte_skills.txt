prt_church,173,23,4	script	Cleric	79,{
	mes "[Acolyte Klift]";
	mes "Ahh . Hermanos ! La tarea de";
	mes "cuidar a nuestras ovejas perdidas";
	mes "te agobia y te cansa ?";
	mes "Estoy aqui para ayudarte.";
	next;
	switch (select("^3355FF' Sobre la habilidad oculta del acolito '^000000 ..:^3355FF' Entrenamiento de Holy Light '^000000:Terminar conversacion")) {
	case 1:
		mes "[Acolyte Klift]";
		mes "Nuestros miembros del clero";
		mes "aprenden una habilidad de forma";
		mes "natural a medida que maduran. Cuando se acercan";
		mes "a sus anos de experiencia como acolito";
		mes "obtienen esta habilidad especial.";
		next;
		mes "[Acolyte Klift]";
		mes "Ante un gran peligro y";
		mes "una lucha interminable contra el mal,";
		mes "nuestros hermanos necesitan mucha ayuda.";
		mes "^3355FF' Holy Light '^000000 es esa habilidad.";
		mes "Para obtener esta habilidad por ti mismo,";
		mes "requiere algo de trabajo.";
		next;
		mes "[Acolyte Klift]";
		mes "Para obtener la habilidad, debes encontrar estos objetos.";
		mes "^FF33551 Opal^000000";
		mes "^FF33551 Crystal Blue^000000";
		mes "^FF33551 Rosary^000000 ";
		next;
		mes "[Acolyte Klift]";
		mes "Para poder usar esta habilidad,";
		mes "debes tener suficiente experiencia.";
		mes "^FF3355 Job Level 30^000000";
		mes "es necesario para aprender esta habilidad.";
		next;
		mes "[Acolyte Klift]";
		mes "En cuanto a un sacerdote, debido a su";
		mes "ya vasta cantidad de experiencia,";
		mes "pueden obtener esta habilidad en cualquier";
		mes "nivel de trabajo.";
		close;
	case 2:
		if (getskilllv("AL_HOLYLIGHT") == 1) {
			mes "[Acolyte Klift]";
			mes "Hermano, ya posees";
			mes "la habilidad de ` Holy Light '.";
			mes "No puedes obtener una habilidad que";
			mes "ya posees ..";
			mes "Rezo para que estes usando";
			mes "esta habilidad para hacer el bien . .";
			close;
		}
		if ((countitem(727) > 0) && (countitem(991) > 0) && (countitem(2608) > 0) && (JobLevel > 29 || (BaseJob == Job_Priest || BaseJob == Job_Monk)) && (BaseClass == Job_Acolyte)) {
			mes "[Acolyte Klift]";
			mes "Tu fe ha demostrado ser digna";
			mes "para que obtengas la habilidad ' Holy Light '.";
			mes "Tu habilidad es adecuada";
			mes "para usar esta tecnica.";
			mes "Usala sabiamente. . .";
			next;
			delitem 727,1; //White_Jewel
			delitem 991,1; //Crystal_Blue
			delitem 2608,1; //Rosary
			skill "AL_HOLYLIGHT",1,SKILL_PERM;
			mes "[Acolyte Klift]";
			mes "Ahora conoces ' Holy Light '";
			mes "Que uses esta habilidad solo con";
			mes "la mejor conducta . . . . .";
			close;
		}
		mes "[Acolyte Klift]";
		mes "Oh, esta claro. . .";
		mes "Aun no estas listo para";
		mes "recibir la habilidad ' Holy Light '.";
		next;
		mes "[Acolyte Klift]";
		mes "Deberias escuchar atentamente lo que";
		mes "es necesario para esta habilidad.";
		mes "Si escuchas con atencion,";
		mes "podras saber que te falta.";
		mes "Para que puedas mejorar.";
		close;
	case 3:
		mes "[Acolyte Klift]";
		mes ". . . . .";
		mes "Entiendo tu entusiasmo.";
		mes "Aun tienes mucho tiempo para";
		mes "practicar y ganar experiencia.";
		mes "Bendiciones para ti . . . . .";
		close;
	}
}
