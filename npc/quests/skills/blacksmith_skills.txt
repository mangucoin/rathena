			mes "importancia al valor de un objeto";
			mes "en terminos de zeny, podras";
			mes "deshacerte de ellos mucho";
			mes "mas facilmente. Mirame hacer esto.";
			mes "Ves? Ahora tu tambien puedes hacerlo.";
			next;
		}
		mes "[Goodman]";
		mes "La habilidad que aprendiste se";
		mes "llama Greed, pero espero";
		mes "que la uses de una manera que";
		mes "sea diferente a su nombre.";
		mes "No olvides eso, y espero";
		mes "que forjes verdaderas obras maestras.";
		skill "BS_GREED",1,SKILL_PERM;
		if (Upper == 1)
			BLACK_SK2 = 4;	// note: official script is using another variable when a transcendent character is doing the quest for the first time
		else
			BLACK_SK2 = 2;
		close;
	}
	mes "[Goodman]";
	mes "Como has estado usando";
	mes "la habilidad Greed? Espero que";
	mes "haya sido conveniente en";
	mes "tus esfuerzos por forjar bienes";
	mes "de mayor calidad para todos";
	mes "tus camaradas.";
	if (getskilllv("BS_GREED") == 0)	// debug
		skill "BS_GREED",1,SKILL_PERM;
	close;
}
