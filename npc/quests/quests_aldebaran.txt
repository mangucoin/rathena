	mes "[Trader]";
	mes "~Quien es este hombre misterioso?";
	mes "~Yo, el enigmatico y encantador 'Trader?'";
	next;
	mes "[Trader]";
	mes "Viajando por el Continente de Midgard, con todos sus maravillosos sombreros? Nadie lo sabe...";
	next;
	mes "[Trader]";
	mes "~Con que proposito colecciono e intercambio estos sombreros de todo el mundo? Elige un sombrero, acercate a resolver el misterio...";
	next;
	mes " (1) ^3355FFDoctor Band^000000:";
	mes "1 Red Bandana + 50 Irons + 1 Cracked Diamond + 3500 Zeny";
	mes " (2)^3355FFFeather Bonnet^000000:";
	mes "1 Romantic Gent + 300 Feather of Birds + 500 Zeny";
	mes " (3) ^3355FFPhantom of Opera^000000:";
	mes "20 Iron + 1 Singing Plant + 5000 Zeny ";
	mes " (4) ^3355FFSakkat^000000:";
	mes "120 Trunks + 10000 Zeny ";
	next;
	switch(select(" Doctor Band : Feather Bonnet : Phantom of Opera : Sakkat ")) {
	case 1:
		if ((countitem(2275) > 0) && (countitem(998) > 49) && (countitem(733) > 0) && (Zeny > 3499)) {
			delitem 2275,1; //Red_Bandana
			delitem 998,50; //Iron
			delitem 733,1; //Crystal_Jewel___
			set Zeny, Zeny-3500;
			mes "[Trader]";
			mes "~Hm! No tienes licencia medica, verdad? Esta bien, he oido hablar de un medico clandestino sin licencia que realizaba milagros medicos! Pero... eso podria haber sido un comic.";
			next;
			mes "[Trader]";
			mes "Oh, como sea. Solo no te dejes atrapar.";
			getitem 2273,1; //Doctor_Cap
			close;
		}
		else {
			mes "[Trader]";
			mes "Oye tu~";
			mes "Revisa los requisitos de nuevo.";
			mes "No pareces tonto. Asi que vamos, se serio.";
			close;
		}
	case 2:
		if ((countitem(2247) > 0) && (countitem(916) > 299) && (Zeny > 499)) {
			delitem 2247,1; //Oldman's_Romance
			delitem 916,300; //Feather_Of_Birds
			set Zeny, Zeny-500;
			mes "[Trader]";
			mes "~Ooh~! Tienes buen gusto para la moda. Se que te costo mucho reunir todo esto, pero este sombrero lo vale. Tomalo. ~Todo lo que necesitas ahora es un abrigo de piel y un baston!";
			getitem 5018,1; //Feather_Bonnet
			close;
		}
		else {
			mes "[Trader]";
			mes "Oye tu~";
			mes "Ve a revisar mis requisitos de nuevo. No pareces tonto. Vamos, se serio.";
			close;
		}
	case 3:
		if ((countitem(998) > 19) && (countitem(707) > 0) && (Zeny > 4999)){ //Iron Singing_Plant
			delitem 998,20; //Iron
			delitem 707,1; //Singing_Plant
			set Zeny, Zeny-5000;
			mes "[Trader]";
			mes "~Este? Es un poco aterrador, aunque tiene cierta cualidad romantica. ~Que piensas? ~Te gusta? Esta bien, tomalo, es tuyo!";
			getitem 2281,1; //Phantom_Of_Opera
			close;
		}
		else {
			mes "[Trader]";
			mes "Tonto. Ve a revisar los requisitos de nuevo. ^3355FFPhantom of Opera^000000 no es facil de conseguir. Asi que vamos, se serio.";
			close;
		}
	case 4:
		if ((countitem(1019) > 119) && (Zeny > 9999)){ //Wooden_Block
			delitem 1019,120; //Wooden_Block
			set Zeny, Zeny-10000;
			mes "[Trader]";
			mes "Si tienes la oportunidad de visitar la Aldea de las Tierras Altas, Payon, por favor ve y conoce al Artesano de Sakkat.";
			next;
			mes "[Trader]";
			mes "Nunca le ha vendido Sakkat a comerciantes que no sea yo, ya que solo yo puedo reconocer su calidad. Debido a su rareza, el Sakkat se ha convertido en un producto muy unico y excepcional. ~Esta bien! Tomalo, es tuyo!";
			getitem 2280,1; //Sahkkat
			close;
		}
		else {
			mes "[Trader]";
			mes "~TONTO~ Revisa mis requisitos de nuevo. Vamos, no pareces idiota, asi que se serio~";
			close;
		}
	}
}
