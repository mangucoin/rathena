payon_in03,188,146,5	script	Marx Hansen#juice	86,{
	if (checkweight(1201,1) == 0) {
		mes "- Espera un momento! -";
		mes "- Actualmente llevas -";
		mes "- demasiados objetos contigo. -";
		mes "- Por favor vuelve despues -";
		mes "- de guardar algunos objetos en el almacen Kafra. -";
		close;
	}
	if (MISC_QUEST&1) {
		mes "[Merchant Marx Hansen]";
		mes "Bienvenido.";
		mes "~Viniste a";
		mes "procesar frutas tambien?";
		next;
		switch(select("Hacer Jugo.:Hablar y obtener informacion sobre el procesamiento de frutas.:Cancelar.")) {
		case 1:
			mes "[Merchant Marx Hansen]";
			mes "~Que tipo de jugo de frutas te gustaria hacer?";
			next;
			switch(select("Jugo de Manzana:Jugo de Banana:Jugo de Zanahoria:Jugo de Uva:Cancelar")) {
			case 1:
				set .@fruit,512;
				set .@juice,531;
				break;
			case 2:
				set .@fruit,513;
				set .@juice,532;
				break;
			case 3:
				set .@fruit,515;
				set .@juice,534;
				break;
			case 4:
				set .@fruit,514;
				set .@juice,533;
				break;
			case 5:
				mes "[Merchant Marx Hansen]";
				mes "Bueno entonces...";
				mes "Nos vemos la proxima.";
				close;
			}
			if (countitem(.@fruit) == 0 || countitem(713) == 0 || Zeny < 3) {
				mes "[Merchant Marx Hansen]";
				mes "Oh no...";
				mes "No tienes todos los materiales necesarios. Para hacer "+getitemname(.@juice)+", necesito 1 "+getitemname(.@fruit)+" y 1 Empty Bottle. Tambien necesitare una tarifa de 3 zeny.";
				next;
				mes "[Merchant Marx Hansen]";
				mes "Cuando hayas preparado todo, licuare la fruta para darte un delicioso jugo.";
				close;
			}

			mes "[Merchant Marx Hansen]";
			mes "~Cuantos te gustaria?";
			next;
			switch(select("Tantos como pueda.:Quiero una cantidad especifica.:Cancelar.")) {
			case 1:
				set .@make,countitem(.@fruit);
				if (countitem(713) < .@make) set .@make, countitem(713);
				if (Zeny/3 < .@make) set .@make, Zeny/3;
				break;
			case 2:
				mes "[Merchant Marx Hansen]";
				mes "Elige un numero menor a 100. Si no quieres, pon '0'. Puedes hacer hasta " + countitem(.@fruit) + " botellas de jugo.";
				next;
				while(1) {
					input .@input;
					if (.@input == 0) {
						mes "[Merchant Marx Hansen]";
						mes "Bueno entonces...";
						mes "Vuelve pronto.";
						close;
					}
					else if (.@input > 100) {
						mes "[Merchant Marx Hansen]";
						mes "Mas de 100 botellas es imposible. Elige una cantidad diferente.";
						next;
					}
					else {
						break;
					}
				}
				set .@make,.@input;
				break;
			case 3:
				mes "[Merchant Marx Hansen]";
				mes "Bueno entonces...";
				mes "Vuelve pronto.";
				close;
			}

			set .@total_zeny,3 * .@make;

			if (countitem(.@fruit) < .@make || countitem(713) < .@make || Zeny < .@total_zeny) {
				mes "[Merchant Marx Hansen]";
				mes "Oh no...";
				mes "No tienes todos los materiales necesarios. No puedo hacer nada en esta situacion. Supongo que debes reunir lo que necesitas.";
				close;
			}
			delitem .@fruit,.@make;
			delitem 713,.@make;
			set Zeny, Zeny-.@total_zeny;
			getitem .@juice,.@make;

			mes "[Merchant Marx Hansen]";
			mes "~Aqui tienes! Jugo fresco y delicioso como lo prometi. Deberia ser muy refrescante y agradable.";
			next;
			mes "[Merchant Marx Hansen]";
			mes "Bueno entonces...";
			mes "Vuelve pronto.";
			close;
		case 2:
			mes "[Merchant Marx Hansen]";
			mes "Antes de que los humanos pudieran desarrollar sociedades tan vastas, recolectaban frutas de los arboles para sobrevivir. Las frutas pudieron haber sido la bendicion de la naturaleza que nos permitio existir en el mundo.";
			next;
			mes "[Merchant Marx Hansen]";
			mes "Desde que la vida se volvio tan prospera, la generacion mas joven parece ya no comer frutas. Asi que, comence a pensar en formas de hacer la fruta mas facil de comer.";
			next;
			mes "[Merchant Marx Hansen]";
			mes "Me di cuenta de que cuando haces jugo de frutas, es mas conveniente de consumir y tiene un sabor mucho mejor.";
			next;
			mes "# Informacion de Jugos de Frutas #";
			mes "^CC4E5C- Jugo de Manzana -^000000";
			mes "Apple x 1, Empty Bottle x 1, 3 zeny.";
			mes "^E3CF57- Jugo de Banana -^000000";
			mes "Banana x 1, Empty Bottle x 1, 3 zeny.";
			mes "^ED9121- Jugo de Zanahoria -^000000";
			mes "Carrot x 1, Empty Bottle x 1, 3 zeny.";
			mes "^CC00FF- Jugo de Uva -^000000";
			mes "Grape x 1, Empty Bottle x 1, 3 zeny.";
			close;
		case 3:
			mes "[Merchant Marx Hansen]";
			mes "~Oye!";
			mes "Si visitas";
			mes "a alguien, ~hablale!";
			close;
		}
	}
	else {
		mes "[Merchant Marx Hansen]";
		mes "Bienvenido.";
		mes "~En que puedo ayudarte?";
		next;
		if (select("Hablar:Cancelar") == 1) {
			mes "[Merchant Marx Hansen]";
			mes "Antes de que los humanos pudieran desarrollar sociedades tan vastas, recolectaban frutas de los arboles para sobrevivir. Las frutas pudieron haber sido la bendicion de la naturaleza que nos permitio existir en el mundo.";
			next;
			mes "[Merchant Marx Hansen]";
			mes "Desde que la vida se volvio tan prospera, la generacion mas joven parece ya no comer frutas. Asi que, comence a pensar en formas de hacer la fruta mas facil de comer.";
			next;
			mes "[Merchant Marx Hansen]";
			mes "Me di cuenta de que cuando haces jugo de frutas, es mas conveniente de consumir y tiene un sabor mucho mejor.";
			close;
		}
		mes "[Merchant Marx Hansen]";
		mes "~Oye!";
		mes "Si visitas";
		mes "a alguien, ~hablale!";
		close;
	}
}
