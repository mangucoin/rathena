alberta,26,229,0	script	Kafra Employee#bunny	83,{
	if (BUNYBND == 1) {
		mes "[Kafra Employee]";
		mes "~Hola~!";
		mes "~Te gustaria";
		mes "participar en el";
		mes "evento especial de Kafra";
		mes "Corporation ^529DFFBunny Band^000000?";
		next;
		switch(select("Claro, traje los objetos.:Informacion del evento:Cancelar")) {
		case 1:
			mes "[Kafra Employee]";
			mes "~Muy bien~";
			mes "Dejame revisar";
			mes "si trajiste";
			mes "todos los objetos...";
			next;
			setarray .@Items[0],949,100,706,1,722,1,2213,1;
			for(set .@i,0; .@i<8; set .@i,.@i+2)
				if (countitem(.@Items[.@i]) < .@Items[.@i+1]) {
					mes "[Kafra Employee]";
					mes "Ooh, lo siento";
					mes "pero necesitas";
					mes "traer al menos";
					mes callfunc("F_InsertPlural",.@Items[.@i+1],getitemname(.@Items[.@i]))+".";
					close;
				}
			mes "[Kafra Employee]";
			mes "Genial, veo que";
			mes "has reunido";
			mes "todo lo que necesito para";
			mes "hacer la Bunny Band.";
			mes "Por favor espera un momento";
			mes "mientras la armo...";
			next;
			for(set .@i,0; .@i<8; set .@i,.@i+2)
				if (countitem(.@Items[.@i]) < .@Items[.@i+1]) {
					mes "[Kafra Employee]";
					mes "~Hm? Lo siento,";
					mes "pero en realidad no puedo";
					mes "hacer esto ahora. Necesitas";
					mes ""+callfunc("F_InsertPlural",.@Items[.@i+1],getitemname(.@Items[.@i]));
					mes "para que pueda armar";
					mes "esta Bunny Band...";
					close;
				}
			delitem 949,100; //Feather
			delitem 706,1; //Four_Leaf_Clover
			delitem 722,1; //Scarlet_Jewel
			delitem 2213,1; //Cat_Hairband
			mes "[Kafra Employee]";
			mes "Ah, ~aqui tienes~";
			mes "~La Bunny Band perfecta!";
			mes "Bueno, espero que la disfrutes.";
			getitem 2214,1; //Bunny_Band
			set BUNYBND,0;
			next;
			mes "[Kafra Employee]";
			mes "Gracias por";
			mes "participar en este";
			mes "evento especial y por";
			mes "tu continuo apoyo";
			mes "a los Servicios Kafra~";
			close;
		case 2:
			goto L_Info;
		case 3:
			close;
		}
	}
	mes "[Kafra Employee]";
	mes "~Hola~! ~Te gustaria";
	mes "unirte a nuestro evento especial ^529DFFBunny";
	mes "Band^000000 organizado por Kafra";
	mes "Corporation y patrocinado por";
	mes "el Gremio de Mercaderes de Alberta?";
	next;
	switch(select("Unirme al evento:Informacion del evento:Cancelar")) {
	case 1:
		set BUNYBND,1;
		mes "[Kafra Employee]";
		mes "~Genial! ~Gracias por";
		mes "participar! Si aun";
		mes "no lo sabias,";
		mes "necesitas reunir estos";
		mes "objetos si quieres que te arme";
		mes "una Bunny Band...";
		next;
		mes "[Kafra Employee]";
		mes "100 Feathers,";
		mes "1 Four-Leaf Clover,";
		mes "1 Kitty Band y";
		mes "1 Pearl. ~Eso es todo!";
		mes "~Buena suerte y estare";
		mes "esperandote aqui~";
		close;
	case 2:
		break;
	case 3:
		close;
	}
L_Info:
	mes "[Kafra Employee]";
	mes "Para agradecer a nuestros";
	mes "valiosos clientes, Kafra Corporation";
	mes "ha preparado un evento especial";
	mes "donde una Empleada Kafra armara";
	mes "Bunny Bands para los aventureros";
	mes "que traigan los objetos requeridos.";
	next;
	mes "[Kafra Employee]";
	mes "Para este objeto especial,";
	mes "unico en su tipo,";
	mes "trae 100 Feathers,";
	mes "1 Four-Leaf Clover,";
	mes "1 Kitty Band y";
	mes "1 Pearl.";
	next;
	mes "[Kafra Employee]";
	mes "Cuando estes listo,";
	mes "regresa y traeme";
	mes "esos objetos.";
	mes "(Patrocinado por el";
	mes "Gremio de Mercaderes de Alberta).";
	close;
}
