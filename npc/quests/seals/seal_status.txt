prontera,113,294,3	script	Sign Post#god	837,{
	mes "======== Estado de Sellos Divinos ========";
	setarray .@seal$[1], "Sleipnir","Megingjard","Brisingamen","Mjolnir";
	for(set .@i,1; .@i<=4; set .@i,.@i+1) {
		set .@val, getd("$God"+.@i);
		if (.@val == 0)
			set .@status$,"No Visto";
		else if (.@val < $@god_check1)
			set .@status$,"Activo";
		else if (.@val < $@god_check2)
			set .@status$,"Aparecido";
		else
			set .@status$,"Liberado";
		mes .@seal$[.@i]+" Sello: "+.@status$;
	}
	mes " ";
	mes "======= Tu Estado de Sellos ========";
	if (god_sl_1 < 51)
		mes "^ff0000Sello de Sleipnir: No Completado^000000";
	else
		mes "^00ff00Sello de Sleipnir: Completado^000000";
	if (god_eremes < 28)
		mes "^ff0000Sello de Megingjard: No Completado^000000";
	else
		mes "^00ff00Sello de Megingjard: Completado^000000";
	if (god_brising < 50)
		mes "^ff0000Sello de Brisingamen: No Completado^000000";
	else
		mes "^00ff00Sello de Brisingamen: Completado^000000";
	if (god_mjo_0 < 11)
		mes "^ff0000Sello de Mjolnir: No Completado^000000";
	else
		mes "^00ff00Sello de Mjolnir: Completado^000000";
	close;
}
