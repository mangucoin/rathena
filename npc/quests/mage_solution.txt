pay_arche,122,100,0	script	Dollshoi	88,{
	mes "[Mage Guildsman]";
	mes "~Quieres una Solucion? Hmm, dame 50 Zeny y al menos 1 Tubo de Ensayo Vacio.";
	next;
	if (select("De acuerdo, trato hecho.:No, olvidalo.") == 1) {
		mes "[Mage Guildsman]";
		if (Zeny < 50) {
			mes "~Oye! No tienes suficiente dinero para cubrir mi cargo de 50 Zeny.";
			close;
		}
		if (countitem(1092) == 0) {
			mes "~No puedes llevar soluciones sin una botella! Traeme un Tubo de Ensayo Vacio.";
			close;
		}
		delitem 1092,1; //Empty_Cylinder
		set Zeny, Zeny-50;
		getitem 1089,1; //Payon_Potion
	}
	close;
}
