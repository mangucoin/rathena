// Edgar's Offer :: iz_npc
//============================================================
-	script	::Edgar_izlude	-1,{
	if (MISC_QUEST & 16) {
		mes "[Edgar]";
		mes "Entonces, vas a Alberta de nuevo? Dejame hacerte el mismo descuento y cobrarte solo 250 Zeny, igual que la ultima vez.";
		mes "Que te parece?";
		next;
		if (select("Dale~!:Por que eres tan amable conmigo?!") == 1) {
			if (Zeny < 250) {
				mes "[Edgar]";
				mes "Ehm...";
				mes "Esto no es";
				mes "suficiente dinero.";
				mes "Por que no vas";
				mes "a conseguir mas?";
				close ;
			}
			set Zeny, Zeny-250;
			warp "alberta",195,164;
			end;
		}
		mes "[Edgar]";
		mes "Es solo mi forma de ser. Ademas, tu sonrisa picara me recuerda a mi querido hijo rubio que se fue de casa hace anos para convertirse en marinero en su propio barco. Que Dios lo bendiga, donde sea que este.";
		close;
	}
	mes "[Edgar]";
	mes "Mi pueblo, Izlude, esta conectado con Alberta por el puerto del Oeste. Hay tanto trafico entre nosotros que casi me vuelvo Albertiano. Jejeje~";
	next;
	mes "[Edgar]";
	mes "Hay un tipo que conozco bastante bien, Phelix, que vive en Alberta. Ese tipo es muy tacano... Cobra por todo!";
	next;
	mes "[Edgar]";
	mes "Pero en realidad es buen tipo y le gusta ayudar a los demas. Tiene buen corazon y te dara su apoyo si le pagas lo que pide.";
	next;
	mes "[Edgar]";
	mes "Ultimamente, la gente en Alberta dice que realmente esta tratando de ayudar a la gente y que su demanda de Jellopies es solo una fachada. Bueno, deberias echarle un vistazo a lo que ofrece.";
	next;
	if (select("Puedes decirme como llegar a Alberta?:Terminar conversacion") == 1) {
		mes "[Edgar]";
		mes "Eh? Bueno, puedes usar tus pies y simplemente caminar. Pero si tienes dinero, te sugiero que tomes un barco.";
		next;
		if (select("Okay, entendido.:Pero estoy harto de caminar y no tengo dinero!") == 1) {
			mes "[Edgar]";
			mes "Dale, cuidate~";
			close;
		}
		mes "[Edgar]";
		mes "Okay...";
		mes "No quieres caminar Y no tienes dinero, pero aun asi quieres ir? Ay por Dios...";
		next;
		mes "[Edgar]";
		mes "Bueno, esta bien. Yo, siendo capitan de un barco, puedo llevarte a un precio mas bajo. Que tal 250 Zeny?";
		next;
		if (select("Dale~!:Bah, que estafa!!") == 1) {
			set MISC_QUEST,MISC_QUEST | 16;
			if (Zeny < 250) {
				mes "[Edgar]";
				mes "Ehm...";
				mes "Esto no es";
				mes "suficiente dinero.";
				mes "Ve y consigue";
				mes "un poco mas.";
				close;
			}
			set Zeny, Zeny-250;
			warp "alberta",195,164;
			end;
		}
		mes "[Edgar]";
		mes "Ay ay ay,";
		mes "si piensas";
		mes "que eso es una estafa...";
		close;
	}
	mes "[Edgar]";
	mes "Si, esta bien.";
	mes "Nos vemos~";
	close;
}
