	mes "Esta es la Asistencia de Sonrisas.";
	mes "~En que puedo ayudarte?";
	next;
	switch(select("^3355FFMr. Smile^000000 ?:Construir ^3355FFMr. Smile^000000 :Salir")) {
	case 1:
		mes "[Smile Girl]";
		mes "Un evento nacional organizado por orden de ^5577FFSu majestad Tristram III^000000,";
		mes "que pretende animar a la nacion del Reino de Rune-Midgarts";
		mes " a jugar en un ambiente mas agradable!";
		mes "Soy ^3355FF' Asistencia de Sonrisas '^000000,";
		mes ".. quien lidera el evento nacional bajo el nombre de ";
		mes "^3355FFSonrisas por todo el Reino de Rune-Midgarts~^000000";
		next;
		mes "[Smile Girl]";
		mes "Con objetos simples y faciles de conseguir,";
		mes "puedo proporcionarte";
		mes "^3355FF' Mr. Smile '^000000.";
		mes "Los requisitos son los siguientes.";
		next;
		mes "[Smile Girl]";
		mes "^3355FF10 Jellopy^000000";
		mes "^3355FF10 Fluff^000000";
		mes "^3355FF10 Clover^000000";
		next;
		mes "[Smile Girl]";
		mes "Con este evento";
		mes "todos seran felices y sonreiran,";
		mes "reuniendose con otras personas,";
		mes "y trataran de hacer de Ragnarok el juego mas divertido del mundo.";
		close;
	case 2:
		if (countitem(909) > 9 && countitem(914) > 9 && countitem(705) > 9) {
			mes "[Smile Girl]";
			mes "~Felicidades!";
			mes "Ahora por favor toma este Mr.Smile.";
			next;
			delitem 909,10; //Jellopy
			delitem 914,10; //Fluff
			delitem 705,10; //Clover
			getitem 2278,1; //Mr_Smile
			mes "[Smile Girl]";
			mes "Su majestad, Tristram III";
			mes "ha prometido hacer su mejor esfuerzo para que Ragnarok sea mejor y mas divertido.";
			next;
			mes "El hecho mencionado anteriormente";
			mes "fue anunciado por";
			mes "la Oficina de Informacion Publica del Reino de Rune-Midgarts.";
			close;
		}
		else {
			mes "[Smile Girl]";
			mes "Oh - desafortunadamente";
			mes "no has traido";
			mes "suficientes objetos para Mr. Smile.";
			mes "^3355FF10 Jellopy^000000";
			mes "^3355FF10 Fluff^000000";
			mes "^3355FF10 Clover^000000";
			mes "Por favor revisa los requisitos de arriba.";
			close;
		}
	case 3:
		mes "[Smile Girl]";
		mes "Gracias por visitarnos.";
		mes "Nosotros..";
		mes "El Reino de Rune-Midgarts";
		mes "siempre tratamos de hacer de Ragnarok";
		mes "un juego mejor y mas divertido.";
		next;
		mes "[Smile Girl]";
		mes "Te pedimos sinceramente";
		mes "que cooperes.";
		mes "El hecho mencionado anteriormente";
		mes "fue anunciado por la Oficina de Informacion Publica del Reino de Rune-Midgarts.";
		close;
	}
}
