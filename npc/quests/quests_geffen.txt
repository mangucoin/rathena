// Blacksmith: 'Welding Mask' Quest
//============================================================
geffen_in,144,166,4	script	Blacksmith	63,{
	mes "[Blacksmith]";
	mes "Nunca oiste de mi? Pues, soy el Herrero Veterano de estas tierras. He pasado 30 anios en este fuego infernal, doblando el acero a mi voluntad de hierro!";
	next;
	mes "[Blacksmith]";
	mes "Pero...";
	mes "Estoy envejeciendo y mi destreza comienza a desvanecerse con la edad. Asi que invente algo para los jovenes y prometedores Herreros, con la esperanza de que tambien dominen mi oficio...";
	next;
	switch(select("Que es?:...")) {
	case 1:
		mes "[Blacksmith]";
		mes "Los Herreros siempre debemos manipular el acero bajo calor extremo, y el aire caliente y los metales fundidos son mas de lo que la mayoria puede soportar.";
		next;
		mes "[Blacksmith]";
		mes "Por el bien de nuestro oficio, nuestros hermosos pero masculinos rostros estan en riesgo...";
		next;
		mes "[Blacksmith]";
		mes "Pero con esta maravilla de la tecnologia, no hay de que preocuparse! Contempla, la ^3355FFMascara de Soldadura^000000 !!";
		next;
		mes "[Blacksmith]";
		mes "Funciona cubriendo tu cara con una gruesa placa de metal. De verdad, es un objeto que todos deberian querer! Si no para ti, seria buena idea darselo a un amigo que estudie herreria.";
		next;
		mes "[Blacksmith]";
		mes "Que dices? Si me das ^2244FF50 Steel^000000 y ^4422FFsolo 2000 Zeny^000000, te lo hare de inmediato~!!";
		next;
		switch(select("Hmm... No esta mal. De acuerdo.:Como funciona otra vez?")) {
		case 1:
			if ((countitem(999) > 49) && (Zeny > 1999)){
				mes "[Blacksmith]";
				mes "Ah, veo que aprecias el trabajo de un Herrero! Muy bien, lo hare de inmediato!!";
				next;
				delitem 999,50;  //Steel
				set Zeny,Zeny-2000;
				getitem 2292,1; // Welding_Mask
				mes "[Blacksmith]";
				mes "Jajaja~! Dale buen uso!";
				close;
			}
			else {
				mes "[Blacksmith]";
				mes "Hmmm...";
				mes "No tienes los materiales que necesito.";
				next;
				mes "[Blacksmith]";
				mes "Un Herrero comun interpretaria esto como falta de respeto, pero te perdonare. Solo no olvides lo que necesito si quieres que te haga esta mascara.";
				close;
			}
		case 2:
			mes "[Blacksmith]";
			mes "La ^3355FFMascara de Soldadura^000000 protege tu cara cubriendola con una gruesa placa de metal. Esta invencion es el resultado de 30 anios de trabajo incansable en el martillo y el yunque.";
			close;
		}
	case 2:
		mes "[Blacksmith]";
		mes "Ah...!";
		mes "Supongo que es la primera vez que conoces a un Herrero tan grande como yo, y estas sin palabras de la impresion. Por favor, tomate tu tiempo y habla cuando estes listo.";
		next;
		mes "^3355FFEste tipo parece";
		mes "un poco lleno";
		mes "de si mismo.^000000";
		close;
	}
}

// Number Match Game :: gef_yagu
//============================================================
geffen,122,77,4	script	Nia#yagu	700,{
	mes "[Nia]";
	mes "Hola, soy Nia, la quinta";
	mes "discipula de Mawong, el";
	mes "maestro de los juegos. Ahora,";
	mes "te gustaria jugar un juego de";
	mes "Adivina el Numero conmigo?";
	next;
	switch(select("Adivina el Numero?:Practicar Adivina el Numero:Jugar Adivina el Numero:Poseedor del Record:Cancelar")) {
	case 1:
		mes "[Nia]";
		mes "Ah, bueno, el objetivo del";
		mes "juego Adivina el Numero es";
		mes "adivinar el numero que estoy";
		mes "pensando. Por supuesto, hay";
		mes "algunas reglas que limitan los";
		mes "numeros que puedo elegir.";
		next;
		mes "[Nia]";
		mes "Todos los numeros que";
		mes "invento deben ser de 3 digitos,";
		mes "no puedo usar 0 en ninguno";
		mes "de los digitos, y no puedo";
		mes "usar el mismo numero en";
		mes "ninguno de los digitos mas de una vez.";
		next;
		mes "[Nia]";
		mes "Entonces, por ejemplo, no";
		mes "puedo usar los numeros 103,";
		mes "112, 252, o 701. Esos numeros";
		mes "o tienen un 0, o usan el";
		mes "mismo numero en los digitos";
		mes "mas de una vez. Entendido?";
		next;
		mes "[Nia]";
		mes "Ahora, solo tienes 5 intentos,";
		mes "y despues de cada intento, te";
		mes "dire cuantos digitos adivinaste";
		mes "correctamente, seguido del numero de digitos correctos en";
		mes "la posicion correcta.";
		next;
		mes "[Nia]";
		mes "Por ejemplo, digamos que el";
		mes "numero que invente es 168.";
		mes "Si tu primer intento es 678, entonces adivinaste dos digitos,";
		mes "6 y 8, correctamente. Sin embargo,";
		mes "solo el 8 esta en la posicion correcta.";
		next;
		mes "[Nia]";
		mes "Sabes, seria buena idea";
		mes "si pruebas el Modo Practica";
		mes "primero. Luego, cuando te";
		mes "sientas mas comodo, juega el";
		mes "juego real. No olvides que la velocidad cuenta para tu puntaje!";
		next;
		mes "[Nia]";
		mes "Ah, si logras hacer un";
		mes "nuevo record en el juego";
		mes "Adivina el Numero, tu";
		mes "nombre sera registrado,";
		mes "asi que no pierdas esta";
		mes "oportunidad de fama y quiza fortuna~";
		close;
	case 2:
		mes "[Nia]";
		mes "Genial, hagamos una practica";
		mes "de Adivina el Numero. Por favor";
		mes "intenta adivinar mi numero, si?";
		mes "Si quieres cancelar, ingresa un numero mayor a 1,000.";
		set .@practice,1;
		break;
	case 3:
		mes "[Nia]";
		mes "Genial, juguemos Adivina el Numero!";
		mes "Tendras 5 oportunidades para adivinar el numero que estoy pensando.";
		mes "Si quieres cancelar, ingresa un numero mayor a 1,000.";
		set .@practice,0;
		break;
	case 4:
		set .@name_record$, $050908_minus1_yagu$;
		if (.@name_record$ == "") //custom
			set .@name_record$,"(null)";
		set .@score_record, $050908_minus1_yagu;
		set .@score_min, .@score_record / 60;
		set .@score_sec, .@score_record % 60;
		mes "[Nia]";
		mes "^ff0000" + .@name_record$ + "^000000 es el";
		mes "poseedor del record del juego";
		mes "Adivina el Numero con un";
		mes "tiempo de ^ff0000" + .@score_min + " minutos, " + .@score_sec + " segundos^000000.";
		close;
	case 5:
		mes "[Nia]";
		mes "Cuidate mucho,";
		mes "aventurero. Si alguna vez";
		mes "tienes ganas de jugar";
		mes "Adivina el Numero, ven";
		mes "a hablar conmigo, si? Nos vemos~";
		close;
	}
	next;

	// Generate 3 digits of a number.
	while(1) {
		set .@yagu100, rand(1,9);
		set .@yagu10, rand(1,9);
		set .@yagu1, rand(1,9);
		if ((.@yagu100 != .@yagu10) && (.@yagu100 != .@yagu1) && (.@yagu10 != .@yagu1)) {
			// Store the actual number (unofficial).
			set .@yagu, (100*.@yagu100) + (10*.@yagu10) + .@yagu1;
			break;
		}
	}

	// The official time calculation is based on ServerTime,
	// an int HHMMSS equivalent to atoi(gettimestr("%H%M%S",7)).
	// UNIX time is really a better way to do this.
	set .@start_time, gettimetick(2);

	while(1) {

		// Guess a number that meets all conditions.
		while(1) {
			if (input(.@input,100,999) != 0) {
				mes "[Nia]";
				mes "Has cancelado el";
				mes "juego Adivina el Numero.";
				close;
			}
			set .@input100, .@input / 100;
			set .@input10, (.@input % 100) / 10;
			set .@input1, .@input % 10;
			if ((.@input100 > 0) && (.@input10 > 0) && (.@input1 > 0)) {
				if ((.@input100 != .@input10) && (.@input100 != .@input1) && (.@input10 != .@input1))
					break;
			}
			mes "[Nia]";
			mes "Ingresaste uno de los";
			mes "digitos como 0, o pusiste";
			mes "el mismo valor numerico en";
			mes "los digitos mas de una vez.";
			mes "Por favor ingresa otro intento.";
			next;
		}

		set .@retry, .@retry+1;
		mes "[Nia]";
		mes "^ff0000"+strcharinfo(0)+"^000000,";
		mes "tu intento es ^0000ff"+.@input+"^000000.";
		mes "Dame un momento para";
		mes "calcular tus resultados.";
		next;

		// Correct digits in correct locations.
		set .@strike,0;
		if (.@yagu100 == .@input100)
			set .@strike, .@strike+1;
		if (.@yagu10 == .@input10)
			set .@strike, .@strike+1;
		if (.@yagu1 == .@input1)
			set .@strike, .@strike+1;

		// Correct digits in incorrect locations.
		set .@ball,0;
		if ((.@yagu100 == .@input10) || (.@yagu100 == .@input1))
			set .@ball, .@ball+1;
		if ((.@yagu10 == .@input100) || (.@yagu10 == .@input1))
			set .@ball, .@ball+1;
		if ((.@yagu1 == .@input100) || (.@yagu1 == .@input10))
			set .@ball, .@ball+1;

		if (.@strike == 3) {
			if (.@practice) {
				mes "[Nia]";
				mes "Si, el numero que";
				mes "pense era ^ff0000"+.@yagu+"^000000!";
				mes "Felicidades, acabas";
				mes "de ganar esta partida";
				mes "de practica! Ahora deberias";
				mes "estar listo para lo real~";
				close;
			} else {

				// Calculate final time and check against the previous record.
				set .@end_time, gettimetick(2);
				if (.@end_time < .@start_time)
					set .@end_time, .@end_time + 3600;
				set .@total_time, .@end_time - .@start_time;
				set .@total_min, .@total_time / 60;
				set .@total_sec, .@total_time % 60;
				set .@name_record$, $050908_minus1_yagu$;
				if (.@name_record$ == "") //custom
					set .@name_record$,"(null)";
				set .@score_record, $050908_minus1_yagu;
				set .@score_min, .@score_record / 60;
				set .@score_sec, .@score_record % 60;

				if (.@score_record >= .@total_time || !.@score_record) { //custom (second condition)
					set $050908_minus1_yagu$, strcharinfo(0);
					set $050908_minus1_yagu, .@total_time;
					mes "[Nia]";
					mes "Genial! Adivinaste";
					mes "mi numero, que era";
					mes "^ff0000" + .@yagu + "^000000. Terminaste en";
					mes "^ff0000" + .@total_min + " minutos, " + .@total_sec + " segundos^000000,";
					mes "rompiendo el record anterior";
					mes "de ^ff0000" + .@score_min + " minutos, " + .@score_sec + " segundos^000000.";
					next;
					mes "[Nia]";
					mes "^ff0000" + .@name_record$ + "^000000";
					mes "establecio ese viejo record,";
					mes "pero ahora tu eres la";
					mes "nueva persona a vencer en el";
					mes "juego. Felicidades";
					mes "por establecer un nuevo record!";
					close;
				}
				mes "[Nia]";
				mes "Buen trabajo, adivinaste";
				mes "mi numero! La respuesta, por";
				mes "supuesto, era ^ff0000" + .@yagu + "^000000. Lograste";
				mes "adivinar este numero correctamente";
				mes "en ^ff0000" + .@total_min + " minutos, " + .@total_sec + " segundos^000000.";
				next;
				mes "[Nia]";
				mes "^ff0000" + .@name_record$ + "^000000 es";
				mes "el actual poseedor del record";
				mes "de Adivina el Numero con un";
				mes "tiempo de ^ff0000" + .@score_min + " minutos, " + .@score_sec + " segundos^000000.";
				mes "Buena suerte, y ojala puedas";
				mes "romper este record algun dia~";
				close;
			}
		} else {
			if (.@practice) {
				mes "[Nia]";
				mes "Tu intento tiene ^ff0000" + .@ball + "^000000 de los";
				mes "numeros correctos. ^ff0000" + .@strike + "^000000 de los";
				mes "digitos en tu intento tienen";
				mes "el numero correcto en la";
				mes "posicion correcta del digito.";
				mes "Bien, intentalo de nuevo~";
				next;
			} else {
				mes "[Nia]";
				mes "Tu ultimo intento tenia ^ff0000" + .@ball + "^000000";
				mes "de los numeros correctos,";
				mes "y ^ff0000" + .@strike + "^000000 digitos en tu intento";
				mes "tenian el numero correcto en";
				mes "la posicion correcta del digito.";
				switch(.@retry) {
					case 1: mes "Te quedan cuatro intentos."; break;
					case 2: mes "Te quedan tres intentos."; break;
					case 3: mes "Te quedan dos intentos."; break;
					case 4: mes "Solo te queda un intento..."; break;
					case 5: mes "La respuesta era ^ff0000" + .@yagu + "^000000."; close;
				}
				next;
			}
		}
	}
}
