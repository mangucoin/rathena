		mes "^4D4DFF1 Holy Water^000000, y";
		mes "^4D4DFF1 Chicken Egg^000000.";
		close;
	case 2:
		mes "[Nerlen]";
		mes "Bien, cuantos";
		mes "Holy Eggs te gustaria";
		mes "que te hiciera?";
		mes "Solo puedo hacer hasta";
		mes "10 Holy Eggs a la vez.";
		mes "Ingresa ''0'' para cancelar.";
		next;
		while(1) {
			input .@input;
			if (.@input == 0) {
				mes "[Nerlen]";
				mes "Oh, cambiaste de";
				mes "opinion? Sientete libre de visitarme";
				mes "cuando quieras que te";
				mes "haga algunos Holy Eggs~";
				close;
			}
			else if (.@input < 0 || .@input > 10) {
				mes "[Nerlen]";
				mes "Lo siento, pero solo puedo";
				mes "hacer hasta 10 Holy Eggs";
				mes "a la vez. Si querias";
				mes "cancelar, solo ingresa ''0.''";
				next;
			}
			else {
				if ((countitem(509) < .@input) || (countitem(511) < .@input) || (countitem(508) < .@input) || (countitem(523) < .@input) || (countitem(7605) < .@input)) {

					mes "Holy Eggs. Necesitas lo";

					mes "cada Holy Egg para ti.";
					next;
					mes "[Nerlen]";
					mes "^4D4DFF1 White Herb^000000,";
					mes "^4D4DFF1 Green Herb^000000,";
					mes "^4D4DFF1 Yellow Herb^000000,";
					mes "^4D4DFF1 Holy Water^000000, y";
					mes "^4D4DFF1 Chicken Egg^000000.";
					close;
				}
				else {
					break;
				}
			}
		}
		if (checkweight(12019,.@input) == 0) {
			mes "[Nerlen]";
			mes "Lo siento, pero no";
			mes "podras cargar";
			mes "tantos Holy Eggs ahora...";
			mes "Mejor libera mas";
			mes "espacio en tu Inventario primero.";
			close;
		}
		else {
			delitem 509,.@input; //White_Herb
			delitem 511,.@input; //Green_Herb
			delitem 508,.@input; //Yellow_Herb
			delitem 523,.@input; //Holy_Water
			delitem 7605,.@input; //Egg
			getitem 12019,.@input; //Holy_Egg
			mes "[Nerlen]";
			mes "Gracias!";
			mes "Espero que disfrutes";
			mes "estos Holy Eggs~";
			mes "Espero que estes con el amor de Dios.";
			close;
		}
	case 3:
		mes "[Nerlen]";
		mes "Bien, entiendo.";
		mes "Que Dios te bendiga.";
		close;
	}
}
