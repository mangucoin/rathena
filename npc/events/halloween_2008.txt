payon,162,176,4	script	Halloween Magician#iRO08	704,{
	if (Hallow08 < 1) {
		mes "[Mago de Halloween]";
		mes "Kkkkkkkkk!";
		mes "Tengo un evento especial de Halloween que pone a prueba tu suerte y agilidad.";
		mes "¿Te interesa?";
		next;
		mes "[Mago de Halloween]";
		mes "¡Vamos! ¡No seas cobarde!";
		mes "¡Si juntas suficientes tickets puedes obtener buenos premios!";
		mes "¿Entonces qué dices?";
		next;
		goto MainMenu;
	}
	if (Hallow08Kill == 1) {
		mes "[Mago de Halloween]";
		mes "¿Eres un miedoso o qué?!";
		mes "Sabes que quieres intentarlo de nuevo...";
		mes "¿Conoces las reglas?";
		next;
		set Hallow08Kill,0;
		if(select("Si, las conozco.:No, no las conozco.") == 2) {
			goto Rules;
		}
		goto Participate;
	}
	if (Hallow08Kill == 2) {
		mes "[Mago de Halloween]";
		mes "Oh, ¡bien hecho! ¡Eres bueno!";
		mes "¿No es divertido con zombies??";
		mes "¡Sabes, los zombies también fueron personas!";
		mes "¡Ja!";
		mes "Kkkkkkk.";
		next;
		getitem 7941,1;
		set Hallow08Kill,0;
		mes "[Mago de Halloween]";
		mes "Como te prometí";
		mes "Puedes cambiar los tickets de Halloween por objetos geniales.";
		close;
	}
	if (Hallow08 > 0) {
		mes "[Mago de Halloween]";
		mes "Bueno, ¿quieres escuchar las reglas de nuevo o simplemente volver a la acción...";
		next;
		goto MainMenu2;
	}

		Rules:
			mes "[Mago de Halloween]";
			mes "Esta aldea es como un Payon virtual.";
			mes "Hay zombies y ghouls merodeando y tres salidas al sur, pero solo una funciona.";
			mes "Eso te toca descubrirlo a ti.";
			next;
			mes "[Mago de Halloween]";
			mes "No puedes usar ninguna habilidad para matar a los ghouls o zombies.";
			mes "Y una cosa más...";
			mes "no deberías olvidar...";
			next;
			mes "[Mago de Halloween]";
			mes "Todos los participantes deben ir sin nada puesto.";
			mes "Pon todas tus pertenencias en tu almacén y vuelve aquí cuando tu peso sea '0'.";
			next;
			mes "[Mago de Halloween]";
			mes "¡Ah, y una cosa más!";
			mes "No puedes estar montado en un PecoPeco ni tener un Carrito.";
			mes "Si los tienes, los removeré antes de que entres.";
			mes "¿Entendido?";
			next;
			mes "[Mago de Halloween]";
			mes "Recuerda, hay tres salidas pero solo una funciona al azar, los zombies y ghouls que merodean no pueden ser matados y no puedes llevar nada puesto.";
			next;
			if (Hallow08 > 0) {
				mes "[Mago de Halloween]";
				mes "Oye...";
				mes "Vuelve cuando estés listo.";
				close;
			} else {
				goto MainMenu;
			}

		Participate:
			mes "[Mago de Halloween]";
			mes "Ok, estás listo.";
			mes "Déjame revisar tu peso.";
			next;
			if (Weight > 0) {
				mes "[Mago de Halloween]";
				mes "¡Cielos!";
				mes "Siempre hay una oveja negra en todas partes.";
				next;
				mes "[Mago de Halloween]";
				mes "¿Crees que no me daría cuenta de que tu peso está por encima de '0'?";
				mes "Estás pasado de peso...";
				close;
			} else {
				mes "[Mago de Halloween]";
				mes "Pareces listo para ir, y tu peso está perfecto.";
				next;
				mes "[Mago de Halloween]";
				mes "He estado bastante ocupado con mi trabajo, así que estoy agotado.";
				next;
				mes "[Mago de Halloween]";
				mes "A veces se me olvida enviarte allá...";
				next;
				mes "[Mago de Halloween]";
				mes "Espero que regreses bien.";
				close2;
				if (Hallow08 < 1) {
					set Hallow08,1;
				}
				set Hallow08Kill,1;
				set @Hallow08Warp, rand(1,3);
				percentheal -98,0;
				setriding 0;
				setcart 0;
				warp "evt_zombie",155,246;
				end;
			}

		TicketExchange:
			mes "[Mago de Halloween]";
			mes "¿Quieres cambiar tickets por premios?";
			mes "¡Buen trabajo! ¡Kkkkkk!";
			next;
			mes "[Mago de Halloween]";
			mes "Déjame decirte qué objetos puedes intercambiar.";
			next;
			mes "[Mago de Halloween]";
			mes "5 tickets por Pumpkin Pie.";
			mes "20 tickets por Pumpkin-Head.";
			mes "50 tickets por Old Blue Box.";
			mes "70 tickets por Old Purple Box.";
			mes "200 tickets por Old Card Album.";
			next;
			mes "[Mago de Halloween]";
			mes "¿Qué te gustaría intercambiar?";
			next;
			switch(select("Pumpkin Pie:Pumpkin-Head:Old Blue Box:Old Purple Box:Old Card Album")) {
			case 1:
				if (countitem(7941) < 5) {
					goto NotEnough;
				} else {
					delitem 7941,5;
					getitem 12192,1;
					goto Enough;
				}
				break;
			case 2:
				if (countitem(7941) < 20) {
					goto NotEnough;
				} else {
					delitem 7941,20;
					getitem 5134,1;
					goto Enough;
				}
				break;
			case 3:
				if (countitem(7941) < 50) {
					goto NotEnough;
				} else {
					delitem 7941,50;
					getitem 603,1;
					goto Enough;
				}
				break;
			case 4:
				if (countitem(7941) < 70) {
					goto NotEnough;
				} else {
					delitem 7941,70;
					getitem 617,1;
					goto Enough;
				}
				break;
			case 5:
				if (countitem(7941) < 200) {
					goto NotEnough;
				} else {
					delitem 7941,200;
					getitem 616,1;
					goto Enough;
				}
			}

		NextTime:
			mes "[Mago de Halloween]";
			mes "Ok, nos vemos entonces.";
			mes "Kkkkkkkk.";
			close;

	MainMenu:
		menu "Explícamelo.",Rules,"Quiero participar.",Participate,"Quiero cambiar tickets por premios.",TicketExchange,"Volveré la próxima vez.",NextTime;

	MainMenu2:
		menu "¡Llévame de vuelta ahora!",Participate,"Por favor, dime las reglas",Rules,"Quiero cambiar tickets por premios.",TicketExchange,"Volveré la próxima vez.",NextTime;

	NotEnough:
		mes "[Mago de Halloween]";
		mes "¡No tienes suficientes tickets!";
		mes "¿Ni siquiera sabes contar?";
		mes "Por favor ven aquí con la cantidad correcta de tickets.";
		close;

	Enough:
		mes "[Mago de Halloween]";
		mes "Aquí tienes.";
		mes "¿Necesitas...";
		mes "algo más?";
		next;
		if (Hallow08 == 1) {
			goto MainMenu2;
		} else {
			goto MainMenu;
		}
}
