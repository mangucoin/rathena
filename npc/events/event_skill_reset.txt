//= Skill Reset with payment, minimum BaseLv.60 to reset skill.
//= Each 1 BaseLv need 20.000 Zeny, so be careful to use skill.
//============================================================
yuno,138,187,4	script	Hypnotist Teacher	124,{
	set @npcname$,"[^D5A500Hipnotista^000000]";
	mes @npcname$;

	if(MISC_QUEST & 1024){
		mes "Ya te dije que solo puedes completar este evento una vez.";
		close;
	}
	if(BaseLevel < 60) {
		mes "Por favor regresa cuando alcances BaseLv 60 o superior.";
		close;
	}
	if(SkillPoint != 0) {
		mes "Necesitaras usar todos tus puntos de habilidad si quieres que continue.";
		mes "Por favor vuelve pronto!";
		close;
	}
	if(Weight || checkfalcon() || checkcart() || checkriding()) {
		mes "Te gustaria reiniciar habilidades?";
		mes "Lo siento, pero...";
		next;
		mes @npcname$;
		if(Weight) {
			mes "No puedes reiniciar habilidades";
			mes "cuando llevas";
			mes "algun objeto.";
		} else if(checkcart())
			mes "Por favor, deja tu carrito y continuaremos.";
		else if(checkfalcon())
			mes "Por favor, libera tu Falcon y continuaremos.";
		else if(checkriding())
			mes "Por favor, libera tu PecoPeco y continuaremos.";
		next;
		mes @npcname$;
		mes "Vuelve pronto!";
		close;
	}
	mes "Hola, soy el Reiniciador de Habilidades.";
	mes "Tu nombre es ^0080FF"+strcharinfo(0)+"^000000.";
	mes "En que puedo ayudarte?";
	next;
	menu "^009500Informacion sobre reiniciar habilidades.^000000",L_Info,"^00B6FFReiniciar habilidades.^000000",L_Reset,"^000088No importa^000000",-;

	mes @npcname$;
	mes "Ya sabes donde encontrarme,";
	mes "si alguna vez quieres un reinicio!!";
	close;

L_Info:
	mes @npcname$;
	mes "Este reinicio de habilidades NO ES GRATIS!!";
	mes "El costo por el reinicio de habilidad es ^D5A50020000 Zeny x BaseLv^000000.";
	mes "Si ...cada BaseLv cuesta 20000 Zeny para reiniciar habilidad.";
	next;
	mes @npcname$;
	mes "Ah si, una cosa mas!";
	mes "Cualquier carrito, falcon o peco que tengas equipado";
	mes "sera removido si reinicias tus habilidades.";
	next;
	mes @npcname$;
	mes "Solo una vez se puede reiniciar los puntos de habilidad";
	mes "Ten cuidado con tus habilidades de aqui en adelante.";
	close;

L_Reset:
	mes @npcname$;
	mes "Antes de comenzar el reinicio de habilidades.";
	mes "Primero tendras que decirme tu Nivel Base.";
	next;
	mes "^D5A500[" + strcharinfo(0) + "]^000000";
	mes "Mi nivel base es ^AA00AANivel " + BaseLevel + "BaseLv.^000000";
	next;
	set @zeny,BaseLevel*20000;
	mes @npcname$;
	mes "El total de zeny para el reinicio de habilidad es ^529DFF" + @zeny + "Zeny^000000 por el servicio de reinicio de habilidad.";
	next;
	if(Zeny < @zeny) {
		mes @npcname$;
		mes "Parece que no tienes suficiente dinero.";
		mes "Ademas esperamos la oportunidad.";
		emotion ET_SCRATCH;
		close;
	}
	set Zeny, Zeny-@zeny;
	set MISC_QUEST,MISC_QUEST | 1024;
	resetskill;
	mes @npcname$;
	mes "Gracias.";
	emotion ET_THANKS;
//	logmes "SKILL RESET EVENT";
	close;
}
