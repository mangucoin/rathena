mjolnir_01,272,363,4	script	Rocks#08StPattysDay	-1,1,1,{
	end;

OnTouch:
	if (StPatrick2008 < 1) {
		mes "- Encontraste un monton de rocas cubiertas de tierra. -";
		close;
	}
	if (StPatrick2008 == 1) {
		mes "- Encontraste un monton de rocas cubiertas de tierra. -";
		next;
		if (countitem(12018) < 200) {
			mes "Las rocas no se mueven.";
			mes "Quizas O'Riley sepa como mover las rocas.";
			close;
		}
		else {
			if (select("Usar Petardos.:Ignorar") == 2) {
				close;
			}
			mes "Enterraste 200 Petardos debajo del monton de rocas.";
			mes "Enciendes la mecha.";
			next;
			mes "*BOOM!*";
			specialeffect2 EF_LORD;
			next;
			mes "Despues de que una nube de polvo y humo se disipo,";
			mes "encontraste una caja entre las rocas y la tierra.";
			mes "Esta caja debe contener los objetos valiosos de O'Riley.";
			mes "Llevemos la caja a O'Riley.";
			close2;
			set StPatrick2008,2;
			delitem 12018,200;
			getitem 7721,1;
		}
	}
	end;
}

	case 1:
		getitem 7915,1;
		end;
	case 2:
		getitem 7915,1;
		getitem 7916,1;
		getitem 7720,1;
		end;
	case 3:
		getitem 7720,1;
		end;
	case 4:
		getitem 7915,1;
		getitem 7916,1;
		end;
	case 5:
		getitem 7916,2;
		end;
	case 6:
		getitem 7915,2;
		end;
	}
}
