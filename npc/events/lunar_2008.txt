payon_in01,193,30,4	script	Rice Mill Grandma#rat	78,{
	mes "[Abuela del Molino de Arroz]";
	if (!lunar_rat) {
		mes "Esas asquerosas";
		mes "criaturas! Correteando";
		mes "por ahi, robandome";
		mes "las cosas! Si tan solo no";
		mes "fueran tan condenadamente rapidas...";
		next;
		select("Que sucede?");
		mes "[Abuela del Molino de Arroz]";
		mes "Ah, solo estaba haciendo";
		mes "unos pasteles de arroz y";
		mes "postres para celebrar";
		mes "el ano nuevo, pero estos";
		mes "animales han estado robando";
		mes "las Bolsas de Arroz que prepare.";
		next;
		select("Bolsas de Arroz?");
		mes "[Abuela del Molino de Arroz]";
		mes "Si, mi hijo necesita esas";
		mes "Bolsas de Arroz para machacar";
		mes "el arroz en un mortero, pero no puedo";
		mes "hacer ningun postre si ni siquiera";
		mes "tengo el arroz. Crees";
		mes "que puedes ayudarme?";
		next;
		if (select("Por supuesto.:Ni te conozco.") == 1) {
			set lunar_rat,1;
			mes "[" + strcharinfo(0)+"]";
			mes "Por supuesto.";
			mes "Que puedo hacer?";
			next;
			mes "[Abuela del Molino de Arroz]";
			mes "Bueno, crees que puedes";
			mes "atrapar a los animales que";
			mes "robaron mis Bolsas de Arroz?";
			mes "Son unos ratones azules y";
			mes "blancos y unos";
			mes "asquerosos topos.";
			next;
			select("No hay problema.");
			mes "[" + strcharinfo(0)+"]";
			mes "No hay problema.";
			mes "Atrapare a esos animales,";
			mes "y traere de vuelta todas las Bolsas";
			mes "de Arroz que pueda encontrar.";
			next;
			mes "[Abuela del Molino de Arroz]";
			mes "Oh, gracias!";
			mes "Buena suerte atrapando";
			mes "a esas plagas por mi~";
			close;
		}
		mes "[Abuela del Molino de Arroz]";
		mes "Yo... Bueno...";
		mes "Eso es cierto, pero estaba";
		mes "planeando darte algo";
		mes "bonito a cambio de";
		mes "tu ayuda. Se que no eres";
		mes "una mala persona, asi que...";
		close;
	}
	else if ((lunar_rat >= 1 && lunar_rat <= 2) || lunar_rat == 4) {
		if (countitem(7770) >= 1 && lunar_rat == 2) {
			mes "Oh, ese es el";
			mes "Arroz Dulce que hizo mi hijo?";
			mes "Me lo dejarias?";
			mes "Por favor espera un momento,";
			mes "y te hare un postre~";
			delitem 7770,1; //Sweet_rice
			set lunar_rat,3;
			close;
		}
		else if (lunar_rat == 4) {
			if (countitem(7770) >= 1) {
				mes "Oh, ese es el";
				mes "Arroz Dulce que hizo mi hijo?";
				mes "Me lo dejarias?";
				mes "Por favor espera un momento,";
				mes "y te hare un postre~";
				delitem 7770,1; //Sweet_rice
				set lunar_rat,5;
				close;
			}
			mes "Oh, donde estan todas las";
			mes "Bolsas de Arroz? Las necesitamos";
			mes "para hacer mas pasteles de arroz...";
			close;
		}
		else if (countitem(7869) < 1 && lunar_rat <= 2) {
			mes "Ooh, esos ratones blancos y";
			mes "azules podrian haber";
			mes "robado mis Bolsas de Arroz.";
			mes "Esos topos probablemente tambien";
			mes "se las llevaron. Que criaturas";
			mes "tan desagradables, no?";
			close;
		}
		mes "Oh, esa es una de mis";
		mes "Bolsas de Arroz? Disculpa,";
		mes "pero el dolor de mi cadera..";
		mes "Te importaria ser";
		mes "un amor y llevarle";
		mes "eso a mi hijo por mi?";
		set lunar_rat,2;
		close;
	}
	else if (lunar_rat == 3 || lunar_rat == 5) {
		set .@reward,rand(1,100);
		if (.@reward <= 5) {
			if (lunar_rat == 3) {
				getitem 9038,1; // New_Year_Doll_Egg
				set lunar_rat,4;
			}
			else if (lunar_rat == 5) {
				getitem 668,1; // Red_Envelope
			}
		}
		else if (.@reward >= 6 && .@reward <= 10) getitem 12198,2; // Lucky_Rice_Cake
		else if (.@reward >= 11 && .@reward <= 30) getitem 12195,3; // Plain_Rice_Cake
		else if (.@reward >= 31 && .@reward <= 40) getitem 12196,2; // Hearty_Rice_Cake
		else if (.@reward >= 41 && .@reward <= 55) getitem 12123,2; // Honey_Pastry
		else if (.@reward >= 56 && .@reward <= 70) getitem 12122,2; // Sesame_Pastry
		else if (.@reward >= 71 && .@reward <= 80) getitem 12124,2; // Rainbow_Cake
		else if (.@reward >= 81 && .@reward <= 90) getitem 12198,2; // Korean_Rice_Cake
		else if (.@reward >= 91 && .@reward <= 100) getitem 12197,3; // Salty_Rice_Cake
		// After getting the egg there is a chance to get an envelope with each turn in
		if (rand(1,20) <= 5 && lunar_rat == 4) getitem 668,1; // Red_Envelope
		// Set the quest accordingly for repeat turn-ins.
		if (lunar_rat == 3) set lunar_rat,2;
		else if (lunar_rat == 5) set lunar_rat,4;
		mes "No es ni de cerca suficiente";
		mes "para pagarte por lo que has";
		mes "hecho por mi, pero me gustaria";
		mes "que te lleves este postre que";
		mes "acabo de hacer. Disfrutalo~";
		next;
		mes "[Abuela del Molino de Arroz]";
		mes "Gracias por el";
		mes "Arroz Dulce! Me asegurare";
		mes "de hacerte algo";
		mes "delicioso si me";
		mes "traes mas, de acuerdo?";
		close;
	}
}
