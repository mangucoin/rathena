xmas_in,100,96,4	script	Father Christmas::Santa2	718,{
	mes "[Santa Claus]";
	if(xmas_npc==0) set xmas_npc, 1; // For Lutie & Bard quest
	if(#event_xmas > 0 && #event_xmas < 30 ) goto L_Start;
	mes "Feliz Navidad!";
	if(Class==0 || #event_xmas>=30 ) close; //anti exploit protection
	mes "Tengo un regalo para ti! Jo Jo Jo!";
	getitem rand(664,667),1; //gives one of 4 gift boxes
	set #event_xmas,#event_xmas+1;
	close;

L_Start:
	mes "Tengo un pequeno problema...";
	mes "Te importaria escuchar?";
	next;
	menu "Escuchar a Santa Claus.",M_0, "Darle a Santa Claus la prueba.",M_1, "Cancelar.",M_End;

	M_0:
		mes "[Santa Claus]";
		mes "Mi problema es este.";
		mes "Parece que hay un hombre alla afuera";
		mes "que me esta suplantando y esparciendo";
		mes "terror por toda la tierra.";
		next;
		mes "[Santa Claus]";
		mes "Como el Grinch de la leyenda, esta tomando";
		mes "todos los juguetes de los ninos y guardandoselos";
		mes "para el mismo.";
		next;
		mes "[Santa Claus]";
		mes "Estoy muy ocupado aqui creando mi lote de";
		mes "juguetes para el proximo ano, asi que no puedo";
		mes "salir a buscarlo yo mismo.";
		mes "Asi que me gustaria que fueras";
		mes "y destruyeras a este hombre por mi.";
		next;
		mes "[Santa Claus]";
		mes "El tiene en su posesion uno de mis";
		mes "sacos magicos, sin embargo, asi que";
		mes "escapara dentro de el a otro lugar cada";
		mes "vez que lo ataques.";
		next;
		mes "[Santa Claus]";
		mes "Sin embargo,";
		mes "En su prisa, tiende a dejar caer cosas.";
		mes "Si por casualidad deja caer uno de sus Calcetines";
		mes "Con Agujeros que usa para robar los";
		mes "juguetes de los pobres ninos, recogelo.";
		next;
		mes "[Santa Claus]";
		mes "Si recolectas 3 de estos, te dare";
		mes "un prototipo de caja misteriosa que";
		mes "he estado guardando cerca del";
		mes "laboratorio. Escupe regalos aleatorios";
		mes "y me ahorra mucho trabajo.";
		close;
	M_1:
		mes "[Santa Claus]";
		if(countitem(7034) < 3) goto L_NotEnuf;
		delitem 7034,3;
		mes "Parece que has estado haciendo un";
		mes "buen trabajo derribando a esos";
		mes "Santas falsos. Sigue asi!";
		next;
		getitem 644,1; //Gift Box Prototype
		set #event_xmas,#event_xmas+1;
		mes "[Santa Claus]";
		mes "Ahi esta tu recompensa.";
		mes "Si consigues 3 mas, te dare otra.";
		mes "Espero que te toque un buen objeto.";
		close;

		L_NotEnuf:
			mes "No tienes suficientes calcetines como prueba.";
			mes "Ve a derribar a esos Santas malvados";
			mes "y consigue mas para mi y te recompensare.";
			close;
	M_End:
		mes "[Santa Claus]";
		mes "Ya veo. Bueno, al menos";
		mes "nos veremos de nuevo en la manana de Navidad.";
		close;

OnInit:
	disablenpc "Santa Claus";
	end;
}
