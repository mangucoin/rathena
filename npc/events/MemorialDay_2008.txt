		mes "-namun kenangan tentang mereka akan bertahan selama tanah yang mereka hormati.-";
		mes "-Daniel Webster-";

			getexp 93750,43750;
		}
		next;
		mes "-Ada pesan lain.-";
		next;
		mes "-Aku mungkin tidak pergi ke tempat yang aku tuju,-";
		mes "-tapi kurasa aku berakhir di tempat yang memang aku butuhkan.-";
		mes "- Douglas Adams.-";
		if (Memorial08 == 9) {
			getexp 93750,43750;
		}
		next;
		mes "-Ini adalah pesan terakhir.-";
		next;
		mes "-Kepahlawanan sejati itu tenang, sangat tidak dramatis.-";
		mes "-Bukan dorongan untuk melampaui semua orang dengan biaya berapa pun,-";

		mes "- Arthur Ashe -";
