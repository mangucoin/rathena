		close;
	}

	else if (Memorial08 == 8) {
		mes "[" + strcharinfo(0) + "]";
		mes "- Esta es una placa vieja y polvorienta.-";
		mes "'Esta debe ser de la que hablaba Lauds.'";
		mes "'Limpiemosla con la toalla.'";
		next;
		set Memorial08,9;
		getexp 93750,43750;
		goto L_CleanPlaque;
	}

	else if (Memorial08 >= 9) {
		goto L_CleanPlaque;
	}

	L_CleanPlaque:
		mes "- Ves un mensaje en la placa que ya fue limpiada.-";
		next;
		mes "-Aunque no hay marmol tallado erigido en su memoria,-";
		mes "-ni piedra tallada que registre sus hazanas,-";
		mes "-sin embargo el recuerdo de ellos perdurara mientras exista la tierra que ellos honraron.-";
		mes "-Daniel Webster-";
		if (Memorial08 == 9) {
			getexp 93750,43750;
		}
		next;
		mes "-Hay otro mensaje.-";
		next;
		mes "-Puede que no haya ido a donde me dirigia,-";
		mes "-pero creo que termine en el lugar donde necesitaba estar.-";
		mes "- Douglas Adams.-";
		if (Memorial08 == 9) {
			getexp 93750,43750;
		}
		next;
		mes "-Este es el ultimo mensaje.-";
		next;
		mes "-El verdadero heroismo es sereno, muy poco dramatico.-";
		mes "-No es el impulso de superar a todos a cualquier costo,-";
		mes "- sino el impulso de servir a los demas a cualquier costo. -";
		mes "- Arthur Ashe -";
		if (Memorial08 == 9) {
			getexp 93750,43750;
			set Memorial08,10;
		}
		close;
}
