	mes "[Riss]";

	mes "A veces deberias ser generoso.";
	mes "No hay tiempo para pensar en cosas pequenas";
	mes "cuando tienes que concentrarte en cosas mas importantes.";
	next;
	mes "[Riss]";
	mes "Hum... de todos modos quiero decir... algo..";
	mes "Mientras viajaba por algunos lugares,";
	mes "encontre un lugar realmente aterrador.";
	next;
	mes "[Riss]";
	mes "La mayoria de las veces cuando llegas a un lugar,";
	mes "hay un monstruo fuerte y temible.";
	mes "No es asi?";
	next;
	mes "[Riss]";
	mes "Pero... pero... ahi dentro...";
	mes "Hay muchos monstruos temibles y fuertes ahi dentro...";
	mes "Eso fue realmente espantoso.";
	next;
	mes "[Riss]";
	mes "Si hubiera reaccionado un poco mas tarde... unos segundos...";
	mes "Podria haber muerto.";
	next;
	mes "[Riss]";
	mes "Que?";
	mes "Quieres ir ahi?";
	mes "Oh~ Chico~ no me entendiste.";
	mes "Ahi dentro.......";
	next;
	mes "[Riss]";
	mes "Uh... ya lo sabes?";
	mes "Aunque conoces el lugar, quieres entrar...";
	mes "Bien, te enviare ahi.";
	mes "Pero despues de que vayas, no me guardes";

	// If the VIP system is enabled, players pay a Reset Stone for 5 entrances (with a choice of warps).
	// If it's disabled, players pay 5,000z per entrance.
	if (VIP_SCRIPT) {
		mes "rencor. Te costara 1 Reset Stone por 5 accesos.";
		next;
		mes "[Riss]";
		if (bossnia_event > 0) {
			set .@type,1;
			mes "Accesos restantes: "+bossnia_event;
		} else if (countitem(6320)) {
			set .@type,2;
			mes "Quieres ir?";
		} else {
			mes "No tienes un ticket ahora.....";
			mes "Asi que vuelve a verme con una Reset Stone despues.";
			close;
		}
		next;
		if(select("Entrar:Salir") == 2)
			close;
		set .@i, select("Primero:Segundo:Tercero:Cuarto");
		switch(select("Warp 1:Warp 2:Warp 3:Warp 4")) {
			case 1: set .@x,31; set .@y,208; break;
			case 2: set .@x,31; set .@y,31; break;
			case 3: set .@x,208; set .@y,31; break;
			case 4: set .@x,208; set .@y,208; break;
		}
		if (.@type == 1)
			set bossnia_event, bossnia_event-1;
		else {
			delitem 6320,1; //Premium_Reset_Stone
			set bossnia_event,4;
		}
		specialeffect2 EF_MAXPOWER;
		warp "bossnia_0"+.@i,.@x,.@y;
		close;
	} else {
		mes "rencor. Ademas cuesta 5,000 zeny.";
		next;
		if (Zeny > 4999) {
			mes "[Riss]";
			mes "Realmente quieres aceptar el desafio?";
			mes "Bien, solo elige el curso.";
			next;
			set .@i, select("Primero:Segundo:Tercero:Cuarto");
			mes "[Riss]";
			mes "Cuidate, chico~";
			mes "No me guardes rencor.";
			close2;
			set Zeny, Zeny - 5000;
			warp "bossnia_0"+.@i,rand(202,204),rand(202,204);
			end;
		} else {
			mes "[Riss]";
			mes "No tienes suficiente dinero...";
			mes "Vuelve cuando tengas al menos 5,000 zeny.";
			close;
		}
	}
}
