mosk_in,21,246,5	script	Pinkamenia	71,{
	set .@n$,"^0000FF[Pinkamenia]^000000";
	if (#V_QUE12==3) {
		set .@LFlowerID,7864;
		setarray .@SweetsID[0],538,558,539,573,559,560,12062,596,597,12414,12319;
		setarray .@SweetsCost[0],1,2,5,10,10,10,15,15,15,20,20;
		mes .@n$;
		mes "Hola "+strcharinfo(0)+", quieres";
		mes "comprar algunos objetos?";
		mes "Tendras que darme Love";
		mes "Flowers por ellos, por supuesto!";
		next;
		if(select("Si, claro!","No, no importa.") == 2) close;
		next;
		mes .@n$;
		mes "Selecciona un objeto.";
		mes "La cantidad de Love Flowers que necesitaras esta entre parentesis.";
		for(set .@i,0; .@i < getarraysize(.@SweetsID); set .@i,.@i+1)
			set .@menu$, .@menu$+"^00AA00["+.@SweetsCost[.@i]+"]^000000 "+getitemname(.@SweetsID[.@i])+":";
		next;
		set .@choice, select(.@menu$)-1;
		mes .@n$;
		if (countitem(.@LFlowerID) < .@SweetsCost[.@choice]) {
			mes "Necesitaras mas Love Flowers si quieres eso!";
			close;
		}
		mes "Estas seguro de que quieres cambiar ^00aa00"+.@SweetsCost[.@choice]+"x Love Flower ^000000 por ^0055FF"+getitemname(.@SweetsID[.@choice])+"^000000?";
		if(select("No, cambie de opinion.:Si, intercambiar!") == 1) close;
		set .@new, .@LFlowerID - .@SweetsCost[.@choice];
		if (.@new < 0) set .@new, 0;
		delitem .@LFlowerID, .@SweetsCost[.@choice];
		getitem .@SweetsID[.@choice],1;
		mes "Diviertete con tu objeto!";
		close;
	} else if (#V_QUE12==2) {
		mes .@n$;
		if (countitem(7864) < 15) {
			mes "Tienes que traerme 15 Love Flowers!";
			close;
		}
		mes "Muchas gracias por";
		mes "recuperar nuestras Flores!";
		delitem 7864,15;
		next;
		mes .@n$;
		mes "Toma esto como un pequeno";
		mes "'agradecimiento'.";
		getexp 500000,400000;
		getitem 617,1;
		getitem 12319,2;
		next;
		mes .@n$;
		mes "Si consigues mas Love";
		mes "Flowers, puedes cambiarlas";
		mes "por algunos dulces aqui.";
		mes "Nos vemos pronto!";
		set #V_QUE12,3;
		close;
	} else if (#V_QUE12==1) {
		mes .@n$;
		mes "Por favor lleva un +8 Cake Hat";
		mes "al Baker Extraordinaire";
		mes "que esta parado junto a mi!";
		close;
	} else if (BaseLevel >= 45) {
		mes .@n$;
		mes "Hola "+strcharinfo(0)+", es";
		mes "Dia de San Valentin y nos";
		mes "encanta entregar dulces";
		mes "chocolates.";
		next;
		mes .@n$;
		mes "Nuestro problema es que";
		mes "necesitamos una persona fuerte";
		mes "como tu que pueda ayudarnos,";
		mes "pero primero tendras";
		mes "que llevar un +8 Cake Hat";
		mes "al Baker Extraordinaire, quien";
		mes "esta parado justo a mi lado!";
		set #V_QUE12,1;
		close;
	}
	mes .@n$;
	mes "Hola "+strcharinfo(0)+"!";
	close;

OnInit:
	// Unknown item drop rate (currently set at 5%)
	setitemscript 5105,"{ bonus bDex,1; bonus bMaxSP,80; bonus3 bAddMonsterDropItem,7864,RC_DemiHuman,500; }",0;
	end;
}

mosk_in,21,244,3	script	Baker Extraordinaire	410,{
	set .@n$,"^0000FF[Baker Extraordinaire]^000000";
	if(#V_QUE12==3) {
		mes .@n$;
		mes "Cambia tus Love Flowers con Pinkamenia!";
		close;
	} else if(#V_QUE12==2) {
		mes .@n$;
		mes "Lleva 15 Love Flowers a Pinkamenia!";
		close;
	} else if(#V_QUE12==1) {
		mes .@n$;
		if (getequipid(EQI_HEAD_TOP) == 5024 && getequiprefinerycnt(EQI_HEAD_TOP) >= 8) {
			mes "Ah, entonces Pinkamenia te dijo";
			mes "que me trajeras el +8 Cake";
			mes "Hat. Ahora voy";
			mes "a cambiar tu +8 Cake";
			mes "Hat por otro Cake Hat.";
			next;
			mes .@n$;
			delitem 5024,1;
			getitem 5105,1;
			equip 5105;
			mes "Ahora, si usas el";
			mes "nuevo Cake Hat, hay";
			mes "una probabilidad de que";
			mes "los Monstruos Demi-Human";
			mes "suelten una Love Flower!";
			next;
			mes .@n$;
			mes "Los mobs robaron nuestras Flores,";
			mes "las cuales necesitamos para crear";
			mes "nuestros chocolates. Ahora consigue";
			mes "15 Love Flowers y llevalas";
			mes "a Pinkamenia!";
			set #V_QUE12,2;
			close;
		} else {
			mes "Lo siento, pero donde esta tu";
			mes "+8 Cake Hat? Traemelo,";
			mes "y recuerda";
			mes "tenerlo equipado!";
			close;
		}
	} else {
		mes .@n$;
		mes "Hola, "+strcharinfo(0)+"!";
		close;
	}
}
