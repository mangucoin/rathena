prontera,147,61,3	script	Driller#pron	87,{
	disable_items;
	if (checkweight(1201,1) == 0) {
		mes "Kamu membawa terlalu banyak item di inventaris. Kunjungi penyimpanan Kafra dan coba lagi.";
		close;
	}
	if (MaxWeight - Weight < 10000) {
		mes "Kamu tidak bisa melanjutkan karena kelebihan berat.";
		close;
	}
	set .@part, EQI_ACC_L;
	if (!getequipisequiped(.@part)) {
		mes "[Driller]";
		mes "Tugasku adalah melubangi slot kartu pada aksesoris RWC Memorial.";
		next;
		mes "[Driller]";
		mes "Maaf, kamu tidak mempunyai item yang dipakai di posisi aksesoris kanan.";
		close;
	}
	mes "[Driller]";
	mes "Tugasku adalah melubangi slot kartu pada aksesoris RWC Memorial. Ademas, solo trabajo con ^ff0000objetos puros^000000, o aquellos que no han sido encantados.";
	next;
	set .@equip_id, getequipid(.@part);
	if (.@equip_id != 2966 && .@equip_id != 2968) {
		mes "[Driller]";
		mes "Namun, aku bisa melihat bahwa aksesoris yang kamu pakai di sisi kanan tidak bisa ditangani. Silakan pasang aksesoris RWC Memorial.";
		close;
	}
	mes "[Driller]";
	mes "Kamu juga harus tahu bahwa membuat slot kartu itu sangat berbahaya. ^ff0000Peluang berhasil sekitar 50%.^000000 Apakah kamu ingin melanjutkan?";
	next;
	if(select("Batal:Ayo!") == 1) {
		mes "[Driller]";
		mes "Sampai jumpa kalau begitu.";
		close;
	}
	if (.@equip_id == 2966) {
		set .@slotted, 2967; //RWC_2012_Ring_
		set .@name$,"RWC 2012 Memorial Ring";
		set .@str$,"ring";
	} else if (.@equip_id == 2968) {
		set .@slotted, 2969; //RWC_2012_Pendant_
		set .@name$,"RWC 2012 Memorial Pendant";
		set .@str$,"pendant";
	} else {
		mes "[Driller]";
		mes "Aku tidak bisa mengidentifikasi item aksesoris yang kamu pakai di tangan kanan. Aku tidak bisa mengerjakannya.";
		close;
	}
	if (getequipcardid(.@part,3) > 0) {
		mes "[Driller]";
		mes "Item ini sudah di-enchant. Aku tidak bisa mengerjakan ini karena melanggar aturan.";
		close;
	}

	// anti-hack
	if (callfunc("F_IsEquipIDHack", .@part, .@equip_id))
		close;

	delequip .@part;
	if (rand(1,10) > 5) {
		getitem .@slotted,1;
		specialeffect2 EF_REPAIRWEAPON;
		mes "[Driller]";
		mes "Hore! Berhasil! "+.@name$+" sekarang punya slot kartu. Coba lihat!";
		close;
	} else {
		specialeffect2 EF_LORD;
		mes "[Driller]";
		mes "Aduh... Lemah sekali "+.@str$+"... Pecah selama prosedur. Aku minta maaf.";
		close;
	}
}

prontera,147,59,3	script	Goldberg#pron	878,{
	disable_items;
	if (checkweight(1201,1) == 0) {
		mes "Kamu membawa terlalu banyak item. Kembali setelah kamu merapikan inventarismu.";
		close;
	}
	if (MaxWeight - Weight < 10000) {
		mes "Kamu tidak bisa melanjutkan karena kelebihan berat.";
		close;
	}
	mes "[Goldberg]";
	mes "Halo! Aku bertugas meng-enchant aksesoris RWC Memorial dengan kekuatan mistis.";
	next;
	set .@part, EQI_ACC_L;
	if (!getequipisequiped(.@part)) {
		mes "[Goldberg]";
		mes "Maaf, kamu tidak mempunyai item yang dipakai di posisi aksesoris kanan.";
		close;
	}
	set .@equip_id, getequipid(.@part);
	if (.@equip_id < 2966 || .@equip_id > 2969) {
		mes "[Goldberg]";
		mes "Namun, aku bisa melihat bahwa aksesoris yang kamu pakai bukan sesuatu yang bisa aku kerjakan. Silakan pasang aksesoris RWC Memorial.";
		close;
	}
	set .@select, select("Maaf, tidak tertarik.:Tolong, perkuat aksesoriku.:Hapus Enchant.")-1;
	if (.@select == 0) {
		mes "[Goldberg]";
		mes "Baiklah, sampai jumpa lain waktu...";
		close;
	}
	setarray .@equip_card[0], getequipcardid(.@part,0),getequipcardid(.@part,1),getequipcardid(.@part,2),getequipcardid(.@part,3);
	if (.@select == 1) {
		switch(getequipid(.@part)) {
		case 2966:
			setarray .@option[0],2,2,4,4;
			break;
		case 2967:
			setarray .@option[0],0,2,4,4;
			break;
		case 2968:
			setarray .@option[0],1,1,3,3;
			break;
		case 2969:
			setarray .@option[0],0,1,3,3;
			break;
		default:
			mes "[Goldberg]";
			mes "Maaf, tapi aku tidak bisa mengerjakan aksesoris yang sedang kamu pakai.";
			mes "Jika kamu memasang aksesoris RWC Memorial di sisi kiri, coba pindahkan ke sisi kanan.";
			close;
		}
		for(set .@i,3; .@i>=0; set .@i,.@i-1) {
			if (.@equip_card[.@i] == 0) {
				set .@slot, .@i;
				set .@op_type, .@option[.@i];
				break;
			}
		}
		switch(.@op_type) {
		case 4:
			mes "[Goldberg]";
			mes "Enchantment mana yang ingin kamu infus?";
			next;
			setarray .@enchant_select[0],1,2,3,4;
			set .@i, select("Batal:Fighting Spirit:ATK (%):Max HP:HP")-2;
			break;
		case 3:
			setarray .@enchant_select[0],5,6,7;
			set .@i, select("Batal:Spell:MATK (%):SP")-2;
			break;
		case 2:
			mes "[Goldberg]";
			mes "^ff0000Ten cuidado! Hay un 25% de probabilidad de que el encantamiento falle. Si esto sucede, el objeto sera destruido.^000000 Enchantment mana yang ingin kamu infus?";
			next;
			setarray .@enchant_select[0],8,9,10,11,12,13,14;
			set .@i, select("Batal:STR:AGI:VIT:INT:DEX:LUK:SP")-2;
			break;
		case 1:
			mes "[Goldberg]";
			mes "^ff0000Hay un 25% de probabilidad de que el encantamiento falle. Si esto sucede, el objeto sera destruido.^000000 Enchantment mana yang ingin kamu infus?";
			next;
			setarray .@enchant_select[0],8,9,10,11,12,13,15,16;
			set .@i, select("Batal:STR:AGI:VIT:INT:DEX:LUK:MHP:HP")-2;
			break;
		case 0:
			mes "[Goldberg]";
			mes "Aksesorismu sudah menerima terlalu banyak enchantment sehingga aku hampir tidak bisa mengerjakannya lagi.";
			close;
		}
		if (.@i == -1) {
			mes "[Goldberg]";
			mes "Baiklah, sampai jumpa lain waktu.";
			close;
		}
		mes "[Goldberg]";
		mes "Kekuatan enchantment akan dipilih secara acak. ^ff0000Setelah diinfus, enchantment tidak bisa dihapus.^000000 Apakah kita lanjutkan?";
		next;
		if(select("Tidak, tolong berhenti.:Ya, silakan lanjutkan.") == 1) {
			mes "[Goldberg]";
			mes "Baiklah, sampai jumpa lain waktu...";
			close;
		}
		set .@enchant_type, .@enchant_select[.@i];
		if (!getequipisequiped(.@part)) {
			mes "[Goldberg]";
			mes "Jangan lepas perlengkapanmu saat aku sedang bekerja, oke?";
			close;
		}
		switch(.@enchant_type) {
		case 1:
			setarray .@enc[0],4811,4810,4809; //Fighting_Spirit1,Fighting_Spirit2,Fighting_Spirit3
			break;
		case 2:
			setarray .@enc[0],4819,4766,4767; //Atk1,Atk2,Atk3
			break;
		case 3:
			setarray .@enc[0],4861,4862,4867; //MHP1,MHP2,MHP3
			break;
		case 4:
			setarray .@enc[0],4795,4796,4797; //HP100,HP200,HP300
			break;
		case 5:
			setarray .@enc[0],4760,4761,4806; //Matk1,Matk2,Matk3
			break;
		case 6:
			setarray .@enc[0],4815,4814,4813; //Spell1,Spell2,Spell3
			break;
		case 7:
			setarray .@enc[0],4870,4800,4871; //SP25,SP50,SP75
			break;
		case 8:
			setarray .@enc[0],4700,4701,4702; //Strength1,Strength2,Strength3
			break;
		case 9:
			setarray .@enc[0],4730,4731,4732; //Agility1,Agility2,Agility3
			break;
		case 10:
			setarray .@enc[0],4740,4741,4742; //Vitality1,Vitality2,Vitality3
			break;
		case 11:
			setarray .@enc[0],4710,4711,4712; //Inteligence1,Inteligence2,Inteligence3
			break;
		case 12:
			setarray .@enc[0],4720,4721,4722; //Dexterity1,Dexterity2,Dexterity3
			break;
		case 13:
			setarray .@enc[0],4750,4751,4752; //Luck1,Luck2,Luck3
			break;
		case 14:
			setarray .@enc[0],4870,4800,4871; //SP25,SP50,SP75
			break;
		case 15:
			setarray .@enc[0],4861,4862,4867; //MHP1,MHP2,MHP3
			break;
		case 16:
			setarray .@enc[0],4795,4796,4797; //HP100,HP200,HP300
			break;
		default:
			mes "[Goldberg]";
			mes "Ada masalah, biarkan aku memeriksanya.";
			close;
		}

		if (.@enchant_type < 8)
			set .@i, rand(1,300); // 0% break chance.
		else
			set .@i, rand(1,400); // 25% break chance.

		     if (.@i < 151) set .@enchant, .@enc[0];
		else if (.@i < 251) set .@enchant, .@enc[1];
		else if (.@i < 301) set .@enchant, .@enc[2];
		else set .@enchant,9;

		// anti-hack
		if (callfunc("F_IsEquipIDHack", .@part, .@equip_id) ||
		    callfunc("F_IsEquipCardHack", .@part, .@equip_card[0], .@equip_card[1], .@equip_card[2], .@equip_card[3]))
			close;

		set .@equip_card[.@slot], .@enchant;
		if (.@slot == 2 && .@enchant == 0) {
			set .@equip_card[3],0;
		} else if (.@slot == 1 && .@enchant == 0) {
			set .@equip_card[2],0;
			set .@equip_card[3],0;
		} else if (.@slot == 0 && .@enchant == 0) {
			set .@equip_card[1],0;
			set .@equip_card[2],0;
			set .@equip_card[3],0;
		}

		set .@equip_refine, getequiprefinerycnt(.@part);
		delequip .@part;
		if (.@enchant == 9) {
			mes "[Goldberg]";
			mes "Ya ampun!";
 			mes "Item tidak cukup kuat untuk menahan enchantment dan akhirnya hancur. Maafkan aku.";
			specialeffect2 EF_LORD;
			close;
		}
		if (.@enchant == 0) { // Should never happen.
			mes "[Goldberg]";
			mes "Oh... Sepertinya ada ketidakstabilan antara semua kekuatan yang diinfus. Ini menyebabkan semua enchantment menghilang. Sayang sekali, tapi silakan coba lagi!";
		} else {
			mes "[Goldberg]";
			mes "Bagus!";
			mes "Enchantment berhasil! Akan diterapkan di socket No.^990000"+(.@slot+1)+"^000000.";
			specialeffect2 EF_REPAIRWEAPON;
		}

//		GetNonSlotItemSock2 .@equip_refine .@equip_id .@equip_card[0] .@equip_card[1] .@equip_card[2] .@equip_card[3]
		getitem2 .@equip_id,1,1,.@equip_refine,0,.@equip_card[0],.@equip_card[1],.@equip_card[2],.@equip_card[3];

		close;
	} else if (.@select == 2) {
		mes "[Goldberg]";
		mes "Aku akan menginisialisasi opsi enchant tanpa melakukan apapun pada kartu yang terpasang. Mau lanjut?";
		next;
		if(select("Aku berhenti.:Ya, tentu, lanjutkan.") == 1) {
			mes "[Goldberg]";
			mes "Kembali jika kamu berubah pikiran.";
			close;
		}
		if (countitem(6665) == 0) {
			mes "[Goldberg]";
			mes "Maaf. Tapi kamu tidak punya kupon Inisialisasi RWC. Bisa cek inventarismu?";
			close;
		}
		if (getiteminfo(.@equip_card[3], ITEMINFO_SUBTYPE) != CARD_ENCHANT) {
			mes "[Goldberg]";
			mes "Hm... perlengkapan ini bersih. Aku tidak bisa menginisialisasi jika tidak ada apa-apa! Cek lagi.";
			close;
		}
		specialeffect2 EF_REPAIRWEAPON;
		mes "[Goldberg]";
		mes "Opsi enchant pada itemmu akan diinisialisasi.";
		delitem 6665,1; //RWC_Inicializer
		delequip .@part;
	
//		GetNonSlotItemSock2 .@equip_refine .@equip_id .@equip_card[0] .@equip_card[1] .@equip_card[2] .@equip_card[3]
		for ( .@i = getiteminfo(.@equip_id, ITEMINFO_SLOT); .@i < MAX_SLOTS; .@i++ ) {
			if (getiteminfo(.@equip_card[.@i], ITEMINFO_SUBTYPE) == CARD_ENCHANT)
				.@equip_card[.@i] = 0;// Armor Enchant System
		}
		getitem2 .@equip_id,1,1,.@equip_refine,0,.@equip_card[0],.@equip_card[1],.@equip_card[2],.@equip_card[3];
		
		close;
	}
}
