// Stephen - Valentine Event Chocolate seller
//============================================================
alberta,26,243,4	script	Stephen	58,{
	mes "[Stephen]";
	mes "Adivina que tengo?";
	mes "Una golosina que no se encuentra facilmente en Midgard....";
	next;
	mes "[Stephen]";
	mes "Chocolate!";
	mes "Asi es, a quien no le gusta el chocolate.... a mi si.";
	mes "Y tienes suerte, porque los vendo por solo 5,000 zeny cada uno!";
	next;
	if(select("Quiero chocolate!:No gracias.")==2) {
		mes "[Stephen]";
		mes "No quieres chocolate?";
		mes "Te lo digo! Te vas a arrepentir!";
		mes "Mejor compra ahora... no encontraras Chocolate como este nunca mas!";
		mes "Piensalo y visitame de nuevo.";
		close; }
	mes "[Stephen]";
	mes "Ja!";
	mes "Lo sabia!";
	mes "Pero no puedo venderte mas de 5 a la vez... pero, si realmente necesitas mas....";
	mes "puedes volver otra vez.";
	mes "Cuantos quieres?";
	next;
	input .@i;
	if (.@i <= 0) close;
	mes "[Stephen]";
	if (.@i > 5)  {
		mes "Lo siento, pero no puedo darte tantos.";
		close; }
	if (Zeny < .@i*5000) {
		mes "Lo siento, pero parece que no puedes pagar por estos.";
		close; }
	set Zeny, Zeny - .@i*5000;
	getitem 558,.@i;
	mes "Aqui tienes!";
	mes "Puedes darselo a alguien como regalo, o disfrutarlo tu mismo!";
	mes "Mmm... dulce chocolate...";
	mes "Visitame cuando quieras...!";
	close;
}
