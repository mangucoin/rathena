function	script	F_GldManager	{
	.@npc_name$ = getarg(0);
	.@map_name$ = getarg(1);
	.@gid = getcastledata(.@map_name$,1);
	.@gm_name$ = getguildmaster(.@gid);

	mes "[ "+ .@npc_name$ +" ]";
	if (.@gid == 0){
		mes "Estoy esperando a mi amo. Valiente aventurero, sigue tu destino!";
		return;
	}
	if (getcharid(2) != .@gid){
		mes "Estoy aqui para seguir las ordenes de ^5533FF" + getguildmaster(.@gid) + "^000000! Oye! Ni siquiera eres parte del gremio!!";
		mes "Donde estan los guardianes? Destruyan a estos intrusos!";
		return;
	}
	if (is_guild_leader() == false){
		mes "Tu no eres ^5533FF" + getguildmaster(.@gid) + "^000000! Estoy aqui para seguir solo las ordenes de ^5533FF" + getguildmaster(.@gid) + "^000000";
		return 0;
	}

	mes "Bienvenido Maestro ^5533FF" + getguildmaster(.@gid) + "^000000 ! Te asistire en todo lo que pueda!";
	next;

	menu "Empleo / Despido de Kafra",M_Kaf, "Entrar a la Sala del Tesoro",M_Treas, "Cancelar",M_End;

	//===========================
	M_Kaf:
	//======
		mes "[ "+ .@npc_name$ +" ]";
		if (GetCastleData(.@map_name$,9) == 1) goto L_Dismiss;
		if (getgdskilllv(.@gid,10001) == 0){
			mes "Maestro, no tienes un contrato con la Compania de Kafra.";
			mes "Para contratar una Kafra, primero debes aprender la habilidad de Gremio ^5533FFContrato con Kafra^000000.";
			return;
		}

		L_Hire:
			mes "Te gustaria emplear los servicios de una Kafra? Necesitaras ^5533FF10,000 Zeny^000000 para hacerlo... ";
			next;
			menu "Emplear Kafra.",-,"Cancelar",sM_KafEnd;

				mes "[ "+ .@npc_name$ +" ]";
				if (Zeny < 10000){
					mes "Maestro, no tienes suficiente dinero para emplear una Kafra. El empleo ha sido cancelado.";
					return;
				}
				set Zeny,Zeny-10000;
				enablenpc "Kafra Staff#"+getarg(4);
				SetCastleData .@map_name$,9,1;
				mes "Has creado un contrato con la Compania de Kafra.";
				next;
				cutin "kafra_01",2;
				mes "[ Kafra Staff ]";
				mes "Como estas? Estoy aqui para brindarte un servicio util! Hare lo mejor que pueda para servirte.";
				next;
				cutin "kafra_01",255;
				mes "[ "+ .@npc_name$ +" ]";
				//mes "Tu contrato de empleo dura ^5533FF1 mes^000000. Despues de que termine este plazo tendras que crear un nuevo contrato.";
				mes "Creo que la Kafra beneficiara a los miembros de nuestro gremio.";
				return;

			sM_KafEnd:
				mes "[ "+ .@npc_name$ +" ]";
				mes "Como desees Maestro. Pero te sugiero que consigamos una Kafra lo antes posible!";
				return;

		L_Dismiss:
			mes "Te gustaria despedir a la Kafra actual?";
			next;
			menu "Despido",-,"Cancelar",sM_KafEnd2;

				cutin "kafra_01",2;
				mes "[ Kafra Staff ]";
				mes "He hecho algo mal? Si lo hice, me perdonarias?";
				next;
				menu "Despedir",-,"Cancelar",ssM_KafEnd2;

					mes "[ Kafra Staff ]";
					mes "Es una lastima que ya no podre servir a tu gremio....";
					next;
					disablenpc "Kafra Staff#"+getarg(4);
					SetCastleData .@map_name$,9,0;
					cutin "kafra_01",255;
					mes "[ "+ .@npc_name$ +" ]";
					mes "La Kafra ha sido despedida. Pero... realmente deberiamos conseguir una Kafra lo antes posible!";
					return;
				ssM_KafEnd2:
					mes "[ Kafra Staff ]";
					mes "Gracias maestro, dare lo mejor de mi! ^^.";
					cutin "kafra_01",255;
					return;
			sM_KafEnd2:
				mes "[ "+ .@npc_name$ +" ]";
				mes "Maestro, creo que deberias mantener a la Kafra actual porque ya esta dando lo mejor de si para servirnos";
				return;

	//=========================
	M_Treas:
	//========
		mes "[ "+ .@npc_name$ +" ]";
		mes "Deseas ir a nuestra Sala del Tesoro? Solo tu, el Maestro del Gremio, tienes permitido entrar a esta sala.";
		next;
		menu "Entrar a la Sala del Tesoro.",-,"Cancelar",sM_TresEnd;

			mes "[ "+ .@npc_name$ +" ]";
			mes "Por favor sigueme por el pasaje secreto.";
			mes "Debes jalar el interruptor secreto para poder salir.";
			next;
			warp .@map_name$,getarg(2),getarg(3);
			return;
		sM_TresEnd:
			mes "[ "+ .@npc_name$ +" ]";
			mes "Los bienes se producen todos los dias.";
			mes "Deberias recogerlos cuando puedas porque podrian desaparecer si los tomas en el momento equivocado.";
			return;


	//==========================
	M_End:
	//=======
		mes "[ "+ .@npc_name$ +" ]";
		mes "Como desees, maestro.";
		return;
}
