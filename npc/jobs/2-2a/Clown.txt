valkyrie,53,54,3	script	Minstrel#Valkyrie	741,{
	if (ADVJOB == 0 || Upper != 1) {
		set .@karma_d,rand(1,10);
		if (.@karma_d > 4) {
			mes "[Minstrel]";
			mes "Felicidades.";
			mes "Honor a los guerreros!";
			close;
		}
		mes "[Minstrel]";
		mes "Quieres cantar";
		mes "una cancion conmigo?";
		mes "Sha la la la la~";
		close;
	}
	else if (ADVJOB == Job_Clown && Class == Job_Archer_High && JobLevel > 39) {
		mes "[Minstrel]";
		mes "El lugubre mundo";
		mes "de los mortales necesita";
		mes "canciones mas alegres.";
		mes "Se las llevaras";
		mes "y cambiaras el rumbo en la";
		mes "batalla contra el mal?";
		next;
		mes "[Minstrel]";
		mes "Haras esto";
		mes "por Midgard...";
		mes "como Minstrel?";
		next;
		if (select("No.:Si.") == 1) {
			mes "[Minstrel]";
			mes "Cuando estes listo,";
			mes "no dudes en regresar.";
			mes "Honor a los guerreros!";
			close;
		}
		if (SkillPoint) {
			mes "[Minstrel]";
			mes "Aun es posible que aprendas mas habilidades. Por favor usa";
			mes "todos tus puntos de habilidad restantes antes de regresar conmigo.";
			close;
		}
		jobchange Job_Clown;
		set ADVJOB,0;
		mes "[Minstrel]";
		mes "Felicidades!";
		mes "Como Minstrel, tus";
		mes "canciones traeran";
		mes "esperanza a tus aliados, y";
		mes "desesperacion a tus enemigos.";
		close;
	}
	mes "[Minstrel]";
	mes "Bienvenido";
	mes "a Valhalla,";
	mes "el Salon del Honor.";
	next;
	mes "[Minstrel]";
	mes "Por favor ponte";
	mes "comodo";
	mes "mientras estes aqui.";
	mes "Honor a los guerreros!";
	close;
}
