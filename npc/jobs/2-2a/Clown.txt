valkyrie,53,54,3	script	Minstrel#Valkyrie	741,{
	if (ADVJOB == 0 || Upper != 1) {
		set .@karma_d,rand(1,10);
		if (.@karma_d > 4) {
			mes "[Minstrel]";
			mes "Felicidades.";
			mes "¡Honor a los guerreros!";
			close;
		}
		mes "[Minstrel]";
		mes "¿Quieres cantar";
		mes "una canción conmigo?";
		mes "Sha la la la la~";
		close;
	}
	else if (ADVJOB == Job_Clown && Class == Job_Archer_High && JobLevel > 39) {
		mes "[Minstrel]";
		mes "El lúgubre mundo";
		mes "de los mortales necesita";
		mes "canciones más alegres.";
		mes "¿Se las llevarás";
		mes "y cambiarás el rumbo en la";
		mes "batalla contra el mal?";
		next;
		mes "[Minstrel]";
		mes "¿Harás esto";
		mes "por Midgard...";
		mes "como Minstrel?";
		next;
		if (select("No.:Sí.") == 1) {
			mes "[Minstrel]";
			mes "Cuando estés listo,";
			mes "no dudes en regresar.";
			mes "¡Honor a los guerreros!";
			close;
		}
		if (SkillPoint) {
			mes "[Minstrel]";
			mes "Aún es posible que aprendas más habilidades. Por favor usa";
			mes "todos tus puntos de habilidad restantes antes de regresar conmigo.";
			close;
		}
		jobchange Job_Clown;
		set ADVJOB,0;
		mes "[Minstrel]";
		mes "¡Felicidades!";
		mes "Como Minstrel, tus";
		mes "canciones traerán";
		mes "esperanza a tus aliados, y";
		mes "desesperación a tus enemigos.";
		close;
	}
	mes "[Minstrel]";
	mes "Bienvenido";
	mes "a Valhalla,";
	mes "el Salón del Honor.";
	next;
	mes "[Minstrel]";
	mes "Por favor ponte";
	mes "cómodo";
	mes "mientras estés aquí.";
	mes "¡Honor a los guerreros!";
	close;
}
