valkyrie,53,42,3	script	Champion#Valkyrie	52,{
	if (ADVJOB == 0 || Upper != 1) {
		set .@karma_d,rand(1,10);
		if (.@karma_d > 4) {
			mes "[Champion]";
			mes "Felicidades.";
			mes "¡Honor a los guerreros!";
			close;
		}
		mes "[Champion]";
		mes "Habilidad.";
		mes "Velocidad.";
		mes "Fuerza.";
		mes "Agilidad.";
		next;
		mes "[Champion]";
		mes "Un Champion puede";
		mes "beneficiarse de todas";
		mes "estas cosas. Pero";
		mes "uno solo puede dominar";
		mes "una cantidad limitada en la vida...";
		close;
	}
	else if (ADVJOB == Job_Champion && Class == Job_Acolyte_High && JobLevel > 39) {
		mes "[Champion]";
		mes "Es el momento.";
		mes "¡Momento de que grandes héroes";
		mes "se levanten contra las";
		mes "fuerzas del mal que azotan";
		mes "el mundo de Midgard!";
		next;
		mes "[Champion]";
		mes "¿Te gustaría";
		mes "convertirte en Champion?";
		next;
		if (select("No.:Sí.") == 1) {
			mes "[Champion]";
			mes "Cuando estés listo,";
			mes "no dudes en regresar.";
			mes "¡Honor a los guerreros!";
			close;
		}
		if (SkillPoint) {
			mes "[Champion]";
			mes "Aún es posible que aprendas más habilidades. Por favor usa";
			mes "todos tus puntos de habilidad restantes antes de regresar conmigo.";
			close;
		}
		jobchange Job_Champion;
		set ADVJOB,0;
		mes "[Champion]";
		mes "¡Felicidades!";
		mes "Vive como Champion,";
		mes "y trae luz al";
		mes "mundo a través de la";
		mes "fuerza de tus puños.";
		close;
	}
	mes "[Champion]";
	mes "Bienvenido";
	mes "a Valhalla,";
	mes "el Salón del Honor.";
	next;
	mes "[Champion]";
	mes "Por favor ponte";
	mes "cómodo";
	mes "mientras estés aquí.";
	mes "¡Honor a los guerreros!";
	close;
}
