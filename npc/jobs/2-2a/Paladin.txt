valkyrie,53,39,3	script	Paladin#Valkyrie	752,{
	if (ADVJOB == 0 || Upper != 1) {
		set .@karma_d,rand(1,10);
		if (.@karma_d > 4) {
			mes "[Paladin]";
			mes "Felicidades.";
			mes "Honor a los guerreros!";
			close;
		}
		mes "[Paladin]";
		mes "No olvides";
		mes "que la Guerra Santa";
		mes "se acerca rapidamente!";
		mes "Debemos prepararnos!";
		mes "Que la luz de la justicia";
		mes "siempre ilumine nuestro camino!";
		close;
	}
	else if (ADVJOB == Job_Paladin && Class == Job_Swordman_High && JobLevel > 39) {
		mes "[Paladin]";
		mes "La Guerra Santa estara";
		mes "sobre nosotros antes de que";
		mes "lo sepamos. Mas que nunca,";
		mes "necesitamos hombres y mujeres";
		mes "fuertes que luchen por lo";
		mes "que es bueno y correcto.";
		next;
		mes "[Paladin]";
		mes "Lucharas del";
		mes "lado de la justicia";
		mes "como Paladin?";
		next;
		if (select("No.:Si.") == 1) {
			mes "[Paladin]";
			mes "Cuando estes listo,";
			mes "no dudes en regresar.";
			mes "Honor a los guerreros!";
			close;
		}
		if (SkillPoint) {
			mes "[Paladin]";
			mes "Aun es posible que aprendas mas habilidades. Por favor usa";
			mes "todos tus puntos de habilidad restantes antes de regresar conmigo.";
			close;
		}
		jobchange Job_Paladin;
		set ADVJOB,0;
		mes "[Paladin]";
		mes "Felicidades.";
		mes "Como Paladin, espero";
		mes "que protejas a los";
		mes "mas debiles que tu, y nos traigas";
		mes "la victoria en la proxima";
		mes "guerra entre el bien y el mal.";
		close;
	}
	mes "[Paladin]";
	mes "Bienvenido";
	mes "a Valhalla,";
	mes "el Salon del Honor.";
	next;
	mes "[Paladin]";
	mes "Por favor ponte";
	mes "comodo";
	mes "mientras estes aqui.";
	mes "Honor a los guerreros!";
	close;
}
