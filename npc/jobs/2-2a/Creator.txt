valkyrie,53,50,3	script	Biochemist#Valkyrie	122,{
	if (Class == Job_Creator && bioeth == 13 && getskilllv("AM_BIOETHICS") == 0) {
		mes "[Biochemist]";
		mes "Ah, has venido a";
		mes "recuperar los recuerdos";
		mes "que se te perdieron? Si, debes";
		mes "estar aqui por los";
		mes "secretos de la vida que";
		mes "alguna vez fueron tuyos...";
		next;
		if (select("Si:No") == 1) {
			mes "[Biochemist]";
			mes "Cierra los ojos y";
			mes "pon tu mente en calma.";
			mes "Regresaremos a tu";
			mes "pasado para recolectar los";
			mes "fragmentos de tus recuerdos";
			mes "perdidos.";
			next;
			mes "[Biochemist]";
			mes "Cuando abras los ojos,";
			mes "recordaras claramente";
			mes "el secreto de la vida. Tambien";
			mes "recordaras el peso de la";
			mes "responsabilidad de usar estos";
			mes "secretos para los fines correctos...";
			next;
			skill "AM_BIOETHICS",1,SKILL_PERM;
			mes "[Biochemist]";
			mes "Abre los ojos...";
			mes "Ahora que has";
			mes "recordado como";
			mes "crear vida artificial, solo";
			mes "te pido que trates a todas tus";
			mes "creaciones con respeto.";
			close;
		}
		mes "[Biochemist]";
		mes "Si deseas";
		mes "recuperar tus recuerdos";
		mes "perdidos, por favor";
		mes "regresa conmigo.";
		mes "El secreto para crear";
		mes "vida no es cosa menor...";
		close;
	}
	else if (ADVJOB == 0 || Upper != 1) {
		set .@karma_d,rand(1,10);
		if (.@karma_d > 4) {
			mes "[Biochemist]";
			mes "Felicidades.";
			mes "Honor a los guerreros!";
			close;
		}
		mes "[Biochemist]";
		mes "Es extrano que";
		mes "alguien como yo este aqui.";
		mes "Pero incluso alguien habil";
		mes "en las formas de la ciencia";
		mes "puede ser un heroe.";
		next;
		mes "[Biochemist]";
		mes "En este caso,";
		mes "no son necesariamente";
		mes "los medios que he usado, sino";
		mes "los fines por los que he";
		mes "luchado lo que me gano";
		mes "un lugar en Valhalla...";
		close;
	}
	else if (ADVJOB == Job_Creator && Class == Job_Merchant_High && JobLevel > 39) {
		mes "[Biochemist]";
		mes "Si...";
		mes "Ya era hora.";
		mes "Necesitamos mas genios";
		mes "como tu en Midgard.";
		next;
		mes "[Biochemist]";
		mes "Te gustaria";
		mes "convertirte en Biochemist?";
		next;
		if (select("No.:Si.") == 1) {
			mes "[Biochemist]";
			mes "Cuando estes listo,";
			mes "no dudes en regresar.";
			mes "Honor a los guerreros!";
			close;
		}
		if (SkillPoint) {
			mes "[Biochemist]";
			mes "Aun es posible que aprendas mas habilidades. Por favor usa";
			mes "todos tus puntos de habilidad restantes antes de regresar conmigo.";
			close;
		}
		jobchange Job_Creator;
		if (bioeth == 13)
			skill "AM_BIOETHICS",1,SKILL_PERM;
		set ADVJOB,0;
		mes "[Biochemist]";
		mes "Felicidades!";
		mes "Como Biochemist,";
		mes "espero que uses tu";
		mes "vasto conocimiento para los";
		mes "propositos correctos.";
		close;
	}
	else {
		mes "[Biochemist]";
		mes "Bienvenido";
		mes "a Valhalla,";
		mes "el Salon del Honor.";
		next;
		mes "[Biochemist]";
		mes "Por favor ponte";
		mes "comodo";
		mes "mientras estes aqui.";
		mes "Honor a los guerreros!";
		close;
	}
}
