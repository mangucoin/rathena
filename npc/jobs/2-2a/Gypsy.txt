valkyrie,53,56,3	script	Gypsy#Valkyrie	101,{
	if (ADVJOB == 0 || Upper != 1) {
		set .@karma_d,rand(1,10);
		if (.@karma_d > 4) {
			mes "[Gypsy]";
			mes "Felicidades.";
			mes "¡Honor a los guerreros!";
			close;
		}
		mes "[Gypsy]";
		mes "Mueve a la izquierda,";
		mes "mueve a la derecha~";
		mes "¡Y paso...!";
		mes "Bailar puede ser";
		mes "más que un pasatiempo.";
		mes "Para mí, ¡es un estilo de vida~";
		close;
	}
	else if (ADVJOB == Job_Gypsy && Class == Job_Archer_High && JobLevel > 39) {
		mes "[Gypsy]";
		mes "La tierra de Midgard";
		mes "necesita mujeres talentosas";
		mes "para cambiar sutilmente el equilibrio";
		mes "en la batalla entre el bien";
		mes "y el mal.";
		next;
		mes "[Gypsy]";
		mes "¿Estás lista";
		mes "para asumir este rol,";
		mes "y convertirte en Gypsy?";
		next;
		if (select("No.:Sí.") == 1) {
			mes "[Gypsy]";
			mes "Cuando estés lista,";
			mes "no dudes en regresar.";
			mes "¡Honor a los guerreros!";
			close;
		}
		if (SkillPoint) {
			mes "[Gypsy]";
			mes "Aún es posible que aprendas más habilidades. Por favor usa";
			mes "todos tus puntos de habilidad restantes antes de regresar conmigo.";
			close;
		}
		jobchange Job_Gypsy;
		set ADVJOB,0;
		mes "[Gypsy]";
		mes "¡Felicidades!";
		mes "Como Gypsy, sé";
		mes "que tus actuaciones";
		mes "cautivarán los corazones de todos";
		mes "los que estarán observando...";
		close;
	}
	mes "[Gypsy]";
	mes "Bienvenida";
	mes "a Valhalla,";
	mes "el Salón del Honor.";
	next;
	mes "[Gypsy]";
	mes "Por favor ponte";
	mes "cómoda";
	mes "mientras estés aquí.";
	mes "¡Honor a los guerreros!";
	close;
}
