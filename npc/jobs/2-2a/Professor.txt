valkyrie,53,47,3	script	Scholar#Valkyrie	743,{
	if (ADVJOB == 0 || Upper != 1) {
		set .@karma_d,rand(1,10);
		if (.@karma_d > 4) {
			mes "[Scholar]";
			mes "Felicidades.";
			mes "¡Honor a los guerreros!";
			close;
		}
		mes "[Scholar]";
		mes "Toma toda una vida...";
		mes "Literalmente toda una vida";
		mes "para acumular el conocimiento";
		mes "necesario para convertirse";
		mes "en Scholar...";
		next;
		mes "[Scholar]";
		mes "Es abrumador.";
		mes "Cuanto más aprendes, más";
		mes "descubres lo que";
		mes "aún no sabes.";
		mes "No hay fin para el";
		mes "proceso de aprender...";
		close;
	}
	else if (ADVJOB == Job_Professor && Class == Job_Mage_High && JobLevel > 39) {
		mes "[Scholar]";
		mes "Midgard no tiene";
		mes "suficientes Scholars para";
		mes "ayudar a iniciar una nueva era";
		mes "de prosperidad. El";
		mes "mundo te necesita...";
		next;
		mes "[Scholar]";
		mes "¿Asumirás esta";
		mes "gran responsabilidad?";
		mes "¿Servirás a Midgard";
		mes "como Scholar?";
		next;
		if (select("No.:Sí.") == 1) {
			mes "[Scholar]";
			mes "Cuando estés listo,";
			mes "no dudes en regresar.";
			mes "¡Honor a los guerreros!";
			close;
		}
		if (SkillPoint) {
			mes "[Scholar]";
			mes "Aún es posible que aprendas más habilidades. Por favor usa";
			mes "todos tus puntos de habilidad restantes antes de regresar conmigo.";
			close;
		}
		jobchange Job_Professor;
		set ADVJOB,0;
		mes "[Scholar]";
		mes "¡Felicidades!";
		mes "Como Professor, espero";
		mes "que tomes un papel";
		mes "activo en llevar";
		mes "la luz del conocimiento";
		mes "donde hay oscuridad.";
		close;
	}
	mes "[Scholar]";
	mes "Bienvenido";
	mes "a Valhalla,";
	mes "el Salón del Honor.";
	next;
	mes "[Scholar]";
	mes "Por favor ponte";
	mes "cómodo";
	mes "mientras estés aquí.";
	mes "¡Honor a los guerreros!";
	close;
}
