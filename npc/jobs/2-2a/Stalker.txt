valkyrie,53,58,3	script	Stalker#Valkyrie	747,{
	if (ADVJOB == 0 || Upper != 1) {
		set .@karma_d,rand(1,10);
		if (.@karma_d > 4) {
			mes "[Stalker]";
			mes "Felicidades.";
			mes "¡Honor a los guerreros!";
			close;
		}
		mes "[Stalker]";
		mes "Je...";
		mes "Es difícil";
		mes "ser un héroe";
		mes "y ser turbio,";
		mes "poco confiable,";
		mes "escurridizo...";
		next;
		mes "[Stalker]";
		mes "Pero cuando";
		mes "las cosas se ponen difíciles";
		mes "mis compañeros saben que pueden";
		mes "contar conmigo.";
		mes "Yo los necesito y";
		mes "ellos me necesitan.";
		close;
	}
	else if (ADVJOB == Job_Stalker && Class == Job_Thief_High && JobLevel > 39) {
		mes "[Stalker]";
		mes "Este mundo necesita";
		mes "más héroes dispuestos";
		mes "a caminar en la línea";
		mes "entre el orden y la anarquía.";
		next;
		mes "[Stalker]";
		mes "¿Estás listo";
		mes "para unirte a las filas";
		mes "de los guerreros más";
		mes "astutos? ¿Estás listo";
		mes "para convertirte en Stalker?";
		next;
		if (select("No.:Sí.") == 1) {
			mes "[Stalker]";
			mes "Cuando estés listo,";
			mes "no dudes en regresar.";
			mes "¡Honor a los guerreros!";
			close;
		}
		if (SkillPoint) {
			mes "[Stalker]";
			mes "Aún es posible que aprendas más habilidades. Por favor usa";
			mes "todos tus puntos de habilidad restantes antes de regresar conmigo.";
			close;
		}
		jobchange Job_Stalker;
		set ADVJOB,0;
		mes "[Stalker]";
		mes "¡Felicidades!";
		mes "Como Stalker, espero";
		mes "que apuñales a la gente correcta";
		mes "por la espalda. ¡Destierra a los";
		mes "malvados usando sus propios";
		mes "métodos ruines!";
		close;
	}
	mes "[Stalker]";
	mes "Bienvenido";
	mes "a Valhalla,";
	mes "el Salón del Honor.";
	next;
	mes "[Stalker]";
	mes "Por favor ponte";
	mes "cómodo";
	mes "mientras estés aquí.";
	mes "¡Honor a los guerreros!";
	close;
}
