valkyrie,53,58,3	script	Stalker#Valkyrie	747,{
	if (ADVJOB == 0 || Upper != 1) {
		set .@karma_d,rand(1,10);
		if (.@karma_d > 4) {
			mes "[Stalker]";
			mes "Felicidades.";
			mes "Honor a los guerreros!";
			close;
		}
		mes "[Stalker]";
		mes "Je...";
		mes "Es dificil";
		mes "ser un heroe";
		mes "y ser turbio,";
		mes "poco confiable,";
		mes "escurridizo...";
		next;
		mes "[Stalker]";
		mes "Pero cuando";
		mes "las cosas se ponen dificiles";
		mes "mis companeros saben que pueden";
		mes "contar conmigo.";
		mes "Yo los necesito y";
		mes "ellos me necesitan.";
		close;
	}
	else if (ADVJOB == Job_Stalker && Class == Job_Thief_High && JobLevel > 39) {
		mes "[Stalker]";
		mes "Este mundo necesita";
		mes "mas heroes dispuestos";
		mes "a caminar en la linea";
		mes "entre el orden y la anarquia.";
		next;
		mes "[Stalker]";
		mes "Estas listo";
		mes "para unirte a las filas";
		mes "de los guerreros mas";
		mes "astutos? Estas listo";
		mes "para convertirte en Stalker?";
		next;
		if (select("No.:Si.") == 1) {
			mes "[Stalker]";
			mes "Cuando estes listo,";
			mes "no dudes en regresar.";
			mes "Honor a los guerreros!";
			close;
		}
		if (SkillPoint) {
			mes "[Stalker]";
			mes "Aun es posible que aprendas mas habilidades. Por favor usa";
			mes "todos tus puntos de habilidad restantes antes de regresar conmigo.";
			close;
		}
		jobchange Job_Stalker;
		set ADVJOB,0;
		mes "[Stalker]";
		mes "Felicidades!";
		mes "Como Stalker, espero";
		mes "que apunales a la gente correcta";
		mes "por la espalda. Destierra a los";
		mes "malvados usando sus propios";
		mes "metodos ruines!";
		close;
	}
	mes "[Stalker]";
	mes "Bienvenido";
	mes "a Valhalla,";
	mes "el Salon del Honor.";
	next;
	mes "[Stalker]";
	mes "Por favor ponte";
	mes "comodo";
	mes "mientras estes aqui.";
	mes "Honor a los guerreros!";
	close;
}
