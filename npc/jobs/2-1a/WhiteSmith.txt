valkyrie,44,50,5	script	MasterSmith#Valkyrie	731,{
	if (ADVJOB == 0 || Upper != 1) {
		set .@karma_d,rand(1,10);
		if (.@karma_d > 4) {
			mes "[MasterSmith]";
			mes "Felicidades.";
			mes "¡Honor a los guerreros!";
			close;
		}
		mes "[MasterSmith]";
		mes "Ah...";
		mes "La cúspide";
		mes "de la artesanía.";
		mes "Ese es el trabajo de";
		mes "un MasterSmith.";
		next;
		mes "[MasterSmith]";
		mes "Una vez que eres";
		mes "el mejor de los mejores,";
		mes "comienzas a forjar con";
		mes "el objetivo de descubrir";
		mes "formas nuevas y mejores de";
		mes "crear...";
		close;
	}
	else if (ADVJOB == Job_WhiteSmith && Class == Job_Merchant_High && JobLevel > 39) {
		mes "[MasterSmith]";
		mes "¡El momento ha llegado!";
		mes "Nuestro mundo necesita";
		mes "aventureros valientes y";
		mes "trabajadores como tú...";
		next;
		mes "[MasterSmith]";
		mes "¿Te gustaría";
		mes "convertirte en MasterSmith?";
		next;
		if (select("No.:Sí.") == 1) {
			mes "[MasterSmith]";
			mes "Cuando estés listo,";
			mes "no dudes en regresar.";
			mes "¡Honor a los guerreros!";
			close;
		}
		if (SkillPoint) {
			mes "[MasterSmith]";
			mes "Aún es posible que aprendas más habilidades. Por favor usa";
			mes "todos tus puntos de habilidad restantes antes de regresar conmigo.";
			close;
		}
		jobchange Job_Whitesmith;
		set ADVJOB,0;
		mes "[MasterSmith]";
		mes "¡Felicidades!";
		mes "Como MasterSmith,";
		mes "espero que forjes";
		mes "un camino hacia un futuro";
		mes "más brillante para Midgard.";
		close;
	}
	mes "[MasterSmith]";
	mes "Bienvenido";
	mes "a Valhalla,";
	mes "el Salón del Honor.";
	next;
	mes "[MasterSmith]";
	mes "Por favor ponte";
	mes "cómodo";
	mes "mientras estés aquí.";
	mes "¡Honor a los guerreros!";
	close;
}
