valkyrie,44,39,5	script	Lord Knight#Valkyrie	56,{
	if (ADVJOB == 0 || Upper != 1) {
		set .@karma_d,rand(1,10);
		if (.@karma_d > 4) {
			mes "[Lord Knight]";
			mes "Felicidades.";
			mes "Honor a los guerreros!";
			close;
		}
		mes "[Lord Knight]";
		mes "Los Knights tenemos una";
		mes "responsabilidad enorme...";
		mes "Servir y proteger.";
		next;
		mes "[Lord Knight]";
		mes "Incluso a costa";
		mes "de nuestras propias vidas,";
		mes "debemos salvaguardar el";
		mes "bienestar de nuestros companeros.";
		close;
	}
	else if (ADVJOB == Job_Lord_Knight && Class == Job_Swordman_High && JobLevel > 39) {
		mes "[Lord Knight]";
		mes "Tu momento ha llegado!";
		mes "El mundo aun te necesita.";
		mes "Por favor continua tu vida";
		mes "como heroe con una nueva apariencia.";
		next;
		mes "[Lord Knight]";
		mes "Te gustaria";
		mes "convertirte en Lord Knight?";
		next;
		if (select("No.:Si.") == 1) {
			mes "[Lord Knight]";
			mes "Cuando estes listo,";
			mes "no dudes en regresar.";
			mes "Honor a los guerreros!";
			close;
		}
		if (SkillPoint) {
			mes "[Lord Knight]";
			mes "Aun es posible que aprendas mas habilidades. Por favor usa";
			mes "todos tus puntos de habilidad restantes antes de regresar conmigo.";
			close;
		}
		jobchange Job_Lord_Knight;
		set ADVJOB,0;
		mes "[Lord Knight]";
		mes "Felicidades!";
		mes "Como Lord Knight,";
		mes "espero que estes";
		mes "al frente de la batalla,";
		mes "y guies a tus aliados a la victoria!";
		close;
	}
	mes "[Lord Knight]";
	mes "Bienvenido";
	mes "a Valhalla,";
	mes "el Salon del Honor.";
	next;
	mes "[Lord Knight]";
	mes "Por favor ponte";
	mes "comodo";
	mes "mientras estes aqui.";
	mes "Honor a los guerreros!";
	close;
}
