valkyrie,44,39,5	script	Lord Knight#Valkyrie	56,{
	if (ADVJOB == 0 || Upper != 1) {
		set .@karma_d,rand(1,10);
		if (.@karma_d > 4) {
			mes "[Lord Knight]";
			mes "Felicidades.";
			mes "¡Honor a los guerreros!";
			close;
		}
		mes "[Lord Knight]";
		mes "Los Knights tenemos una";
		mes "responsabilidad enorme...";
		mes "Servir y proteger.";
		next;
		mes "[Lord Knight]";
		mes "Incluso a costa";
		mes "de nuestras propias vidas,";
		mes "debemos salvaguardar el";
		mes "bienestar de nuestros compañeros.";
		close;
	}
	else if (ADVJOB == Job_Lord_Knight && Class == Job_Swordman_High && JobLevel > 39) {
		mes "[Lord Knight]";
		mes "¡Tu momento ha llegado!";
		mes "El mundo aún te necesita.";
		mes "Por favor continúa tu vida";
		mes "como héroe con una nueva apariencia.";
		next;
		mes "[Lord Knight]";
		mes "¿Te gustaría";
		mes "convertirte en Lord Knight?";
		next;
		if (select("No.:Sí.") == 1) {
			mes "[Lord Knight]";
			mes "Cuando estés listo,";
			mes "no dudes en regresar.";
			mes "¡Honor a los guerreros!";
			close;
		}
		if (SkillPoint) {
			mes "[Lord Knight]";
			mes "Aún es posible que aprendas más habilidades. Por favor usa";
			mes "todos tus puntos de habilidad restantes antes de regresar conmigo.";
			close;
		}
		jobchange Job_Lord_Knight;
		set ADVJOB,0;
		mes "[Lord Knight]";
		mes "¡Felicidades!";
		mes "Como Lord Knight,";
		mes "espero que estés";
		mes "al frente de la batalla,";
		mes "¡y guíes a tus aliados a la victoria!";
		close;
	}
	mes "[Lord Knight]";
	mes "Bienvenido";
	mes "a Valhalla,";
	mes "el Salón del Honor.";
	next;
	mes "[Lord Knight]";
	mes "Por favor ponte";
	mes "cómodo";
	mes "mientras estés aquí.";
	mes "¡Honor a los guerreros!";
	close;
}
