valkyrie,44,55,5	script	Sniper#Valkyrie	727,{
	if (ADVJOB == 0 || Upper != 1) {
		set .@karma_d,rand(1,10);
		if (.@karma_d > 4) {
			mes "[Sniper]";
			mes "Felicidades.";
			mes "¡Honor a los guerreros!";
			close;
		}
		mes "[Sniper]";
		mes "Un tiro.";
		mes "Una muerte.";
		mes "No es tan difícil";
		mes "una vez que desarrollas";
		mes "la visión para ese estilo";
		mes "de combate.";
		close;
	}
	else if (ADVJOB == Job_Sniper && Class == Job_Archer_High && JobLevel > 39) {
		mes "[Sniper]";
		mes "El mundo necesita";
		mes "arqueros poderosos";
		mes "como tú. ¿Estás listo para";
		mes "la gran responsabilidad?";
		next;
		mes "[Sniper]";
		mes "¿Estás dispuesto a";
		mes "dar el siguiente paso y";
		mes "convertirte en Sniper?";
		next;
		if (select("No.:Sí.") == 1) {
			mes "[Sniper]";
			mes "Cuando estés listo,";
			mes "no dudes en regresar.";
			mes "¡Honor a los guerreros!";
			close;
		}
		if (SkillPoint) {
			mes "[Sniper]";
			mes "Aún es posible que aprendas más habilidades. Por favor usa";
			mes "todos tus puntos de habilidad restantes antes de regresar conmigo.";
			close;
		}
		jobchange Job_Sniper;
		set ADVJOB,0;
		mes "[Sniper]";
		mes "¡Felicidades!";
		mes "Como Sniper, espero";
		mes "que los secuaces del mal";
		mes "nunca estén a salvo mientras";
		mes "estén en";
		mes "tu mira.";
		close;
	}
	mes "[Sniper]";
	mes "Bienvenido";
	mes "a Valhalla,";
	mes "el Salón del Honor.";
	next;
	mes "[Sniper]";
	mes "Por favor ponte";
	mes "cómodo";
	mes "mientras estés aquí.";
	mes "¡Honor a los guerreros!";
	close;
}
