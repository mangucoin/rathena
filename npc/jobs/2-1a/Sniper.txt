valkyrie,44,55,5	script	Sniper#Valkyrie	727,{
	if (ADVJOB == 0 || Upper != 1) {
		set .@karma_d,rand(1,10);
		if (.@karma_d > 4) {
			mes "[Sniper]";
			mes "Felicidades.";
			mes "Honor a los guerreros!";
			close;
		}
		mes "[Sniper]";
		mes "Un tiro.";
		mes "Una muerte.";
		mes "No es tan dificil";
		mes "una vez que desarrollas";
		mes "la vision para ese estilo";
		mes "de combate.";
		close;
	}
	else if (ADVJOB == Job_Sniper && Class == Job_Archer_High && JobLevel > 39) {
		mes "[Sniper]";
		mes "El mundo necesita";
		mes "arqueros poderosos";
		mes "como tu. Estas listo para";
		mes "la gran responsabilidad?";
		next;
		mes "[Sniper]";
		mes "Estas dispuesto a";
		mes "dar el siguiente paso y";
		mes "convertirte en Sniper?";
		next;
		if (select("No.:Si.") == 1) {
			mes "[Sniper]";
			mes "Cuando estes listo,";
			mes "no dudes en regresar.";
			mes "Honor a los guerreros!";
			close;
		}
		if (SkillPoint) {
			mes "[Sniper]";
			mes "Aun es posible que aprendas mas habilidades. Por favor usa";
			mes "todos tus puntos de habilidad restantes antes de regresar conmigo.";
			close;
		}
		jobchange Job_Sniper;
		set ADVJOB,0;
		mes "[Sniper]";
		mes "Felicidades!";
		mes "Como Sniper, espero";
		mes "que los secuaces del mal";
		mes "nunca esten a salvo mientras";
		mes "esten en";
		mes "tu mira.";
		close;
	}
	mes "[Sniper]";
	mes "Bienvenido";
	mes "a Valhalla,";
	mes "el Salon del Honor.";
	next;
	mes "[Sniper]";
	mes "Por favor ponte";
	mes "comodo";
	mes "mientras estes aqui.";
	mes "Honor a los guerreros!";
	close;
}
