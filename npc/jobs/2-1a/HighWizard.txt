valkyrie,44,47,5	script	High Wizard#Valkyrie	735,{
	if (ADVJOB == 0 ||  Upper != 1) {
		set .@karma_d,rand(1,10);
		if (.@karma_d > 4) {
			mes "[High Wizard]";
			mes "Felicidades.";
			mes "Honor a los guerreros!";
			close;
		}
		mes "[High Wizard]";
		mes "Los High Wizards tenemos";
		mes "la responsabilidad de";
		mes "usar nuestra magia destructiva";
		mes "para los propositos correctos.";
		next;
		mes "[High Wizard]";
		mes "Se requiere toda una vida de";
		mes "entrenamiento antes de convertirse";
		mes "en High Wizard. Te imaginas";
		mes "que pasaria si nuestro poder";
		mes "cayera en las manos equivocadas?";
		close;
	}
	else if (ADVJOB == Job_High_Wizard && Class == Job_Mage_High && JobLevel > 39) {
		mes "[High Wizard]";
		mes "Es el momento.";
		mes "Y Midgard necesita";
		mes "a quienes puedan";
		mes "manejar la magia mas poderosa...";
		next;
		mes "[High Wizard]";
		mes "Te gustaria";
		mes "convertirte en High Wizard?";
		next;
		if (select("No.:Si.") == 1) {
			mes "[High Wizard]";
			mes "Cuando estes listo,";
			mes "no dudes en regresar.";
			mes "Honor a los guerreros!";
			close;
		}
		if (SkillPoint) {
			mes "[High Wizard]";
			mes "Aun es posible que aprendas mas habilidades. Por favor usa";
			mes "todos tus puntos de habilidad restantes antes de regresar conmigo.";
			close;
		}
		jobchange Job_high_Wizard;
		set ADVJOB,0;
		mes "[High Wizard]";
		mes "Felicidades.";
		mes "Como High Wizard,";
		mes "espero que uses";
		mes "tus poderes para traer";
		mes "paz a los oprimidos.";
		close;
	}
	mes "[High Wizard]";
	mes "Bienvenido";
	mes "a Valhalla,";
	mes "el Salon del Honor.";
	next;
	mes "[High Wizard]";
	mes "Por favor ponte";
	mes "comodo";
	mes "mientras estes aqui.";
	mes "Honor a los guerreros!";
	close;
}
