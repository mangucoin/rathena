valkyrie,44,42,5	script	High Priest#Valkyrie	60,{
	if (ADVJOB == 0 || Upper != 1) {
		set .@karma_d,rand(1,10);
		if (.@karma_d > 4) {
			mes "[High Priest]";
			mes "Felicidades.";
			mes "¡Honor a los guerreros!";
			close;
		}
		mes "[High Priest]";
		mes "A través del poder";
		mes "de la santidad, que";
		mes "encontremos paz, fuerza";
		mes "y protección. Líbranos";
		mes "de las fuerzas del mal...";
		close;
	}
	else if (ADVJOB == Job_High_Priest && Class == Job_Acolyte_High && JobLevel > 39) {
		mes "[High Priest]";
		mes "Nuestro mundo necesita";
		mes "personas de";
		mes "talento y convicción.";
		mes "Por favor continúa tus";
		mes "buenas obras como un héroe";
		mes "de la santidad aún más grande...";
		next;
		mes "[High Priest]";
		mes "¿Te gustaría";
		mes "convertirte en High Priest?";
		next;
		if (select("No.:Sí.") == 1) {
			mes "[High Priest]";
			mes "Cuando estés listo,";
			mes "no dudes en regresar.";
			mes "¡Honor a los guerreros!";
			close;
		}
		if (SkillPoint) {
			mes "[High Priest]";
			mes "Aún es posible que aprendas más habilidades. Por favor usa";
			mes "todos tus puntos de habilidad restantes antes de regresar conmigo.";
			close;
		}
		jobchange Job_High_Priest;
		set ADVJOB,0;
		mes "[High Priest]";
		mes "Felicidades.";
		mes "Como High Priest,";
		mes "espero que guíes";
		mes "a otros por el camino";
		mes "hacia la santidad...";
		close;
	}
	mes "[High Priest]";
	mes "Bienvenido";
	mes "a Valhalla,";
	mes "el Salón del Honor.";
	next;
	mes "[High Priest]";
	mes "Por favor ponte";
	mes "cómodo";
	mes "mientras estés aquí.";
	mes "¡Honor a los guerreros!";
	close;
}
