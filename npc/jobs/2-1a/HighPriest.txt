valkyrie,44,42,5	script	High Priest#Valkyrie	60,{
	if (ADVJOB == 0 || Upper != 1) {
		set .@karma_d,rand(1,10);
		if (.@karma_d > 4) {
			mes "[High Priest]";
			mes "Felicidades.";
			mes "Honor a los guerreros!";
			close;
		}
		mes "[High Priest]";
		mes "A traves del poder";
		mes "de la santidad, que";
		mes "encontremos paz, fuerza";
		mes "y proteccion. Libranos";
		mes "de las fuerzas del mal...";
		close;
	}
	else if (ADVJOB == Job_High_Priest && Class == Job_Acolyte_High && JobLevel > 39) {
		mes "[High Priest]";
		mes "Nuestro mundo necesita";
		mes "personas de";
		mes "talento y conviccion.";
		mes "Por favor continua tus";
		mes "buenas obras como un heroe";
		mes "de la santidad aun mas grande...";
		next;
		mes "[High Priest]";
		mes "Te gustaria";
		mes "convertirte en High Priest?";
		next;
		if (select("No.:Si.") == 1) {
			mes "[High Priest]";
			mes "Cuando estes listo,";
			mes "no dudes en regresar.";
			mes "Honor a los guerreros!";
			close;
		}
		if (SkillPoint) {
			mes "[High Priest]";
			mes "Aun es posible que aprendas mas habilidades. Por favor usa";
			mes "todos tus puntos de habilidad restantes antes de regresar conmigo.";
			close;
		}
		jobchange Job_High_Priest;
		set ADVJOB,0;
		mes "[High Priest]";
		mes "Felicidades.";
		mes "Como High Priest,";
		mes "espero que guies";
		mes "a otros por el camino";
		mes "hacia la santidad...";
		close;
	}
	mes "[High Priest]";
	mes "Bienvenido";
	mes "a Valhalla,";
	mes "el Salon del Honor.";
	next;
	mes "[High Priest]";
	mes "Por favor ponte";
	mes "comodo";
	mes "mientras estes aqui.";
	mes "Honor a los guerreros!";
	close;
}
