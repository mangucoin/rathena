valkyrie,44,58,5	script	Assassin Cross#Valkyrie	725,{
	if (ADVJOB == 0 || Upper != 1) {
		set .@karma_d,rand(1,10);
		if (.@karma_d > 4) {
			mes "[Assassin Cross]";
			mes "Felicidades...";
			next;
			mes "[Assassin Cross]";
			mes "...";
			next;
			mes "[Assassin Cross]";
			mes "...";
			mes "......";
			next;
			mes "[Assassin Cross]";
			mes "...";
			mes "......";
			mes "Honor a";
			mes "los guerreros.";
			close;
		}
		mes "[Assassin Cross]";
		mes "Somos los guerreros";
		mes "del desierto. Nadie";
		mes "nos menosprecia.";
		mes "Nadie...";
		close;
	}
	else if (ADVJOB == Job_Assassin_Cross && Class == Job_Thief_High && JobLevel > 39) {
		mes "[Assassin Cross]";
		mes "El momento ha llegado.";
		mes "El mundo te necesita...";
		mes "Más que nunca.";
		next;
		mes "[Assassin Cross]";
		mes "Te pido que sigas viviendo en las sombras, pero como un Assassin aún más grande con una nueva apariencia.";
		next;
		mes "[Assassin Cross]";
		mes "¿Te convertirás";
		mes "en un Assassin Cross?";
		next;
		if (select("No.:Sí.") == 1) {
			mes "[Assassin Cross]";
			mes "Cuando estés";
			mes "listo, regresa.";
			next;
			mes "[Assassin Cross]";
			mes "Honor a";
			mes "los guerreros.";
			close;
		}
		if (SkillPoint) {
			mes "[Assassin Cross]";
			mes "Aún no has";
			mes "aprendido todo";
			mes "lo que puedes.";
			next;
			mes "[Assassin Cross]";
			mes "Usa todos tus";
			mes "puntos de habilidad";
			mes "y luego regresa.";
			close;
		}
		jobchange Job_Assassin_Cross;
		set ADVJOB,0;
		mes "[Assassin Cross]";
		mes "Felicidades.";
		mes "Como Assassin Cross,";
		mes "espero que luches por un futuro más brillante dentro de la oscuridad.";
		close;
	}
	mes "[Assassin Cross]";
	mes "Bienvenido";
	mes "a Valhalla,";
	mes "el Salón del Honor.";
	next;
	mes "[Assassin Cross]";
	mes "Por favor ponte";
	mes "cómodo";
	mes "mientras estés aquí.";
	mes "Honor a los guerreros.";
	close;
}
