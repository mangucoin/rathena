			mes "poder de la tierra te proteja";
			mes "en todas tus aventuras~";

				getitem 13180,1;		// Novice_Rifle
				getitem 12149,2;		// Bullet_Case
				getitem 12151,1;		// Bullet_Case_Silver
			} else {
				if (rand(2))
					getitem 13100,1;	// Six_Shooter
				else
					getitem 13150,1;	// Branch
			}
			close;
		}
	} else {
		if (Class == Job_Gunslinger) {
			mes "[Master Miller]";
			mes "¡Oh! Cuánto tiempo sin verte,";
			mes "amigo. ¿Cómo has estado?";
			mes "Espero que hayas mantenido";
			mes "tu arma en buen estado.";
			mes "Cuídala, y ella te cuidará";
			mes "a ti. Recuérdalo.";

			mes "[Master Miller]";
			mes "Si no tienes";
			mes "ningún asunto conmigo,";
