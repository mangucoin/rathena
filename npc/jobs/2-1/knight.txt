			mes "Entonces, vamos con el capitan y decidimos sobre tu cambio de trabajo? No te preocupes demasiado. Eres lo suficientemente bueno para ser Knight!";
			close;
		}
		else if (.@knight_t == 10) {
			set KNIGHT_Q,14;
			changequest 9011,9012;
			mes "[Sir Gray]";
			mes "Disfrute hablar contigo. Aunque, hubo algunas";
			mes "cosas que me molestaron...";
			next;
			mes "[Sir Gray]";
			mes "Deberias ir";
			mes "con el capitan";
			mes "para que decidamos";
			mes "sobre tu cambio de trabajo.";
			next;
			mes "[Sir Gray]";
			mes "No te preocupes demasiado, venir a tomar mi prueba significa que los otros tambien te han reconocido.";
			mes "Ve ahora...!";
			close;
		}
		else {
			set KNIGHT_Q,13;
			mes "[Sir Gray]";
			mes "Conversar";
			mes "con jovenes";
			mes "siempre es agradable...";
			next;
			mes "[Sir Gray]";
			mes "Pero parece que tu sueno esta en otro lugar, o que tu enfoque es difuso. Pasa mas tiempo como Swordman, y regresa";
			mes "conmigo despues.";
			next;
			mes "[Sir Gray]";
			mes "Si realmente deseas convertirte en Knight, primero debes cambiar tu perspectiva. Entonces, ya veremos.";
			close;
		}
	}
	else if (KNIGHT_Q == 14) {
		mes "Te dije";
		mes "que fueras";
		mes "con el capitan.";
		next;
		mes "[Sir Gray]";
		mes "Todos";
		mes "tomaran su decision";
		mes "cuidadosamente,";
		mes "asi que ve ahora!";
		close;
	}
	else {
		mes "Oh...";
		mes "Un joven Swordman.";
		mes "Que puedo hacer por ti?";
		next;
		if (select("Me gustaria tomar la prueba para cambiar de trabajo.:Oh, nada.") == 1) {
			mes "[Sir Gray]";
			mes "Jojo~!";
			mes "Hay muchos";
			mes "otros Knights";
			mes "mas jovenes aqui.";
			next;
			mes "[Sir Gray]";
			mes "Si hablas";
			mes "con todos ellos,";
			mes "puede que te";
			mes "evalue tambien.";
			close;
		}
		mes "[Sir Gray]";
		mes "Cuidate!";
		mes "La salud es";
		mes "el tesoro";
		mes "de todo hombre!";
		close;
	}
