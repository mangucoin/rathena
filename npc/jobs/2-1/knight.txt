			case 2:
				mes "[Sir Gray]";
				mes "Hay muchos Knights similares afuera en el mundo. Piensa en ellos como tus superiores y hazles muchas preguntas.";
				next;
				break;
			case 3:
				set .@knight_t,.@knight_t+15;
				mes "[Sir Gray]";
				mes "Oh no. ¿Consideras la riqueza como prioridad de ser Knight? No estamos hechos para eso. Regresa cuando hayas pensado";
				mes "más al respecto...";
				next;
				break;
			}
			break;
		}
		mes "[Sir Gray]";
		mes "Oh no, hemos estado";
		mes "hablando demasiado...";
		mes "Me disculpo por";
		mes "retenerte aquí";
		mes "por tanto tiempo.";
		next;
		if (.@knight_t == 0) {
			set KNIGHT_Q,14;
			changequest 9011,9012;
			mes "[Sir Gray]";
			mes "Disfruté hablar contigo. Me recuerdas a mí mismo cuando era joven recluta. ¿Vamos a hablar con el capitán y decidir sobre tu";
			mes "cambio de trabajo?";
			next;
			mes "[Sir Gray]";
			mes "No te preocupes";
			mes "demasiado, tengo una muy";
			mes "buena opinión de ti.";
			mes "¡Ahora, ve~!";
			close;
		}
		else if (.@knight_t == 5) {
			set KNIGHT_Q,14;
			changequest 9011,9012;
			mes "[Sir Gray]";
			mes "Disfruté hablar contigo. Puedes pensar más sobre los principios de la Caballería una vez que te conviertas en Knight.";
			next;
			mes "[Sir Gray]";
			mes "Entonces, ¿vamos con el capitán y decidimos sobre tu cambio de trabajo? No te preocupes demasiado. ¡Eres lo suficientemente bueno para ser Knight!";
			close;
		}
		else if (.@knight_t == 10) {
			set KNIGHT_Q,14;
			changequest 9011,9012;
			mes "[Sir Gray]";
			mes "Disfruté hablar contigo. Aunque, hubo algunas";
			mes "cosas que me molestaron...";
			next;
			mes "[Sir Gray]";
			mes "Deberías ir";
			mes "con el capitán";
			mes "para que decidamos";
			mes "sobre tu cambio de trabajo.";
			next;
			mes "[Sir Gray]";
			mes "No te preocupes demasiado, venir a tomar mi prueba significa que los otros también te han reconocido.";
			mes "¡Ve ahora...!";
			close;
		}
		else {
			set KNIGHT_Q,13;
			mes "[Sir Gray]";
			mes "Conversar";
			mes "con jóvenes";
			mes "siempre es agradable...";
			next;
			mes "[Sir Gray]";
			mes "Pero parece que tu sueño está en otro lugar, o que tu enfoque es difuso. Pasa más tiempo como Swordman, y regresa";
			mes "conmigo después.";
			next;
			mes "[Sir Gray]";
			mes "Si realmente deseas convertirte en Knight, primero debes cambiar tu perspectiva. Entonces, ya veremos.";
			close;
		}
	}
	else if (KNIGHT_Q == 14) {
		mes "Te dije";
		mes "que fueras";
		mes "con el capitán.";
		next;
		mes "[Sir Gray]";
		mes "Todos";
		mes "tomarán su decisión";
		mes "cuidadosamente,";
		mes "¡así que ve ahora!";
		close;
	}
	else {
		mes "Oh...";
		mes "Un joven Swordman.";
		mes "¿Qué puedo hacer por ti?";
		next;
		if (select("Me gustaría tomar la prueba para cambiar de trabajo.:Oh, nada.") == 1) {
			mes "[Sir Gray]";
			mes "¡Jojo~!";
			mes "Hay muchos";
			mes "otros Knights";
			mes "más jóvenes aquí.";
			next;
			mes "[Sir Gray]";
			mes "Si hablas";
			mes "con todos ellos,";
			mes "puede que te";
			mes "evalúe también.";
			close;
		}
		mes "[Sir Gray]";
		mes "¡Cuídate!";
		mes "¡La salud es";
		mes "el tesoro";
		mes "de todo hombre!";
		close;
	}
